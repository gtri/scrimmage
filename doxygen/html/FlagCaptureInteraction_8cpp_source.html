66bool FlagCaptureInteraction::init(

71
72 auto callback = [&](scrimmage::MessagePtr<sp::Shape> msg) {
73 if (msg->data.id().id() == flag_boundary_id_) {
74 flag_boundary_shape_ = msg->data;
75 flag_boundary_ = sci::Boundary::make_boundary(msg->data);
76 } else if (msg->data.id().id() == capture_boundary_id_) {
77 capture_boundary_shape_ = msg->data;
78 capture_boundary_ = sci::Boundary::make_boundary(msg->data);
79 }
80 };
81 subscribe<sp::Shape>("GlobalNetwork", "Boundary", callback);
82
83 flag_taken_pub_ = advertise("GlobalNetwork", "FlagTaken");
84 flag_captured_pub_ = advertise("GlobalNetwork", "FlagCaptured");
85
86 return true;

89bool FlagCaptureInteraction::step_entity_interaction(
90 std::list<sc::EntityPtr>& ents,
91 double t,
92 double dt) {
93
94 if (flag_boundary_ == nullptr || capture_boundary_ == nullptr) {
95 return true;
96 }
97
98 // If the flag is not currently captured, determine if an entity from the
99 // opposing team is within the flag's boundary
100 if (!flag_taken_) {
101 for (sc::EntityPtr ent : ents) {
102 if (ent->id().team_id() != flag_boundary_shape_.id().team_id()
103 && flag_boundary_->contains(ent->state_truth()->pos())) {
104 flag_taken_ = true;
105 entity_with_flag_ = ent->id();
106
107 auto msg = std::make_shared<sc::Message<sm::FlagTaken>>();
108 msg->data.set_entity_id(ent->id().id());
109 msg->data.set_entity_team_id(ent->id().team_id());
110 msg->data.set_flag_boundary_id(flag_boundary_shape_.id().id());
111 msg->data.set_flag_team_id(flag_boundary_shape_.id().team_id());
112 flag_taken_pub_->publish(msg);
113 // std::cout << "FLAG " << flag_boundary_info_.id.id()
114 // << " TAKEN by : " << ent->id().id() << std::endl;
115 }
116 }
117 } else if (!flag_captured_) {
118 auto it = id_to_ent_map_->find(entity_with_flag_.id());
119 if (it != id_to_ent_map_->end() && it->second != nullptr
120 && it->second->state_truth() != nullptr) {
121 if (capture_boundary_->contains(it->second->state_truth()->pos())) {
122 flag_captured_ = true;
123
124 auto msg = std::make_shared<sc::Message<sm::FlagCaptured>>();
125 msg->data.set_entity_id(entity_with_flag_.id());
126 msg->data.set_entity_team_id(entity_with_flag_.team_id());
127 msg->data.set_flag_boundary_id(flag_boundary_shape_.id().id());
128 msg->data.set_flag_team_id(flag_boundary_shape_.id().team_id());
129 flag_captured_pub_->publish(msg);
130 // std::cout << "FLAG " << flag_boundary_info_.id.id()
131 // << " captured by : " << entity_with_flag_.id() << std::endl;
132 }
133 } else {
134 flag_taken_ = false;
135 }
136 }
137 return true;
138}
