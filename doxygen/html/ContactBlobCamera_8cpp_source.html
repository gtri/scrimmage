215void ContactBlobCamera::draw_frustum(
216 const std::vector<scrimmage_proto::ShapePtr>& frustum_shapes,
217 double x_rot,
218 double y_rot,
219 double z_rot) {
220 if (frustum_shapes.size() != 8) {
221 std::cerr << "ContactBlobCamera::draw_frustum: ERROR: input shape vector must be size 8"
222 << std::endl;
223 return;
224 }
225 double sensor_footprint_height = max_detect_range_ * tan(el_thresh_ / 2) * 2;
226 double sensor_footprint_width = max_detect_range_ * tan(az_thresh_ / 2) * 2;
227 Eigen::Vector3d sensor_UL(
228 max_detect_range_,
229 sensor_footprint_width / 2,
230 -sensor_footprint_height / 2);
231 Eigen::Vector3d sensor_UR(
232 max_detect_range_,
233 sensor_footprint_width / 2,
234 sensor_footprint_height / 2);
235 Eigen::Vector3d sensor_LR(
236 max_detect_range_,
237 -sensor_footprint_width / 2,
238 sensor_footprint_height / 2);
239 Eigen::Vector3d sensor_LL(
240 max_detect_range_,
241 -sensor_footprint_width / 2,
242 -sensor_footprint_height / 2);
243 std::vector<Eigen::Vector3d> sensor_scene_bb =
244 {sensor_UL, sensor_UR, sensor_LR, sensor_LL, sensor_UL};
245
246 // Rotate into sensor frame
247 auto x_rotation = Eigen::AngleAxisd(x_rot, Eigen::Vector3d::UnitX());
248 auto y_rotation = Eigen::AngleAxisd(y_rot, Eigen::Vector3d::UnitY());
249 auto z_rotation = Eigen::AngleAxisd(z_rot, Eigen::Vector3d::UnitZ());
250 auto A =
251 Eigen::Translation3d(parent_->state_truth()->pos()) * z_rotation * y_rotation * x_rotation;
252 for (size_t i = 0; i < sensor_scene_bb.size(); i++) {
253 sensor_scene_bb[i] = A * sensor_scene_bb[i];
254 }
255
256 // Draw scene box
257 auto scene_bb_line = std::make_shared<sp::Shape>();
258 scene_bb_line->set_opacity(1.0);
259 scene_bb_line->set_persistent(false);
260 scene_bb_line->set_persist_duration(0.0);
261 sc::set(scene_bb_line->mutable_color(), 0, 255, 0);
262
263 sc::path_to_lines(
264 sensor_scene_bb,
265 scene_bb_line,
266 std::static_pointer_cast<EntityPlugin>(shared_from_this()));
267 // draw 8 lines for the 4 points
268 const Eigen::Vector3d fov_edge_start = parent_->state_truth()->pos();
269 for (int idx_bb = 0; idx_bb != 4; ++idx_bb) {
270 // box edges are idx 0-3 in frustum, fov edges are 4-7
271 const size_t idx_fov_edge = idx_bb + 4;
272
273 // set box line
274 Eigen::Vector3d box_edge_start;
275 const Eigen::Vector3d box_edge_end = sensor_scene_bb.at(idx_bb);
276 if (idx_bb == 0) {
277 box_edge_start = sensor_scene_bb.back();
278 } else {
279 box_edge_start = sensor_scene_bb.at(idx_bb - 1);
280 }
281 scrimmage::set(frustum_shapes.at(idx_bb)->mutable_line()->mutable_start(), box_edge_start);
282 scrimmage::set(frustum_shapes.at(idx_bb)->mutable_line()->mutable_end(), box_edge_end);
283
284 // set fov line
285 scrimmage::set(
286 frustum_shapes.at(idx_fov_edge)->mutable_line()->mutable_start(),
287 fov_edge_start);
288 scrimmage::set(
289 frustum_shapes.at(idx_fov_edge)->mutable_line()->mutable_end(),
290 box_edge_end);
291 }
292 // draw all lines
293 for (auto&& shape : frustum_shapes) {
294 draw_shape(shape);
295 }

298void ContactBlobCamera::contacts_to_bounding_boxes(
299 const scrimmage::State& sensor_frame,
300 scrimmage::ContactMap& contacts,
301 std::shared_ptr<sc::Message<ContactBlobCameraType>>& msg) {
302
303 for (auto& kv : contacts) {
304 // Filter out (skip) own contact
305 if (kv.second.id().id() == parent_->id().id())
306 continue;
307
308 // Filter out contacts out of range, should use RTree, but rtree still
309 // requires querying of ID from contact lists. (TODO)
310 if ((kv.second.state()->pos() - sensor_frame.pos()).norm() > max_detect_range_) {
311 continue;
312 }
313
314 // Don't "detect" current contact relative to false negative probability
315 double r = parent_->random()->rng_uniform(0.0, 1.0);
316 if (r < fn_prob_)
317 continue;
318
319 // Transform contact into "camera" coordinate system
320 Eigen::Vector3d rel_pos = sensor_frame.rel_pos_local_frame(kv.second.state()->pos());
321
322 if (!in_field_of_view(rel_pos))
323 continue;
324
325 // Convert 3D relative position to 2d image plane position
326 Eigen::Vector2d raster_center = project_rel_3d_to_2d(rel_pos);
327
328 double object_radius = kv.second.radius();
329
330 // Get angle between (vector between camera and object center) and
331 // (vector between object center and line tangent to object circle)
332 double beta = acos(object_radius / rel_pos.norm());
333
334 // Get vector pointing from contact to camera
335 Eigen::Vector2d v_r(-rel_pos(0), -rel_pos(1));
336
337 // Normalize to unit vector and Make length equal to object radius
338 v_r = v_r.normalized() * object_radius;
339
340 // Rotate one direction and add circle's center point
341 Eigen::Vector2d v_r_1 = v_r, v_r_2 = v_r;
342 Eigen::Rotation2D<double> rot1(beta);
343 Eigen::Rotation2D<double> rot2(-beta);
344 v_r_1 = rot1.toRotationMatrix() * v_r_1;
345 v_r_2 = rot2.toRotationMatrix() * v_r_2;
346
347 // Get 3D relative position of rotated vectors
348 Eigen::Vector3d p1(rel_pos(0) + v_r_1(0), rel_pos(1) + v_r_1(1), rel_pos(2));
349
350 Eigen::Vector3d p2(rel_pos(0) + v_r_2(0), rel_pos(1) + v_r_2(1), rel_pos(2));
351
352 // Get 2D position of object boundaries
353 Eigen::Vector2d r1 = project_rel_3d_to_2d(p1);
354 Eigen::Vector2d r2 = project_rel_3d_to_2d(p2);
355
356 // Calculate image radius using distance between object boundaries
357 double object_img_radius = (r1 - r2).norm() / 2.0;
358
359 double object_img_half_width =
360 std::max(1.0, object_img_radius + (*size_noise_[0])(*gener_));
361 double object_img_half_height =
362 std::max(1.0, object_img_radius + (*size_noise_[1])(*gener_));
363
364 // Add noise to position in 2D image plane
365 raster_center(0) += (*pos_noise_[0])(*gener_);
366 raster_center(1) += (*pos_noise_[1])(*gener_);
367
368 // Add bounding box to frame around object
369 cv::Rect rect(
370 std::floor(raster_center(0)) - std::floor(object_img_half_width),
371 std::floor(raster_center(1)) - std::floor(object_img_half_height),
372 std::floor(object_img_half_width * 2) + 1,
373 std::floor(object_img_half_height * 2) + 1);
374
375 if (object_img_radius > 0) {
376 draw_object_with_bounding_box(
377 msg->data.frame,
378 kv.second.id().id(),
379 rect,
380 raster_center,
381 object_img_radius);
382 }
383
384 // Collect all bounding boxes for current detected object
385 msg->data.bounding_boxes[kv.second.id().id()].push_back(rect);
386 }

389void ContactBlobCamera::add_false_positives(
390 std::shared_ptr<scrimmage::Message<ContactBlobCameraType>>& msg) {
391
392 for (int i = 0; i < max_false_positives_; i++) {
393 double r = parent_->random()->rng_uniform(0.0, 1.0);
394 if (r > fp_prob_)
395 continue;
396
397 // Generate random center position in frame
398 Eigen::Vector2d raster_center(
399 parent_->random()->rng_uniform_int(0, img_width_),
400 parent_->random()->rng_uniform_int(0, img_height_));
401
402 double object_img_radius = parent_->random()->rng_uniform(0.1, 20.0);
403
404 // Add bounding box to frame around object
405 cv::Rect rect(
406 std::floor(raster_center(0)) - std::floor(object_img_radius),
407 std::floor(raster_center(1)) - std::floor(object_img_radius),
408 std::floor(object_img_radius * 2) + 1,
409 std::floor(object_img_radius * 2) + 1);
410
411 // Generate a random ID
412 int id = parent_->random()->rng_uniform_int(1, 100);
413
414 if (object_img_radius > 0) {
415 draw_object_with_bounding_box(
416 msg->data.frame,
417 id,
418 rect,
419 raster_center,
420 object_img_radius);
421 }
422 msg->data.bounding_boxes[id].push_back(rect);
423 }

426bool ContactBlobCamera::step() {
427 if ((time_->t() - last_frame_t_) < 1.0 / fps_)
428 return true;
429
430 sc::State sensor_frame;
431 sensor_frame.quat() =
432 static_cast<sc::Quaternion>(parent_->state_truth()->quat() * this->transform()->quat());
433 sensor_frame.pos() = this->transform()->pos() + parent_->state_truth()->pos();
434
435 if (show_frustum_) {
436 draw_frustum(
437 frustum_shapes_,
438 sensor_frame.quat().roll(),
439 sensor_frame.quat().pitch(),
440 sensor_frame.quat().yaw());
441 }
442
443 auto msg = std::make_shared<sc::Message<ContactBlobCameraType>>();
444
445 msg->data.roll = sc::Angles::rad2deg(this->transform()->quat().roll());
446 msg->data.pitch = sc::Angles::rad2deg(this->transform()->quat().pitch());
447 msg->data.yaw = sc::Angles::rad2deg(this->transform()->quat().yaw());
448 msg->data.camera_id = camera_id_;
449 msg->data.img_width = img_width_;
450 msg->data.img_height = img_height_;
451 msg->data.max_detect_range = max_detect_range_;
452 msg->data.focal_length = focal_length_;
453 msg->data.fps = fps_;
454 msg->data.az_thresh = az_thresh_;
455 msg->data.el_thresh = el_thresh_;
456 msg->data.fn_prob = fn_prob_;
457 msg->data.fp_prob = fp_prob_;
458 msg->data.max_false_positives = max_false_positives_;
459
460 msg->data.frame = cv::Mat::zeros(img_height_, img_width_, CV_8UC3);
461
462 if (not ignore_real_entities_) {
463 // Compute bounding boxes for real contacts
464 contacts_to_bounding_boxes(sensor_frame, *(parent_->contacts()), msg);
465 }
466
467 // Compute bounding boxes for added "simulated" contacts
468 contacts_to_bounding_boxes(sensor_frame, sim_contacts_, msg);
469
470 if (show_sim_contacts_ && time_->t() > last_contact_send_time_ + contact_send_dt_) {
471 // draw_sim_contacts(sim_contacts_);
472 for (auto& kv : sim_contacts_) {
473 // Draw a sphere at the expected commanded state
474 Eigen::Vector3d sphere_center(
475 kv.second.state()->pos().x(),
476 kv.second.state()->pos().y(),
477 kv.second.state()->pos().z());
478
479 sc::set(sim_tgt_sphere_->mutable_sphere()->mutable_center(), sphere_center);
480 sc::set(sim_tgt_sphere_->mutable_color(), 0, 255, 255);
481 sim_tgt_sphere_->mutable_sphere()->set_radius(kv.second.radius());
482 sim_tgt_sphere_->set_opacity(0.7);
483 sim_tgt_sphere_->set_persistent(false);
484 draw_shape(sim_tgt_sphere_);
485 }
486
487 last_contact_send_time_ = time_->t();
488 }
489
490 // Add false positives
491 add_false_positives(msg);
492
493 frame_ = msg->data.frame;
494 last_frame_t_ = time_->t();
495
496 if (show_image_) {
497 std::string window_name = window_name_ + "-" + std::to_string(parent_->id().id()) + "-"
498 + std::to_string(camera_id_);
499 cv::imshow(window_name.c_str(), msg->data.frame);
500 cv::waitKey(1);
501 }
502
503 // Log any detections
504 if (log_detections_) {
505 detections_file_ << time_->t();
506 for (auto elem : msg->data.bounding_boxes) {
507 detections_file_ << ", " << elem.first;
508 }
509 detections_file_ << std::endl;
510 }
511
512 pub_->publish(msg);
513 return true;
514}

516Eigen::Vector2d ContactBlobCamera::project_rel_3d_to_2d(Eigen::Vector3d rel_pos) {
517 // 3D to 2D transforms are described here:
518 // http://www.scratchapixel.com/lessons/3d-basic-rendering/computing-pixel-coordinates-of-3d-point/mathematics-computing-2d-coordinates-of-3d-points
519
520 // Convert relative position of contact into computer vision coordinates
521 // for relative coordinates:
522 Eigen::Vector3d pos(rel_pos(1), rel_pos(2), rel_pos(0));
523
524 // Transform to screen / canvas space
525 Eigen::Vector2d s(focal_length_ * -pos(0) / pos(2), focal_length_ * pos(1) / pos(2));
526
527 // Transform to Normalized Device Coordinate (NDC) space
528 Eigen::Vector2d ndc(
529 (s(0) + canvas_width_ / 2.0) / canvas_width_,
530 (s(1) + canvas_height_ / 2.0) / canvas_height_);
531
532 // Tranform to raster space
533 Eigen::Vector2d r(ndc(0) * img_width_, (1 - ndc(1)) * img_height_);
534 return r;
535}

537bool ContactBlobCamera::in_field_of_view(Eigen::Vector3d rel_pos) {
538 double az = atan2(rel_pos(1), rel_pos(0));
539 double norm_xy = sqrt(pow(rel_pos(0), 2) + pow(rel_pos(1), 2));
540 double el = atan2(rel_pos(2), norm_xy);
541 if (std::abs(az) > (az_thresh_ / 2) || std::abs(el) > (el_thresh_ / 2)) {
542 return false;
543 }
544 return true;
545}

580void ContactBlobCamera::set_plugin_params(std::map<std::string, double> params) {
581 if (params["senderId"] != parent_->id().id()) {
582 return;
583 }
584
585 camera_id_ = static_cast<int>(params["camera_id"]);
586
587 img_width_ = static_cast<int>(params["img_width"]);
588 img_height_ = static_cast<int>(params["img_height"]);
589
590 max_detect_range_ = params["max_detect_range"];
591 focal_length_ = params["focal_length"];
592 fps_ = params["fps"];
593 az_thresh_ = params["az_thresh"];
594 el_thresh_ = params["el_thresh"];
595
596 fn_prob_ = params["fn_prob"];
597 fp_prob_ = params["fp_prob"];
598 max_false_positives_ = params["max_false_positives"];
599 std_dev_w_ = params["std_dev_w"];
600 std_dev_h_ = params["std_dev_h"];
601
602 if (std_dev_w_ > img_width_) {
603 std_dev_w_ = 10;
604 } // TODO: use standard deviation values to inform the false positives
605 if (std_dev_h_ > img_height_) {
606 std_dev_h_ = 10;
607 }
608
609 canvas_width_ = 2 * focal_length_ * tan(az_thresh_ / 2.0);
610 canvas_height_ = 2 * focal_length_ * tan(el_thresh_ / 2.0);
611
612 // keep track of values of the parameters
613 char buf[100];
614 snprintf(buf, sizeof(buf), "\n===================================\n");
615 parameters_file_ << buf << std::endl;
