97bool LOSSensor::step() {
98 // Call the super class first
99 bool retVal = RayTrace::step();
100 if (retVal == false) {
101 return retVal;
102 }
103 // Check whether an update is available if not using the collision service
104 if (use_flat_earth_ == true) {
105 if (time_->t() < (last_update_time_ + max_sample_rate())) {
106 // No update, no message sent
107 return retVal;
108 }
109 } else { // Updated rate managed by collision system
110 if (subscribed_ == false) {
111 subscribed_ = true;
112 // Register for the collision callbacks. Assume set up for local network publish
113 auto pc_cb = [&](scrimmage::MessagePtr<sensor::RayTrace::PointCloud> msg) {
114 range_ = msg->data.points[0].point.norm();
115 oor_ = msg->data.points[0].oor;
116 new_data_ = true;
117 };
118 std::string topic_name =
119 std::to_string(parent_->id().id()) + "/" + name() + "/pointcloud";
120 subscribe<sensor::RayTrace::PointCloud>("LocalNetwork", topic_name, pc_cb);
121 printf("LOSSensor: Subscribing with topic name %s\n", topic_name.c_str());
122 }
123 if (new_data_ == false) {
124 return retVal;
125 }
126 }
127
128 // Can update
129 last_update_time_ = time_->t();
130 new_data_ = false;
131
132 // Coordinate systems used:
133 // Local level: East/North/Up
134 // Body: Nose/Left/Up
135
136 bool oor = false;
137 double range = 0.0;
138 if (use_flat_earth_ == true) {
139 // Get appropriate conversion
140 Eigen::Matrix4d tf_m = parent_->state_truth()->tf_matrix(false) * transform()->tf_matrix();
141 // Transform sensor's origin to world coordinates
142 Eigen::Vector4d sensor_pos = tf_m * Eigen::Vector4d(0, 0, 0, 1);
143 Eigen::Vector3d sensor_pos_w = sensor_pos.head<3>() + parent_->state_truth()->pos();
144
145 // Transform ray's end point to world coordinates
146 Eigen::Vector4d ray = tf_m * Eigen::Vector4d(1.0, 0.0, 0.0, 1.0);
147 Eigen::Vector3d ray_w = ray.head<3>();
148
149 // Verify that this will intersect the ground
150 if ((sensor_pos_w(2) < 0.0) || (ray_w(2) >= 0.0)) { // Out of range
151 oor = true;
152 } else { // Potentially in range
153 // Project the unit vector intersection to the ground.
154 range = sensor_pos_w(2) / (-1.0 * ray_w(2));
155 }
156 } else {
157 // Use a service to get range to collision
158 // Without a service available, this returns oor at all times
159 range = range_;
160 oor = oor_;
161 }
162
163 if ((oor == false) && (probability_oor_error_ > 0.0)) {
164 // Check the random error to see whether it could swap in-range/out-of-range.
165 bool swap_in_out_range = (static_cast<double>(std::rand()) / static_cast<double>(RAND_MAX))
166 < probability_oor_error_;
167
168 if ((((range < min_range()) || (range > max_range())) && !swap_in_out_range)
169 || (((range > min_range()) && (range < max_range())) && swap_in_out_range)) {
170 // Out of range
171 oor = true;
172 } else { // In-range
173 // Limit to the range boundaries (in case an error caused an in-range
174 // measurement when it was actually out of range)
175 range = std::min(std::max(range, min_range()), max_range());
176 }
177 }
178
179 // Now that in-range versus out-of-range is known as well as the actual range,
180 // add error.
181 auto msg = std::make_shared<sc::Message<sm::LOSRangeMessage>>();
182 msg->data.set_sensor_id(sensor_id_);
183 if (oor == false) {
184 // Add error
185 if ((static_cast<double>(std::rand()) / static_cast<double>(RAND_MAX))
186 < probability_of_error_) {
187 // Add additional error
188 std::normal_distribution<double> error_dist(0.0, error_sd_);
189 double error = error_dist(generator_);
190 range = std::min(std::max(range + error, min_range()), max_range());
191 }
192
193 // Finally, add the actual noise
194 double total_sd = range_sd_min_ + (range_sd_per_unit_ * range);
195 if (total_sd > 0.0) {
196 std::normal_distribution<double> noise_dist(0.0, total_sd);
197 range = std::min(std::max(range + noise_dist(generator_), min_range()), max_range());
198 }
199 // Set the values to the message
200 msg->data.set_range(range);
201 msg->data.set_range_var(total_sd * total_sd);
202 } else { // Out of range
203 // Set the values to the message
204 msg->data.set_range(oor_return_);
205 msg->data.set_range_var(range_sd_oor_ * range_sd_oor_);
206 }
207
208 // Publish the message and return all good
209 los_pub_->publish(msg);
210 return retVal;
211}
