73bool Predator::step_autonomy(double t, double dt) {
74 if (contacts_->count(follow_id_) == 0) {
75 follow_id_ = -1;
76 }
77
78 if (follow_id_ < 0 || allow_prey_switching_) {
79 // Find nearest entity on other team. Loop through each contact, calculate
80 // distance to entity, save the ID of the entity that is closest.
81 double min_dist = std::numeric_limits<double>::infinity();
82 for (auto it = contacts_->begin(); it != contacts_->end(); it++) {
83 // Skip if this contact isn't on team 1 (prey)
84 // if (it->second.id().team_id() == parent_->id().team_id()) continue;
85 if (it->second.id().team_id() != prey_team_id_)
86 continue;
87
88 // Calculate distance to entity
89 double dist = (it->second.state()->pos() - state_->pos()).norm();
90 if (dist < min_dist) {
91 // If this is the minimum distance, save distance and reference
92 // to entity
93 min_dist = dist;
94 follow_id_ = it->first;
95 }
96 }
97 }
98
99 // If any non-team members are within capture range, publish capture
100 // message
101 for (auto it = contacts_->begin(); it != contacts_->end(); it++) {
102 // Skip if this contact is on the same team
103 if (it->second.id().team_id() == parent_->id().team_id())
104 continue;
105
106 // Calculate distance to entity
107 double dist = (it->second.state()->pos() - state_->pos()).norm();
108 // Publish capture message if within capture range
109 if (dist < capture_range_) {
110 auto msg = std::make_shared<Message<sm::CaptureEntity>>();
111 msg->data.set_source_id(parent_->id().id());
112 msg->data.set_target_id(it->second.id().id());
113 capture_ent_pub_->publish(msg);
114 }
115 }
116
117 // Head toward entity on other team
118 if (contacts_->count(follow_id_) > 0) {
119 // Get a reference to the entity's state.
120 StatePtr ent_state = contacts_->at(follow_id_).state();
121
122 // Calculate max velocity towards target entity:
123 Eigen::Vector3d v = (ent_state->pos() - state_->pos()).normalized() * max_speed_;
124
125 // Convert to spherical coordinates:
126 double desired_heading = atan2(v(1), v(0));
127 double desired_pitch = atan2(v(2), v.head<2>().norm());
128
129 vars_.output(speed_idx_, max_speed_);
130 vars_.output(turn_rate_idx_, Angles::angle_pi(desired_heading - state_->quat().yaw()));
131 vars_.output(pitch_rate_idx_, Angles::angle_pi(desired_pitch + state_->quat().pitch()));
132
133 vars_.output(desired_heading_idx_, desired_heading);
134 vars_.output(desired_speed_idx_, max_speed_);
135 }
136
137 return true;
138}
