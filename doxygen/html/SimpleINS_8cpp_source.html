90bool SimpleINS::step() {
91 auto gener = parent_->random()->gener();
92 auto msg_SS = std::make_shared<sc::Message<double>>();
93 msg_SS->data = sea_state_gps_;
94 pub_seastate_->publish(msg_SS);
95
96 // Make a copy of the current state
97 StateWithCovariance ns(*(parent_->state_truth()));
98
99 auto msg = std::make_shared<Message<StateWithCovariance>>();
100
101 if (init_m_) {
102 m_ = Eigen::MatrixXd::Identity(ns.covariance().rows(), ns.covariance().cols());
103 init_m_ = false;
104 }
105
106 // Acceleration
107 accel_ = ns.vel() - vel_Nminus1;
108 vel_Nminus1 = ns.vel();
109 for (int i = 0; i < 3; i++) {
110 accel_(i) = abs(accel_(i) / .01);
111 if (accel_(i) > 1) {
112 accel_(i) = 1;
113 }
114 }
115
116 // Use gen in order to create a growing covariance
117 double pos_noise_0 = (*pos_noise_[0])(*gener) * (accel_(0));
118 double pos_noise_1 = (*pos_noise_[1])(*gener) * (accel_(1));
119 const int arrayNum[4] = {15, 30, 45, 60};
120
121 if (!gps_fix_) {
122 m_(0, 0) += (pos_noise_0 < 0 ? -1 * pos_noise_0 : pos_noise_0);
123 m_(1, 1) += (pos_noise_1 < 0 ? -1 * pos_noise_1 : pos_noise_1);
124
125 prev_time_ = time_->t();
126 int randIdx = rand() % 4;
127 surface_timer_ = arrayNum[randIdx];
128 if (sea_state_gps_ > 1) {
129 surface_timer_ += 60;
130 }
131 } else {
132 // GPS fix on - wait on surface timer and snap back to an identity matrix
133 m_(0, 0) += (pos_noise_0 < 0 ? -1 * pos_noise_0 : pos_noise_0);
134 m_(1, 1) += (pos_noise_1 < 0 ? -1 * pos_noise_1 : pos_noise_1);
135 // GPS fix on - wait on surface timer and snap back to an identity matrix
136 if (time_->t() - prev_time_ > surface_timer_) {
137 m_ = Eigen::MatrixXd::Identity(ns.covariance().rows(), ns.covariance().cols());
138 // Reset the position error accumulator
139 pos_error_accum_ = Eigen::Vector3d::Zero();
140 }
141 }
142
143 // Create noisy position / velocity
144 for (int i = 0; i < 3; i++) {
145 // std::cout << "Accelration for " << i << ", is: " << (accel_(i)) << std::endl;
146 pos_error_accum_(i) += (*pos_noise_[i])(*gener) * (accel_(i));
147 // msg->data.pos()(i) = ns.pos()(i) + (*pos_noise_[i])(*gener);
148 msg->data.vel()(i) = ns.vel()(i) + (*vel_noise_[i])(*gener);
149 }
150 msg->data.pos() = ns.pos() + pos_error_accum_;
151
152 msg->data.quat() = ns.quat() * Quaternion(Eigen::Vector3d::UnitX(), (*orient_noise_[0])(*gener))
153 * Quaternion(Eigen::Vector3d::UnitY(), (*orient_noise_[1])(*gener))
154 * Quaternion(Eigen::Vector3d::UnitZ(), (*orient_noise_[2])(*gener));
155
156 msg->data.set_covariance(m_);
157
158 double sig_L;
159 double sig_S;
160
161 if (m_(0, 0) < m_(1, 1)) {
162 sig_L = sqrt(m_(1, 1));
163 sig_S = sqrt(m_(0, 0));
164 } else {
165 sig_L = sqrt(m_(0, 0));
166 sig_S = sqrt(m_(1, 1));
167 }
168 double sig_w = sig_S / sig_L;
169 double CEP;
170 if (sig_w < 0.5) {
171 CEP = sig_L * (0.67 + 0.8 * sig_w * sig_w);
172 } else {
173 CEP = 0.59 * sig_L * (1 + sig_w);
174 }
175 auto msg_cep = std::make_shared<sc::Message<double>>();
176 msg_cep->data = CEP;
177 pub_cep_->publish(msg_cep);
178 // Publish StateWCovariance msg
179 pub_->publish(msg);
180
181 // Update the entity's state.
182 parent_->set_state_belief(static_cast<sc::State>(msg->data));
183
184 return true;
185}
