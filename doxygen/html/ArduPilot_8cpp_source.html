170bool ArduPilot::step_autonomy(double t, double dt) {
171
172 // Convert state6dof into ArduPilot fdm_packet and transmit
173 fdm_packet fdm_pkt = state6dof_to_fdm_packet(t, *state_6dof_);
174 try {
175 // TODO: Michael, endian handling?
176 tx_socket_->send_to(ba::buffer(&fdm_pkt, sizeof(fdm_packet)), tx_endpoint_);
177 ntx++;
178 last_ntx++;
179 } catch (std::exception& e) {
180 cerr << "Exception: " << e.what() << "\n";
181 }
182
183 // Let async receive callback run
184 if (asynchonous_mode_) {
185 recv_io_service_.poll();
186 } else {
187 boost::system::error_code error;
188 ba::ip::udp::socket::message_flags flags = 0;
189 std::size_t nbytes = recv_socket_->receive_from(
190 ba::buffer(recv_buffer_),
191 recv_remote_endpoint_,
192 flags,
193 error);
194
195 parse_receive(error, nbytes);
196 }
197
198 // Copy the received servo commands into the desired state
199 servo_pkt_mutex_.lock();
200 for (auto servo_tf : servo_tfs_) {
201 vars_.output(
202 servo_tf.vector_index(),
203 servo_tf.scale(servo_pkt_.servos[servo_tf.axis_index()]));
204 // int prec = 9;
205 // cout << std::setprecision(prec) << "servo_out"<< servo_tf.axis_index() << ": " <<
206 // servo_tf.scale(servo_pkt_.servos[servo_tf.axis_index()]) << endl;
207 }
208 servo_pkt_mutex_.unlock();
209
210 if (t - last_print_t > 5) {
211 const auto t0 = std::chrono::system_clock::now();
212 auto d = std::chrono::duration_cast<std::chrono::microseconds>(t0 - last_print_wall_t);
213 cout << "walltime: " << t0.time_since_epoch().count() << ", simtime: " << t
214 << ", packets tx: " << ntx << ", rx: " << nrx
215 << ", rate: " << double(last_ntx) / double(d.count()) * 1e6 << " hz" << endl;
216 last_print_t = t;
217 last_print_wall_t = t0;
218 last_ntx = 0;
219 }
220
221 return true;
222}

256ArduPilot::fdm_packet ArduPilot::state6dof_to_fdm_packet(
257 double t,
258 sc::motion::RigidBody6DOFState& state) {
259
260 fdm_packet fdm_pkt = {0};
261
262 fdm_pkt.timestamp_us = t * 1e6;
263
264 // x/y/z to lat/lon/alt conversion
265 parent_->projection()->Reverse(
266 state.pos()(0),
267 state.pos()(1),
268 state.pos()(2),
269 fdm_pkt.latitude,
270 fdm_pkt.longitude,
271 fdm_pkt.altitude);
272
273 // convert everything to NED-FRU world and body frames
274
275 // Heading conversion
276 angles_to_gps_.set_angle(sc::Angles::rad2deg(state.quat().yaw()));
277 fdm_pkt.heading = sc::Angles::deg2rad(angles_to_gps_.angle());
278
279 fdm_pkt.speedN = state.vel()(1);
280 fdm_pkt.speedE = state.vel()(0);
281 fdm_pkt.speedD = -state.vel()(2);
282
283 // Body frame linear acceleration FRU
284 if (mavproxy_mode_) {
285 Eigen::Vector3d specific_force_body =
286 state.linear_accel_body() - state.quat().rotate_reverse(Eigen::Vector3d(0, 0, -9.81));
287 fdm_pkt.xAccel = specific_force_body(0);
288 fdm_pkt.yAccel = -specific_force_body(1);
289 fdm_pkt.zAccel = -specific_force_body(2);
290 } else {
291 fdm_pkt.xAccel = state.linear_accel_body()(0);
292 fdm_pkt.yAccel = -state.linear_accel_body()(1);
293 fdm_pkt.zAccel = -state.linear_accel_body()(2);
294 }
295
296 // Global frame, roll, pitch, yaw from NED to FRU
297 fdm_pkt.roll = state.quat().roll();
298 fdm_pkt.pitch = -state.quat().pitch_safe();
299 fdm_pkt.yaw = fdm_pkt.heading;
300
301 if (mavproxy_mode_) {
302 // world frame rotational velocities
303 double p = state.ang_vel_body()(0);
304 double q = -state.ang_vel_body()(1);
305 double r = -state.ang_vel_body()(2);
306 double sphi = sin(fdm_pkt.roll);
307 double cphi = cos(fdm_pkt.roll);
308 double tth = tan(fdm_pkt.pitch);
309 double cth = std::max(.00001, cos(fdm_pkt.pitch));
310 fdm_pkt.rollRate = p + tth * sphi * q + tth * cphi * r;
311 fdm_pkt.pitchRate = cphi * q - sphi * r;
312 fdm_pkt.yawRate = sphi / cth * q + cphi / cth * r;
313 } else {
314 // Body frame rotational velocities FRU
315 fdm_pkt.rollRate = state.ang_vel_body()(0);
316 fdm_pkt.pitchRate = -state.ang_vel_body()(1);
317 fdm_pkt.yawRate = -state.ang_vel_body()(2);
318 }
319
320 // Airspeed is magnitude of velocity vector for now
321 fdm_pkt.airspeed = (state.vel() + state.wind()).norm();
322 if (std::isnan(fdm_pkt.airspeed)) {
323 // The norm of a small vector can be NaN, so most likely, the airspeed
324 // should be zero in this case.
325 fdm_pkt.airspeed = 0.0;
326 }
327
328 if (std::isinf(fdm_pkt.airspeed)) {
329 std::cout << "ArduPilot: Warning: velocity or wind contains infinite value." << std::endl;
330 fdm_pkt.airspeed = 0.0;
331 }
332
333#if 0
334 cout << "--------------------------------------------------------" << endl;
335 cout << " State information being sent to ArduPilot" << endl;
336 cout << "--------------------------------------------------------" << endl;
337 int prec = 9;
338 cout << std::setprecision(prec) << "timestamp_us: " << fdm_pkt.timestamp_us << endl;
339 cout << std::setprecision(prec) << "Latitude: " << fdm_pkt.latitude << endl;
340 cout << std::setprecision(prec) << "Longitude: " << fdm_pkt.longitude << endl;
341 cout << std::setprecision(prec) << "Altitude: " << fdm_pkt.altitude << endl;
342 cout << std::setprecision(prec) << "Airspeed: " << fdm_pkt.airspeed << endl;
343 cout << std::setprecision(prec) << "Roll: " << fdm_pkt.roll << endl;
344 cout << std::setprecision(prec) << "Pitch: " << fdm_pkt.pitch << endl;
345 cout << std::setprecision(prec) << "Yaw: " << fdm_pkt.yaw << endl;
346 cout << std::setprecision(prec) << "xAccel: " << fdm_pkt.xAccel << endl;
347 cout << std::setprecision(prec) << "yAccel: " << fdm_pkt.yAccel << endl;
348 cout << std::setprecision(prec) << "zAccel: " << fdm_pkt.zAccel << endl;
349 cout << std::setprecision(prec) << "speedN: " << fdm_pkt.speedN << endl;
350 cout << std::setprecision(prec) << "speedE: " << fdm_pkt.speedE << endl;
351 cout << std::setprecision(prec) << "speedD: " << fdm_pkt.speedD << endl;
352 cout << std::setprecision(prec) << "heading: " << fdm_pkt.heading << endl;
353 cout << std::setprecision(prec) << "rollRate: " << fdm_pkt.rollRate << endl;
354 cout << std::setprecision(prec) << "pitchRate: " << fdm_pkt.pitchRate << endl;
355 cout << std::setprecision(prec) << "yawRate: " << fdm_pkt.yawRate << endl;
356#endif
357
358 return fdm_pkt;
359}
