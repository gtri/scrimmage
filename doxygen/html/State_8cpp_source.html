101bool State::InFieldOfView(State& other, double fov_width, double fov_height) const {
102 Eigen::Vector3d rel_pos = this->rel_pos_local_frame(other.pos());
103 double az = atan2(rel_pos(1), rel_pos(0));
104 double norm_xy = sqrt(pow(rel_pos(0), 2) + pow(rel_pos(1), 2));
105 double el = atan2(rel_pos(2), norm_xy);
106 return std::abs(az) < fov_width / 2 && std::abs(el) < fov_height / 2;
107}

125double State::rel_az(const Eigen::Vector3d& other) const {
126 Eigen::Vector3d diff = other - pos_;
127 double az = atan2(diff(1), diff(0));

131Eigen::Matrix4d State::tf_matrix(bool enable_translate) {
132 // Construct 4x4 matrix of the form:
133 // |rot|rot|rot| sx |
134 // |rot|rot|rot| sy |
135 // |rot|rot|rot| sz |
136 // | x | y | z | 1 |
137 // scales are zero (sx, sy, sz)
138 Eigen::Matrix4d mat4 = Eigen::Matrix4d::Identity();
139 mat4.block(0, 0, 3, 3) = quat_.toRotationMatrix();
140
141 if (enable_translate) {
142 mat4.block(3, 0, 1, 3) = pos_;
143 }
144 return mat4;
145}
