74bool SingleIntegrator::step(double /*t*/, double dt) {
75 Eigen::Vector3d desired_vel(
76 vars_.input(vel_x_idx_),
77 vars_.input(vel_y_idx_),
78 vars_.input(vel_z_idx_));
79
80 Eigen::Vector3d& vel = state_->vel();
81
82 // If the max_speed is less than 0, directly apply the desired velocity,
83 // otherwise, normalize the velocity and multiply by the max_speed.
84 if (max_speed_ < 0) {
85 vel = desired_vel;
86 } else {
87 vel = desired_vel.normalized() * max_speed_;
88 }
89
90 if (vel.hasNaN()) {
91 vel = Eigen::Vector3d::Zero();
92 }
93
94 // Propogate the state
95 state_->pos() = state_->pos() + vel * dt;
96
97 double yaw = override_heading_ ? vars_.input(desired_heading_idx_) : atan2(vel(1), vel(0));
98 double pitch = atan2(vel(2), vel.head<2>().norm());
99 state_->quat().set(0, pitch, yaw);
100
101 return true;
102}
