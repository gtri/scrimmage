267bool FixedWing6DOF::step(double time, double dt) {
268 // Get inputs and saturate
269 throttle_ = clamp(vars_.input(throttle_idx_), -1.0, 1.0);
270 thrust_ = scale<double>(
271 throttle_,
272 throttle_input_min_,
273 throttle_input_max_,
274 thrust_min_,
275 thrust_max_);
276
277 delta_elevator_ = clamp(vars_.input(elevator_idx_), delta_elevator_min_, delta_elevator_max_);
278 delta_aileron_ = clamp(vars_.input(aileron_idx_), delta_aileron_min_, delta_aileron_max_);
279 delta_rudder_ = clamp(vars_.input(rudder_idx_), delta_rudder_min_, delta_rudder_max_);
280
281 quat_body_ = rot_180_x_axis_ * sc::Quaternion(state_->quat() * rot_180_x_axis_);
282 quat_body_.normalize();
283
284 Eigen::Vector3d local_lin_vel(state_->vel()(0), -state_->vel()(1), -state_->vel()(2));
285 local_lin_vel = quat_body_.rotate_reverse(local_lin_vel);
286
287 x_[U] = local_lin_vel(0);
288 x_[V] = local_lin_vel(1);
289 x_[W] = local_lin_vel(2);
290
291 Eigen::Vector3d local_ang_vel(
292 state_->ang_vel()(0),
293 -state_->ang_vel()(1),
294 -state_->ang_vel()(2));
295 local_ang_vel = quat_body_.rotate_reverse(local_ang_vel);
296
297 x_[P] = local_ang_vel(0);
298 x_[Q] = local_ang_vel(1);
299 x_[R] = local_ang_vel(2);
300
301 x_[Uw] = state_->vel()(0);
302 x_[Vw] = state_->vel()(1);
303 x_[Ww] = state_->vel()(2);
304
305 x_[Xw] = state_->pos()(0);
306 x_[Yw] = state_->pos()(1);
307 x_[Zw] = state_->pos()(2);
308
309 x_[q0] = quat_body_.w();
310 x_[q1] = quat_body_.x();
311 x_[q2] = quat_body_.y();
312 x_[q3] = quat_body_.z();
313
314 x_[U_dot] = linear_accel_body_(0);
315 x_[V_dot] = linear_accel_body_(1);
316 x_[W_dot] = linear_accel_body_(2);
317 x_[P_dot] = ang_accel_body_(0);
318 x_[Q_dot] = ang_accel_body_(1);
319 x_[R_dot] = ang_accel_body_(2);
320
321 // Cache values to calculate changes:
322 Eigen::Vector3d prev_linear_vel_ENU(state_->vel());
323 Eigen::Vector3d prev_angular_vel(x_[P], x_[Q], x_[R]);
324
325 // wind frame velocity
326 Eigen::Vector3d wind_B(wind_(0), -wind_(1), -wind_(2));
327 wind_B = quat_body_.rotate_reverse(wind_B);
328 Eigen::Vector3d vel_rel_wind(x_[U] + wind_B(0), x_[V] + wind_B(1), x_[W] + wind_B(2));
329
330 alpha_ = atan2(vel_rel_wind(2), vel_rel_wind(0)); // angle of attack
331 // TODO: Need to compute alpha_dot without using alpha_prev_
332 alpha_dot_ = 0;
333 // alpha_dot_ = (alpha_ - alpha_prev_) / dt;
334 // alpha_prev_ = alpha_;
335
336 if (time > launch_time_)
337 launch_command_ = 1;
338
339 // Eigen::Vector3d F_catapult_NED(0, 0, 0);
340 Eigen::Vector3d F_catapult(0, 0, 0);
341
342 double launch_duration = launch_speed_ / launch_accel_;
343 switch (launch_state_) {
344 case PRELAUNCH:
345 if (launch_command_) {
346 launch_state_ = LAUNCH;
347 launch_start_t_ = time;
348 }
349 break;
350 case LAUNCH:
351 F_catapult(0) = launch_accel_ * mass_;
352 if (launch_duration < time - launch_start_t_) {
353 launch_state_ = POSTLAUNCH;
354 }
355 break;
356 case POSTLAUNCH:
357 // normal operation
358 break;
359 }
360
361 // Apply any external forces (todo)
362 Eigen::Vector3d ext_force_NED(ext_force_(0), -ext_force_(1), -ext_force_(2));
363 ext_force_ = Eigen::Vector3d::Zero(); // reset ext_force_ member variable
364 force_ext_body_ = quat_body_.rotate_reverse(ext_force_NED);
365 force_ext_body_ += F_catapult;
366
367 if (!skip_propagation_) {
368 ode_step(dt);
369 }
370 skip_propagation_ = false;
371
372 quat_body_.set(x_[q0], x_[q1], x_[q2], x_[q3]);
373 quat_body_.normalize();
374
375 linear_vel_body_ << x_[U], x_[V], x_[W];
376 ang_vel_body_ << x_[P], x_[Q], x_[R];
377
378 // Calculate change in velocity to populate acceleration elements
379 Eigen::Vector3d linear_vel_ENU(x_[Uw], x_[Vw], x_[Ww]);
380 Eigen::Vector3d linear_acc_ENU = (linear_vel_ENU - prev_linear_vel_ENU) / dt;
381 Eigen::Vector3d angular_vel(x_[P], x_[Q], x_[R]);
382 Eigen::Vector3d angular_acc = (angular_vel - prev_angular_vel) / dt;
383 Eigen::Vector3d angular_acc_FLU(angular_acc(0), -angular_acc(1), -angular_acc(2));
384
385 // Rotate back to Z-axis pointing up
386 state_->quat() = rot_180_x_axis_ * sc::Quaternion(quat_body_ * rot_180_x_axis_);
387 state_->quat().normalize();
388
389 Eigen::Vector3d angvel_b_e_bodyRef = quat_body_.rotate(angular_vel);
390 Eigen::Vector3d angvel_b_e_ENU;
391 angvel_b_e_ENU << angvel_b_e_bodyRef(0), -angvel_b_e_bodyRef(1), -angvel_b_e_bodyRef(2);
392
393 state_->pos() = Eigen::Vector3d(x_[Xw], x_[Yw], x_[Zw]);
394 state_->vel() = linear_vel_ENU;
395 state_->ang_vel() = angvel_b_e_ENU;
396
397 linear_accel_body_ = state_->quat().rotate_reverse(linear_acc_ENU);
398 ang_accel_body_ = angular_acc_FLU;
399
400 // draw velocity
401 if (draw_vel_) {
402 sc::ShapePtr line(new sp::Shape());
403 line->set_opacity(1.0);
404 sc::set(line->mutable_color(), 255, 0, 0);
405 sc::set(line->mutable_line()->mutable_start(), state_->pos());
406 sc::set(line->mutable_line()->mutable_end(), state_->pos() + state_->vel());
407 draw_shape(line);
408 }
409
410 // draw angular velocity
411 if (draw_ang_vel_) {
412 sc::ShapePtr line(new sp::Shape());
413 line->set_opacity(1.0);
414 sc::set(line->mutable_color(), 0, 255, 0);
415 sc::set(line->mutable_line()->mutable_start(), state_->pos());
416 sc::set(line->mutable_line()->mutable_end(), state_->pos() + state_->ang_vel());
417 draw_shape(line);
418 }
419
420 if (write_csv_) {
421 double beta = atan2(x_[V], x_[U]); // side slip
422 // Log state to CSV
423 csv_.append(
424 sc::CSV::Pairs{
425 {"t", time},
426 {"x", x_[Xw]},
427 {"y", x_[Yw]},
428 {"z", x_[Zw]},
429 {"U", x_[U]},
430 {"V", x_[V]},
431 {"W", x_[W]},
432 {"alpha", alpha_},
433 {"alpha_dot", alpha_dot_},
434 {"beta", beta},
435 {"P", x_[P]},
436 {"Q", x_[Q]},
437 {"R", x_[R]},
438 {"Ax_b", linear_accel_body_(0)},
439 {"Ay_b", linear_accel_body_(1)},
440 {"Az_b", linear_accel_body_(2)},
441 {"AngAccelx_b", ang_accel_body_(0)},
442 {"AngAccely_b", ang_accel_body_(1)},
443 {"AngAccelz_b", ang_accel_body_(2)},
444 {"roll", quat_body_.roll()},
445 {"pitch", quat_body_.pitch()},
446 {"yaw", quat_body_.yaw()},
447 {"throttle", throttle_},
448 {"thrust", thrust_},
449 {"elevator", delta_elevator_},
450 {"aileron", delta_aileron_},
451 {"rudder", delta_rudder_}});
452 }
453 return true;

456void FixedWing6DOF::model(const vector_t& x, vector_t& dxdt, double t) {
457 // Calculate velocity magnitude (handle zero velocity)
458 Eigen::Vector3d wind_B(wind_(0), -wind_(1), -wind_(2));
459 wind_B = quat_body_.rotate_reverse(wind_B);
460 Eigen::Vector3d vel_rel_wind(x_[U] + wind_B(0), x_[V] + wind_B(1), x_[W] + wind_B(2));
461 double V_tau = vel_rel_wind.norm();
462 if (std::abs(V_tau) < std::numeric_limits<double>::epsilon()) {
463 V_tau = 0.00001;
464 }
465
466 // Calculate commonly used quantities
467 double beta = asin(clamp(vel_rel_wind(1) / V_tau, -1.0, 1.0)); // side slip
468 // double delta_Ve = 0.05*V_tau; // wind velocity across tail of aircraft (approximation)
469 // double VtauVe = pow((V_tau + delta_Ve)/V_tau, 2);
470 double pVtS = rho_ * pow(V_tau, 2) * S_ / 2.0;
471
472 // Calculate lift, drag, and side_force magnitudes
473 double CL =
474 (C_L0_ + C_L_alpha_ * alpha_ + C_LQ_ * x_[Q] * c_ / (2 * V_tau)
475 + C_L_alpha_dot_ * alpha_dot_ * c_ / (2 * V_tau) + C_L_delta_elevator_ * delta_elevator_);
476 double lift = CL * pVtS;
477
478 double drag = (C_D0_ + C_D_alpha_ * std::abs(alpha_) + CL * CL / (M_PI * AR_ * e_)
479 + C_D_delta_elevator_ * std::abs(delta_elevator_))
480 * pVtS;
481
482 double side_force = (C_Y_beta_ * beta + C_Y_delta_rudder_ * delta_rudder_) * pVtS;
483
484 // Bring lift, drag, side_force magnitudes into body frame
485 Eigen::Vector3d F_aero(
486 lift * sin(alpha_) - drag * cos(alpha_) - side_force * sin(beta),
487 side_force * cos(beta),
488 -lift * cos(alpha_) - drag * sin(alpha_));
489
490 // Calculate force from weight in body frame:
491 Eigen::Vector3d gravity_vector(0, 0, +mass_ * g_);
492 Eigen::Vector3d F_weight = quat_body_.rotate_reverse(gravity_vector);
493
494 Eigen::Vector3d F_thrust(thrust_, 0, 0);
495 Eigen::Vector3d F_total = F_weight + F_thrust + F_aero;
496
497 // simple ground contact model
498 Eigen::Vector3d F_ground_W(0, 0, 0);
499 Eigen::Vector3d F_ground(0, 0, 0);
500 if (x_[Zw] < 0 && use_ground_model_) {
501 double wn = 20;
502 double Kp = wn * wn * mass_;
503 double Kd = 2 * wn;
504
505 F_ground_W[2] = std::max(0.0, -Kp * (x_[Zw] - 0.0) - Kd * (x_[Ww] - 0.0));
506
507 F_ground(0) = F_ground_W(0);
508 F_ground(1) = -F_ground_W(1);
509 F_ground(2) = -F_ground_W(2);
510 F_ground = quat_body_.rotate_reverse(F_ground);
511
512 F_total += F_ground;
513 }
514 if (POSTLAUNCH != launch_state_) {
515 F_total = Eigen::Vector3d::Zero();
516 ;
517 }
518 F_total += force_ext_body_;
519
520 // Calculate body frame linear velocities
521 dxdt[U] = x[V] * x[R] - x[W] * x[Q] + F_total(0) / mass_;
522 dxdt[V] = x[W] * x[P] - x[U] * x[R] + F_total(1) / mass_;
523 dxdt[W] = x[U] * x[Q] - x[V] * x[P] + F_total(2) / mass_;
524
525 // Calculate moments;
526 Eigen::Vector3d Moments_thrust(0, 0, 0); // no moment from thrust
527 Eigen::Vector3d Moments_torque(0, 0, 0); // no moment from torque
528 Eigen::Vector3d Moments_gyro(0, 0, 0); // no moment from gyro effect
529
530 // Calculate moments from aerodynamic forces
531 Eigen::Vector3d Moments_aero(
532 (C_L_beta_ * beta + C_LP_ * x_[P] * b_ / (2 * V_tau) + C_LR_ * x_[R] * b_ / (2 * V_tau)
533 + C_L_delta_aileron_ * delta_aileron_ + C_L_delta_rudder_ * delta_rudder_)
534 * pVtS * b_,
535 (C_M0_ + C_M_alpha_ * alpha_ + C_MQ_ * x_[Q] * c_ / (2 * V_tau)
536 + C_M_alpha_dot_ * alpha_dot_ * c_ / (2 * V_tau) + C_M_delta_elevator_ * delta_elevator_)
537 * pVtS * c_,
538 (C_N_beta_ * beta + C_NP_ * x_[P] * b_ / (2 * V_tau) + C_NR_ * x_[R] * b_ / (2 * V_tau)
539 + C_N_delta_aileron_ * delta_aileron_ + C_N_delta_rudder_ * delta_rudder_)
540 * pVtS * b_);
541
542 // Sum moments
543 Eigen::Vector3d Moments_total = Moments_aero + Moments_thrust + Moments_torque + Moments_gyro;
544
545 if (POSTLAUNCH != launch_state_) {
546 Moments_total = Eigen::Vector3d::Zero();
547 }
548
549 // Calculate rotational velocites
550 Eigen::Vector3d pqr(x_[P], x_[Q], x_[R]);
551 Eigen::Vector3d pqr_dot = I_inv_ * (Moments_total - pqr.cross(I_ * pqr));
552 dxdt[P] = pqr_dot(0);
553 dxdt[Q] = pqr_dot(1);
554 dxdt[R] = pqr_dot(2);
555
556 // Compute quaternion derivatives
557 double lambda = 1 - (pow(x[q0], 2) + pow(x[q1], 2) + pow(x[q2], 2) + pow(x[q3], 2));
558 dxdt[q0] = -0.5 * (x[q1] * x[P] + x[q2] * x[Q] + x[q3] * x[R]) + lambda * x[q0];
559 dxdt[q1] = +0.5 * (x[q0] * x[P] + x[q2] * x[R] - x[q3] * x[Q]) + lambda * x[q1];
560 dxdt[q2] = +0.5 * (x[q0] * x[Q] + x[q3] * x[P] - x[q1] * x[R]) + lambda * x[q2];
561 dxdt[q3] = +0.5 * (x[q0] * x[R] + x[q1] * x[Q] - x[q2] * x[P]) + lambda * x[q3];
562
563 // Normalize quaternion
564 sc::Quaternion quat(x[q0], x[q1], x[q2], x[q3]);
565 quat.normalize();
566
567 // Integrate local velocities to compute local positions
568 Eigen::Vector3d vel_local(x[U], x[V], x[W]);
569 Eigen::Vector3d vel_world = quat.rotate(vel_local); // rot * vel_local;
570 dxdt[Xw] = vel_world(0);
571 dxdt[Yw] = -vel_world(1); // Due to rotated frame
572 dxdt[Zw] = -vel_world(2); // Due to rotated frame
573
574 // Integrate local accelerations to compute global velocities
575 Eigen::Vector3d acc_local = F_total / mass_;
576 Eigen::Vector3d acc_world = quat.rotate(acc_local); // rot * acc_local;
577 dxdt[Uw] = acc_world(0);
578 dxdt[Vw] = -acc_world(1); // Due to rotated frame
579 dxdt[Ww] = -acc_world(2); // Due to rotated frame
580}
