50PublisherPtr PubSub::advertise(

53 const unsigned int& max_queue_size,

55 EntityPluginPtr plugin) {

61 PublisherPtr pub =
62 std::make_shared<Publisher>(topic, max_queue_size, enable_queue_size, plugin);
63 pub_map_[network_name][topic].push_back(pub);

67boost::optional<std::list<NetworkDevicePtr>> PubSub::find_devices(
68 const std::string& network_name,
69 const std::string& topic_name,
70 TopicMap& devs) {
71
72 auto it_network = devs.find(network_name);
73 if (it_network == devs.end()) {
74 LOG_WARN("Failed to find network while setting up device.\n"
75 << "Network name: " << network_name << "\n"
76 << "Topic name: " << topic_name);
77 } else {
78 auto it_topic_pub = it_network->second.find(topic_name);
79 if (it_topic_pub == it_network->second.end()) {
80 LOG_WARN("Failed to find topic while setting up device.\n"
81 << "Network name: " << network_name << "\n"
82 << "Topic name: " << topic_name);
83 } else {
84 return boost::optional<std::list<NetworkDevicePtr>>(it_topic_pub->second);
85 }
86 }
87 return boost::none;
88}
