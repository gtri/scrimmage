118scrimmage::Quaternion FlightGearMultiplayer::fromLonLatRad(float lon, float lat) {
119 // TODO : This transformation was taken from the Flight Gear Multiplayer
120 // server, we should convert this to using our own quaternion class, but
121 // for now, this works.
122
123 // Eigen::AngleAxisd R_a(sc::Angles::deg2rad(lon), Eigen::Vector3d::UnitZ());
124 // Eigen::AngleAxisd R_b(sc::Angles::deg2rad(lat),
125 // Eigen::Vector3d::UnitX()); Eigen::Quaterniond rot2 = R_a * R_b;
126
127 float zd2 = static_cast<float>(0.5) * lon;
128 float yd2 =
129 static_cast<float>(-0.25) * static_cast<float>(M_PI) - static_cast<float>(0.5) * lat;
130 float Szd2 = sin(zd2);
131 float Syd2 = sin(yd2);
132 float Czd2 = cos(zd2);
133 float Cyd2 = cos(yd2);
134 scrimmage::Quaternion quat(Czd2 * Cyd2, -Szd2 * Syd2, Czd2 * Syd2, Szd2 * Cyd2);
135 return quat;
136}

138bool FlightGearMultiplayer::step_autonomy(double t, double dt) {
139 // Create the flight gear position msg
140 pos_msg_.time = XDR_encode64<float>(time_->t());
141 pos_msg_.lag = XDR_encode64<float>(time_->t());
142
143 // Compute ECEF coordinates from x/y/z lat/lon/alt
144 double lat, lon, alt;
145 parent_->projection()
146 ->Reverse(state_->pos()(0), state_->pos()(1), state_->pos()(2), lat, lon, alt);
147 double ECEF_X, ECEF_Y, ECEF_Z;
148 earth_->Forward(lat, lon, alt, ECEF_X, ECEF_Y, ECEF_Z);
149
150 pos_msg_.position[0] = XDR_encode64<double>(ECEF_X);
151 pos_msg_.position[1] = XDR_encode64<double>(ECEF_Y);
152 pos_msg_.position[2] = XDR_encode64<double>(ECEF_Z);
153
154 // Convert local heading to GPS heading
155 angles_to_jsbsim_.set_angle(sc::Angles::rad2deg(state_->quat().yaw()));
156 sc::Quaternion local_quat(
157 state_->quat().roll(),
158 state_->quat().pitch(),
159 sc::Angles::deg2rad(angles_to_jsbsim_.angle()));
160
161 // Convert local quaternion to ECEF coordinate
162 scrimmage::Quaternion rot =
163 this->fromLonLatRad(sc::Angles::deg2rad(lon), sc::Angles::deg2rad(lat));
164
165 // The FlightGear Multiplayer packet expects the orientation to be in
166 // angle-axis form, where the axis magnitude is the rotation angle.
167 Eigen::AngleAxisd ang_axis(rot * local_quat);
168 Eigen::Vector3d ang_axis_mod = ang_axis.axis().normalized() * ang_axis.angle();
169
170 pos_msg_.orientation[0] = XDR_encode<float>(ang_axis_mod(0));
171 pos_msg_.orientation[1] = XDR_encode<float>(ang_axis_mod(1));
172 pos_msg_.orientation[2] = XDR_encode<float>(ang_axis_mod(2));
173
174 // Copy header and position data structs into memory buffer
175 char* msg = new char[msg_size_];
176 std::memcpy(msg, &header_msg_, sizeof(T_MsgHdr));
177 std::memcpy(msg + sizeof(T_MsgHdr), &pos_msg_, sizeof(T_PositionMsg));
178
179 data_socket_->sendto(msg, msg_size_, 0, &net_address_);
180 delete[] msg;
181
182 return true;
183}
