63bool AvoidWalls::step_autonomy(double t, double dt) {
64 // Find closest point and move away from it
65 bool all_close_points = true;
66 std::list<Eigen::Vector3d> points;
67 for (sensor::RayTrace::PCPoint& p : point_cloud_.points) {
68 if (p.point.norm() < avoid_distance_) {
69 points.push_back(p.point);
70 } else {
71 all_close_points = false;
72 }
73 }
74
75 if (!points.empty()) {
76 std::vector<Eigen::Vector3d> O_vecs;
77 for (Eigen::Vector3d& p : points) {
78 // Transform the point to global coordinate space
79 double dist = p.norm();
80 p = state_->quat().rotate(p);
81 p += state_->pos();
82
83 // Get vector pointing towards point
84 Eigen::Vector3d diff = p - state_->pos();
85 double O_mag = avoid_distance_ - dist;
86
87 Eigen::Vector3d O_dir = -O_mag * diff.normalized();
88 O_vecs.push_back(O_dir);
89 }
90
91 // Normalize each repulsion vector and sum
92 Eigen::Vector3d O_vec(0, 0, 0);
93 for (auto it = O_vecs.begin(); it != O_vecs.end(); it++) {
94 if (it->hasNaN()) {
95 continue; // ignore misbehaved vectors
96 }
97 O_vec += *it;
98 }
99
100 Eigen::Vector3d dir;
101 // Just turn left if all points are too close (reduces chattering)
102 if (all_close_points) {
103 dir = state_->quat().rotate(Eigen::Vector3d(0, -1, 0));
104 } else {
105 dir = O_vec.normalized();
106 }
107 dir *= 10;
108
109 const double heading = atan2(dir(1), dir(0));
110
111 vars_.output(heading_idx_, heading);
112 vars_.output(speed_idx_, 10);
113 } else {
114 vars_.output(heading_idx_, state_->quat().yaw());
115 }
116
117 return true;
118}
