125bool RapidXMLParserDocument::parse_document(const std::string& filename) {
126 fs::path path{filename};
127 if (fs::is_symlink(path)) {
128 path = fs::read_symlink(path);
129 }
130 assert(fs::exists(path) && fs::is_regular_file(path));
131 std::ifstream file(path.string());
132 std::stringstream buffer;
133 buffer << file.rdbuf();
134 file.close();
135 std::string filecontent_str = buffer.str();
136 std::vector<char> filecontent(filecontent_str.cbegin(), filecontent_str.cend());
137 filename_ = path.string();
138 return parse_document(filecontent);
139}

141bool RapidXMLParserDocument::parse_document(const std::vector<char>& filecontent) {
142 filecontent_ = filecontent;
143 // Note: This parse function can hard fail (seg fault, no exception) on
144 // badly formatted xml data. Sometimes it'll except, sometimes not.
145 // Exceptions are caught by the caller (MissionParse) which has Logger access.
146 doc_.parse<rapidxml::parse_non_destructive>(filecontent_.data());
147 return true;
148}
