88void SimpleCaptureMetrics::calc_team_scores() {
89 // Create the score if there isn't one yet
90 for (auto& team_id : teams_) {
91 if (team_scores_map_.count(team_id) == 0) {
92 Score score;
93 score.set_weights(params_);
94 team_scores_map_[team_id] = score;
95 }
96 }
97
98 for (auto& kv : scores_) {
99 Score& score = kv.second;
100
101 int team_id = (*id_to_team_map_)[kv.first];
102
103 team_scores_map_[team_id].add_count("TeamCapture", score.count("TeamCapture"));
104 team_scores_map_[team_id].add_count("NonTeamCapture", score.count("NonTeamCapture"));
105 }
106
107 for (auto& kv : team_scores_map_) {
108 int team_id = kv.first;
109 Score& score = kv.second;
110 team_metrics_[team_id]["TeamCapture"] = score.count("TeamCapture");
111 team_metrics_[team_id]["NonTeamCapture"] = score.count("NonTeamCapture");
112
113 double s = score.score();
114 team_scores_[team_id] = s;
115 }
116
117 // list the headers we want put in the csv file
118 headers_.push_back("TeamCapture");
119 headers_.push_back("NonTeamCapture");
120}
