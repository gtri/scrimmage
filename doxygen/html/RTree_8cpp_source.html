60void RTree::add(const Eigen::Vector3d& pos, const ID& id) {
61 point p(pos(0), pos(1), pos(2));
62 std::pair<point, ID> pair(p, id);
63 rtree_->insert(pair);
64
65 int team_id = id.team_id();
66 auto it = rtree_team_.find(team_id);
67 if (it == rtree_team_.end()) {
68 rtreePtr rtree = std::make_shared<rtree_t>(bgi::dynamic_rstar(size_));
69 it = rtree_team_.insert(std::make_pair(team_id, rtree)).first;
70 }
71 it->second->insert(pair);
72}

74void results_to_neighbors(std::list<point_id_t>& results, std::vector<ID>& neighbors, int self_id) {

76 neighbors.reserve(results.size());

78 for (point_id_t& pt_id : results) {

80 neighbors.push_back(pt_id.second);

82 }
83 } else {

90void RTree::nearest_n_neighbors(
91 const Eigen::Vector3d& pos,
92 std::vector<ID>& neighbors,
93 unsigned int n,
94 int self_id,
95 int team_id) const {
96 std::list<point_id_t> results;
97 point sought(pos(0), pos(1), pos(2));
98
99 if (self_id != -1) {
100 // assume that if an id is given then it will be located at pos so
101 // would always be in the neighborhood
102 n += 1;
103 }
104
105 if (team_id == -1) {
106 rtree_->query(bgi::nearest(sought, n), std::front_inserter(results));
107 } else {
108 auto it = rtree_team_.find(team_id);
109 if (it == rtree_team_.end()) {
110 neighbors.clear();
111 return;
112 } else {
113 it->second->query(bgi::nearest(sought, n), std::front_inserter(results));
114 }
115 }
116 results_to_neighbors(results, neighbors, self_id);
117}

119void RTree::neighbors_in_range(
120 const Eigen::Vector3d& pos,
121 std::vector<ID>& neighbors,
122 double dist,
123 int self_id,
124 int team_id) const {
125 // see here: http://stackoverflow.com/a/22910447
126 std::list<point_id_t> results;
127 double x = pos(0);
128 double y = pos(1);
129 double z = pos(2);
130 point sought(x, y, z);
131
132 bg::model::box<point> box(
133 point(x - dist, y - dist, z - dist),
134 point(x + dist, y + dist, z + dist));
135
136 auto dist_func = [&](point_id_t const& v) { return bg::distance(v.first, sought) < dist; };
137
138 if (team_id == -1) {
139 rtree_->query(bgi::within(box) && bgi::satisfies(dist_func), std::front_inserter(results));
140 } else {
141 auto it = rtree_team_.find(team_id);
142 if (it == rtree_team_.end()) {
143 neighbors.clear();
144 return;
145 } else {
146 it->second->query(
147 bgi::within(box) && bgi::satisfies(dist_func),
148 std::front_inserter(results));
149 }
150 }
151
152 results_to_neighbors(results, neighbors, self_id);
153}
