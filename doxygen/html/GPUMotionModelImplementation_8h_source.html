117 void init_new_entities(double time) {
118 // "Propagate" the motion model with dt=0. This should not update the
119 // motion, but should ensure that all states are properly initalized.
120 if (to_init_.size() > 0) {
121 if (input_num_items_ == 0) {
122 input_num_items_ = vars_[to_init_[0]].input()->size();
123 }
124 GPUMapBuffer<T> states{queue_, CL_MEM_READ_WRITE};
125 GPUMapBuffer<T> inputs{queue_, CL_MEM_WRITE_ONLY};
126 collect_states(to_init_, states, inputs);
127 propagate(states, inputs, to_init_.size(), time, 0, 1);
128 distribute_states(to_init_, states);
129 entities_.insert(entities_.end(), to_init_.begin(), to_init_.end());
130 to_init_.clear();
131 }
132 }

134 bool step(double time, double dt, std::size_t iterations) {
135 remove_inactive();
136 bool success = true;
137 if (entities_.size() != 0) {
138 collect_states(entities_, states_, inputs_);
139 success = propagate(states_, inputs_, entities_.size(), time, dt, iterations);
140 distribute_states(entities_, states_);
141 }
142 return success;

145 protected:
146 void collect_states(
147 std::vector<EntityPtr>& entities,
148 GPUMapBuffer<T>& states,
149 GPUMapBuffer<T>& inputs) {
150 states.resize(STATE_NUM_ITEMS * entities.size());
151 inputs.resize(input_num_items_ * entities.size());
152
153 // Map our device buffers into host memory to write to them;
154 // We don't care about any possible data in our buffers. Invalidate
155 // the region we are writing to.
156 states.map(CL_MAP_WRITE_INVALIDATE_REGION);
157 inputs.map(CL_MAP_WRITE_INVALIDATE_REGION);
158 for (auto entityptr_it = entities.cbegin(); entityptr_it != entities.cend();
159 ++entityptr_it) {
160
161 // For now, assume that controllers have already been run.
162 // Eventually I think we want to include the controller with the kernel.
163 // This also assumes that all entities are homogeneous in their state
164 // and controller output
165 EntityPtr entity = *entityptr_it;
166 VariableIO& model_input = vars_[entity];
167
168 // We only need to copy state information when a new entity is added.
169 // Otherwise the state information of the device is consistent
170 // with the host, and we just copy that data back to the host after
171 // every step.
172 StatePtr state = entity->state_truth();
173 Eigen::Vector3d& pos = state->pos();
174 Eigen::Vector3d& vel = state->vel();
175 Eigen::Vector3d& ang_vel = state->ang_vel();
176 Quaternion& quat = state->quat();
177 states.insert(
178 states.cend(),
179 {
180 static_cast<T>(pos(0)),
181 static_cast<T>(pos(1)),
182 static_cast<T>(pos(2)),
183 static_cast<T>(vel(0)),
184 static_cast<T>(vel(1)),
185 static_cast<T>(vel(2)),
186 static_cast<T>(ang_vel(0)),
187 static_cast<T>(ang_vel(1)),
188 static_cast<T>(ang_vel(2)),
189 static_cast<T>(quat.w()),
190 static_cast<T>(quat.x()),
191 static_cast<T>(quat.y()),
192 static_cast<T>(quat.z()),
193 });
194 // We always need to copy motion inputs as they change on the host.
195 for (int i = 0; i < model_input.input()->size(); ++i) {
196 inputs.push_back(static_cast<T>(model_input.input(i)));
197 }
198 }
199
200 // Unmap buffers to commit changes to device.
201 states.unmap();
202 inputs.unmap();
203 }

233

236
237 err = queue_.enqueueNDRangeKernel(

239 cl::NullRange,

241 cl::NullRange);

249 void distribute_states(std::vector<EntityPtr>& entities, GPUMapBuffer<T>& states) {
250 // Map state information back to host memory to copy to entites
251 auto state_info = [&](std::size_t entity_index, std::size_t state_index) {
252 return states.at(STATE_NUM_ITEMS * entity_index + state_index);
253 };
254
255 states.map(CL_MAP_READ);
256
257 std::size_t entity_ind;
258 for (auto entity_ptr_it = entities.begin(); entity_ptr_it != entities.end();
259 ++entity_ptr_it) {
260 entity_ind = std::distance(entities.begin(), entity_ptr_it);
261 StatePtr state = (*entity_ptr_it)->state_truth();
262 state->pos() << state_info(entity_ind, X), state_info(entity_ind, Y),
263 state_info(entity_ind, Z);
264 state->vel() << state_info(entity_ind, X_VEL), state_info(entity_ind, Y_VEL),
265 state_info(entity_ind, Z_VEL);
266 state->ang_vel() << state_info(entity_ind, X_ANG_VEL),
267 state_info(entity_ind, Y_ANG_VEL), state_info(entity_ind, Z_ANG_VEL);
268 state->quat().w() = state_info(entity_ind, QUAT_W);
269 state->quat().x() = state_info(entity_ind, QUAT_X);
270 state->quat().y() = state_info(entity_ind, QUAT_Y);
271 state->quat().z() = state_info(entity_ind, QUAT_Z);
272 }
273
274 states.unmap();
275 }
