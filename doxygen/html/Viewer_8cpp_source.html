108bool Viewer::run() {
109 double update_rate = 50; // Hz
110
111 if (enable_network_) {
112 outgoing_interface_->init_network(Interface::client, remote_ip_, remote_port_);
113 network_thread_ = std::thread(
114 &Interface::init_network,
115 &(*incoming_interface_),
116 Interface::server,
117 local_ip_,
118 local_port_);
119 network_thread_.detach();
120 } else {
121 incoming_interface_->set_mode(Interface::shared);
122 outgoing_interface_->set_mode(Interface::shared);
123 }
124
125 // Render and interact
126 renderWindow_->Render();
127
128 // Initialize must be called prior to creating timer events.
129 renderWindowInteractor_->Initialize();
130
131 if (full_screen_) {
132 // Don't use "FullScreenOn" or SetFullScreen().
133 // This uses the entire screen, prevents displaying the window name
134 // and makes it difficult to resize the window later.
135 renderWindow_->SetSize(renderWindow_->GetScreenSize());
136 }
137
138 // Sign up to receive TimerEvent
139 vtkSmartPointer<scrimmage::Updater> updater = vtkSmartPointer<scrimmage::Updater>::New();
140 renderWindowInteractor_->AddObserver(vtkCommand::TimerEvent, updater);
141 updater->set_renderer(renderer_);
142 updater->set_rwi(renderWindowInteractor_);
143 updater->set_incoming_interface(incoming_interface_);
144 updater->set_outgoing_interface(outgoing_interface_);
145 updater->set_max_update_rate(update_rate);
146 updater->set_init_scale(init_scale_);
147 updater->reset_scale();
148
149 std::string camera_pos_str = get<std::string>("pos", camera_params_, "0, 1, 200");
150
151 std::vector<double> camera_pos;
152 if (!str2container(camera_pos_str, ",", camera_pos, 3)) {
153 std::cout << "camera_position should have 3 comma separated entries" << std::endl;
154 return false;
155 }
156
157 std::string camera_focal_pos_str = get<std::string>("focal_point", camera_params_, "0, 0, 0");
158
159 std::vector<double> camera_focal_pos;
160 if (!str2container(camera_focal_pos_str, ",", camera_focal_pos, 3)) {
161 std::cout << "camera_focal_point should have 3 comma separated entries" << std::endl;
162 return false;
163 }
164
165 updater->set_camera_reset_params(
166 camera_pos[0],
167 camera_pos[1],
168 camera_pos[2],
169 camera_focal_pos[0],
170 camera_focal_pos[1],
171 camera_focal_pos[2]);
172 updater->set_show_fps(get("show_fps", camera_params_, false));
173
174 updater->set_follow_id(get("follow_id", camera_params_, 1) - 1);
175
176 std::string view_mode =
177 boost::to_upper_copy(get<std::string>("mode", camera_params_, "follow"));
178
179 if (view_mode == "FOLLOW") {
180 updater->set_view_mode(Updater::ViewMode::FOLLOW);
181 } else if (view_mode == "FREE") {
182 updater->set_view_mode(Updater::ViewMode::FREE);
183 updater->set_reset_camera();
184 } else if (view_mode == "OFFSET") {
185 updater->set_view_mode(Updater::ViewMode::OFFSET);
186 } else {
187 std::cout << "Unrecognized attribute \"" << view_mode << "\" for camera_view_mode"
188 << std::endl;
189 updater->set_view_mode(Updater::ViewMode::FOLLOW);
190 }
191
192 updater->init(log_dir_, dt_);
193
194 cam_int_->set_updater(updater);
195
196 renderWindowInteractor_->CreateRepeatingTimer(1.0 / update_rate * 1e3); // ms
197
198 // Start the interaction and timer
199 renderWindowInteractor_->Start();
200
201 // Shutdown the updater
202 updater->shutting_down();
203
204 return true;
205}
