183bool Multirotor::step(double time, double dt) {
184 for (int i = 0; i < motor_idx_vec_.size(); i++) {
185 ctrl_u_(i) = boost::algorithm::clamp(vars_.input(motor_idx_vec_(i)), wmin_, wmax_);
186 }
187
188 // TODO: convert global linear velocity and angular velocity into local
189 // velocities
190
191 x_[q0] = state_->quat().w();
192 x_[q1] = state_->quat().x();
193 x_[q2] = state_->quat().y();
194 x_[q3] = state_->quat().z();
195
196 x_[Uw] = state_->vel()(0);
197 x_[Vw] = state_->vel()(1);
198 x_[Ww] = state_->vel()(2);
199
200 x_[Xw] = state_->pos()(0);
201 x_[Yw] = state_->pos()(1);
202 x_[Zw] = state_->pos()(2);
203
204 // Cache values to calculate changes:
205 Eigen::Vector3d prev_linear_vel_w(x_[Uw], x_[Vw], x_[Ww]);
206 Eigen::Vector3d prev_angular_vel(x_[P], x_[Q], x_[R]);
207
208 // Apply any external forces
209 force_ext_body_ = state_->quat().rotate_reverse(ext_force_);
210 ext_force_ = Eigen::Vector3d::Zero(); // reset ext_force_ member variable
211
212 ode_step(dt); // step the motion model ODE solver
213
214 state_->quat().set(x_[q0], x_[q1], x_[q2], x_[q3]);
215 state_->quat().normalize();
216
217 // Calculate change in velocity to populate acceleration elements
218 Eigen::Vector3d linear_vel_w(x_[Uw], x_[Vw], x_[Ww]);
219 Eigen::Vector3d linear_acc_w = (linear_vel_w - prev_linear_vel_w) / dt;
220 Eigen::Vector3d linear_acc = state_->quat().rotate_reverse(linear_acc_w);
221 Eigen::Vector3d angular_vel(x_[P], x_[Q], x_[R]);
222 Eigen::Vector3d angular_acc = (angular_vel - prev_angular_vel) / dt;
223
224 // Convert local coordinates to world coordinates
225 state_->pos() << x_[Xw], x_[Yw], x_[Zw];
226 state_->vel() << x_[Uw], x_[Vw], x_[Ww];
227 state_->ang_vel() = state_->quat().rotate(angular_vel);
228 linear_accel_body_ = linear_acc;
229 ang_accel_body_ = angular_acc;
230
231 // draw velocity
232 if (show_shapes_) {
233 sc::ShapePtr line(new sp::Shape());
234 line->set_opacity(1.0);
235 sc::set(line->mutable_color(), 255, 255, 0);
236 sc::set(line->mutable_line()->mutable_start(), state_->pos());
237 sc::set(line->mutable_line()->mutable_end(), state_->pos() + state_->vel());
238 draw_shape(line);
239 }
240
241 // draw angular velocity
242 if (show_shapes_) {
243 sc::ShapePtr line(new sp::Shape());
244 line->set_opacity(1.0);
245 sc::set(line->mutable_color(), 0, 255, 255);
246 sc::set(line->mutable_line()->mutable_start(), state_->pos());
247 sc::set(line->mutable_line()->mutable_end(), state_->pos() + state_->ang_vel() * 10);
248 draw_shape(line);
249 }
250
251 if (write_csv_) {
252 // Log state to CSV
253 csv_.append(
254 sc::CSV::Pairs{
255 {"t", time},
256 {"x", x_[Xw]},
257 {"y", x_[Yw]},
258 {"z", x_[Zw]},
259 {"U", x_[U]},
260 {"V", x_[V]},
261 {"W", x_[W]},
262 {"P", x_[P]},
263 {"Q", x_[Q]},
264 {"R", x_[R]},
265 {"AXb", linear_accel_body_(0)},
266 {"AYb", linear_accel_body_(1)},
267 {"AZb", linear_accel_body_(2)},
268 {"WXDOTb", ang_accel_body_(0)},
269 {"WYDOTb", ang_accel_body_(1)},
270 {"WZDOTb", ang_accel_body_(2)},
271 {"roll", state_->quat().roll()},
272 {"pitch", state_->quat().pitch()},
273 {"yaw", state_->quat().yaw()},
274 {"w_1", ctrl_u_(0)},
275 {"w_2", ctrl_u_(1)},
276 {"w_3", ctrl_u_(2)},
277 {"w_4", ctrl_u_(3)}});
278 }
279
280 return true;

283void Multirotor::model(const vector_t& x, vector_t& dxdt, double t) {
284 // Omega values for each rotor
285 Eigen::VectorXd omega = ctrl_u_;
286 Eigen::VectorXd omega_sq = omega.cwiseProduct(omega);
287
288 // Calculate force from combined rotor thrust
289 Eigen::Vector3d F_thrust(0, 0, 0);
290 Eigen::VectorXd rotor_thrust(rotors_.size());
291 Eigen::VectorXd rotor_torque(rotors_.size());
292
293 for (unsigned int i = 0; i < rotors_.size(); i++) {
294 rotor_thrust(i) = c_T_ * omega_sq(i);
295 if (std::abs(omega[i] - wmin_)
296 < std::numeric_limits<double>::epsilon()) { // if minimum throttle, servo is off
297 rotor_thrust[i] = 0.0;
298 }
299 rotor_torque(i) = rotors_[i].direction() * c_Q_ * omega_sq(i);
300 F_thrust(2) += rotor_thrust(i);
301 }
302
303 // Calculate force from weight in body frame:
304 Eigen::Vector3d gravity_vector(0, 0, -mass_ * g_);
305 Eigen::Vector3d F_weight = state_->quat().rotate_reverse(gravity_vector);
306
307 // Calculate force from drag (TODO: Check source)
308 Eigen::Vector3d vel_body(x_[U], x_[V], x_[W]);
309 double vel_mag = vel_body.norm();
310 Eigen::Vector3d F_drag = vel_body * (-0.5 * c_D_ * vel_mag);
311
312 // Calculate normal force
313 force_ext_body_[2] = force_ext_body_[2] - F_thrust[2];
314 if (force_ext_body_[2] > 0) { // only true if contacting ground
315 force_ext_body_[2] =
316 -1.0 * (F_weight[2]) - F_thrust[2]; // correctly calculate normal force
317 }
318
319 if (force_ext_body_[2] < 0) { // normal force only pushes up from the ground
320 force_ext_body_[2] = 0.0;
321 }
322
323 // Calculate total force
324 Eigen::Vector3d F_total = F_thrust + F_weight + F_drag + force_ext_body_;
325
326 dxdt[U] = x[V] * x[R] - x[W] * x[Q] + F_total(0) / mass_;
327 dxdt[V] = x[W] * x[P] - x[U] * x[R] + F_total(1) / mass_;
328 dxdt[W] = x[U] * x[Q] - x[V] * x[P] + F_total(2) / mass_;
329
330 // Calculate moments from thrust
331 Eigen::Vector3d Moments_thrust(0, 0, 0); // L, M, N
332
333 for (unsigned int i = 0; i < rotors_.size(); i++) {
334 double length = rotors_[i].offset_length();
335 double xy_angle = rotors_[i].xy_angle();
336
337 Moments_thrust(0) += rotor_thrust(i) * length * sin(xy_angle);
338 Moments_thrust(1) -= rotor_thrust(i) * length * cos(xy_angle);
339 Moments_thrust(2) += rotor_torque(i);
340 }
341
342 Eigen::Vector3d Moments_total = Moments_thrust;
343
344 Eigen::Vector3d pqr(x_[P], x_[Q], x_[R]);
345 Eigen::Vector3d pqr_dot = I_inv_ * (Moments_total - pqr.cross(I_ * pqr));
346 dxdt[P] = pqr_dot(0);
347 dxdt[Q] = pqr_dot(1);
348 dxdt[R] = pqr_dot(2);
349
350 // Update quaternion from angular accelerations
351 double lambda = 1 - (pow(x[q0], 2) + pow(x[q1], 2) + pow(x[q2], 2) + pow(x[q3], 2));
352 dxdt[q0] = -0.5 * (x[q1] * x[P] + x[q2] * x[Q] + x[q3] * x[R]) + lambda * x[q0];
353 dxdt[q1] = +0.5 * (x[q0] * x[P] + x[q2] * x[R] - x[q3] * x[Q]) + lambda * x[q1];
354 dxdt[q2] = +0.5 * (x[q0] * x[Q] + x[q3] * x[P] - x[q1] * x[R]) + lambda * x[q2];
355 dxdt[q3] = +0.5 * (x[q0] * x[R] + x[q1] * x[Q] - x[q2] * x[P]) + lambda * x[q3];
356
357 // Local position / velocity to global
358 // Normalize quaternion
359 sc::Quaternion quat(x[q0], x[q1], x[q2], x[q3]);
360 quat.w() = x[q0];
361 quat.x() = x[q1];
362 quat.y() = x[q2];
363 quat.z() = x[q3];
364 quat.normalize();
365
366 Eigen::Vector3d vel_local(x[U], x[V], x[W]);
367 Eigen::Vector3d vel_world = quat.rotate(vel_local);
368 dxdt[Xw] = vel_world(0);
369 dxdt[Yw] = vel_world(1);
370 dxdt[Zw] = vel_world(2);
371
373
374 Eigen::Vector3d acc_local = F_total / mass_; // this is in body frame
375 Eigen::Vector3d acc_world = quat.rotate(acc_local);
376 dxdt[Uw] = acc_world(0);
377 dxdt[Vw] = acc_world(1);
378 dxdt[Ww] = acc_world(2);
379}
