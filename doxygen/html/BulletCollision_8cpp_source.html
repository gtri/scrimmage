394bool BulletCollision::step_entity_interaction(std::list<sc::EntityPtr>& ents, double t, double dt) {
395 // Update positions of all objects
396 for (auto& kv : *id_to_ent_map_) {
397 auto it_object = objects_.find(kv.first);
398 if (it_object != objects_.end()) {
399 sc::EntityPtr& ent = kv.second;
400 it_object->second.object->getWorldTransform().setOrigin(btVector3(
401 (btScalar)ent->state_truth()->pos()(0),
402 (btScalar)ent->state_truth()->pos()(1),
403 (btScalar)ent->state_truth()->pos()(2)));
404
405 if (show_collision_shapes_) {
406 sc::set(
407 it_object->second.shape->mutable_sphere()->mutable_center(),
408 ent->state_truth()->pos());
409 draw_shape(objects_[kv.first].shape);
410 }
411 }
412 }
413
414 // If an entity no longer exists in the ent map, remove it's shape from the
415 // bullet environment
416 for (auto& kv : objects_) {
417 auto it_ent = id_to_ent_map_->find(kv.first);
418 if (it_ent == id_to_ent_map_->end()) {
419 remove_object(kv.first);
420 }
421 }
422
423 // For each entity's ray-based sensors, compute point clouds
424 for (auto& kv : pc_descs_) {
425 sc::EntityPtr& own_ent = (*id_to_ent_map_)[kv.first];
426 Eigen::Vector3d own_pos = own_ent->state_truth()->pos();
427
428 // For each ray sensor on a single entity
429 for (auto& kv2 : kv.second) {
430 sensor::RayTrace::PointCloud& pc = kv2.second->point_cloud;
431
432 if (t > (pc.max_sample_rate + kv2.second->last_update_time)) {
433 kv2.second->last_update_time = time_->t();
434
435 auto msg = std::make_shared<sc::Message<RayTrace::PointCloud>>();
436 msg->data.max_range = pc.max_range;
437 msg->data.min_range = pc.min_range;
438 msg->data.rays = pc.get_rays();
439
440 // Compute transformation matrix from entity's frame to sensor's
441 // frame.
442 sc::SensorPtr& sensor = own_ent->sensors()[kv2.first];
443 Eigen::Matrix4d tf_m =
444 own_ent->state_truth()->tf_matrix(false) * sensor->transform()->tf_matrix();
445
446 // For each ray in the sensor
447 unsigned int i = 0;
448 for (RayTrace::PCPoint& pcpoint : pc.points) {
449 Eigen::Vector3d original_ray = pcpoint.point;
450 // Transform sensor's origin to world coordinates
451 Eigen::Vector4d sensor_pos = tf_m * Eigen::Vector4d(0, 0, 0, 1);
452 Eigen::Vector3d sensor_pos_w = sensor_pos.head<3>() + own_pos;
453
454 // Transform ray's end point to world coordinates
455 Eigen::Vector4d ray =
456 tf_m
457 * Eigen::Vector4d(original_ray(0), original_ray(1), original_ray(2), 1);
458 Eigen::Vector3d ray_w = ray.head<3>() + own_pos;
459
460 // Create bullet vectors
461 btVector3 btFrom(sensor_pos_w(0), sensor_pos_w(1), sensor_pos_w(2));
462 btVector3 btTo(ray_w(0), ray_w(1), ray_w(2));
463
464 // Perform ray casting
465 btCollisionWorld::ClosestRayResultCallback res(btFrom, btTo);
466 bt_collision_world->rayTest(btFrom, btTo, res);
467
468 // Points in the RayTrace message are defined with respect to
469 // the LIDAR sensor's coordinate frame. Use original ray's
470 // direction, shorten to length of detection ray, if a
471 // collision occurred.
472 Eigen::Vector3d ray_end;
473 if (res.hasHit()) {
474 Eigen::Vector3d hit_point(
475 res.m_hitPointWorld.x(),
476 res.m_hitPointWorld.y(),
477 res.m_hitPointWorld.z());
478 Eigen::Vector3d return_vec =
479 original_ray.normalized() * (hit_point - sensor_pos_w).norm();
480 msg->data.points.push_back(RayTrace::PCPoint(
481 return_vec,
482 255,
483 ((return_vec.norm() > msg->data.max_range)
484 || (return_vec.norm() < msg->data.min_range))));
485 ray_end = hit_point;
486 } else {
487 msg->data.points.push_back(RayTrace::PCPoint(original_ray, 255, true));
488 ray_end = ray_w;
489 }
490
491 if (show_rays_) {
492 std::unique_ptr<PointCloudDescription>& pc_desc = kv2.second;
493 if (res.hasHit()) {
494 sc::set(pc_desc->shapes[i]->mutable_color(), 255, 0, 0);
495 pc_desc->shapes[i]->set_opacity(1.0);
496 } else {
497 sc::set(pc_desc->shapes[i]->mutable_color(), 0, 0, 255);
498 pc_desc->shapes[i]->set_opacity(0.5);
499 }
500 sc::set(pc_desc->shapes[i]->mutable_line()->mutable_start(), sensor_pos_w);
501 sc::set(pc_desc->shapes[i]->mutable_line()->mutable_end(), ray_end);
502 draw_shape(pc_desc->shapes[i]);
503 }
504 i++;
505 }
506 kv2.second->pub->publish(msg);
507 }
508 }
509 }
510
511 if (enable_collision_detection_) {
512 bt_collision_world->performDiscreteCollisionDetection();
513
514 int numManifolds = bt_collision_world->getDispatcher()->getNumManifolds();
515
516 // For each contact manifold
517 for (int i = 0; i < numManifolds; i++) {
518 btPersistentManifold* contactManifold =
519 bt_collision_world->getDispatcher()->getManifoldByIndexInternal(i);
520 const btCollisionObject* obA =
521 static_cast<const btCollisionObject*>(contactManifold->getBody0());
522 const btCollisionObject* obB =
523 static_cast<const btCollisionObject*>(contactManifold->getBody1());
524 contactManifold->refreshContactPoints(
525 obA->getWorldTransform(),
526 obB->getWorldTransform());
527 int numContacts = contactManifold->getNumContacts();
528
529 // For each contact point in that manifold
530 for (int j = 0; j < numContacts; j++) {
531 // Get the contact information
532 btManifoldPoint& pt = contactManifold->getContactPoint(j);
533
534 Eigen::Vector3d normal_B(
535 pt.m_normalWorldOnB.x(),
536 pt.m_normalWorldOnB.y(),
537 pt.m_normalWorldOnB.z());
538
539 if (id_to_ent_map_->count(obB->getUserIndex()) > 0) {
540 (*id_to_ent_map_)[obB->getUserIndex()]->motion()->set_external_force(-normal_B);
541 }
542 if (id_to_ent_map_->count(obA->getUserIndex()) > 0) {
543 (*id_to_ent_map_)[obA->getUserIndex()]->motion()->set_external_force(normal_B);
544 }
545 }
546
547 if (remove_on_collision_ && numContacts > 0) {
548 bool ob_a_is_ent, ob_b_is_ent;
549 int obj_a_id = obA->getUserIndex();
550 int obj_b_id = obB->getUserIndex();
551
552 sc::EntityPtr ent_a, ent_b;
553 std::tie(ob_a_is_ent, ent_a) = get_entity(obj_a_id);
554 std::tie(ob_b_is_ent, ent_b) = get_entity(obj_b_id);
555
556 // We need to determine if a collision occured because two
557 // entities collided or if the entity collided with a
558 // non-entity (terrain, shape obstacle, etc.). If both objects
559 // were removed, two entities collided with each other. If only
560 // one object was removed, it collided with the terrain or
561 // another static obstacle.
562 if (ob_a_is_ent && ob_b_is_ent) {
563 bool same_team = ent_a->id().team_id() == ent_b->id().team_id();
564 if (enable_team_collisions_ && same_team) {
565 // Construct the TeamCollision message
566 auto msg = std::make_shared<Message<sm::TeamCollision>>();
567 msg->data.set_entity_id_1(ent_a->id().id());
568 msg->data.set_entity_id_2(ent_b->id().id());
569 team_collision_pub_->publish(msg);
570
571 // Remove the entities
572 remove_entity_object(obj_a_id);
573 remove_entity_object(obj_b_id);
574
575 } else if (enable_non_team_collisions_ && !same_team) {
576 // Construct the NonTeamCollision message
577 auto msg = std::make_shared<Message<sm::NonTeamCollision>>();
578 msg->data.set_entity_id_1(ent_a->id().id());
579 msg->data.set_entity_id_2(ent_b->id().id());
580 non_team_collision_pub_->publish(msg);
581
582 // Remove the entities
583 remove_entity_object(obj_a_id);
584 remove_entity_object(obj_b_id);
585 }
586
587 } else if (ob_a_is_ent && ent_a->is_alive()) {
588 // Construct the GroundCollision message for entity A
589 auto msg = std::make_shared<sc::Message<sm::GroundCollision>>();
590 msg->data.set_entity_id(ent_a->id().id());
591 collision_pub_->publish(msg);
592
593 remove_entity_object(obj_a_id);
594
595 } else if (ob_b_is_ent && ent_b->is_alive()) {
596 // Construct the GroundCollision message for entity B
597 auto msg = std::make_shared<sc::Message<sm::GroundCollision>>();
598 msg->data.set_entity_id(ent_b->id().id());
599 collision_pub_->publish(msg);
600
601 remove_entity_object(obj_b_id);
602 }
603 }
604 }
605 }
606 return true;
607}

609bool BulletCollision::get_ray_tracing(
610 scrimmage::MessageBasePtr request,
611 scrimmage::MessageBasePtr& response) {
612 auto request_cast = std::dynamic_pointer_cast<sc::Message<RayTrace::PointCloudWithId>>(request);
613
614 if (request_cast == nullptr) {
615 std::cout << "Could not cast to sc::Message<RayTrace::PointCloudId> request" << std::endl;
616 return false;
617 }
618
619 int entity_id = request_cast->data.entity_id;
620 std::string sensor_name = request_cast->data.sensor_name;
621 sc::EntityPtr& own_ent = (*id_to_ent_map_)[entity_id];
622 Eigen::Vector3d own_pos = own_ent->state_truth()->pos();
623 // Compute transformation matrix from entity's frame to sensor's
624 // frame.
625 sc::SensorPtr& sensor = own_ent->sensors()[sensor_name];
626 Eigen::Matrix4d tf_m =
627 own_ent->state_truth()->tf_matrix(false) * sensor->transform()->tf_matrix();
628 auto response_cast = std::make_shared<sc::Message<RayTrace::PointCloud>>();
629 response_cast->data.max_range = request_cast->data.max_range;
630 response_cast->data.min_range = request_cast->data.min_range;
631 response_cast->data.rays = request_cast->data.get_rays();
632
633 // Create the points from the rays
634 for (auto& ray : response_cast->data.rays) {
635 Eigen::Vector3d r(response_cast->data.max_range, 0, 0);
636 sc::Quaternion rot_vert(Eigen::Vector3d(0, 1, 0), ray.elevation_rad);
637 r = rot_vert.rotate(r);
638
639 sc::Quaternion rot_horiz(Eigen::Vector3d(0, 0, 1), ray.azimuth_rad);
640 r = rot_horiz.rotate(r);
641
642 request_cast->data.points.push_back(RayTrace::PCPoint(r));
643 }
644
645 // For each ray in the sensor
646 for (RayTrace::PCPoint& pcpoint : request_cast->data.points) {
647 Eigen::Vector3d original_ray = pcpoint.point;
648 // Transform sensor's origin to world coordinates
649 Eigen::Vector4d sensor_pos = tf_m * Eigen::Vector4d(0, 0, 0, 1);
650 Eigen::Vector3d sensor_pos_w = sensor_pos.head<3>() + own_pos;
651
652 Eigen::Vector3d ray_w;
653 if (request_cast->data.world_frame == false) {
654 // Transform ray's end point to world coordinates
655 Eigen::Vector4d ray =
656 tf_m * Eigen::Vector4d(original_ray(0), original_ray(1), original_ray(2), 1);
657 ray_w = ray.head<3>() + own_pos;
658 } else {
659 // Already in world frame
660 ray_w = original_ray;
661 }
662
663 // Create bullet vectors
664 btVector3 btFrom(sensor_pos_w(0), sensor_pos_w(1), sensor_pos_w(2));
665 btVector3 btTo(ray_w(0), ray_w(1), ray_w(2));
666
667 // Perform ray casting
668 btCollisionWorld::ClosestRayResultCallback res(btFrom, btTo);
669 bt_collision_world->rayTest(btFrom, btTo, res);
670
671 // Points in the RayTrace message are defined with respect to
672 // the LIDAR sensor's coordinate frame. Use original ray's
673 // direction, shorten to length of detection ray, if a
674 // collision occurred.
675 Eigen::Vector3d ray_end;
676 if (res.hasHit()) {
677 Eigen::Vector3d hit_point(
678 res.m_hitPointWorld.x(),
679 res.m_hitPointWorld.y(),
680 res.m_hitPointWorld.z());
681 Eigen::Vector3d return_vec =
682 original_ray.normalized() * (hit_point - sensor_pos_w).norm();
683 response_cast->data.points.push_back(RayTrace::PCPoint(
684 return_vec,
685 255,
686 ((return_vec.norm() > response_cast->data.max_range)
687 || (return_vec.norm() < response_cast->data.min_range))));
688 ray_end = hit_point;
689 } else {
690 response_cast->data.points.push_back(RayTrace::PCPoint(original_ray, 255, true));
691 ray_end = ray_w;
692 }
693 }
694 // Save the response
695 response = response_cast;
696
697 // Success
698 return true;
699}
