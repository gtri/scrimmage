218bool Straight::step_autonomy(double t, double dt) {
219 if (gen_ents_) {
220 if (time_->t() > (prev_gen_time_ + 2.0)) {
221 prev_gen_time_ = time_->t();
222
223 // Create a state for the new entity
224 State s(*state_);
225
226 // Rotate the heading of the new entity by 90 degrees and offset
227 // the initial position to the left of our current position to
228 // avoid collisions.
229 Eigen::AngleAxisd rot_90_z(M_PI / 2.0, Eigen::Vector3d::UnitZ());
230 s.pos() = s.pos() + s.quat() * rot_90_z * (Eigen::Vector3d::UnitX() * 10);
231 s.quat() = rot_90_z * s.quat();
232
233 // Create the GenerateEntity message
234 auto msg = std::make_shared<Message<scrimmage_msgs::GenerateEntity>>();
235 sc::set(msg->data.mutable_state(), s); // Copy the new state
236
237 // The entity_tag must match the "tag" XML attribute of the entity
238 // to be generated in the mission file.
239 msg->data.set_entity_tag("gen_straight");
240
241 // Modify the entity's color
242 auto kv_color = msg->data.add_entity_param();
243 kv_color->set_key("color");
244 kv_color->set_value("255, 255, 0");
245
246 // Modify the entity's visual model
247 auto kv_visual = msg->data.add_entity_param();
248 kv_visual->set_key("visual_model");
249 kv_visual->set_value("sphere");
250
251 pub_gen_ents_->publish(msg); // Publish the GenerateEntity message
252 }
253 }
254
255 if (show_text_label_) {
256 // An example of changing a shape's property
257 if (t > 1.0 && t < (1.0 + dt)) {
258 text_shape_->set_opacity(0.5);
259 text_shape_->mutable_text()->set_text("Goodbye, SCRIMMAGE!");
260 draw_shape(text_shape_);
261 }
262
263 // An example of removing a shape
264 if (t > 5.0 && t < (5.0 + dt)) {
265 text_shape_->set_persistent(false);
266 draw_shape(text_shape_);
267 }
268 }
269
270 // Read data from sensors...
271 if (!noisy_state_set_) {
272 noisy_state_ = *state_;
273 }
274
275 if (boundary_ != nullptr && enable_boundary_control_) {
276 if (!boundary_->contains(noisy_state_.pos())) {
277 // Project goal through center of boundary
278 Eigen::Vector3d center = boundary_->center();
279 center(2) = noisy_state_.pos()(2); // maintain altitude
280 Eigen::Vector3d diff = center - noisy_state_.pos();
281 goal_ = noisy_state_.pos() + diff.normalized() * 1e6;
282 }
283 }
284
285 Eigen::Vector3d diff = goal_ - noisy_state_.pos();
286 Eigen::Vector3d v = speed_ * diff.normalized();
287
289 // Convert desired velocity to desired speed, heading, and pitch controls
291 double heading = Angles::angle_2pi(atan2(v(1), v(0)));
292 vars_.output(desired_alt_idx_, goal_(2));
293 vars_.output(desired_speed_idx_, v.norm());
294 vars_.output(desired_heading_idx_, heading);
295
296 noisy_state_set_ = false;
297 return true;
298}
