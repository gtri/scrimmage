195boost::optional<std::string> run_test(
196 const std::string& mission,
197 const bool& init_python,
198 const bool& shutdown_python) {
199 SimControl simcontrol;
200 if (not simcontrol.init(mission, init_python)) {
201 cout << "Failed to initialize SimControl." << endl;
202 return boost::none;
203 }
204
205 // Handle kill signals
206 struct sigaction sa;
207 memset(&sa, 0, sizeof(sa));
208 shutdown_handler = [&](int /*s*/) {
209 cout << endl << "Exiting gracefully" << endl;
210 simcontrol.force_exit();
211 };
212 sa.sa_handler = signal_handler;
213 sigfillset(&sa.sa_mask);
214 sigaction(SIGINT, &sa, NULL);
215 sigaction(SIGTERM, &sa, NULL);
216
217 simcontrol.mp()->set_time_warp(-1);
218 simcontrol.mp()->set_enable_gui(false);
219
220 simcontrol.pause(false);
221 if (not simcontrol.run()) {
222 return boost::none;
223 }
224
225 // Extract the log directory
226 auto out = boost::optional<std::string>{simcontrol.mp()->log_dir()};
227
228 // Shutdown SimControl
229 if (not simcontrol.shutdown(shutdown_python)) {
230 return boost::none;
231 }
232 return out;
233}
