45void CameraInterface::OnKeyPress() {
46 // Get the keypress
47 vtkRenderWindowInteractor* rwi = this->Interactor;
48 std::string key = rwi->GetKeySym();
49
50 if (last_key_ == "semicolon") {

53 } else if (key == "Right") {
54 updater_->inc_follow();

56 updater_->dec_follow();
57 } else if (key == "a") {
58 updater_->next_mode();
59 } else if (key == "u") {
60 updater_->undo_camera();
61 } else if (key == "c") {
62 updater_->request_cached();
63 } else if (key == "t") {
64 updater_->toggle_trails();
65 } else if (key == "bracketleft") {
66 updater_->dec_warp();
67 } else if (key == "bracketright") {
68 updater_->inc_warp();
69 } else if (key == "b") {
70 updater_->toggle_pause();
71 } else if (key == "space") {
72 updater_->single_step();
73 } else if (key == "plus" || key == "equal") {
74 updater_->inc_scale();
75 } else if (key == "minus") {
76 updater_->dec_scale();
77 } else if (key == "0") {
78 updater_->reset_scale();
79 } else if (key == "r") {
80 updater_->reset_view();
81 } else if (key == "z") {
82 updater_->inc_follow_offset();
83 } else if (key == "Z") {
84 updater_->dec_follow_offset();
85 } else if (key == "A") {
86 updater_->set_reset_camera();
87 } else if (key == "n") {
88 updater_->dec_label_scale();
89 } else if (key == "N") {
90 updater_->inc_label_scale();
91 } else if (key == "o") {
92 enable_object_draw_ = not enable_object_draw_;
93 } else if (key == "h") {
94 updater_->toggle_helpmenu();
95 } else if (key == "semicolon") {
96 last_key_ = key;
97 } else {
98 // cout << "key: " << key << endl;
99 }
