288bool External::step(double t) {
289 mutex.lock();
290 double dt = std::isnan(last_t_) ? 0 : t - last_t_;
291 time_->set_t(t);
292 time_->set_dt(dt);
293 last_t_ = t;
294 mutex.unlock();
295
296 if (!this->call_update_contacts(t)) {
297 LOG_ERROR("External::step failed during call_update_contacts");
298 return false;
299 }
300
301 mutex.lock();
302
304 // run main loops of plugins
306 for (AutonomyPtr autonomy : entity_->autonomies()) {
307 autonomy->step_autonomy(t, dt);
308 }
309
310 for (const auto& kv : entity_->sensors()) {
311 SensorPtr sensor = kv.second;
312 sensor->step();
313 }
314
315 entity_->setup_desired_state();
316
317 int num_steps = std::max(1.0, std::round(dt / motion_dt_));
318 double temp_t = t - dt;
319 for (int i = 0; i < num_steps; i++) {
320 for (ControllerPtr controller : entity_->controllers()) {
321 controller->step(temp_t, motion_dt_);
322 }
323 if (enable_motion_) {
324 entity_->motion()->step(temp_t, motion_dt_);
325 }
326 temp_t += motion_dt_;
327 }
328
329 if (!ent_inters_.empty()) {
330 update_ents();
331 for (EntityInteractionPtr ent_inter : ent_inters_) {
332 ent_inter->step_entity_interaction(ents_, t, dt);
333 }
334 }
335
336 for (MetricsPtr metrics : metrics_) {
337 metrics->step_metrics(t, dt);
338 }
339
340 if (!send_messages()) {
341 LOG_ERROR("External::step failed during send_messages");
342 mutex.unlock();
343 return false;
344 }
345
346 // shapes
347 scrimmage_proto::Shapes shapes;
348 shapes.set_time(t);
349 auto add_clear_shapes = [&](auto p) {
350 auto add_shape = [&](auto& shape) {
351 *shapes.add_shape() = *shape;
352
353 shape_queue_.push(shape);
354
355 auto max_size = this->shape_queue_max_size();
356 if (shape_queue_.size() > max_size) {
357 shape_queue_.pop();
358 }
359 };
360 br::for_each(p->shapes(), add_shape);
361 p->shapes().clear(); // don't let shapes build up
362 };
363
364 br::for_each(entity_->autonomies(), add_clear_shapes);
365 br::for_each(entity_->controllers(), add_clear_shapes);
366 br::for_each(entity_->sensors() | ba::map_values, add_clear_shapes);
367 br::for_each(ent_inters_, add_clear_shapes);
368 br::for_each(metrics_, add_clear_shapes);
369
370 log_->save_frame(create_frame(t, entity_->contacts()));
371 log_->save_shapes(shapes);
372
373 // Send shapes over configured grpc network interfaces
374 for (auto interface : outgoing_interfaces_) {
375 interface->send_shapes(shapes);
376 interface->send_frame(t, entity_->contacts());
377 }
378
379 mutex.unlock();
380 return true;
381}

387bool External::send_messages() {
388 // Send messages that are published by a SCRIMMAGE plugin to the external
389 // network (e.g., ROS, MOOS)
390 auto to_publisher = [&](auto& network_device) {
391 return std::dynamic_pointer_cast<Publisher>(network_device);
392 };
393 auto has_callback = [&](auto& pub) { return pub && pub->callback; };
394
395 for (auto& topic_device_kv : pubsub_->pubs() | ba::map_values) {
396 for (auto& dev_list : topic_device_kv | ba::map_values) {
397 for (auto pub : dev_list | ba::transformed(to_publisher) | ba::filtered(has_callback)) {
398 for (auto msg : pub->pop_msgs()) {
399 pub->callback(msg);
400 }
401 }
402 }
403 }
404
405 // any messages that are not linked to the external network can now
406 // be sent via the internal scrimmage messaging system
407 auto& pubs = pubsub_->pubs();
408 auto& subs = pubsub_->subs();
409 for (auto& network : *networks_ | ba::map_values) {
410 auto name = network->name();
411 if (!network->step(pubs[name], subs[name])) {
412 LOG_ERROR("Network failed: " << name);
413 return false;
414 }
415 }
416
417 br::for_each(entity_->autonomies(), run_callbacks);
418 br::for_each(entity_->controllers(), run_callbacks);
419 br::for_each(ent_inters_, run_callbacks);
420 br::for_each(metrics_, run_callbacks);
421 br::for_each(entity_->sensors() | ba::map_values, run_callbacks);
422 return true;
423}

425bool External::call_update_contacts(double t) {
426 mutex.lock();
427 if (update_contacts_task.update(t).first) {
428 auto rtree = entity_->rtree(); // rtree is a shared_ptr
429 if (!rtree) {
430 LOG_ERROR("call_update_contacts failed: rtree is null - entity not properly initialized");
431 mutex.unlock();
432 return false;
433 }
434 rtree->init(entity_->contacts()->size());
435 for (auto& kv : *entity_->contacts()) {
436 rtree->add(kv.second.state()->pos(), kv.second.id());
437 }
438 update_ents();
439 }
440 mutex.unlock();
441 return true;
442}

448void External::update_ents() {
449 if (ent_inters_.empty() && !metrics_.empty())
450 return;
451
452 auto& contacts = *entity_->contacts();
453
454 id_to_team_map_->clear();
455 id_to_ent_map_->clear();
456 ents_.clear();
457 for (auto& kv : contacts) {
458 ID& id = kv.second.id();
459 (*id_to_team_map_)[id.id()] = id.team_id();
460
461 auto ent = id.id() == entity_->id().id() ? entity_ : std::make_shared<Entity>();
462 ent->id() = id;
463 if (ent->state_truth()) {
464 *ent->state_truth() = *kv.second.state();
465 } else {
466 ent->state_truth() = kv.second.state();
467 }
468 ents_.push_back(ent);
469
470 (*id_to_ent_map_)[id.id()] = ent;
471 }
472}
