122Eigen::Vector3d ROSIMUSensor::gravity_ned_from_lla(Eigen::Vector3d lla) {
123 Eigen::Vector3d g;
124
125 lla(0) = Angles::deg2rad(lla(0));
126 lla(1) = Angles::deg2rad(lla(1));
127
128 double sLat = sin(lla(0));
129 double cLat = cos(lla(0));
130
131 double N = earth_radius / sqrt(1.0 - ecc2 * sLat * sLat);
132 double x = (N + lla(2)) * cLat * cos(lla(1));
133 double y = (N + lla(2)) * cLat * sin(lla(1));
134 double z = ((b2 / a2) * N + lla(2)) * sLat;
135
136 double x2 = x * x;
137 double y2 = y * y;
138 double z2 = z * z;
139 double r2 = x2 + y2 + z2 - e2;
140 double u2 = 0.5 * r2 * (1.0 + sqrt(1.0 + 4.0 * e2 * z2 / (r2 * r2)));
141 double u = sqrt(u2);
142 double v2 = u2 + e2;
143 double v = sqrt(v2);
144
145 double Beta = atan2(z * v, u * sqrt(x2 + y2));
146 double sBeta = sin(Beta);
147 double sBeta2 = sBeta * sBeta;
148 double cBeta = cos(Beta);
149
150 double w = sqrt(u2 + e2 * sBeta2) / v;
151 double q = 0.5 * ((1.0 + 3.0 * u2 / e2) * atan2(e, u) - 3.0 * u / e);
152 double q0 =
153 0.5
154 * ((1.0 + 3.0 * b2 / e2) * atan2(e, earth_semiminor_axis) - 3.0 * earth_semiminor_axis / e);
155 double qp = 3.0 * (1.0 + u2 / e2) * (1.0 - (u / e) * atan2(e, u)) - 1.0;
156
157 // % //Mass Attraction Only
158 double Gu =
159 -(wgs84_grav / v2 + omega2 * a2 * e * qp * (0.5 * sBeta2 - 1.0 / 6.0) / (q0 * v2)) / w;
160 double Gb = omega2 * sBeta * cBeta * a2 * q / (w * q0 * v);
161 double Psi = atan2(z, sqrt(x2 + y2));
162 double sPsi = sin(Psi);
163 double cPsi = cos(Psi);
164 double Alfa = lla(0) - Psi;
165 double sAlfa = sin(Alfa);
166 double cAlfa = cos(Alfa);
167 double Gr = Gu * (cPsi * cBeta * u / v + sPsi * sBeta) / w
168 + Gb * (sPsi * cBeta * u / v - cPsi * sBeta) / w;
169 double Gp = -Gu * (sPsi * cBeta * u / v - cPsi * sBeta) / w
170 + Gb * (cPsi * cBeta * u / v + sPsi * sBeta) / w;
171
172 g(0) = -Gr * sAlfa + Gp * cAlfa;
173 g(1) = 0.0;
174 g(2) = -Gr * cAlfa - Gp * sAlfa;
175
176 return g;
177}

179Eigen::Matrix3d ROSIMUSensor::ecef_to_ned_rotation(double lat, double lon) {
180 lat = Angles::deg2rad(lat);
181 lon = Angles::deg2rad(lon);
182
183 double sLat = sin(lat);
184 double sLon = sin(lon);
185 double cLat = cos(lat);
186 double cLon = cos(lon);
187
188 Eigen::Matrix3d CEcefToNed;
189 CEcefToNed(0, 0) = -sLat * cLon;
190 CEcefToNed(0, 1) = -sLat * sLon;
191 CEcefToNed(0, 2) = cLat;
192
193 CEcefToNed(1, 0) = -sLon;
194 CEcefToNed(1, 1) = cLon;
195 CEcefToNed(1, 2) = 0.0;
196
197 CEcefToNed(2, 0) = -cLat * cLon;
198 CEcefToNed(2, 1) = -cLat * sLon;
199 CEcefToNed(2, 2) = -sLat;
200
201 return CEcefToNed;
202}

204Eigen::Matrix3d ROSIMUSensor::skew_sym(Eigen::Vector3d vector) {
205 Eigen::Matrix3d outputMatrix;
206 outputMatrix(0, 0) = 0.0;
207 outputMatrix(0, 1) = -vector(2);
208 outputMatrix(0, 2) = vector(1);
209 outputMatrix(1, 0) = vector(2);
210 outputMatrix(1, 1) = 0.0;
211 outputMatrix(1, 2) = -vector(0);
212 outputMatrix(2, 0) = -vector(1);
213 outputMatrix(2, 1) = vector(0);
214 outputMatrix(2, 2) = 0.0;
215
216 return outputMatrix;
217}

219Eigen::Vector3d ROSIMUSensor::inv_skew_sym(Eigen::Matrix3d inputMatrix) {
220 Eigen::Vector3d OutputVector;
221 OutputVector(0) = 0.5 * (-inputMatrix(1, 2) + inputMatrix(2, 1));
222 OutputVector(1) = 0.5 * (inputMatrix(0, 2) - inputMatrix(2, 0));
223 OutputVector(2) = 0.5 * (-inputMatrix(0, 1) + inputMatrix(1, 0));
224
225 return OutputVector;
226}

228Eigen::Matrix3d ROSIMUSensor::enu_to_ecef_rotation(double lat, double lon) {
229 Eigen::Matrix3d enuToNed;
230 enuToNed << 0, 1, 0, 1, 0, 0, 0, 0, -1;
231 Eigen::Matrix3d rotationMatrix = ecef_to_ned_rotation(lat, lon);
232 // note don't use .transpose(), bad things happen
233 // https://eigen.tuxfamily.org/dox/group__TopicAliasing.html
234 rotationMatrix.transposeInPlace();
235 rotationMatrix = (rotationMatrix * enuToNed).eval();
236 return rotationMatrix;
237}

239Eigen::Vector3d ROSIMUSensor::get_delta_theta(
240 Eigen::Quaterniond qBodyToECEFt1Hat,
241 Eigen::Quaterniond qBodyToECEFt2,
242 double inertialDeltaT) {
243 Eigen::Matrix3d CBodyToECEFt1Hat(qBodyToECEFt1Hat);
244 Eigen::Matrix3d CBodyToECEFt2(qBodyToECEFt2);
245 Eigen::Matrix3d CBodyt2ToBodyt1 = CBodyToECEFt1Hat.transpose() * CBodyToECEFt2;
246
247 Eigen::Matrix3d Ak = CBodyt2ToBodyt1;
248 Eigen::Vector3d dTHat = inv_skew_sym(CBodyt2ToBodyt1);
249 Eigen::Vector3d errVect;
250 for (int nn = 0; nn < 3; nn++) {
251 Eigen::Matrix3d AkHat = skew_sym(dTHat).exp();
252 errVect = inv_skew_sym(AkHat.transpose() * Ak);
253 dTHat = errVect + dTHat;
254 }
255 // % .0035 deg/Hr is the quoted gyro bias for the HG9900, a navigation grade
256 // % IMU
257 if (errVect.norm() >= 0.0035 * (M_PI / 180.0) * (1 / 3600.0) * (inertialDeltaT) * (1 / 100.0)) {
258 cout << "Warning: Check delta-theta calculations inside get_delta_theta!!!" << endl;
259 }
260 Eigen::Vector3d DeltaThetaBodyWRTECEFInBody = dTHat;
261
262 Eigen::Vector3d EarthRateVector;
263 EarthRateVector << 0, 0, earth_rate;
264 Eigen::Vector3d deltaThetaBodyWRTInertialInBody =
265 DeltaThetaBodyWRTECEFInBody
266 + inertialDeltaT * CBodyToECEFt1Hat.transpose() * EarthRateVector;
267
268 return deltaThetaBodyWRTInertialInBody;
269}

271Eigen::Quaterniond ROSIMUSensor::omega_to_q_dot(
272 Eigen::Quaterniond qBToA,
273 Eigen::Vector3d omegaABInB) {
274 Eigen::Quaterniond Temp4By1;
275 Temp4By1.w() = 0;
276 Temp4By1.vec() = omegaABInB;
277 Eigen::Quaterniond qBToADot = qBToA * Temp4By1;
278 qBToADot.coeffs() *= 0.5;

282Eigen::Quaterniond ROSIMUSensor::integrate_quaternion(
283 Eigen::Quaterniond qBToA,
284 Eigen::Vector3d deltaThetasBFrame) {
285 Eigen::Quaterniond QPredictor(
286 qBToA.coeffs() + omega_to_q_dot(qBToA, deltaThetasBFrame).coeffs());
287 Eigen::Quaterniond QCorrector(
288 qBToA.coeffs() + omega_to_q_dot(QPredictor, deltaThetasBFrame).coeffs());
289 // Perform the addition
290 Eigen::Quaterniond sum;
291 sum.w() = QPredictor.w() + QCorrector.w(); // Add the scalar portion
292 sum.vec() = QPredictor.vec() + QCorrector.vec(); // Add the vector portion
293 Eigen::Quaterniond qBToAUpdated(0.5 * sum.coeffs()); // (QPredictor + QCorrector);
294 qBToAUpdated.coeffs() = qBToAUpdated.coeffs() / qBToAUpdated.norm();
295 return qBToAUpdated;
296}

298Eigen::Quaterniond ROSIMUSensor::propagate_quaternion(
299 Eigen::Quaterniond qBodyToECEFt1Hat,
300 Eigen::Vector3d deltaThetaBodyWRTInertialInBody,
301 double inertialDeltaT) {
302 Eigen::Matrix3d CBodyToECEFt1Hat(qBodyToECEFt1Hat);
303
304 Eigen::Vector3d EarthRateVector;
305 EarthRateVector << 0, 0, earth_rate;
306 Eigen::Vector3d deltaThetaBodyWRTECEFInBodyHat =
307 deltaThetaBodyWRTInertialInBody
308 - inertialDeltaT * CBodyToECEFt1Hat.transpose() * EarthRateVector;
309 Eigen::Quaterniond qBodyToECEFt2 =
310 integrate_quaternion(qBodyToECEFt1Hat, deltaThetaBodyWRTECEFInBodyHat);
311 return qBodyToECEFt2;
312}

314Eigen::Vector3d ROSIMUSensor::get_deltaV(
315 Eigen::Vector3d posECEF,
316 Eigen::Vector3d velECEF,
317 Eigen::Quaterniond qBodyToECEF,
318 double InertialDeltaT) {
319 double k1 = -1, k2 = -1.25;
320
321 Eigen::Vector3d Acct1 = (velECEF - vel_t1) / InertialDeltaT;
322 Eigen::Vector3d PosLLAt1 = ecef_to_lla(pos_ECEF_t1);
323 c_ECEF_To_NED_t1 = ecef_to_ned_rotation(PosLLAt1(0), PosLLAt1(1));
324 Eigen::Vector3d GravityNED = gravity_ned_from_lla(PosLLAt1);
325
326 Eigen::Vector3d GravityECEF = c_ECEF_To_NED_t1.transpose() * GravityNED;
327 Eigen::Matrix3d CECEFToBodyTruth(prev_qBody_to_ECEF);
328 CECEFToBodyTruth.transposeInPlace();
329
330 Eigen::Vector3d EarthRateVector;
331 EarthRateVector << 0, 0, earth_rate;
332 Eigen::Matrix3d EarthRateSkewSym = skew_sym(EarthRateVector);
333 Eigen::Vector3d deltaV = CECEFToBodyTruth
334 * (Acct1 + 2.0 * EarthRateSkewSym * vel_t1
335 + EarthRateSkewSym * EarthRateSkewSym * pos_ECEF_t1 - GravityECEF);
336 deltaV = deltaV * InertialDeltaT;
337
338 Eigen::Matrix3d CBodyToECEFt1Hat(qBody_to_ECEF_hat);
339 Eigen::Vector3d LLA = ecef_to_lla(p_hat);
340 GravityNED = gravity_ned_from_lla(LLA);
341
342 Eigen::Matrix3d CECEFToNED = ecef_to_ned_rotation(LLA(0), LLA(1));
343 GravityECEF = CECEFToNED.transpose() * GravityNED;
344
345 Eigen::Vector3d PErr = pos_ECEF_t1 - p_hat;
346 Eigen::Vector3d VErr = vel_t1 - v_hat;
347
348 Eigen::Vector3d AAdd = -k1 * PErr - k2 * VErr;
349 deltaV = (deltaV + InertialDeltaT * CBodyToECEFt1Hat.transpose() * AAdd);
350
351 Eigen::Vector3d Ahat = CBodyToECEFt1Hat * deltaV / InertialDeltaT - 2 * EarthRateSkewSym * v_hat
352 - EarthRateSkewSym * EarthRateSkewSym * p_hat + GravityECEF;
353 Eigen::Vector3d tmpVhat = v_hat + (InertialDeltaT)*Ahat;
354 p_hat = p_hat + (InertialDeltaT)*v_hat + 0.5 * (InertialDeltaT * InertialDeltaT) * Ahat;
355 v_hat = tmpVhat;
356
357 return deltaV;
358}

360bool ROSIMUSensor::step() {
361 // Obtain current state information
362 sc::StatePtr& state = parent_->state_truth();
363 double time_now = time_->t();
364 double dt = time_now - prev_time_;
365 prev_time_ = time_now;
366
367 // Fill IMU Message
368 sensor_msgs::Imu imu_msg;
369
370 // Header
371 std_msgs::Header header; // empty header
372 // TODO: header.frame = ? // No system for ROS Frame ID's yet
373 header.stamp = ros::Time::now(); // time
374 imu_msg.header = header;
375
376 // get the geodetic origin specified in the mission file
377 MissionParsePtr mp_ = parent_->mp();
378 double lat = mp_->latitude_origin();
379 double lon = mp_->longitude_origin();
380 double alt = mp_->altitude_origin();
381
382 // convert origin to ECEF and derive a rotation between ENU and ECEF coordinates
383 Eigen::Vector3d ecefOrigin = lla_to_ecef(lat, lon, alt);
384 Eigen::Matrix3d enuToEcef = enu_to_ecef_rotation(lat, lon);
385 Eigen::Quaterniond qEnuToEcef(enuToEcef);
386
387 // convert the current position (ENU) to ECEF
388 Eigen::Vector3d ecefPos = state->pos();
389 ecefPos = enuToEcef * ecefPos; // rotate
390 ecefPos = ecefPos + ecefOrigin; // translate
391
392 // convert (ENU) velocity to ECEF
393 Eigen::Vector3d ecefVel = state->vel();
394 ecefVel = enuToEcef * ecefVel; // rotate
395
396 // calculate the body to ECEF orientation
397 Eigen::Quaterniond qbodyToEnu = state->quat();
398 Eigen::Quaterniond qbodyToECEF = qEnuToEcef * qbodyToEnu;
399
400 // calculate angular velocity and linear acceleration based on this and the previous frames
401 if (!first_sample_collected) { // for the first sample, just store it, we need 2 samples to
402 // perform a calculation
403 pos_ECEF_t1 = ecefPos;
404 vel_t1 = ecefVel;
405 prev_qBody_to_ECEF = qbodyToECEF;
406 p_hat = pos_ECEF_t1;
407 v_hat = vel_t1;
408 qBody_to_ECEF_hat = qbodyToECEF;
409 first_sample_collected = true; // first sample has been collected
410 } else {
411 // calculate deltas
412 Eigen::Vector3d deltaV = get_deltaV(ecefPos, ecefVel, qbodyToECEF, dt);
413 Eigen::Vector3d deltaTheta = get_delta_theta(qBody_to_ECEF_hat, qbodyToECEF, dt);
414
415 // update values for next frame
416 qBody_to_ECEF_hat = propagate_quaternion(qBody_to_ECEF_hat, deltaTheta, dt);
417 pos_ECEF_t1 = ecefPos;
418 vel_t1 = ecefVel;
419 prev_qBody_to_ECEF = qbodyToECEF;
420
421 // Populate the message with orientation
422 imu_msg.orientation.x = qbodyToECEF.x();
423 imu_msg.orientation.y = qbodyToECEF.y();
424 imu_msg.orientation.z = qbodyToECEF.z();
425 imu_msg.orientation.w = qbodyToECEF.w();
426
427 if (error_sim == nullptr) {
428 // Angular Velocity
429 imu_msg.angular_velocity.x = deltaTheta.x();
430 imu_msg.angular_velocity.y = deltaTheta.y();
431 imu_msg.angular_velocity.z = deltaTheta.z();
432
433 // Linear Acceleration
434 imu_msg.linear_acceleration.x = deltaV.x();
435 imu_msg.linear_acceleration.y = deltaV.y();
436 imu_msg.linear_acceleration.z = deltaV.z();
437 } else {
438 // add error to the deltaV and deltaTheta values
439 NoisyIMUData noisyData = error_sim->EachCycle(error_budget, deltaV, deltaTheta);
440
441 // Angular Velocity
442 imu_msg.angular_velocity.x = noisyData.noisyDeltaTheta.x();
443 imu_msg.angular_velocity.y = noisyData.noisyDeltaTheta.y();
444 imu_msg.angular_velocity.z = noisyData.noisyDeltaTheta.z();
445
446 // Linear Acceleration
447 imu_msg.linear_acceleration.x = noisyData.noisyDeltaV.x();
448 imu_msg.linear_acceleration.y = noisyData.noisyDeltaV.y();
449 imu_msg.linear_acceleration.z = noisyData.noisyDeltaV.z();
450 }
451
452 // Publish IMU information
453 imu_pub_.publish(imu_msg);
454
455 // Write IMU data to CSV
456 // Write the CSV file to the root log directory file name = imu_data.csv
457 if (!csv.output_is_open()) {
458 cout << "File isn't open. Can't append to CSV" << endl;
459 }
460 csv.append(
461 sc::CSV::Pairs{
462 {"time", time_now},
463 {"dt", dt},
464 {"ECEF_POSX", ecefPos.x()},
465 {"ECEF_POSY", ecefPos.y()},
466 {"ECEF_POSZ", ecefPos.z()},
467 {"ECEF_VELX", ecefVel.x()},
468 {"ECEF_VELY", ecefVel.y()},
469 {"ECEF_VELZ", ecefVel.z()},
470 {"bodyToEcef_X", qbodyToECEF.x()},
471 {"bodyToEcef_Y", qbodyToECEF.y()},
472 {"bodyToEcef_Z", qbodyToECEF.z()},
473 {"bodyToEcef_W", qbodyToECEF.w()},
474 {"dV_X", deltaV.x()},
475 {"dV_Y", deltaV.y()},
476 {"dV_Z", deltaV.z()},
477 {"dTheta_X", deltaTheta.x()},
478 {"dTheta_Y", deltaTheta.y()},
479 {"dTheta_Z", deltaTheta.z()},
480 {"Noisy_dV_X", imu_msg.linear_acceleration.x},
481 {"Noisy_dV_Y", imu_msg.linear_acceleration.y},
482 {"Noisy_dV_Z", imu_msg.linear_acceleration.z},
483 {"Noisy_dTheta_X", imu_msg.angular_velocity.x},
484 {"Noisy_dTheta_Y", imu_msg.angular_velocity.y},
485 {"Noisy_dTheta_Z", imu_msg.angular_velocity.z}},
486 true,
487 true);
488 }
489 return true;
490}
