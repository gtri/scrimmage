117bool TerrainMap::generate_linear() {
118 // Force the altitude center to be at the midpoint between z_max and z_min
119 center_(2) = (z_max_ + z_min_) / 2.0;
120
121 // Initialize the map, such that row 0 is at z_min and the last row is at
122 // z_max, with a linear interpolation across the rows. Also, add noise to
123 // each height value.
124 double z_step = (z_max_ - z_min_) / num_y_rows_;
125
126 for (unsigned int row = 0; row < num_y_rows_; ++row) {
127 double height = z_step * row;
128 for (unsigned int col = 0; col < num_x_cols_; ++col) {
129 grid_[row][col].height = height + (*rng_)(*gener_);
130 grid_[row][col].is_set = true;
131 }
132 }
133 center_height_adjust();
134 clamp_height();
135 return true;
136}

138bool TerrainMap::generate_random_walk() {
139 for (unsigned int row = 0; row < num_y_rows_; ++row) {
140 for (unsigned int col = 0; col < num_x_cols_; ++col) {
141 // If this node is already set, skip (continue)
142 if (grid_[row][col].is_set) {
143 continue;
144 }
145 // Get the average of the neighbors that are already set
146 double height_avg = get_neighbor_avg(row, col);
147 grid_[row][col].height = height_avg + (*rng_)(*gener_);
148 grid_[row][col].is_set = true;
149 }
150 }
151 center_height_adjust();
152 clamp_height();
153 return true;
154}

190void TerrainMap::center_height_adjust() {
191 // Make sure the grid's center is located at the appropriate height.
192 double offset =
193 center_(2) - grid_[std::round(num_y_rows_ / 2.0)][std::round(num_x_cols_ / 2.0)].height;
194 for (unsigned int row = 0; row < num_y_rows_; ++row) {
195 for (unsigned int col = 0; col < num_x_cols_; ++col) {
196 grid_[row][col].height += offset;
197 }
198 }

201void TerrainMap::clamp_height() {
202 // Ensure all height values fall within z_min and z_max
203 for (unsigned int row = 0; row < num_y_rows_; ++row) {
204 for (unsigned int col = 0; col < num_x_cols_; ++col) {
205 grid_[row][col].height = clamp(grid_[row][col].height, z_min_, z_max_);

210double TerrainMap::get_neighbor_avg(const int& row, const int& col) {
211 unsigned int neighbors = 0;
212 double height_sum = 0;
213
214 if (row > 0) {
215 height_sum += grid_[row - 1][col].height;
216 ++neighbors;
217 }
218 if (col > 0) {
219 height_sum += grid_[row][col - 1].height;
220 ++neighbors;
221 }
222 if (row > 0 && col > 0) {
223 height_sum += grid_[row - 1][col - 1].height;
224 ++neighbors;
225 }
226
227 return (neighbors == 0) ? 0 : height_sum / neighbors;
228}

230scrimmage::ShapePtr TerrainMap::shape() {
231 // Create the shape associated with this terrain
232 sc::ShapePtr shape = std::make_shared<sp::Shape>();
233 shape->set_persistent(true);
234 sc::set(shape->mutable_color(), color_(0), color_(1), color_(2));
235 shape->mutable_pointcloud()->set_size(3);
236
237 for (unsigned int row = 0; row < num_y_rows_; ++row) {
238 double y = center_(1) - y_length_ / 2.0 + row * y_resolution_;
239 for (unsigned int col = 0; col < num_x_cols_; ++col) {
240 sp::Vector3d* p = shape->mutable_pointcloud()->add_point();
241 double x = center_(0) - x_length_ / 2.0 + col * x_resolution_;
242 sc::set(p, x, y, grid_[row][col].height);
243 }
244 }
245 return shape;
246}

268boost::optional<double> TerrainMap::height_at(const double& x, const double& y) {
269 int row = std::floor((y + y_length_ / 2.0 - center_(1)) / y_resolution_);
270 int col = std::floor((x + x_length_ / 2.0 - center_(0)) / x_resolution_);
271 if (row < 0 || row >= static_cast<int>(num_y_rows_) || col < 0
272 || col >= static_cast<int>(num_x_cols_) || not grid_[row][col].is_set) {
273 return boost::optional<double>{};
274 }
275 return grid_[row][col].height;
276}
