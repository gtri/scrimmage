150bool DubinsAirplane3D::step(double t, double dt) {
151 // Get inputs and saturate
152 speed_ = boost::algorithm::clamp(vars_.input(desired_speed_idx_), speed_min_, speed_max_);
153 pitch_ = vars_.input(desired_pitch_idx_);
154 roll_ = vars_.input(desired_roll_idx_);
155
156 x_[Uw] = state_->vel()(0);
157 x_[Vw] = state_->vel()(1);
158 x_[Ww] = state_->vel()(2);
159
160 x_[Xw] = state_->pos()(0);
161 x_[Yw] = state_->pos()(1);
162 x_[Zw] = state_->pos()(2);
163
164 // state_->quat().normalize();
165 state_->quat().set(roll_, pitch_, state_->quat().yaw());
166 quat_local_ = state_->quat() * quat_world_inverse_;
167 quat_local_.normalize();
168 x_[q0] = quat_local_.w();
169 x_[q1] = quat_local_.x();
170 x_[q2] = quat_local_.y();
171 x_[q3] = quat_local_.z();
172
173 Eigen::Vector3d force_body = quat_local_.rotate_reverse(ext_force_);
174 Eigen::Vector3d ext_moment_body = ext_moment_;
175 ext_force_ = Eigen::Vector3d::Zero();
176 ext_moment_ = Eigen::Vector3d::Zero();
177
178 x_[U] = speed_ + force_body(0) / mass_;
179 x_[V] = force_body(1) / mass_;
180 x_[W] = force_body(2) / mass_;
181
182 double turn_rate = 0;
183 if (std::abs(speed_) >= std::numeric_limits<double>::epsilon()) {
184 turn_rate = -g_ / speed_ * tan(roll_);
185 }
186
187 x_[P] = ext_moment_body(0) / mass_;
188 x_[Q] = ext_moment_body(1) / mass_;
189 x_[R] = ext_moment_body(2) / mass_ + turn_rate;
190
191 ode_step(dt);
192
193 // Normalize quaternion
194 quat_local_.w() = x_[q0];
195 quat_local_.x() = x_[q1];
196 quat_local_.y() = x_[q2];
197 quat_local_.z() = x_[q3];
198 quat_local_.normalize();
199
200 x_[q0] = quat_local_.w();
201 x_[q1] = quat_local_.x();
202 x_[q2] = quat_local_.y();
203 x_[q3] = quat_local_.z();
204
205 Eigen::Vector3d vel_local(x_[U], x_[V], x_[W]);
206
207 // Convert local coordinates to world coordinates
208 state_->quat() = quat_local_ * quat_world_;
209 state_->quat().normalize();
210 state_->pos() << x_[Xw], x_[Yw], x_[Zw];
211 state_->vel() << state_->quat().toRotationMatrix() * vel_local;
212
213 speed_ = vel_local.norm();
214
215 if (write_csv_) {
216 // Log state to CSV
217 csv_.append(sc::CSV::Pairs{
218 {"t", t},
219 {"x", x_[Xw]},
220 {"y", x_[Yw]},
221 {"z", x_[Zw]},
222 {"U", x_[U]},
223 {"V", x_[V]},
224 {"W", x_[W]},
225 {"P", x_[P]},
226 {"Q", x_[Q]},
227 {"R", x_[R]},
228 {"roll", state_->quat().roll()},
229 {"pitch", state_->quat().pitch()},
230 {"yaw", state_->quat().yaw()},
231 {"speed", speed_},
232 {"Uw", state_->vel()(0)},
233 {"Vw", state_->vel()(1)},
234 {"Ww", state_->vel()(2)}});
235 }
236
237 return true;
238}

240void DubinsAirplane3D::model(const vector_t& x, vector_t& dxdt, double t) {
241 dxdt[U] = 0;
242 dxdt[V] = 0;
243 dxdt[W] = 0;
244
245 dxdt[P] = 0;
246 dxdt[Q] = 0;
247 dxdt[R] = 0;
248
249 double lambda = 1 - (pow(x[q0], 2) + pow(x[q1], 2) + pow(x[q2], 2) + pow(x[q3], 2));
250 dxdt[q0] = -0.5 * (x[q1] * x[P] + x[q2] * x[Q] + x[q3] * x[R]) + lambda * x[q0];
251 dxdt[q1] = +0.5 * (x[q0] * x[P] + x[q2] * x[R] - x[q3] * x[Q]) + lambda * x[q1];
252 dxdt[q2] = +0.5 * (x[q0] * x[Q] + x[q3] * x[P] - x[q1] * x[R]) + lambda * x[q2];
253 dxdt[q3] = +0.5 * (x[q0] * x[R] + x[q1] * x[Q] - x[q2] * x[P]) + lambda * x[q3];
254
255 // Local position / velocity to global
256 // Normalize quaternion
257 sc::Quaternion quat(x[q0], x[q1], x[q2], x[q3]);
258 quat.normalize();
259
260 quat = quat * quat_world_;
261 quat.normalize();
262
263 // Convert local positions and velocities into global coordinates
264 Eigen::Matrix3d rot = quat.toRotationMatrix();
265
266 Eigen::Vector3d vel_local(x[U], x[V], x[W]);
267 Eigen::Vector3d vel_world = rot * vel_local;
268 dxdt[Xw] = vel_world(0);
269 dxdt[Yw] = vel_world(1);
270 dxdt[Zw] = vel_world(2);
271}
