46bool supports_fp64(const cl::Device& device) {
47 cl_int err;
48 constexpr cl_device_fp_config MIN_FP_CAPABILITIES =
49 (CL_FP_FMA | CL_FP_ROUND_TO_NEAREST | CL_FP_ROUND_TO_ZERO | CL_FP_ROUND_TO_INF);
50 cl_device_fp_config fp_config = device.getInfo<CL_DEVICE_DOUBLE_FP_CONFIG>(&err);
51 if (check_error(err, "Error Querying Device for Floating Point Information.")) {
52 return false;
53 }
54 return (fp_config & MIN_FP_CAPABILITIES) == MIN_FP_CAPABILITIES;
55}

87bool check_error(cl_int err, const std::string&& msg) {
88 if (err != 0) {
89 fprintf(
90 stderr,
91 "OpenCL Error: %s (CODE: %d): %s\n",
92 CL_ERROR_MESSAGES.at(err).c_str(),
93 err,
94 msg.c_str());
95 return true;
96 }
97 return false;

100std::optional<std::size_t> cacheline_size(const cl::Device& device) {
101 // Check if device has cache
102 cl_int err;
103 if (device.getInfo<CL_DEVICE_GLOBAL_MEM_CACHE_TYPE>(&err) == CL_NONE) {
104 check_error(err, "Error Querying Device Cache Type");
105 return std::nullopt;
106 }
107
108 std::size_t cacheline_size = device.getInfo<CL_DEVICE_GLOBAL_MEM_CACHELINE_SIZE>(&err);
109
110 if (check_error(err, "Error Querying Device Cacheline Size")) {
111 return std::nullopt;
112 }
113 return std::make_optional(cacheline_size);
114}
