93void NetworkDevice::enforce_queue_size() {
94 if (enable_queue_size_) {
95 if (msg_list_.size() > max_queue_size_) {
96 mutex_.lock();
97 auto erase_end = msg_list_.begin();
98 std::advance(erase_end, msg_list_.size() - max_queue_size_);
99 msg_list_.erase(msg_list_.begin(), erase_end);
100
101 // enforce size constraint on undelivered messages
102 erase_end = undelivered_msg_list_.begin();
103 std::advance(erase_end, undelivered_msg_list_.size() - max_queue_size_);
104 undelivered_msg_list_.erase(undelivered_msg_list_.begin(), erase_end);
105
106 mutex_.unlock();
107 }
108 }

132void NetworkDevice::add_undelivered_msg(MessageBasePtr msg, const bool& is_stochastic_delay) {
133 mutex_.lock();
134 if (!is_stochastic_delay) {
135 // in case of a single, deterministic delay, just add msg to end of queue
136 undelivered_msg_list_.push_back(msg);
137 } else {
138 // if delay is stochastic, sort messages by delivery time, with first-to-deliver in front
139 std::list<MessageBasePtr>::reverse_iterator it = undelivered_msg_list_.rbegin();
140 for (; it != undelivered_msg_list_.rend();) {
141 if ((*it)->time <= msg->time) {
142 break;
143 } else {
144 ++it;
145 }
146 }
147 undelivered_msg_list_.insert(it.base(), msg);
148 }
149 mutex_.unlock();
150}
