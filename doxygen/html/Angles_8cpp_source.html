155void Angles::set_angle(double angle) {
156 // Norm the angle
157 angle = angle_360(angle);
158
159 if (in_direction_ == out_direction_) {
160 // If the angle increase is in the same direction
161 // Just add or subtract a specific offset
162 int rots = static_cast<int>(in_zero_) - static_cast<int>(out_zero_);
163 angle_ = angle_360(angle + rots * 90);
164 } else {
165 // If the coordinates are in different CW / CCW frames
166
167 // Determine number of rotations
168 // The sign is different depending on if the in-direction is
169 // CW or CCW
170 int rots;
171 if (in_direction_ == Rotate::CW) {
172 rots = static_cast<int>(in_zero_) - static_cast<int>(out_zero_);
173 } else {
174 rots = static_cast<int>(out_zero_) - static_cast<int>(in_zero_);
175 }
176
177 // If the coordinates are different CW/CCW, but the zero is on
178 // the same axis, need to set rotations to 4 to get (90*4 = 360)
179 if (rots == 0) {
180 rots = 4;
181 }
182 angle_ = angle_360(rots * 90 - angle);
183 }
