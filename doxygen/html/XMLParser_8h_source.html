167 /*
168 * Iterates over every Node that has this node as a parent.
169 * Will perform an in-order walk of the tree with this parents root
170 */
171 class RecursiveIterator {
172 public:
173 using iterator_category = std::forward_iterator_tag;
174 using difference_type = std::ptrdiff_t;
175 using value_type = T;
176 using pointer = T*;
177 using reference = T&;
178
179 explicit RecursiveIterator(T node) : node_{node} {}
180
181 RecursiveIterator(const RecursiveIterator& other)
182 : node_{other.node_}, ancestors_{other.ancestors_} {}
183 ~RecursiveIterator() {}
184
185 reference operator*() { return node_; }
186 pointer operator->() { return &node_; }
187
188 RecursiveIterator& operator++() {
189 if (node_ != T{nullptr}) { // Avoid incrementing if this is the end node
190 T child = node_.get_first_node();
191 T sibling = node_.get_next_sibling();
192 if (child.is_valid()) {
193 ancestors_.push_back(node_);
194 node_ = child;
195 } else if (sibling.is_valid()) {
196 node_ = sibling;
197 } else {
198 // No child or sibling.
199 node_ = T{nullptr};
200 while (!ancestors_.empty() && (!node_.is_valid())) {
201 T parent = ancestors_.back();
202 ancestors_.pop_back();
203 node_ = parent.get_next_sibling();
204 }
205 }
206 if (!node_.is_valid()) {
207 node_ = T{nullptr};
208 }
209 }
210 return *this;
211 }
212
213 RecursiveIterator& operator++(int) {
214 RecursiveIterator& old{*this};
215 ++(*this);
216 return old;
217 }
218
219 friend bool operator==(const RecursiveIterator& lhs, const RecursiveIterator& rhs) {
220 return (lhs.node_ == rhs.node_) || (!lhs.node_.is_valid() && !rhs.node_.is_valid());
221 }
222
223 friend bool operator!=(const RecursiveIterator& lhs, const RecursiveIterator& rhs) {
224 return !(lhs == rhs);
225 }
226
227 private:
228 bool is_valid() { return node_ != nullptr && node_->is_valid(); }

230 T node_;
231 std::vector<T> ancestors_;
232 };

341std::ostream& operator<<(std::ostream& out, const XMLParserNode<T>& xml_node) {
342 using pos_type = typename std::ostream::pos_type;
343 if (xml_node.is_valid()) {
344 pos_type cur_pos;
345
346 std::string name = xml_node.name();
347 assert(!name.empty());
348 std::ostringstream ss;
349
350 ss << "<" << name << " ";
351 for (auto attr_it = xml_node.attr_begin(); attr_it != xml_node.attr_end(); ++attr_it) {
352 ss << *attr_it << " ";
353 }
354 cur_pos = ss.tellp();
355 assert(cur_pos != pos_type(-1));
356 ss.seekp(cur_pos - pos_type(1));
357 ss << ">";
358 auto child_it = xml_node.child_begin();
359 auto child_end = xml_node.child_end();
360 bool has_children = child_it != child_end;
361 if (has_children) {
362 for (; child_it != child_end; ++child_it) {
363 ss << "\n" << *child_it;
364 }
365 }
366 ss << "</" << name << ">";
367 out << ss.str();
368 }
369 return out;
370}
