30fp4_t quat_from_euler(fp_t roll, fp_t pitch, fp_t yaw) {
31 fp4_t q;
32
33 fp_t sr = sin(roll / 2);
34 fp_t cr = cos(roll / 2);
35 fp_t sy = sin(yaw / 2);
36 fp_t cy = cos(yaw / 2);
37 fp_t sp = sin(pitch / 2);
38 fp_t cp = cos(pitch / 2);

40 q.w = cr * cp * cy + sr * sp * sy;
