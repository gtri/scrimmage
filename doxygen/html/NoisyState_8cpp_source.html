54void NoisyState::init(std::map<std::string, std::string>& params) {
55 auto add_noise = [&](std::string prefix, auto& noise_vec) {
56 for (int i = 0; i < 3; i++) {
57 std::string tag_name = prefix + "_" + std::to_string(i);
58 std::vector<double> vec;
59 double mean, stdev;
60 std::tie(mean, stdev) = get_vec(tag_name, params, " ", vec, 2)
61 ? std::make_pair(vec[0], vec[1])
62 : std::make_pair(0.0, 1.0);
63 noise_vec.push_back(parent_->random()->make_rng_normal(mean, stdev));
64 }
65 };
66
67 add_noise("pos_noise", pos_noise_);
68 add_noise("vel_noise", vel_noise_);
69 add_noise("orient_noise", orient_noise_);
70
71 pub_ = advertise("LocalNetwork", "StateWithCovariance");
72
73 // Move the enity's state pointer to use the state instance provided by
74 // this plugin.
75 parent_->set_state_belief(parent_->state_truth());
76
77 return;
78}

80bool NoisyState::step() {
81 auto gener = parent_->random()->gener();
82
83 // Make a copy of the current ground truth state
84 StateWithCovariance ns(*(parent_->state_truth()));
85
86 auto msg = std::make_shared<Message<StateWithCovariance>>();
87
88 for (int i = 0; i < 3; i++) {
89 msg->data.pos()(i) = ns.pos()(i) + (*pos_noise_[i])(*gener);
90 msg->data.vel()(i) = ns.vel()(i) + (*vel_noise_[i])(*gener);
91 }
92
93 // TODO: Test this math.
94 // add noise in roll, pitch, yaw order
95 msg->data.quat() = ns.quat() * Quaternion(Eigen::Vector3d::UnitX(), (*orient_noise_[0])(*gener))
96 * Quaternion(Eigen::Vector3d::UnitY(), (*orient_noise_[1])(*gener))
97 * Quaternion(Eigen::Vector3d::UnitZ(), (*orient_noise_[2])(*gener));
98
99 pub_->publish(msg);
100
101 // Update the entity's state.
102 parent_->set_state_belief(static_cast<sc::State>(msg->data));
103
104 return true;
105}
