112bool TrajectoryRecordPlayback::step_autonomy(double t, double dt) {
113 if (!enable_playback_) {
114 // Get previous autonomy's desired_state
115 auto it = std::find_if(
116 parent_->autonomies().rbegin(),
117 parent_->autonomies().rend(),
118 [&](auto autonomy) { return autonomy->get_is_controlling(); });
119
120 csv_.append(sc::CSV::Pairs{
121 {"t", t},
122 {"x", state_->pos()(0)},
123 {"y", state_->pos()(1)},
124 {"z", state_->pos()(2)},
125 {"vx", state_->vel()(0)},
126 {"vy", state_->vel()(1)},
127 {"vz", state_->vel()(2)},
128 {"roll", state_->quat().roll()},
129 {"pitch", state_->quat().pitch()},
130 {"yaw", state_->quat().yaw()},
131 {"x_d", (*(*it)->desired_state()).pos()(0)},
132 {"y_d", (*(*it)->desired_state()).pos()(1)},
133 {"z_d", (*(*it)->desired_state()).pos()(2)},
134 {"vx_d", (*(*it)->desired_state()).vel()(0)},
135 {"vy_d", (*(*it)->desired_state()).vel()(1)},
136 {"vz_d", (*(*it)->desired_state()).vel()(2)},
137 {"roll_d", (*(*it)->desired_state()).quat().roll()},
138 {"pitch_d", (*(*it)->desired_state()).quat().pitch()},
139 {"yaw_d", (*(*it)->desired_state()).quat().yaw()},
140 });
141
142 return true;
143 }
144
145 // Playback the trajectory
146 (*desired_state_) = it_traj_->desired_state();
147
148 // Increment trajectory reference
149 ++it_traj_;
150
151 if (it_traj_ == trajs_.end()) {
152 if (remove_at_end_) {
153 // Remove entity from simulation
154 parent_->set_health_points(0);
155 } else {
156 // Remain at last trajectory command if end is reached
157 --it_traj_;
158 }
159 }
160
161 return true;
162}
