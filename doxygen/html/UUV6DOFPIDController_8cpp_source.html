80bool UUV6DOFPIDController::step(double t, double dt) {
81 heading_pid_.set_setpoint(vars_.input(desired_heading_idx_));
82 double u_rudder = heading_pid_.step(time_->dt(), parent()->state_belief()->quat().yaw());
83
84 speed_pid_.set_setpoint(vars_.input(desired_speed_idx_));
85 double u_throttle = speed_pid_.step(time_->dt(), parent()->state_belief()->vel().norm());
86
87 // Reconstruct original velocity vector (close, but not exact)
88 double x_vel =
89 vars_.input(desired_speed_idx_) / sqrt(1 + pow(tan(vars_.input(desired_heading_idx_)), 2));
90 double y_vel = x_vel * tan(vars_.input(desired_heading_idx_));
91 Eigen::Vector3d vel(
92 x_vel,
93 y_vel,
94 vars_.input(desired_altitude_idx_) - parent()->state_belief()->pos()(2));
95 vel = vel.normalized() * vars_.input(desired_speed_idx_);
96
97 // Track desired pitch
98 double desired_pitch = atan2(vel(2), vel.head<2>().norm());
99 pitch_pid_.set_setpoint(desired_pitch);
100 double u_elevator = -pitch_pid_.step(time_->dt(), -parent()->state_belief()->quat().pitch());
101
102 vars_.output(rudder_idx_, u_rudder);
103 vars_.output(elevator_idx_, u_elevator);
104 vars_.output(throttle_idx_, u_throttle);
105
106 return true;
107}
