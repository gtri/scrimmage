88bool Unicycle::step(double t, double dt) {
89 // Get inputs and saturate
90 velocity_ = clamp(vars_.input(speed_idx_), -vel_max_, vel_max_);
91 turn_rate_ = clamp(vars_.input(turn_rate_idx_), -turn_rate_max_, turn_rate_max_);
92 if (use_pitch_) {
93 pitch_rate_ = clamp(vars_.input(pitch_rate_idx_), -pitch_rate_max_, pitch_rate_max_);
94 } else {
95 velocity_z_ = clamp(vars_.input(velocity_z_idx_), -velocity_z_max_, velocity_z_max_);
96 }
97
98 double prev_x = x_[X];
99 double prev_y = x_[Y];
100 double prev_z = x_[Z];
101
102 ode_step(dt);
103
104 double dx = (x_[X] - prev_x) / dt;
105 double dy = (x_[Y] - prev_y) / dt;
106 double dz = (x_[Z] - prev_z) / dt;
107
108 state_->vel()(0) = dx;
109 state_->vel()(1) = dy;
110 state_->vel()(2) = dz;
111
112 state_->pos()(0) = x_[X];
113 state_->pos()(1) = x_[Y];
114 state_->pos()(2) = x_[Z];
115
116 double roll = 0; // TODO: simulate roll if enabled
117 if (enable_roll_ && turn_rate_ != 0) {
118 // see https://en.wikipedia.org/wiki/Standard_rate_turn
119 const double radius = velocity_ / turn_rate_;
120 roll = -atan2(pow(velocity_, 2) / radius, g_);
121 }
122 state_->quat().set(roll, x_[PITCH], x_[YAW]);
123
124 return true;
125}

127void Unicycle::model(const vector_t& x, vector_t& dxdt, double t) {
128 double xy_speed = velocity_ * cos(x[PITCH]);
129 dxdt[X] = xy_speed * cos(x[YAW]);
130 dxdt[Y] = xy_speed * sin(x[YAW]);
131 dxdt[YAW] = turn_rate_;
132 if (use_pitch_) {
133 dxdt[Z] = -velocity_ * sin(x[PITCH]);
134 dxdt[PITCH] = pitch_rate_;
135 } else {
136 dxdt[Z] = velocity_z_;
137 }
