63bool UnicycleControllerPoint::step(double t, double dt) {

65 Eigen::Vector2d pos_xy = parent()->state_belief()->pos().head<2>();
66 Eigen::Vector2d des_vel = gain_ * (des_pos_xy - pos_xy);
67
68 double th = parent()->state_belief()->quat().yaw();
69
70 Eigen::Matrix2d M;
71 M << cos(th), sin(th), -sin(th) / l_, cos(th) / l_;
72 Eigen::Vector2d u_2d = M * des_vel;
73
74 vars_.output(speed_idx_out_, u_2d(0));
