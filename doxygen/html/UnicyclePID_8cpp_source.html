95bool UnicyclePID::step(double t, double dt) {
96 heading_pid_.set_setpoint(vars_.input(desired_heading_idx_));
97 vars_.output(
98 turn_rate_idx_,
99 heading_pid_.step(time_->dt(), parent()->state_belief()->quat().yaw()));
100
101 // Reconstruct original velocity vector (close, but not exact
102 double x_vel =
103 vars_.input(desired_speed_idx_) / sqrt(1 + pow(tan(vars_.input(desired_heading_idx_)), 2));
104 double y_vel = x_vel * tan(vars_.input(desired_heading_idx_));
105 Eigen::Vector3d vel(
106 x_vel,
107 y_vel,
108 vars_.input(desired_alt_idx_) - parent()->state_belief()->pos()(2));
109 vel = vel.normalized() * vars_.input(desired_speed_idx_);
110
111 // Track desired pitch
112 double desired_pitch = atan2(vel(2), vel.head<2>().norm());
113 pitch_pid_.set_setpoint(desired_pitch);
114 vars_.output(
115 pitch_rate_idx_,
116 -pitch_pid_.step(time_->dt(), -parent()->state_belief()->quat().pitch()));
117
118 // Track zero roll
119 roll_pid_.set_setpoint(0);
120 vars_.output(
121 roll_rate_idx_,
122 -roll_pid_.step(time_->dt(), -parent()->state_belief()->quat().roll()));
123
124 if (show_shapes_) {
125 line_shape_->set_persistent(true);
126 sc::set(line_shape_->mutable_color(), 255, 0, 0);
127 line_shape_->set_opacity(1.0);
128 sc::set(line_shape_->mutable_line()->mutable_start(), parent()->state_belief()->pos());
129 sc::set(line_shape_->mutable_line()->mutable_end(), vel + parent()->state_belief()->pos());
130 draw_shape(line_shape_);
131 }
132
133 // Only use PID if controlling acceleration, otherwise, we directly set the
134 // forward speed
135 double desired_speed = vars_.input(desired_speed_idx_);
136 if (use_accel_) {
137 speed_pid_.set_setpoint(desired_speed);
138 desired_speed = speed_pid_.step(time_->dt(), parent()->state_belief()->vel().norm());
139 }
140 vars_.output(speed_idx_, desired_speed);
141
142 return true;
143}
