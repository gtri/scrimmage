88bool TakeFlag::step_autonomy(double t, double dt) {
89
90 Eigen::Vector3d desired_point = state_->pos();
91 if (!has_flag_) {
92 // If we don't have the flag yet, head towards it!
93 auto it = boundaries_.find(flag_boundary_id_);
94 if (it != boundaries_.end()) {
95 desired_point = std::get<1>(it->second)->center();
96 }
97 } else {
98 // Once we have the flag, head towards the return boundary
99 auto it = boundaries_.find(capture_boundary_id_);
100 if (it != boundaries_.end()) {
101 desired_point = std::get<1>(it->second)->center();
102 }
103 }
104
105 // Convert the desired point into a desired vector, relative to our own
106 // position.
107 Eigen::Vector3d desired_velocity = desired_point - state_->pos();
108
109 vars_.output(output_vel_x_idx_, desired_velocity(0));
110 vars_.output(output_vel_y_idx_, desired_velocity(1));
111 vars_.output(output_vel_z_idx_, desired_velocity(2));
112
113 return true;
114}
