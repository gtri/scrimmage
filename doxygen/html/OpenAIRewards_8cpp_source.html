66void OpenAIRewards::init(std::map<std::string, std::string>& /*params*/) {
67 auto cb = [&](auto msg) {
68 size_t id;
69 double reward;
70 std::tie(id, reward) = msg->data;
71 auto it = rewards_.find(id);
72 if (it == rewards_.end()) {
73 rewards_[id] = reward;
74 } else {
75 it->second += reward;
76 }
77 print_team_summary_ = true;
78 };
79 subscribe<std::pair<size_t, double>>("GlobalNetwork", "reward", cb);
80}
