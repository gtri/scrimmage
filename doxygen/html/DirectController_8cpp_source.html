58void DirectController::init(std::map<std::string, std::string>& params) {
59 // Discover the output variables that point to the input variables for the
60 // motion model. Setup input variables that mirror the output variables.
61 for (auto& kv : vars_.output_variable_index()) {
62 int out_idx = vars_.declare(kv.first, VariableIO::Direction::Out);
63 int in_idx = vars_.declare(kv.first, VariableIO::Direction::In);
64 io_map_[out_idx] = in_idx;
65 }
66}
