49void Logger::init(const std::string& path) {
50 std::lock_guard<std::mutex> lock(mutex_);
51 if (file_.is_open()) file_.close();
52 file_.open(path, std::ios::app);
53
54 // Flush any pending messages that were buffered before init
55 if (file_.is_open() && !pending_.empty()) {
56 for (const auto& pm : pending_) {
57 file_ << pm.formatted << std::endl;
58 }
59 file_.flush();
60 pending_.clear();
61 }
62 dir_initialized_ = true;
63}

101void Logger::write(const char* level, const char* file, int line,
102 const std::string& msg) {
103 // Strip path to filename only
104 const char* filename = std::strrchr(file, '/');
105 filename = filename ? filename + 1 : file;
106
107 // Full output for log file (with annotations)
108 std::ostringstream ss;
109 ss << "[" << level << "] ";
110 if (filename && filename[0] != '\0') {
111 ss << filename << ":" << line << " - ";
112 }
113 ss << msg;
114 std::string file_output = ss.str();
115
116 std::lock_guard<std::mutex> lock(mutex_);
117
118 // ANSI color codes for console output
119 const char* color_start = "";
120 const char* color_end = "\033[0m";
121 if (std::strcmp(level, "ERROR") == 0) {
122 color_start = "\033[31m"; // Red
123 } else if (std::strcmp(level, "WARN") == 0) {
124 color_start = "\033[33m"; // Yellow
125 } else {
126 color_end = ""; // No reset needed if no color
127 }
128
129 // Console: plain message with color only
130 std::cout << color_start << msg << color_end << std::endl;
131
132 // If not initialized yet, buffer the message for later
133 if (!dir_initialized_) {
134 PendingMessage pm;
135 pm.level = level;
136 pm.file = file;
137 pm.line = line;
138 pm.msg = msg;
139 pm.formatted = file_output;
140 pending_.push_back(std::move(pm));
141 } else if (file_.is_open()) {
142 file_ << file_output << std::endl; // Full annotations in file
143 }
144
145 // Flush immediately on ERROR to avoid losing logs on crash
146 if (std::strcmp(level, "ERROR") == 0) {
147 std::cout.flush();
148 if (dir_initialized_ && file_.is_open()) file_.flush();
149 }
150}
