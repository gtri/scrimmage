83 void compute_dots() {
84 // Source:
85 // http://math.stackexchange.com/questions/1472049/check-if-a-point-is-inside-a-rectangular-shaped-area-3d
86 u = points_[0] - points_[1];
87 v = points_[0] - points_[3];

98 bool contains(Eigen::Vector3d p) override {
99 double u_dot_p = u.dot(p);
100 double v_dot_p = v.dot(p);
101 double w_dot_p = w.dot(p);
102
103 if ((u_dot_P0 > u_dot_p) && (u_dot_p > u_dot_P1) && (v_dot_P0 > v_dot_p)
104 && (v_dot_p > v_dot_P3) && (w_dot_P0 > w_dot_p) && (w_dot_p > w_dot_P4)) {
105 return true;
106 }
107 return false;
108 }

110 void set_points(std::vector<Eigen::Vector3d>& points) {
111 points_ = points;
112 compute_dots();
113
114 Eigen::Vector3d xy_center = (points[0] + points[2]) / 2;
115 double alt_center = (points[0](2) + points[5](2)) / 2;
116 center_ << xy_center(0), xy_center(1), alt_center;
117
118 // Compute min / max values for x, y, z
119 Eigen::Vector3d mins(
120 std::numeric_limits<double>::infinity(),
121 std::numeric_limits<double>::infinity(),
122 std::numeric_limits<double>::infinity());
123 Eigen::Vector3d maxs(
124 -std::numeric_limits<double>::infinity(),
125 -std::numeric_limits<double>::infinity(),
126 -std::numeric_limits<double>::infinity());
127 for (Eigen::Vector3d p : points) {
128 for (int i = 0; i < 3; i++) {
129 if (p(i) < mins(i))
130 mins(i) = p(i);
131 if (p(i) > maxs(i))
132 maxs(i) = p(i);
133 }
134 }
