268bool JSBSimControl::step(double time, double dt) {
269
270 throttle_ = ba::clamp(vars_.input(throttle_idx_), -1.0, 1.0);
271 delta_elevator_ = ba::clamp(vars_.input(elevator_idx_), -1.0, 1.0);
272 delta_aileron_ = ba::clamp(vars_.input(aileron_idx_), -1.0, 1.0);
273 delta_rudder_ = ba::clamp(vars_.input(rudder_idx_), -1.0, 1.0);
274
275 // TODO: for some reason, jsb sim does not like it when there is an immediate thottle input
276 if (time < .05)
277 throttle_ = 0;
278
279 ap_aileron_cmd_node_->setDoubleValue(delta_aileron_);
280 ap_elevator_cmd_node_->setDoubleValue(delta_elevator_);
281 ap_rudder_cmd_node_->setDoubleValue(delta_rudder_);
282 ap_throttle_cmd_node_->setDoubleValue(throttle_);

286
288 // Save state
289 parent_->projection()->Forward(
290 latitude_node_->getDoubleValue(),
291 longitude_node_->getDoubleValue(),
292 altitude_node_->getDoubleValue() * feet2meters,

294 state_->pos()(1),
295 state_->pos()(2));
296
297 angles_from_jsbsim_.set_angle(ang::rad2deg(yaw_node_->getDoubleValue()));
298
299 state_->quat().set(
300 roll_node_->getDoubleValue(),
301 -pitch_node_->getDoubleValue(),

351 sc::set(line->mutable_line()->mutable_start(), state_->pos());

353 draw_shape(line);
