63bool EnforceBoundaryInteraction::init(
64 std::map<std::string, std::string>& mission_params,
65 std::map<std::string, std::string>& plugin_params) {
66
67 std::vector<int> active_boundary_ids;
68 if (!sc::get_vec<int>("active_boundary_ids", plugin_params, ", ", active_boundary_ids)) {
69 std::cout << "Failed to parse 'active_boundary_ids'" << endl;
70 return false;
71 } else {
72 active_boundary_ids_ =
73 std::set<int>(active_boundary_ids.begin(), active_boundary_ids.end());
74 }
75
76 auto callback = [&](scrimmage::MessagePtr<sp::Shape> msg) {
77 if (active_boundary_ids_.count(msg->data.id().id()) != 0) {
78 boundaries_.push_back(sci::Boundary::make_boundary(msg->data));
79 }
80 };
81 subscribe<sp::Shape>("GlobalNetwork", "Boundary", callback);
82
83 return true;
84}

86bool EnforceBoundaryInteraction::step_entity_interaction(
87 std::list<sc::EntityPtr>& ents,
88 double t,
89 double dt) {
90
91 // If the boundaries haven't been received yet, ignore, just return.
92 if (boundaries_.empty())
93 return true;
94
95 // The entity must be within at least one of the active boundaries,
96 // otherwise, it is removed from the simulation
97 for (sc::EntityPtr ent : ents) {
98 bool within_a_boundary = false;
99 for (std::shared_ptr<sci::BoundaryBase> boundary : boundaries_) {
100 if (boundary->contains(ent->state_truth()->pos())) {
101 within_a_boundary = true;
102 }
103 }
104 if (!within_a_boundary) {
105 ent->collision();
106 }
107 }
108 return true;
109}
