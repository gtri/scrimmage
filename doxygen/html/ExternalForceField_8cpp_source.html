65bool ExternalForceField::init(

68
69 pub_ = advertise("GlobalNetwork", "Force_Field");
70 // Parse the force type
71 std::string force_type_str = sc::get("force_type", plugin_params, "constant");
72 if (force_type_str == "constant") {
73 force_type_ = Constant;
74
75 // Parse the constant force
76 std::vector<double> force_vec;
77
78 if (str2container(
79 sc::get("constant_force", plugin_params, "0, 0, 0"),
80 ", ",
81 force_vec,
82 3)) {
83 if (force_vec[0] == 99) {
84 mag_ = sc::get("constant_mag", plugin_params, mag_);
85 ang_ = sc::get("constant_dir", plugin_params, ang_);
86 ang_ *= M_PI / 180;
87 // calc x y force, set force_v, publish global mag, angle
88 force_vec[0] = mag_ * cos(ang_);
89 force_vec[1] = mag_ * sin(ang_);
90 force_vec[2] = 0;
91 }
92 // std::cout << "Force is [0]: " << force_vec[0] << ", [1]: " << force_vec[1] <<
93 // std::endl;
94 force_ << force_vec[0], force_vec[1], force_vec[2];
95 }
96 } else if (force_type_str == "variable") {
97 force_type_ = Variable;
98
99 // Parse the force change period
100 std::vector<double> force_change_period_vec;
101 if (str2container(
102 sc::get("force_change_period", plugin_params, "0.0, 1.0"),
103 ", ",
104 force_change_period_vec,
105 2)) {
106 force_change_period_noise_ = parent_->random()->make_rng_normal(
107 force_change_period_vec[0],
108 force_change_period_vec[1]);
109 } else {
110 cout << "Failed to parse force_change_period" << endl;
111 return false;
112 }
113
114 // Parse the variable force magnitudes
115 for (int i = 0; i < 3; i++) {
116 std::vector<double> var_force_vec;
117 if (str2container(
118 sc::get("variable_force_" + std::to_string(i), plugin_params, "0.0, 1.0"),
119 ", ",
120 var_force_vec,
121 2)) {
122 auto force_noise =
123 parent_->random()->make_rng_normal(var_force_vec[0], var_force_vec[1]);
124 force_noise_.push_back(force_noise);
125 } else {
126 cout << "Failed to parse variable_force_" << i << endl;
127 return false;
128 }
129 }
130
131 // Sample the first variable force
132 sample_force();
133
134 } else {
135 cout << "Invalid force type: " << force_type_str << endl;
136 return false;
137 }
138
139 // moment periods
140 roll_period_ = sc::get("moment_roll_period", plugin_params, roll_period_);
141 pitch_period_ = sc::get("moment_pitch_period", plugin_params, pitch_period_);
142 yaw_period_ = sc::get("moment_yaw_period", plugin_params, yaw_period_);
143
144 // moment amplitudes

146 pitch_amp_ = sc::get("moment_pitch_amplitude", plugin_params, pitch_amp_);

152bool ExternalForceField::step_entity_interaction(
153 std::list<sc::EntityPtr>& ents,
154 double t,
155 double dt) {
156 if (force_type_ == Variable) {
157 if (time_->t() >= next_sample_time_) {
158 sample_force();
159 }
160 }
161
162 auto msg_force = std::make_shared<sc::Message<Eigen::Vector3d>>();
163 msg_force->data = force_;
164 pub_->publish(msg_force);
165
166 moment_(0) = roll_amp_ * sin(time_->t() * 2 * M_PI * 1.0 / roll_period_);
167 moment_(1) = pitch_amp_ * sin(time_->t() * 2 * M_PI * 1.0 / pitch_period_);
168 moment_(2) = yaw_amp_ * sin(time_->t() * 2 * M_PI * 1.0 / yaw_period_);
169
170 for (auto& ent : ents) {
171 ent->motion()->set_external_force(force_);
172
173 // Only apply moment force within z-boundary. Scale magnitude.
174 if (ent->state_truth()->pos()(2) >= moment_enable_min_z_
175 && ent->state_truth()->pos()(2) <= moment_enable_max_z_) {
176 double scale = (ent->state_truth()->pos()(2) - moment_enable_min_z_)
177 / (moment_enable_max_z_ - moment_enable_min_z_);
178 ent->motion()->set_external_moment(moment_ * scale);
179 }
180 }
181 return true;
182}

184void ExternalForceField::sample_force() {
185 auto gener = parent_->random()->gener();
186
187 // Sample the force
188 for (int i = 0; i < 3; i++) {
189 force_[i] = (*force_noise_[i])(*gener);
190 }
191
192 // Determine when the next sample time will take place
193 next_sample_time_ = time_->t() + (*force_change_period_noise_)(*gener);
194}
