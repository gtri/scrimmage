539 + C_N_delta_aileron_ * delta_aileron_ + C_N_delta_rudder_ * delta_rudder_)
540 * pVtS * b_);

543 Eigen::Vector3d Moments_total = Moments_aero + Moments_thrust + Moments_torque + Moments_gyro;
544

546 Moments_total = Eigen::Vector3d::Zero();

553 dxdt[Q] = pqr_dot(1);
