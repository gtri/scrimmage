77bool DoubleIntegratorControllerVelYaw::step(double t, double dt) {
78 alt_pid_.set_setpoint(vars_.input(desired_alt_idx_));
79 double alt_u = alt_pid_.step(dt, parent()->state_belief()->pos_const()(2));
80
81 double heading = vars_.input(desired_heading_idx_);
82 speed_pid_.set_setpoint(vars_.input(desired_speed_idx_));
83 double acc_u = speed_pid_.step(dt, parent()->state_belief()->vel().norm());
84 Eigen::Vector2d dir;
85 dir << cos(heading), sin(heading);
86
87 // Rotate acceleration into forward direction of double integrator
88 Eigen::Vector2d acc_vec = dir * acc_u;
89
90 vars_.output(acc_x_idx_, acc_vec(0));
91 vars_.output(acc_y_idx_, acc_vec(1));
92 vars_.output(acc_z_idx_, alt_u);
93
94 yaw_pid_.set_setpoint(vars_.input(desired_heading_idx_));
95 double yaw_rate = yaw_pid_.step(dt, parent()->state_belief()->quat().yaw());
96 vars_.output(turn_rate_idx_, yaw_rate);
97
98 return true;
