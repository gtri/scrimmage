91void ROSClockServer::check_rosmaster() {
92 bool rosmaster_state = ros::master::check();
93 if (prev_rosmaster_state_ == false && rosmaster_state == true) {
94 ros::param::set("/use_sim_time", true);
95 // time to reinitialize
96 nh_ = std::make_shared<ros::NodeHandle>(); // this actually starts the node
97 // Create Publisher
98 pub_mutex_.lock();
99 clock_pub_.shutdown();
100 clock_pub_ = nh_->advertise<rosgraph_msgs::Clock>("/clock", 1);
101 pub_mutex_.unlock();
102 }
103 prev_rosmaster_state_ = rosmaster_state;
104}

113void ROSClockServer::publish_clock_msg(const double& t) {
114 // Add the current simulation time to the system time at the start of the
115 // simulation.
116 std::chrono::time_point<std::chrono::system_clock, std::chrono::duration<double>>
117 sim_time_point = sim_start_time_ + std::chrono::duration<double>(t);
118
119 // Convert the time_point into a duration that represents double time since
120 // the epoch.
121 std::chrono::duration<double> time_since_epoch = sim_time_point.time_since_epoch();
122 double sim_time = time_since_epoch.count();
123
124 // Extract the seconds and nanoseconds to construct the ros::Time message
125 uint32_t sec = std::floor(sim_time);
126 uint32_t nsec = (sim_time - static_cast<double>(sec)) * 1e9;
127
128 // Create the ros Clock message and publish it
129 rosgraph_msgs::Clock clock_msg;
130 clock_msg.clock = ros::Time(sec, nsec);
131
132 pub_mutex_.lock();
133 // Check the publisher init since rosmaster might not have started yet (see
134 // check_rosmaster()).
135 if (clock_pub_) {
136 clock_pub_.publish(clock_msg);
137 }
138 pub_mutex_.unlock();
139}
