96template <typename T>
97T str2container(const std::string& str, const std::string& delims) {
98 T out;
99 std::vector<std::string> tokens;
100 split(tokens, str, delims);
101
102 for (std::string& t : tokens) {
103 if (t.length() > 0) {
104 out.insert(out.end(), convert<typename T::value_type>(t));
105 }
106 }
107 return out;
108}
