100template <typename T>
101T str2container(const std::string& str, const std::string& delims) {
102 T out;
103 std::vector<std::string> tokens;
104 split(tokens, str, delims);
105
106 for (std::string& t : tokens) {
107 if (t.length() > 0) {
108 out.insert(out.end(), convert<typename T::value_type>(t));
109 }
110 }
111 return out;
112}
