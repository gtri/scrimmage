71void AvoidEntityMS::avoidance_vectors(ContactMap& contacts, std::vector<Eigen::Vector3d>& O_vecs) {
72 for (auto it = contacts.begin(); it != contacts.end(); it++) {
73 // Ignore own position / id
74 if (it->second.id().id() == parent_->id().id()) {
75 continue;
76 }
77
78 if (!avoid_non_team_ && it->second.id().team_id() != parent_->id().team_id()) {
79 continue;
80 }
81
82 Eigen::Vector3d diff = it->second.state()->pos() - state_->pos();
83
84 double O_mag = 0;
85 double dist = diff.norm();
86
87 if (dist > sphere_of_influence_) {
88 O_mag = 0;
89 } else if (minimum_range_ < dist && dist <= sphere_of_influence_) {
90 O_mag = (sphere_of_influence_ - dist) / (sphere_of_influence_ - minimum_range_);
91 } else if (dist <= minimum_range_) {
92 O_mag = 1e10;
93 }
94
95 Eigen::Vector3d O_dir = -O_mag * diff.normalized();
96 O_vecs.push_back(O_dir);
97 }
98}

100bool AvoidEntityMS::step_autonomy(double t, double dt) {
101 // Compute repulsion vector from each robot contact
102 std::vector<Eigen::Vector3d> O_vecs;
103
104 if (use_truth_contacts_) {
105 avoidance_vectors(*contacts_, O_vecs);
106 }
107 if (use_noisy_contacts_) {
108 avoidance_vectors(noisy_contacts_, O_vecs);
109 }
110
111 // Normalize each repulsion vector and sum
112 desired_vector_ << 0, 0, 0;
113 for (auto it = O_vecs.begin(); it != O_vecs.end(); it++) {
114 if (it->hasNaN()) {
115 continue; // ignore misbehaved vectors
116 }
117 desired_vector_ += *it;
118 }
119
120 desired_vector_ *= max_vector_length_;
121
122 if (show_shapes_) {
123 // Draw the sphere of influence
124 if (circle_shape_ == nullptr) {
125 circle_shape_ = sc::shape::make_circle(
126 state_->pos(),
127 state_->quat(),
128 sphere_of_influence_,
129 Eigen::Vector3d(0, 255, 0),
130 0.2);
131 }
132 sc::set(circle_shape_->mutable_circle()->mutable_center(), state_->pos());
133 draw_shape(circle_shape_);
134
135 if (line_shape_ == nullptr) {
136 line_shape_ = sc::shape::make_line(
137 state_->pos(),
138 desired_vector_ + state_->pos(),
139 Eigen::Vector3d(0, 0, 0),
140 0.75);
141 }
142 sc::set(line_shape_->mutable_line()->mutable_start(), state_->pos());
143 sc::set(line_shape_->mutable_line()->mutable_end(), desired_vector_ + state_->pos());
144 draw_shape(line_shape_);
145 }
146
147 return true;
148}
