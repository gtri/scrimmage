167bool LibXML2ParserDocument::parse_document(const std::string& filename) {
168 fs::path path{filename};
169 if (fs::is_symlink(path)) {
170 path = fs::read_symlink(path);
171 }
172 assert(fs::exists(path) && fs::is_regular_file(path));
173 std::ifstream file(path.string());
174 std::stringstream buffer;
175 buffer << file.rdbuf();
176 file.close();
177 std::string filecontent_str = buffer.str();
178 filecontents_.assign(filecontent_str.cbegin(), filecontent_str.cend());
179 filename_ = path.string();
180 return parse_document(filecontents_);
181}
