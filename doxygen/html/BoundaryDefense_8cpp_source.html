78bool BoundaryDefense::step_autonomy(double t, double dt) {
79 // Get the active boundary to defend
80 auto it = boundaries_.find(boundary_id_);
81 if (it == boundaries_.end()) {
82 return true;
83 }
84
85 // Find the closest entity within the boundary that is not part of the
86 // boundary's team.
87 double min_dist = std::numeric_limits<double>::infinity();
88 sc::StatePtr closest = nullptr;
89 for (auto& kv : *contacts_) {
90 sc::Contact& cnt = kv.second;
91
92 // Ignore same team
93 if (cnt.id().team_id() == std::get<0>(it->second).id().team_id()) {
94 continue;
95 }
96
97 // Is the entity within the boundary?
98 if (std::get<1>(it->second)->contains(cnt.state()->pos())) {
99 if (closest == nullptr || (state_->pos() - closest->pos()).norm() < min_dist) {
100 closest = cnt.state();
101 min_dist = (state_->pos() - closest->pos()).norm();
102 }
103 }
104 }
105
106 Eigen::Vector3d desired_point;
107 if (closest != nullptr) {
108 // Chase the entity
109 desired_point = closest->pos() + closest->vel() * time_->dt();
110 } else {
111 // If there are no entities to chase, move to center of boundary that
112 // we are trying to defend
113 desired_point = std::get<1>(it->second)->center();
114 }
115
116 // Convert the desired point into a desired vector, relative to our own
117 // position.
118 Eigen::Vector3d desired_velocity = desired_point - state_->pos();
119
120 vars_.output(output_vel_x_idx_, desired_velocity(0));
121 vars_.output(output_vel_y_idx_, desired_velocity(1));
122 vars_.output(output_vel_z_idx_, desired_velocity(2));
123
124 return true;
125}
