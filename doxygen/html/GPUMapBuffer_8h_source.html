68 // Returns iterator to pos after insertion
69 template <typename InputIterator>
70 iterator insert(const_iterator pos, InputIterator start, InputIterator end) {
71 std::ptrdiff_t offset = pos - cbegin();
72 std::ptrdiff_t num_additional = std::distance(start, end);
73 if (sizeof(T) * (num_additional + size_) > capacity_) {
74 throw std::out_of_range{
75 "Tried to add more elements to MapBuffer than "
76 "MapBuffer has capacity for."};
77 }
78
79 std::size_t num_to_copy = cend() - pos;
80 for (std::size_t i = 0; i < num_to_copy; ++i) {
81 host_buffer_[offset + num_to_copy + i] = host_buffer_[offset + i];
82 }
83
84 for (auto it = start; it != end; ++it) {
85 std::size_t index = offset + std::distance(start, it);
86 host_buffer_[index] = static_cast<T>(*it);
87 }
88 size_ += num_additional;
89 return iterator(&host_buffer_[offset + num_additional]);
90 }

179 /*
180 * _map_buffer and _unmap_buffer are where device memory gets mapped/unmapped
181 * into host memory. N.B. The size of the buffer is set to the capacity of the
182 * device_buffer when it is mapped into memory with the flag
183 * CL_MAP_WRITE_INVALIDATE_REGION. This invalidates anything in the memory
184 * region we are writing to. As such, this acts as though we are creating a
185 * new host buffer.
186 *
187 */
188
189 cl_int _map_buffer(cl_mem_flags map_flags) {
190 cl_int err = CL_SUCCESS;
191 host_buffer_ = static_cast<T*>(queue_.enqueueMapBuffer(
192 device_buffer_,
193 BLOCKING_MAP,
194 map_flags,
195 OFFSET,
196 capacity_,
197 nullptr,
198 nullptr,
199 &err));
200 if (map_flags == CL_MAP_WRITE_INVALIDATE_REGION) {
201 size_ = 0;
202 }
203 assert(host_buffer_ != nullptr);
204 return err;
205 }
206
207 cl_int _unmap_buffer() {
208 cl_int err = CL_SUCCESS;
209 if (host_buffer_ != nullptr) {
210 err = queue_.enqueueUnmapMemObject(device_buffer_, host_buffer_);
211 host_buffer_ = nullptr;
212 }
213 return err;
214 }

216 // Returns the next largest value that is a multiple of 64 bytes

218 std::size_t next_largest_alignment_size(std::size_t bytes) {
219 if (bytes == 0) {
220 return buffer_alignment_;
221 }
222 return (1 + ((bytes - 1) / buffer_alignment_)) * buffer_alignment_;
223 }
