214bool Log::parse(std::string filename, Log::FileType type) {
215 if (!fs::exists(fs::path(filename))) {
216 std::string path = fs::path(filename).parent_path().string();
217 bool success = false;
218 if (type == FRAMES) {
219 success = parse_proto(
220 [&]() { return std::make_shared<sp::Frame>(); },
221 frames_,
222 path,
223 "frame");
224 for (auto& frame : frames_) {
225 scrimmage_frames_.push_back(proto_2_frame(*frame));
226 }
227 } else {
228 success = parse_proto(
229 [&]() { return std::make_shared<sp::Shapes>(); },
230 shapes_,
231 path,
232 "shapes");
233 }
234 return success;
235 }
236
237 int input_fd = open(filename.c_str(), O_RDONLY);
238 if (input_fd == -1) {
239 cout << "Failed to open file: " << filename.c_str() << endl;
240 return false;
241 }
242
243 ZeroCopyInputStreamPtr input =
244 std::make_shared<google::protobuf::io::FileInputStream>(input_fd);
245
246 if (!input) {
247 cout << "Failed to open FileInputStream. Filename: " << filename.c_str() << endl;
248 }
249
250 if (type == FRAMES) {
251 parse_frames(filename, input);
252 } else if (type == SHAPES) {
253 parse_shapes(filename, input);
254 } else if (type == UTMTERRAIN) {
255 parse_utm_terrain(filename, input);
256 } else if (type == CONTACTVISUAL) {
257 parse_contact_visual(filename, input);
258 } else if (type == MSG) {
259 // parse_contact_visual(filename, input);
260 cout << "No parser for messages yet. " << type << endl;
261 } else {
262 cout << "Log parse(): Invalid parse type: " << type << endl;
263 }
264 close(input_fd);
265
266 return true;
267}

269bool Log::parse_frames(std::string filename, ZeroCopyInputStreamPtr input) {
270 frames_.clear();
271 scrimmage_frames_.clear();
272 bool success = false, clean_eof = false;
273 do {
274 std::shared_ptr<scrimmage_proto::Frame> frame = std::make_shared<scrimmage_proto::Frame>();
275 success = this->readDelimitedFrom(filename, input, frame, clean_eof);
276 if (clean_eof || !success)
277 break;
278 frames_.push_back(frame);
279 scrimmage_frames_.push_back(proto_2_frame(*frame));
280 } while (success);
281
282 if (!clean_eof) {
283 cout << "Frames - WARNING: Clean end-of-file not detected." << endl;
284 }
285 return true;
286}

342bool Log::writeDelimitedTo(
343 const google::protobuf::MessageLite& message,
344 ZeroCopyOutputStreamPtr rawOutput) {
345 if (mode_ == READ || !enable_log_ || drop_bin_logging_) {
346 return true;
347 }
348
349 // We create a new coded stream for each message. Don't worry, this is fast.
350 google::protobuf::io::CodedOutputStream output(rawOutput.get());
351
352 // Write the size.
353 const size_t raw_size = message.ByteSizeLong();
354 if (raw_size > std::numeric_limits<int>::max()) {
355 std::cerr << __FILE__ << "(" << __LINE__ << "): Message size larger than max integer value"
356 << std::endl;
357 return false;
358 }
359 const int size = static_cast<int>(raw_size);
360 output.WriteVarint32(size);
361
362 // cout << "Writing message of size: " << size << endl;
363
364 uint8_t* buffer = output.GetDirectBufferForNBytesAndAdvance(size);
365 if (buffer != NULL) {
366 // Optimization: The message fits in one buffer, so use the faster
367 // direct-to-array serialization path.
368 message.SerializeWithCachedSizesToArray(buffer);
369 } else {
370 // Slightly-slower path when the message is multiple buffers.
371 message.SerializeWithCachedSizes(&output);
372 if (output.HadError()) {
373 return false;
374 }
375 }
376
377 return true;
378}

380bool Log::readDelimitedFrom(
381 const std::string& filename,
382 ZeroCopyInputStreamPtr rawInput,
383 MessageLitePtr message,
384 bool& clean_eof) {
385 // see here for the implementation: http://stackoverflow.com/a/22927149
386
387 // We create a new coded stream for each message. Don't worry, this is fast,
388 // and it makes sure the 64MB total size limit is imposed per-message rather
389 // than on the whole stream. (See the CodedInputStream interface for more
390 // info on this limit.)
391 google::protobuf::io::CodedInputStream input(rawInput.get());
392 const int start = input.CurrentPosition();
393 clean_eof = false;
394
395 // Read the size.
396 uint32_t size;
397 if (!input.ReadVarint32(&size)) {
398 clean_eof = (input.CurrentPosition() == start);
399 return false;
400 }
401 // Tell the stream not to read beyond that size.
402 google::protobuf::io::CodedInputStream::Limit limit = input.PushLimit(size);
403
404 // Parse the message.
405 if (!message->MergeFromCodedStream(&input)) {
406 cout << "Error: MergeFromCodedStream failed: " << filename << endl;
407 return false;
408 }
409
410 if (!input.ConsumedEntireMessage()) {
411 cout << "Error: ConsumedEntireMessage() failed: " << filename << endl;
412 return false;
413 }
414
415 // Release the limit.
416 input.PopLimit(limit);
417
418 return true;
419}
