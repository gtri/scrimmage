62void ROSAltimeter::init(std::map<std::string, std::string>& params) {
63 // Setup robot namespace
64 vehicle_name_ = sc::get<std::string>("vehicle_name", params, "none");
65 if (vehicle_name_ == "none") {
66 ros_namespace_ = sc::get<std::string>("ros_namespace_prefix", params, "robot");
67 ros_namespace_ += std::to_string(parent_->id().id());
68 } else {
69 ros_namespace_ = vehicle_name_;
70 }
71
72 if (!ros::isInitialized()) {
73 int argc = 0;
74 // scrimmage handles it's own SIGINT/SIGTERM shutdown in main.cpp
75 ros::init(argc, NULL, "scrimmage", ros::init_options::NoSigintHandler);
76 }
77 nh_ = std::make_shared<ros::NodeHandle>();
78
79 // Create Publisher
80 altimeter_pub_ = nh_->advertise<mavros_msgs::Altitude>(ros_namespace_ + "/altimeter", 1);
81
82 // Scrimmage is in East North Up (ENU)
83 // For you to get the ROS data in North East Down (NED): switch x and y, and negate z outside
84 // scrimmage. For more information on the MavROS message format view this page:
85 // https://mavlink.io/en/messages/common.html#ALTITUDE
86
87 // Monotonic Altitude should only be set on initialization and not changed.
88 // this altitude value = above sea level (meters)
90 // This altitude measure is initialized on system boot and monotonic (it is never reset, but
91 // represents the local altitude change). The only guarantee on this field is that it will never
92 // be reset and is consistent within a flight. The recommended value for this field is the
93 // uncorrected barometric altitude at boot time. This altitude will also drift and vary between
94 // flights.
95 sc::StatePtr& state = parent_->state_truth();
96 double lat_init, lon_init, alt_init;
97 parent_->projection()
98 ->Reverse(state->pos()(0), state->pos()(1), state->pos()(2), lat_init, lon_init, alt_init);
99 monotonic_ = static_cast<float>(alt_init);
100}

102bool ROSAltimeter::step() {
103 // Obtain current state information
104 sc::StatePtr& state = parent_->state_truth();
105
106 // Scrimmage is in East North Up (ENU)
107 // For you to get the ROS data in North East Down (NED): switch x and y, and negate z outside
108 // scrimmage. For more information on the MavROS message format view this page:
109 // https://mavlink.io/en/messages/common.html#ALTITUDE
110
111 // Fill Altitude Message
112 mavros_msgs::Altitude alt_msg;
113 // Scrimmage is in cartesian, use Geographic lib to get lat, long, alt for monotonic
114 double lat, lon, alt;
115 // cartesian(x, y, z) to (lat, long, alt) using Geographic lib
116 parent_->projection()
117 ->Reverse(state->pos()(0), state->pos()(1), state->pos()(2), lat, lon, alt);
118 alt_msg.monotonic = monotonic_;
119 // If you print here monotonic will look like it's changing, but if you ROSTopic echo it will be
120 // a static number. cout << alt_msg.monotonic << endl;
121
122 // amsl = above sea level (meters)
123 // get amsl from the same place as monotonic altitude, but this DOES change throughout the
124 // simulation
126 // This altitude measure is strictly above mean sea level and might be non-monotonic (it might
127 // reset on events like GPS lock or when a new QNH value is set). It should be the altitude to
128 // which global altitude waypoints are compared to. Note that it is *not* the GPS altitude,
129 // however, most GPS modules already output MSL by default and not the WGS84 altitude.
130 alt_msg.amsl = static_cast<float>(alt);
131
132 // local = scrimmage z position of state
134 // This is the local altitude in the local coordinate frame. It is not the altitude above home,
135 // but in reference to the coordinate origin (0, 0, 0). It is up-positive.
136 alt_msg.local = state->pos()(2);
137
138 // local - monotonic = relative
140 // This is the altitude above the home position. It resets on each change of the current home
141 // position.
142 alt_msg.relative = alt_msg.local - alt_msg.monotonic;
143
144 // alt_terrain = amsl
146 // This is the altitude above terrain. It might be fed by a terrain database or an altimeter.
147 // Values smaller than -1000 should be interpreted as unknown.
148 alt_msg.terrain = alt_msg.amsl;
149
150 // clearance = same as terrain // laser alt in scrimmage one day?
152 // This is not the altitude, but the clear space below the system according to the fused
153 // clearance estimate. It generally should max out at the maximum range of e.g. the laser
154 // altimeter. It is generally a moving target. A negative value indicates no measurement
155 // available.
156 alt_msg.bottom_clearance = alt_msg.amsl;
157
158 // Header
159 std_msgs::Header header; // empty header
160 // TODO: header.frame = ? // No system for ROS Frame ID's yet
161 header.stamp = ros::Time::now(); // time
162 alt_msg.header = header;
163
164 // Publish Altimeter information
165 altimeter_pub_.publish(alt_msg);
166
167 return true;
168}
