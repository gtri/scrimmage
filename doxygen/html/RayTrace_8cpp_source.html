64void RayTrace::init(std::map<std::string, std::string>& params) {
65 // Use the same generator as the parent so that the simulation is
66 // completely deterministic with respect to the simulation seed.
67 gener_ = parent_->random()->gener();
68
69 // Create three independent gaussian noise generators. They will use the
70 // same generator seed.
71 for (int i = 0; i < 3; i++) {
72 std::string tag_name = "pos_noise_" + std::to_string(i);
73 std::vector<double> vec;
74 bool status = sc::get_vec(tag_name, params, " ", vec, 2);
75 if (status) {
76 pos_noise_.push_back(parent_->random()->make_rng_normal(vec[0], vec[1]));
77 } else {
78 pos_noise_.push_back(parent_->random()->make_rng_normal(0, 1));
79 }
80 }
81
82 max_range_ = sc::get<double>("max_range", params, 1);
83 min_range_ = sc::get<double>("min_range", params, 1);
84 max_sample_rate_ = sc::get<double>("max_sample_rate", params, 1);
85 auto_ray_tracing_ = sc::get<bool>("auto_ray_tracing", params, true);
86
87 // Create the ray definitions
88 double angle_res_vert = sc::Angles::deg2rad(sc::get<double>("angle_res_vert", params, 1));
89 double angle_res_horiz = sc::Angles::deg2rad(sc::get<double>("angle_res_horiz", params, 1));
90 int num_rays_vert = sc::get<int>("num_rays_vert", params, 1);
91 int num_rays_horiz = sc::get<int>("num_rays_horiz", params, 1);
92 rays_.clear();
93
94 double fov_horiz = angle_res_horiz * (num_rays_horiz - 1);
95 double fov_vert = angle_res_vert * (num_rays_vert - 1);
96 double start_angle_horiz = -fov_horiz / 2.0;
97 double start_angle_vert = -fov_vert / 2.0;
98
99 double angle_vert = start_angle_vert;
100 for (int v = 0; v < num_rays_vert; v++) {
101 double angle_horiz = start_angle_horiz;
102 for (int h = 0; h < num_rays_horiz; h++) {
103 rays_.push_back(PCRay(angle_horiz, angle_vert));
104
105 angle_horiz += angle_res_horiz;
106 }
107 angle_vert += angle_res_vert;
108 }
109
110 // Create the publisher
111 pub_ = advertise("LocalNetwork", "RayTrace");
112}
