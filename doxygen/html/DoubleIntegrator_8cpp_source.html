96bool DoubleIntegrator::step(double t, double dt) {
97 acc_vec_(0) = vars_.input(acc_x_idx_);
98 acc_vec_(1) = vars_.input(acc_y_idx_);
99 acc_vec_(2) = vars_.input(acc_z_idx_);
100
101 // Maintain direction when clamping acceleration
102 if (acc_vec_.norm() > max_acc_) {
103 acc_vec_ = acc_vec_.normalized() * max_acc_;
104 }
105 turn_rate_ = clamp(vars_.input(turn_rate_idx_), -max_yaw_acc_, max_yaw_acc_);
106 ode_step(dt);
107
108 // Maintain direction when clamping velocity
109 Eigen::Vector3d vel_;
110 vel_ << x_[VX], x_[VY], x_[VZ];
111 if (vel_.norm() > max_vel_) {
112 vel_ = vel_.normalized() * max_vel_;
113 }
114 x_[VX] = vel_[0];
115 x_[VY] = vel_[1];
116 x_[VZ] = vel_[2];
117
118 state_->pos() << x_[X], x_[Y], x_[Z];
119 state_->vel() << x_[VX], x_[VY], x_[VZ];
120 if (motion_model_sets_yaw_) {
121 if (x_[VY] != 0 || x_[VX] != 0) {
122 state_->quat().set(0, 0, atan2(x_[VY], x_[VX]));
123 }
124 } else if (sim_copter_orientation_) {
125 // Get global velocity vector into local coordinate frame
126 Eigen::Vector3d vel_global(x_[VX], x_[VY], x_[VZ]);
127 Eigen::Vector3d vel_local = state_->quat().rotate_reverse(vel_global);
128
129 double pitch = vel_local(0) / max_vel_ * sim_copter_max_pitch_;
130 double roll = vel_local(1) / max_vel_ * sim_copter_max_roll_;
131 state_->quat().set(-roll, pitch, x_[YAW]);
132 } else {
133 state_->quat().set(0, 0, turn_rate_);
134 }
135 return true;
136}
