57void MotionBattery::init(std::map<std::string, std::string>& params) {
58 // Discover the output variables that point to the input variables for the
59 // motion model. Setup input variables that mirror the output variables.
60 for (auto& kv : vars_.output_variable_index()) {
61 int out_idx = vars_.declare(kv.first, VariableIO::Direction::Out);
62 int in_idx = vars_.declare(kv.first, VariableIO::Direction::In);
63 io_map_[kv.first] = std::make_unique<VarLimit>(in_idx, out_idx, 0.0, 0.0, false);
64 }
65
66 double charge = sc::get<double>("charge_initial", params, 1.0);
67 double charge_max = sc::get<double>("charge_max", params, 1.0);
68 double charge_min = sc::get<double>("charge_min", params, 0.0);
69 battery_ = Battery(charge_min, charge_max, charge);
70
71 std::vector<std::vector<std::string>> vecs;
72 if (get_vec_of_vecs(sc::get<std::string>("depletion_map", params, ""), vecs)) {
73 for (auto& vec : vecs) {
74 if (vec.size() != 5) {
75 LOG_ERROR("Invalid depletion mapping");
76 for (const std::string& s : vec) {
77 LOG_ERROR(" " << s);
78 }
79 continue;
80 }
81
82 std::string name = vec[0];
83 auto it_name = io_map_.find(name);
84 if (it_name == io_map_.end()) {
85 LOG_ERROR("MotionBattery: Invalid variable name: " << name);
86 } else {
87 it_name->second->depletion_rate = std::stod(vec[1]);
88 it_name->second->limit_when_depleted = str2bool(vec[3]);
89 it_name->second->value_when_depleted = std::stod(vec[4]);
90 }
91 }
92 } else {
93 LOG_ERROR("Failed to parse depletion_map");
94 }
95
96 // Charging subscriber
97 auto callback_charge_added = [&](scrimmage::MessagePtr<sm::Charge> msg) {
98 if (parent_->id().id() == msg->data.id()) {
99 battery_.add_charge(msg->data.charge_amount());
100 }
101 };
102 subscribe<sm::Charge>("GlobalNetwork", "ChargeAdded", callback_charge_added);
103
104 // Service call to get battery charge
105 parent_->services()["get_battery_charge"] =
106 std::bind(&MotionBattery::get_battery_charge, this, pl::_1, pl::_2);
107
108 publish_charge_ = sc::get<bool>("publish_charge", params, publish_charge_);
109 pub_charge_percentage_ = advertise("LocalNetwork", "ChargePercentage");
110}

112bool MotionBattery::step(double t, double dt) {
113 if (publish_charge_) {
114 auto msg = std::make_shared<sc::Message<double>>(battery_.charge_percentage());
115 pub_charge_percentage_->publish(msg);
116 }
117
118 // Loop over all variables in the io_map_, apply charge calculation, and
119 // limit output if required
120 for (auto& kv : io_map_) {
121 double value = vars_.input(kv.second->input_idx);
122 double depletion = kv.second->calc_depletion(value, time_->dt());
123 if (!battery_.deplete(depletion)) {
124 // If deplete() returns false, the battery is depleted.
125 parent()->collision();
126 if (kv.second->limit_when_depleted) {
127 // Limit the output value if specified
128 value = kv.second->value_when_depleted;
129 }
130 }
131 vars_.output(kv.second->output_idx, value);
132 }
133 return true;
134}
