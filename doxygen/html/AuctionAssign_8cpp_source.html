114bool AuctionAssign::step_autonomy(double t, double dt) {

117
118 auto msg = std::make_shared<sc::Message<auction::StartAuction>>();

122 auction_started_ = true;
123 auction_in_prog_ = true;
124 auction_start_time_ = t;
125 }
126
127 if (auction_in_prog_ && t > auction_start_time_ + auction_max_time_ && id_ == 1) {
128 cout << "======================================" << endl;

131 cout << "Bid: " << max_bid_ << endl;

133 auction_in_prog_ = false;
134 auto msg = std::make_shared<sc::Message<auction::BidAuction>>();
135 msg->data.set_sender_id(max_bid_champ_);
136 msg->data.set_bid(max_bid_);
137 result_auction_pub_->publish(msg);
