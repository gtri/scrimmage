66void MissionParse::set_overrides(const std::string& overrides) {
67 std::string overrides_no_space = remove_whitespace(overrides);
68
69 // Iterate over comma-separated overrides.
70 std::vector<std::string> overrides_tokens;
71 split(overrides_tokens, overrides_no_space, ", ");
72 for (auto& overrides_str : overrides_tokens) {
73 // Parse each key=value pair
74 std::vector<std::string> kv_tokens;
75 split(kv_tokens, overrides_str, ":= ");
76
77 // Only add to the map if the key:value was parsed correctly
78 if (kv_tokens.size() == 2) {
79 overrides_map_[kv_tokens[0]] = kv_tokens[1];
80 }
81 }
82}

100bool MissionParse::read_file_content(const std::string& filename) {
101 mission_filename_ = expand_user(filename);
102 // First, explicitly search for the mission file.
103 if (!fs::exists(mission_filename_)) {
104 // If the file doesn't exist, search for the mission file under the
105 // SCRIMMAGE_MISSION_PATH.
106 FileSearch file_search;
107 std::string result = "";
108 bool status =
109 file_search
110 .find_file(mission_filename_, "xml", "SCRIMMAGE_MISSION_PATH", result, false);
111 if (!status) {
112 // The mission file wasn't found. Exit.
113 cout << "SCRIMMAGE mission file not found: " << mission_filename_ << endl;
114 return false;
115 }
116 // The mission file was found, save its path.
117 mission_filename_ = result;
118 }
119
120 std::ifstream file(mission_filename_.c_str());
121 if (!file.is_open()) {
122 std::cout << "Failed to open mission file: " << mission_filename_ << endl;
123 return false;
124 }
125
126 std::stringstream buffer;
127 buffer << file.rdbuf();
128 file.close();
129 mission_file_content_ = buffer.str();
130 return true;
131}

133std::string MissionParse::replace_overrides(std::string str) {
134 // Search and replace any overrides of the form ${key=value} in the mission
135 // file
136 for (auto& kv : overrides_map_) {
137 std::regex reg("\\$\\{" + kv.first + "=(.+?)\\}");
138 str = std::regex_replace(str, reg, kv.second);
139 }
140
141 // Replace our xml variables of the form ${var=default} with the default
142 // value
143 std::string fmt{"$1"};
144 std::regex reg("\\$\\{.+?=(.+?)\\}");
145 str = std::regex_replace(str, reg, fmt);
146 return str;
147}

151 doc.set_filename(mission_filename_);
152 // doc.parse requires a null terminated string that it can modify.

154 mission_file_content_vec.assign(

657 // If there are overrides in count, catch them and substitue
658 gen_info.total_count = parse_int(script_info["count"]);
659 gen_info.gen_count = gen_info.total_count;
660 } else {
661 script_info["count"] = std::to_string(gen_info.total_count);
662 }
663
664 // Is this entity using generate_rate?
665 if (script_info.count("generate_rate") > 0 && script_info.count("generate_count") > 0) {
666 // Tokenize rate based on "/"
667 std::vector<double> values;
668 bool status = str2container(script_info["generate_rate"], "/", values, 2);
669 double rate = -1;
670 if (status) {
671 rate = values[0] / values[1];
672 } else {
673 rate = parse_double(script_info["generate_rate"]);
674 }
675
676 double start_time = 0;
677 if (script_info.count("generate_start_time") > 0) {
678 start_time = parse_double(script_info["generate_start_time"]);
679 }
680
681 if (script_info.count("generate_time_variance") > 0) {
682 gen_info.time_variance = parse_double(script_info["generate_time_variance"]);
683 }
684
685 std::string gen_count_str = replace_overrides(script_info["generate_count"]);
686 int gen_count = parse_int(gen_count_str);
687 if (rate > 0 && gen_count > 0) {
688 gen_info.start_time = start_time;
689 gen_info.gen_count = gen_count;
690 gen_info.rate = rate;
691
692 } else if (rate > 0 && gen_count <= 0) {
693 cout << "WARNING: Not using entity generator."
694 << "generate_rate defined, but generate_count is "
695 << "less than or equal to zero" << endl;
696 }
697 }
698
699 std::vector<double> start_times;
700 for (int i = 0; i < gen_info.gen_count; i++) {
701 start_times.push_back(gen_info.start_time);
702 }
703 next_gen_times_[ent_desc_id] = start_times;
704 gen_info_[ent_desc_id] = gen_info;
705
706 // If the entity block has a "tag" attribute, save the mapping from the
707 // tag to the entity block ID
708 auto tag_attr = script_node.first_attribute("tag");
709 if (tag_attr.is_valid()) {
710 entity_tag_to_id_[tag_attr.value()] = ent_desc_id;
711 }
712
713 entity_descs_[ent_desc_id++] = script_info;
714 }
715
716 parse_terrain();
717
718 // Parse the output_type
719 auto it_output_type = params_.find("output_type");
720 if (it_output_type != params_.end()) {
721 output_types_ = str2container<std::set<std::string>>(it_output_type->second, ", ");
722 }
723 // If "all" is a possible output_type, include all output_types
724 if (output_types_.find("all") != output_types_.end()) {
725 output_types_ = possible_output_types_;
726 }
727
728 // Log the bins by default. Don't log the bins if the tag is defined in
729 // the mission file and it is set to true.
730 no_bin_logging_ =
731 (params_.count("no_bin_logging") > 0 && str2bool(params_["no_bin_logging"]) == true);
732
733 return true;
734}

736bool MissionParse::create_log_dir() {
737 // Create the root_log_dir_ if it doesn't exist:
738 if (not fs::exists(fs::path(root_log_dir_))
739 && not fs::create_directories(fs::path(root_log_dir_))) {
740 cout << "Failed to create the root_log_dir: " << root_log_dir_ << endl;
741 return false;
742 }
743
744 bool log_dir_created = false;
745 std::string log_dir_original = log_dir_;
746 int attempts = 0;
747 while (!log_dir_created && attempts < 1e4) {
748 log_dir_ = log_dir_original;
749 // If the log directory already exists, append a number to it:
750 if (!use_exact_log_path_ && fs::exists(fs::path(log_dir_))) {
751 int ct = 1;
752 std::string log_dir_tmp;
753 do {
754 log_dir_tmp = log_dir_ + "_" + std::to_string(ct++);
755 } while (fs::exists(fs::path(log_dir_tmp)));
756 log_dir_ = log_dir_tmp;
757 }
758
759 if (fs::create_directories(fs::path(log_dir_))) {
760 log_dir_created = true;
761 }
762 attempts++;
763 }
764
765 if (!log_dir_created) {
766 if (!use_exact_log_path_) {
767 cout << "Unable to create log directory: " << log_dir_ << endl;
768 return false;
769 } else {
770 cout << "Unable to guarantee that the directory for this run is unique and does not "
771 "already contain scrimmage files that would be overwritten: "
772 << log_dir_ << std::endl;
773 return false;
774 }
775 }
776
777 // Copy the input scenario xml file to the output directory
778 if (fs::exists(mission_filename_)) {
779 fs::copy_file(fs::path(mission_filename_), fs::path(log_dir_ + "/mission.orig.xml"));
780 }
781
782 // Write the parsed content to the output directory
783 std::ofstream content_out(log_dir_ + "/mission.xml");
784 content_out << mission_file_content_;
785 content_out.close();
786
787 // Create the latest log directory by default. Don't create the latest
788 // directory if the tag is defined in the mission file and it is set to
789 // false. Can't create latest dir with an exact path because our assumptions
790 // of where to put it are broken.
791 bool create_latest_dir = not(
792 params_.count("create_latest_dir") > 0 && str2bool(params_["create_latest_dir"]) == false);
793
794 if (create_latest_dir && !use_exact_log_path_) {
795 boost::system::error_code ec;
796 auto print_error = [&]() {
797 cout << "Error code value: " << ec.value() << endl;
798 cout << "Error code name: " << ec.category().name() << endl;
799 cout << "Error message: " << ec.message() << endl;
800 };
801 auto fs_err = [&]() { return ec != boost::system::errc::success; };
802
803 // Create a "latest" symlink to the directory
804 // First, remove the latest symlink if it exists
805 fs::path latest_sym(root_log_dir_ + std::string("/latest"));
806 if (fs::is_symlink(latest_sym)) {
807 fs::remove(latest_sym, ec);
808 if (fs_err()) {
809 cout << "WARNING: could not remove symlink to latest directory" << endl;
810 print_error();
811 }
812 }
813
814 // Create the symlink
815 fs::create_directory_symlink(fs::path(log_dir_), latest_sym, ec);
816 if (fs_err()) {
817 cout << "WARNING: Unable to create latest log file symlink" << endl;
818 cout << "Couldn't create symlink log directory: " << latest_sym.string() << " -> "
819 << fs::path(log_dir_).string() << endl;
820 print_error();
821 }
822 }
823 return true;
824}

858bool MissionParse::parse_terrain() {
859 ConfigParse terrain_parse;
860 utm_terrain_ = std::make_shared<scrimmage_proto::UTMTerrain>();
861
862 // Check for grid settings
863 if (params_.count("grid_spacing") > 0) {
864 utm_terrain_->set_grid_spacing(std::stod(params_["grid_spacing"]));
865 } else {
866 utm_terrain_->set_grid_spacing(100);
867 }
868
869 if (params_.count("grid_size") > 0) {
870 utm_terrain_->set_grid_size(std::stod(params_["grid_size"]));
871 } else {
872 utm_terrain_->set_grid_size(10000);
873 }
874
875 // Check for origin settings
876 if (params_.count("origin_length") > 0) {
877 utm_terrain_->set_origin_length(std::stod(params_["origin_length"]));
878 } else {
879 utm_terrain_->set_origin_length(1);
880 }
881
882 if (params_.count("show_origin") > 0) {
883 utm_terrain_->set_show_origin(str2bool(params_["show_origin"]));
884 } else {
885 utm_terrain_->set_show_origin(false);
886 }
887
888 set(utm_terrain_->mutable_background(), background_color_);
889
890 utm_terrain_->set_enable_grid(true);
891 utm_terrain_->set_enable_terrain(false);
892 utm_terrain_->set_enable_extrusion(false);
893
894 if (params_.count("terrain") > 0
895 && find_terrain_files(params_["terrain"], terrain_parse, utm_terrain_)) {
896 double x_easting;
897 double y_northing;
898 int zone;
899 bool northp;
900 double gamma, k_scale;
901 GeographicLib::UTMUPS::Forward(
902 latitude_origin(),
903 longitude_origin(),
904 zone,
905 northp,
906 x_easting,
907 y_northing,
908 gamma,
909 k_scale);
910
911 // Make sure the projected zone and hemisphere match the input xml
912 if (((northp && boost::to_upper_copy(terrain_parse.params()["hemisphere"]) == "NORTH")
913 || (!northp && boost::to_upper_copy(terrain_parse.params()["hemisphere"]) == "SOUTH"))
914 && zone == get("zone", terrain_parse.params(), -2)) {
915 double origin_offset_x = 0;
916 double origin_offset_y = 0;
917 double origin_offset_z = 0;
918
919 if (terrain_parse.params().count("origin_offset_x") == 1
920 && terrain_parse.params().count("origin_offset_y") == 1
921 && terrain_parse.params().count("origin_offset_z") == 1) {
922 origin_offset_x = stod(terrain_parse.params()["origin_offset_x"]);
923 origin_offset_y = stod(terrain_parse.params()["origin_offset_y"]);
924 origin_offset_z = stod(terrain_parse.params()["origin_offset_z"]);
925 }
926
927 utm_terrain_->set_x_translate(x_easting - origin_offset_x);
928 utm_terrain_->set_y_translate(y_northing - origin_offset_y);
929 utm_terrain_->set_z_translate(altitude_origin() - origin_offset_z);
930 utm_terrain_->set_z_translate(altitude_origin());
931 utm_terrain_->set_system(terrain_parse.params()["system"]);
932 utm_terrain_->set_zone(zone);
933 utm_terrain_->set_hemisphere(terrain_parse.params()["hemisphere"]);
934 utm_terrain_->set_enable_grid(
935 get<bool>("enable_grid", terrain_parse.params(), "false"));
936 utm_terrain_->set_enable_terrain(true);
937 return true;
938
939 } else {
940 cout << "============================================" << endl;
941 cout << "Invalid XML Terrain settings: " << endl;
942 cout << "Hemisphere: " << terrain_parse.params()["hemisphere"] << endl;
943 cout << "Zone: " << get("zone", terrain_parse.params(), -2) << endl;
944 cout << "--------------------------------------------" << endl;
945 cout << "Geographic lib output: " << endl;
946 cout << "x_easting: " << x_easting << endl;
947 cout << "y_northing: " << y_northing << endl;
948 cout << "Zone: " << zone << endl;
949 cout << "Northern hemisphere?: " << northp << endl;
950 }
951 }
952 return false;
953}
