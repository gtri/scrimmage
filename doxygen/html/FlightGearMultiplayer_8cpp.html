120scrimmage::Quaternion FlightGearMultiplayer::fromLonLatRad(float lon, float lat) {
121 // TODO : This transformation was taken from the Flight Gear Multiplayer
122 // server, we should convert this to using our own quaternion class, but
123 // for now, this works.
124
125 // Eigen::AngleAxisd R_a(sc::Angles::deg2rad(lon), Eigen::Vector3d::UnitZ());
126 // Eigen::AngleAxisd R_b(sc::Angles::deg2rad(lat),
127 // Eigen::Vector3d::UnitX()); Eigen::Quaterniond rot2 = R_a * R_b;
128
129 float zd2 = static_cast<float>(0.5) * lon;
130 float yd2 =
131 static_cast<float>(-0.25) * static_cast<float>(M_PI) - static_cast<float>(0.5) * lat;
132 float Szd2 = sin(zd2);
133 float Syd2 = sin(yd2);
134 float Czd2 = cos(zd2);
135 float Cyd2 = cos(yd2);
136 scrimmage::Quaternion quat(Czd2 * Cyd2, -Szd2 * Syd2, Czd2 * Syd2, Szd2 * Cyd2);
137 return quat;
138}
