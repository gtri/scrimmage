121scrimmage::Quaternion FlightGearMultiplayer::fromLonLatRad(float lon, float lat) {
122 // TODO : This transformation was taken from the Flight Gear Multiplayer
123 // server, we should convert this to using our own quaternion class, but
124 // for now, this works.
125
126 // Eigen::AngleAxisd R_a(sc::Angles::deg2rad(lon), Eigen::Vector3d::UnitZ());
127 // Eigen::AngleAxisd R_b(sc::Angles::deg2rad(lat),
128 // Eigen::Vector3d::UnitX()); Eigen::Quaterniond rot2 = R_a * R_b;
129
130 float zd2 = static_cast<float>(0.5) * lon;
131 float yd2 =
132 static_cast<float>(-0.25) * static_cast<float>(M_PI) - static_cast<float>(0.5) * lat;
133 float Szd2 = sin(zd2);
134 float Syd2 = sin(yd2);
135 float Czd2 = cos(zd2);
136 float Cyd2 = cos(yd2);
137 scrimmage::Quaternion quat(Czd2 * Cyd2, -Szd2 * Syd2, Czd2 * Syd2, Szd2 * Cyd2);
138 return quat;
139}
