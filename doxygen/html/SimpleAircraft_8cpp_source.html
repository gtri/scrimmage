100bool SimpleAircraft::step(double time, double dt) {
101 // Need to saturate state variables before model runs
102 x_[ROLL] = clamp(x_[ROLL], -max_roll_, max_roll_);
103 x_[PITCH] = clamp(x_[PITCH], -max_pitch_, max_pitch_);
104 x_[SPEED] = clamp(x_[SPEED], min_velocity_, max_velocity_);
105
106 ode_step(dt);
107
108 state_->pos() << x_[X], x_[Y], x_[Z];
109 state_->quat().set(-x_[ROLL], x_[PITCH], x_[YAW]);
110 state_->vel() << x_[SPEED] * cos(x_[YAW]) * cos(x_[PITCH]),
111 x_[SPEED] * sin(x_[YAW]) * cos(x_[PITCH]), x_[SPEED] * sin(x_[PITCH]);
112
113 return true;
114}

116void SimpleAircraft::model(const vector_t& x, vector_t& dxdt, double t) {
124 double throttle = vars_.input(throttle_idx_);
125 double roll_rate = vars_.input(roll_rate_idx_);
126 double pitch_rate = vars_.input(pitch_rate_idx_);
127
128 // Saturate control inputs
129 throttle = clamp(throttle, -100.0, 100.0);
130 roll_rate = clamp(roll_rate, -max_roll_rate_, max_roll_rate_);
131 pitch_rate = clamp(pitch_rate, -max_pitch_rate_, max_pitch_rate_);
132
133 double xy_speed = x[SPEED] * cos(x[PITCH]);
134 dxdt[X] = xy_speed * cos(x[YAW]);
135 dxdt[Y] = xy_speed * sin(x[YAW]);
136 dxdt[Z] = -sin(x[PITCH]) * x[SPEED];
137 dxdt[ROLL] = roll_rate;
138 dxdt[PITCH] = pitch_rate;
139 // Adjust the length based on the speed
140 double currentLength = length_ + lengthSlopePerSpeed_ * (x[SPEED] - speedTarget_);
141 dxdt[YAW] = x[SPEED] / currentLength * tan(x[ROLL]);
142
143 dxdt[SPEED] = throttle / 5;
144}
