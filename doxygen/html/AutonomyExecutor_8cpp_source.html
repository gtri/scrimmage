60void AutonomyExecutor::init(std::map<std::string, std::string>& params) {
61 show_shapes_ = sc::get("show_shapes", params, false);
62
63 // Discover the output variables that point to the input variables for the
64 // motion model. Setup input variables that mirror the output variables.
65 for (auto& kv : vars_.output_variable_index()) {
66 int out_idx = vars_.declare(kv.first, VariableIO::Direction::Out);
67 int in_idx = vars_.declare(kv.first, VariableIO::Direction::In);
68 io_map_[out_idx] = in_idx;
69 }
70
71 // Subscribe to state information
72 std::string state_topic_name = sc::get<std::string>("state_topic_name", params, "State");
73 std::string network_name = sc::get<std::string>("network_name", params, "LocalNetwork");
74
75 auto state_callback = [&](scrimmage::MessagePtr<std::string> msg) {
76 // If the state didn't change, return immediately.
77 if (current_state_ == msg->data) {
78 return;
79 }
80
81 current_state_ = msg->data;
82
83 // Get the list of autonomies that run in this new state
84 auto autonomy_list = autonomies_.find(current_state_);
85 if (autonomy_list != autonomies_.end()) {
86 running_autonomies_ = autonomy_list->second;
87 } else {
88 running_autonomies_ = default_autonomies_;
89 }
90
91 // Call each autonomy's init method to let it know that it is now
92 // running.
93 for (sc::AutonomyPtr autonomy : running_autonomies_) {
94 if (!call_init(autonomy->name(), autonomy)) {
95 cout << "AutonomyExecutor: Failed to reload init for " << autonomy->name() << endl;
96 }
97 }
98 };
99 subscribe<std::string>(network_name, state_topic_name, state_callback);
100
101 // Parse the autonomy plugins
102 std::string autonomies_str = sc::get<std::string>("autonomies", params, "");
103 std::vector<std::vector<std::string>> vecs_of_vecs;
104 sc::get_vec_of_vecs(autonomies_str, vecs_of_vecs, " ");
105 for (std::vector<std::string> vecs : vecs_of_vecs) {
106 if (vecs.size() < 1) {
107 std::cout << "Autonomy name missing." << std::endl;
108 continue;
109 }
110
111 std::string autonomy_name = "";
112 std::map<std::string, std::string> autonomy_params;
113 int i = 0;
114 for (std::string str : vecs) {
115 if (i == 0) {
116 autonomy_name = str;
117 } else {
118 // Parse the autonomy parameters (e.g., param_name='value')
119 // Split the param_name and value, with equals sign in between
120 std::vector<std::string> tokens;
121 boost::split(tokens, str, boost::is_any_of("="));
122 if (tokens.size() == 2) {
123 // Remove the quotes from the value
124 tokens[1].erase(
125 std::remove_if(
126 tokens[1].begin(),
127 tokens[1].end(),
128 [](unsigned char x) { return (x == '\"') || (x == '\''); }),
129 tokens[1].end());
130 autonomy_params[tokens[0]] = tokens[1];
131 }
132 }
133 ++i;
134 }
135
136 sc::ConfigParse config_parse;
137 PluginStatus<Autonomy> status = parent_->plugin_manager()->make_plugin<Autonomy>(
138 "scrimmage::Autonomy",
139 autonomy_name,
140 *(parent_->file_search()),
141 config_parse,
142 autonomy_params,
143 std::set<std::string>{});
144 if (status.status == PluginStatus<Autonomy>::cast_failed) {
145 cout << "AutonomyExecutor Failed to open autonomy plugin: " << autonomy_name << endl;
146 } else if (status.status == PluginStatus<Autonomy>::loaded) {
147 // connect the initialized autonomy plugin's variable output with
148 // the variable input to the controller. This is similar to
149 // connect(autonomy->vars(), controller->vars());
150 // but without the reference to the controller->vars()
151 AutonomyPtr autonomy = status.plugin;
152 autonomy->vars().output_variable_index() = vars_.output_variable_index();
153 autonomy->vars().set_output(vars_.output());
154
155 // Initialize the autonomy
156 autonomy->set_rtree(rtree_);
157 autonomy->set_parent(parent_);
158 autonomy->set_projection(proj_);
159 autonomy->set_pubsub(parent_->pubsub());
160 autonomy->set_time(time_);
161 autonomy->set_state(state_);
162 autonomy->set_contacts(contacts_);
163 autonomy->set_is_controlling(true);
164 autonomy->set_name(autonomy_name);
165 autonomy->init(config_parse.params());
166
167 autonomies_config_[autonomy_name] = config_parse;
168
169 // Determine which states in which this autonomy is active
170 std::vector<std::string> states;
171 if (sc::get_vec("states", config_parse.params(), " ,", states)) {
172 // This is a autonomy that only runs in these states
173 for (std::string state : states) {
174 autonomies_[state].push_back(autonomy);
175 }
176 } else {
177 default_autonomies_.push_back(autonomy);
178 }
179 }
180 }
181
182 // Add the default autonomies to each declared state
183 for (scrimmage::AutonomyPtr autonomy : default_autonomies_) {
184 for (auto& kv : autonomies_) {
185 kv.second.push_back(autonomy);
186 }
187 }
188
189 running_autonomies_ = default_autonomies_;
190}

192bool AutonomyExecutor::step_autonomy(double t, double dt) {
193 for (sc::AutonomyPtr autonomy : running_autonomies_) {
194 // cout << "Running... " << autonomy->name() << endl;
195 autonomy->shapes().clear();
196
197 for (SubscriberBasePtr& sub : autonomy->subs()) {
198 for (auto msg : sub->pop_msgs<sc::MessageBase>()) {
199 sub->accept(msg);
200 }
201 }
202
203 if (!autonomy->step_autonomy(time_->t(), time_->dt())) {
204 cout << "AutonomyExecutor: autonomy error- " << autonomy->name();
205 }
206
207 if (show_shapes_) {
208 std::for_each(autonomy->shapes().begin(), autonomy->shapes().end(), [&](auto& s) {
209 this->draw_shape(s);
210 });
211 }
212 }
213 return true;
214}
