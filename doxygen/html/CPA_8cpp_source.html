72bool CPA::step_metrics(double t, double dt) {
73 // First verify that within time constraints, if they are set
74 bool outside_time =
75 (((min_time_s_ >= 0.0) && (t < min_time_s_)) || ((max_time_s_ > 0.0) && (t > max_time_s_)));
76 if (!outside_time) {
77 if (!initialized_) {
78 for (auto& kv : *id_to_ent_map_) {
79 cpa_map_[kv.first] = CPAData();
80 }
81 std::string log_dir = ((*id_to_ent_map_)[1])->mp()->log_dir();
82 csv_.open_output(log_dir + "/" + "cpa.csv");
83 csv_.set_column_headers(
84 scrimmage::CSV::Headers{"entity", "cpa", "closest_entity", "time"});
85 initialized_ = true;
86 }
87
88 for (auto& kv : *id_to_ent_map_) {
89 // Checks if a specific team ID is not being checked or if the entity has the specific
90 // team ID
91 if ((team_id_ <= 0) || (kv.second->id().team_id() == team_id_)) {
92 for (auto& kv2 : *id_to_ent_map_) {
93 if (kv != kv2) {
94 // Checks if teams are not being checked or if the teams are the same
95 if (!team_cpa_
96 || (kv.second->id().team_id() == kv2.second->id().team_id())) {
97 if (kv.second->state_truth() && kv2.second->state_truth()) {
98 double cur_distance = (kv.second->state_truth()->pos()
99 - kv2.second->state_truth()->pos())
100 .norm();
101 if (cur_distance < cpa_map_[kv.first].distance()) {
102 cpa_map_[kv.first].set_distance(cur_distance);
103 cpa_map_[kv.first].set_closest_entity(kv2.first);
104 cpa_map_[kv.first].set_time(t);
105 }
106 }
107 }
108 }
109 }
110 }
111 }
112 }
113 return true;
114}
