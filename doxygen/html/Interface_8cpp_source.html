61bool Interface::init_network(Interface::Mode_t mode, const std::string& ip, int port) {
62 mode_ = mode;
63 ip_ = ip;
64 port_ = port;
65
66 if (mode_ == server) {
67#if ENABLE_GRPC == 1
68 std::string result = ip_ + ":" + std::to_string(port_);
69 ScrimmageServiceImpl frame_service(this);
70 grpc::ServerBuilder builder;
71 builder.AddListeningPort(result, grpc::InsecureServerCredentials());
72 builder.RegisterService(&frame_service);
73 server_ = builder.BuildAndStart();
74 std::stringstream ss;
75 ss << "Server listening on " << result;
76 LOG_INFO(ss.str());
77
78 // start thread with shutdown option
79 // https://stackoverflow.com/a/36182429
80 auto serving_thread = std::thread(&Interface::start_server, this);
81 auto future = frame_service.exit_requested.get_future();
82 future.wait();
83 server_->Shutdown();
84 serving_thread.join();
85#else
86 LOG_WARN("WARNING: GRPC DISABLED!");
87#endif
88 } else if (mode_ == client) {
89#if ENABLE_GRPC
90 std::string result = ip_ + ":" + std::to_string(port_);
91 std::shared_ptr<Channel> channel(
92 grpc::CreateChannel(result, grpc::InsecureChannelCredentials()));
93 std::unique_ptr<scrimmage_proto::ScrimmageService::Stub> frame_temp(
94 scrimmage_proto::ScrimmageService::NewStub(channel));
95 scrimmage_stub_ = std::move(frame_temp);
96 LOG_INFO("Client connecting to " << result);
97#else
98 LOG_WARN("WARNING: GRPC DISABLED!");
99#endif
100 }
101 return true;
102}

369void Interface::send_cached() {
370 // Disable caching during retransmission, so we don't double save
371 // cached data
372 caching_enabled_ = false;
373
374 // Resend terrain data...
375 send_utm_terrain(utm_terrain_cache_);
376
377 // Resend contact visual data...
378 for (auto cv : contact_visual_cache_) {
379 send_contact_visual(cv);
380 }
381
382 // Back to normal caching mode
383 caching_enabled_ = true;
384}
