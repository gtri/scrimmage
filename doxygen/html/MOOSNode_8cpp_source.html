69bool MOOSNode::OnNewMail(MOOSMSG_LIST& Mail) {
70 MOOSMSG_LIST::iterator q;
71 for (q = Mail.begin(); q != Mail.end(); ++q) {
72 CMOOSMsg& msg = *q;
73 std::string key = msg.GetKey();
74 double dval = msg.GetDouble();
75
76 if (key == "DESIRED_HEADING") {
77 desired_mutex_.lock();
78 desired_.quat().set(0, 0, sc::Angles::deg2rad(dval));
79 desired_mutex_.unlock();
80 } else if (key == "DESIRED_SPEED") {
81 desired_mutex_.lock();
82 desired_.vel() = Eigen::Vector3d::UnitX() * dval;
83 desired_mutex_.unlock();
84 } else if (key == "DESIRED_DEPTH") {
85 desired_mutex_.lock();
86 desired_.pos() = -Eigen::Vector3d::UnitZ() * dval;
87 desired_mutex_.unlock();
88 } else if (key == "IVPHELM_STATE") {
89 if (!deployed_) {
90 Notify("DEPLOY", "true");
91 Notify("RETURN", "false");
92 Notify("MOOS_MANUAL_OVERRIDE", "false");
93 deployed_mutex_.lock();
94 deployed_ = true;
95 deployed_mutex_.unlock();
96 }
97 }
98 }
99 return (true);
100}

195 moos_var = "SENSOR_CONTACT";
196 Notify(moos_var, record.getSpec());
197 break;

201 }
