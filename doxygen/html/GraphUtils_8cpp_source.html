53std::tuple<std::shared_ptr<scrimmage_proto::Shape>, NodePairShapeMap, NodeShapeMap> draw_graph(
54 scrimmage_msgs::Graph& graph,
55 const std::unordered_map<uint64_t, scrimmage_proto::Vector3d>& node_idx_to_pos,
56 DrawNodeLabels draw_node_labels,
57 EntityPluginPtr plugin) {
58
59 const std::vector<int> black{0, 0, 0};
60 const std::vector<int> white{255, 255, 255};
61 const std::vector<int> blue{0, 0, 255};
62
63 NodePairShapeMap edge_shapes;
64
65 for (auto& e : graph.edges()) {
66 auto edge_shape = std::make_shared<sp::Shape>();
67
68 set(edge_shape->mutable_color(), black);
69 edge_shape->set_opacity(1.0);
70 edge_shape->set_persistent(true);
71
72 auto& p1 = node_idx_to_pos.at(e.start_node_id());
73 auto& p2 = node_idx_to_pos.at(e.end_node_id());
74
75 set(edge_shape->mutable_line()->mutable_start(), p1.x(), p1.y(), p1.z());
76 set(edge_shape->mutable_line()->mutable_end(), p2.x(), p2.y(), p2.z());
77 plugin->draw_shape(edge_shape);
78
79 edge_shapes[{e.start_node_id(), e.end_node_id()}] = edge_shape;
80 }
81
82 auto node_shape = std::make_shared<sp::Shape>();
83 node_shape->set_persistent(true);
84 auto to_pt = [&](auto& n) { return n.point(); };
85 for (const auto& pt : graph.nodes() | ba::transformed(to_pt)) {
86 set(node_shape->mutable_pointcloud()->add_point(), pt.x(), pt.y(), pt.z());
87 set(node_shape->mutable_pointcloud()->add_color(), blue);
88 }
89 node_shape->mutable_pointcloud()->set_size(6);
90 plugin->draw_shape(node_shape);
91
92 NodeShapeMap text_shapes;
93
94 for (const auto& node : graph.nodes()) {
95 auto text_shape = std::make_shared<sp::Shape>();
96 text_shape->set_persistent(true);
97 text_shape->set_opacity(1.0);
98 set(text_shape->mutable_color(), white);
99
100 if (draw_node_labels == DrawNodeLabels::YES) {
101 Eigen::Vector3d pos(node.point().x(), node.point().y(), node.point().z());
102 pos(0) += 5;
103 set(text_shape->mutable_text()->mutable_center(), pos);
104 text_shape->mutable_text()->set_scale(200);
105
106 text_shape->mutable_text()->set_text(std::to_string(node.id()));
107 plugin->draw_shape(text_shape);
108 text_shapes[node.id()] = text_shape;
109 }
110 }
111 return std::make_tuple(node_shape, edge_shapes, text_shapes);
112}
