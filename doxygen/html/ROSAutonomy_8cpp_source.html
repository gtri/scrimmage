105bool ROSAutonomy::step_autonomy(double t, double dt) {
106 // Update ROS time
107 ros::Time ros_time = ros::Time::now();
108
109 ros::spinOnce(); // check for new ROS messages
110
111 // Convert scrimmage point cloud into ROS laser scan message
112 // double fov_half = pcl_.num_rays_horiz * pcl_.angle_res_horiz / 2.0;
113 sensor_msgs::LaserScan laser_msg;
114 // laser_msg.angle_min = -fov_half;
115 // laser_msg.angle_max = fov_half;
116 // laser_msg.angle_increment = pcl_.angle_res_horiz;
117 laser_msg.time_increment = 0;
118 laser_msg.scan_time = 0;
119 laser_msg.range_min = pcl_.min_range;
120 laser_msg.range_max = pcl_.max_range;
121 laser_msg.ranges.resize(pcl_.points.size());
122 laser_msg.intensities.resize(pcl_.points.size());
123
124 int i = 0;
125 for (sc::sensor::RayTrace::PCPoint& p : pcl_.points) {
126 laser_msg.ranges[i] = p.point.norm();
127 laser_msg.intensities[i] = p.intensity;
128 i++;
129 }
130 laser_msg.header.stamp = ros_time;
131 laser_msg.header.frame_id = ros_namespace_ + "/base_laser";
132 base_scan_pub_.publish(laser_msg);
133
135 // Publish odometry transform (odom -> base_link)
136 odom_trans_.header.stamp = ros_time;
137 odom_trans_.transform.translation.x = state_->pos()(0);
138 odom_trans_.transform.translation.y = state_->pos()(1);
139 odom_trans_.transform.translation.z = state_->pos()(2);
140
141 geometry_msgs::Quaternion odom_quat = tf::createQuaternionMsgFromYaw(state_->quat().yaw());
142 odom_trans_.transform.rotation = odom_quat;
143 odom_broadcaster_->sendTransform(odom_trans_);
144
145 // Publish odometry message
146 nav_msgs::Odometry odom;
147 odom.header.stamp = ros_time;
148 odom.header.frame_id = ros_namespace_ + "/odom";
149
150 // set the position
151 odom.pose.pose.position.x = state_->pos()(0);
152 odom.pose.pose.position.y = state_->pos()(1);
153 odom.pose.pose.position.z = state_->pos()(2);
154 odom.pose.pose.orientation = odom_quat;
155
156 // set the velocity
157 odom.child_frame_id = ros_namespace_ + "/base_link";
158 odom.twist.twist.linear.x = state_->vel()(0);
159 odom.twist.twist.linear.y = state_->vel()(1);
160 odom.twist.twist.linear.z = state_->vel()(2);
161 odom.twist.twist.angular.z = 0; // TODO
162
163 odom_pub_.publish(odom);
164
166 // Publish odometry transform (base_link -> laser_scan)
167 laser_trans_.header.stamp = ros_time;
168 laser_broadcaster_->sendTransform(laser_trans_);
169
170 // Send commands to low-level controller
171 vars_.output(speed_idx_, cmd_vel_.linear.x);
172 vars_.output(turn_rate_idx_, cmd_vel_.angular.z);
173 vars_.output(pitch_rate_idx_, 0);
174 vars_.output(velocity_z_idx_, 0);
175
176 return true;
177}
