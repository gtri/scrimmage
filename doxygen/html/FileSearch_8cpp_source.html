77bool FileSearch::find_file(
78 const std::string& search,
79 std::string ext,
80 const std::string& env_var,
81 std::string& result,
82 bool verbose) {
83 // Find the xml file.
84 // Search order:
85 // 1. search could be the full path
86 // 2. search could be just the name of the plugin (e.g., QuadTest_plugin)
87 // a. Is the file located in the plugin path environment variable?
88
89 // Determine if the passed in file exists (could be full path)
90 std::list<std::string> filenames;
91 if (!ext.empty() && ext[0] != '.') {
92 ext = std::string(".") + ext;
93 }
94 std::string search_filename = search;
95
96 // If the search term already ends in the extension, don't add the extension
97 // Otherwise, add the extension
98 if (!boost::algorithm::ends_with(search, ext)) {
99 search_filename = search + ext;
100 }
101
102 auto dbg = [&](std::string msg) {
103 if (verbose)
104 std::cout << "find_file: " << msg << std::endl;
105 };
106 dbg(std::string("looking for ") + search_filename);
107
108 if (!fs::exists(search)) {
109 // files[search_filename] = list of full paths
110 dbg(std::string("not an absolute path, checking recursively in ") + env_var);
111 std::unordered_map<std::string, std::list<std::string>> files;
112 char native_path_sep = fs::path("/").make_preferred().native().at(0);
113 std::size_t last_seperator = search_filename.find_last_of(native_path_sep);
114 if (last_seperator != std::string::npos) {
115 search_filename = search_filename.substr(last_seperator + 1);
116 }
117
118 find_files(env_var, ext, files, verbose);
119 filenames = files[search_filename];
120 } else {
121 filenames.push_back(search);
122 }
123
124 if (filenames.empty()) {
125 dbg(std::string("Failed to find xml filename: ") + search);
126 return false;
127 }
128
129 // Use the last XML file that was found
130 result = filenames.back();
131
132 if (filenames.size() > 1) {
133 std::cout << "===============================================" << std::endl;
134 std::cout << "WARNING: Multiple XML files with same name found" << std::endl;
135 for (std::string& full_path : filenames) {
136 std::cout << full_path << std::endl;
137 }
138 std::cout << "Using XML file at: " << result << std::endl;
139 std::cout << "===============================================" << std::endl;
140 }
141 return true;
142}

144// Give an environment variable and a file extension:
145// Find the absolute path to all files in environment variable paths
146void FileSearch::find_files(
147 std::string env_var,
148 const std::string& ext,
149 std::unordered_map<std::string, std::list<std::string>>& out,
150 bool verbose) {
151 auto dbg = [&](std::string msg) {
152 if (verbose)
153 std::cout << "find_files: " << msg << std::endl;
154 };
155
156 auto cache_it = cache_.find(env_var);
157 auto ext_it = cache_[env_var].find(ext);
158 if (cache_it != cache_.end()) {
159 if (ext_it != cache_it->second.end()) {
160 out = ext_it->second;
161 return;
162 } else {
163 cache_it->second[ext];
164 ext_it = cache_it->second.find(ext);
165 }
166 } else {
167 cache_[env_var][ext];
168 ext_it = cache_[env_var].find(ext);
169 }
170
171 out.clear();
172
173 // Get the environment variable
174 std::string env_path;
175 if (env_var.find("/") == std::string::npos) {
176 const char* env_p = std::getenv(env_var.c_str());
177 if (env_p == NULL) {
178 std::cout << env_var << " environment variable not set" << std::endl;
179 return;
180 }
181
182 env_path = std::string(env_p);
183 dbg(env_var + " = " + env_path);
184 } else {
185 // assume is is a path rather than an environment variable
186 // since slashes cannot be in an environment variable
187 env_path = env_var;
188 }
189
190 // Tokenize the path and loop through each directory
191 boost::char_separator<char> sep(":");
192 boost::tokenizer<boost::char_separator<char>> tokens(env_path, sep);
193
194 // we are checking recursively anyway so remove
195 // more specific directories
196 std::vector<std::string> tok(tokens.begin(), tokens.end());
197 br::sort(tok);
198 auto it = tok.begin();
199
200 char native_path_sep = fs::path("/").make_preferred().native().at(0);
201
202 while (it != tok.end()) {
203 auto starts_with = [&](std::string& s) {
204 return (*it == s)
205 || (boost::starts_with(s, *it) && s.size() > it->size()
206 && s.at(it->size()) == native_path_sep);
207 };
208 tok.erase(std::remove_if(std::next(it), tok.end(), starts_with), tok.end());
209 it++;
210 }
211
212 dbg(std::string("not found in cache, looping recursively in ") + env_path);
213 for (const std::string& t : tok) {
214 // Search for all files in the current directory with
215 // the extension
216 fs::path root = t;
217
218 if (fs::exists(root) && fs::is_directory(root)) {
219 dbg(t);
220
221 fs::recursive_directory_iterator it(root);
222 fs::recursive_directory_iterator endit;
223
224 while (it != endit) {
225 fs::path path = it->path();
226 if (fs::is_regular_file(*it) && path.extension() == ext) {
227 std::string fname = path.filename().string();
228 std::string full_path = fs::absolute(path).string();
229 dbg(std::string(" ") + fname);
230 ext_it->second[fname].push_back(full_path);
231 }
232 ++it;
233 }
234 } else if (env_path != env_var) {
235 std::cout << "Search path doesn't exist: " << t << std::endl;
236 }
237 }
238
239 out = ext_it->second;
240 return;
241}
