90std::string get_sha(std::string& path) {
91 std::string cd_cmd = "cd " + path + " && ";
92 std::string sha_cmd = cd_cmd + "git rev-parse HEAD | tr -d '\n'";
93 std::string status_cmd = cd_cmd + "git status --porcelain | wc -l";
94
95 FILE* sha_file = popen(sha_cmd.c_str(), "r");
96 FILE* status_file = popen(status_cmd.c_str(), "r");
97
98 char sha[41], status[3];
99 bool success = fgets(sha, 40, sha_file) != NULL && fgets(status, 2, status_file) != NULL;
100
101 pclose(sha_file);
102 pclose(status_file);
103
104 if (!success) {
105 return "";
106 } else if (status[0] == '0') {
107 return std::string(sha);
108 } else {
109 return std::string(sha) + "-dirty";
110 }
111}

123void filter_line(
124 int downsampling_factor,
125 int num_points,
126 std::vector<Eigen::Vector3d>& path,
127 std::vector<Eigen::Vector3d>& filtered_path) {
128
129 int curvature_sz = path.size() / downsampling_factor;
130 std::list<std::pair<int, double>> curvature;
131
132 auto idx = [=](int i) { return downsampling_factor * i; };
133 for (int i = 1; i < curvature_sz - 1; i++) {
134 Eigen::Vector3d& pt_prev = path[idx(i - 1)];
135 Eigen::Vector3d& pt = path[idx(i)];
136 Eigen::Vector3d& pt_next = path[idx(i + 1)];
137
138 double curv = (pt_next - 2 * pt + pt_prev).squaredNorm();
139 curvature.push_back(std::make_pair(idx(i), curv));
140 }
141
142 using Pair = std::pair<int, double>;
143 curvature.sort([](Pair& a, Pair& b) { return a.second > b.second; });
144 curvature.erase(std::next(curvature.begin(), num_points), curvature.end());
145 curvature.sort([](Pair& a, Pair& b) { return a.first < b.first; });
146
147 filtered_path.clear();
148 filtered_path.reserve(curvature.size() + 2);
149 filtered_path.push_back(path[0]);
150
151 for (Pair& p : curvature) {
152 filtered_path.push_back(path[p.first]);
153 }
154
155 filtered_path.push_back(path.back());
156}

179std::vector<double> linspace(double low, double high, uint32_t n) {
180 std::vector<double> out;
181 if (n > 0) {
182 out.reserve(n);
183 out.push_back(low);
184 if (n > 1) {
185 const double step = (high - low) / (n - 1);
186 for (uint32_t i = 1; i < n - 1; i++) {
187 out.push_back(low + step * i);
188 }
189 out.push_back(high);
190 }
191 }
192 return out;
193}
