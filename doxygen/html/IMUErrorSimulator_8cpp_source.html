50void IMUErrorSimulator::PerformInitialRandomDraws(IMUErrorBudgetTemplate& errorBudget) {
51 // Accel Error Terms
52 // Get random draws for accel scale factor errors:
53 AccelScaleFactorErr.x() =
54 (1e-6) * errorBudget.XAccelAxisScaleFactorPPMStdDev * InitDist(InitRNG);
55 AccelScaleFactorErr.y() =
56 (1e-6) * errorBudget.YAccelAxisScaleFactorPPMStdDev * InitDist(InitRNG);
57 AccelScaleFactorErr.z() =
58 (1e-6) * errorBudget.ZAccelAxisScaleFactorPPMStdDev * InitDist(InitRNG);
59 // Put scale factor errors in the matrix Ma:
60 Ma = AccelScaleFactorErr.asDiagonal();
61
62 // Do initial random draws for accel turn-on bias:
63 AccelBiasTurnOn.x() = errorBudget.XAccelAxisTurnOnBiasMPS2StdDev * InitDist(InitRNG);
64 AccelBiasTurnOn.y() = errorBudget.YAccelAxisTurnOnBiasMPS2StdDev * InitDist(InitRNG);
65 AccelBiasTurnOn.z() = errorBudget.ZAccelAxisTurnOnBiasMPS2StdDev * InitDist(InitRNG);
66
67 // Gyro Error Terms
68 // Get random draws for gyro scale factor errors:
69 GyroScaleFactorErr.x() = (1e-6) * errorBudget.XGyroAxisScaleFactorPPMStdDev * InitDist(InitRNG);
70 GyroScaleFactorErr.y() = (1e-6) * errorBudget.YGyroAxisScaleFactorPPMStdDev * InitDist(InitRNG);
71 GyroScaleFactorErr.z() = (1e-6) * errorBudget.ZGyroAxisScaleFactorPPMStdDev * InitDist(InitRNG);
72 // Place gyro scale factor errors in matrix Mg:
73 Mg = GyroScaleFactorErr.asDiagonal();
74
75 // Get random draws for gyro turn-on biases
76 GyroBiasTurnOn.x() = errorBudget.XGyroAxisTurnOnBiasRadPerSecStdDev * InitDist(InitRNG);
77 GyroBiasTurnOn.y() = errorBudget.YGyroAxisTurnOnBiasRadPerSecStdDev * InitDist(InitRNG);
78 GyroBiasTurnOn.z() = errorBudget.ZGyroAxisTurnOnBiasRadPerSecStdDev * InitDist(InitRNG);
79}

96 errorBudget.YAccelAxisInRunBiasMPS2StdDev

98 AccelBiasInRunStdDev.z() =

153NoisyIMUData IMUErrorSimulator::EachCycle(
154 IMUErrorBudgetTemplate& errorBudget,
155 Eigen::Vector3d InputDeltaVBodyWRTInertialInBody,
156 Eigen::Vector3d InputDeltaThetaBodyWRTInertialInBody) {
157 Eigen::Vector3d quantizedAccelOutput, OutputBodyFrameAccelErrors, quantizedGyroOutput,
158 OutputBodyFrameRateGyroErrors;
159 NoisyIMUData returnValue;
160
161 if (!InRunBiasOption) {
162 TotalAccelBias = AccelBiasTurnOn + AccelBiasInRun;
163 AccelBiasInRun =
164 AccelBiasPhi.array() * AccelBiasInRun.array()
165 + AccelBiasInRunStdDev.array() * RandomVector(AccelBiasDist, AccelBiasRNG).array();
166 }
167
168 AccelVRWMSDraws = AccelVRWMSStdDev.array() * RandomVector(AccelVRWDist, AccelVRWRNG).array();
169
170 // Calculate errors to be added to the accel data
171 OutputBodyFrameAccelErrors =
172 TotalAccelBias + errorBudget.SampleFrequency * AccelVRWMSDraws
173 + errorBudget.SampleFrequency * Ma * InputDeltaVBodyWRTInertialInBody;
174
175 // Add errors to accel data
176 Eigen::Vector3d OutputNoisyDeltaVBodyWRTInertialInBody =
177 InputDeltaVBodyWRTInertialInBody + NominalDeltaT * OutputBodyFrameAccelErrors;
178
179 // Take the noisy delta-V's calculated above in double precision
180 // and quantize them based on the accel spec.
181 quantizedAccelOutput =
182 errorBudget.AccelDeltaVQuantizationMPS
183 * ((OutputNoisyDeltaVBodyWRTInertialInBody + AccelQuantizationResiduals).array()
184 / errorBudget.AccelDeltaVQuantizationMPS)
185 .round();
186 // Calculate residuals from quantization process and later add them to
187 // the next accel measurements to be quantized
188 AccelQuantizationResiduals =
189 OutputNoisyDeltaVBodyWRTInertialInBody + AccelQuantizationResiduals - quantizedAccelOutput;
190
191 // Calculate totality of errors added to accel data:
192 OutputBodyFrameAccelErrors =
193 errorBudget.SampleFrequency
194 * (quantizedAccelOutput - InputDeltaVBodyWRTInertialInBody).array();
195
196 // Gyro
197 // Calculate the total gyro bias, i.e. turn-on + in-run
198 if (InRunBiasOption == false) {
199 TotalGyroBias = GyroBiasTurnOn + GyroBiasInRun;
200 // create the next in-run gyro bias error
201 GyroBiasInRun =
202 GyroBiasPhi.array() * GyroBiasInRun.array()
203 + GyroBiasInRunStdDev.array() * RandomVector(GyroBiasDist, GyroBiasRNG).array();
204 }
205 // do random draw for gyro angel random walk:
206 GyroARWRadDraws = GyroARWRadStdDev.array() * RandomVector(GyroARWDist, GyroARWRNG).array();
207
208 // Calculate errors to be added to the gyro data
209 OutputBodyFrameRateGyroErrors =
210 TotalGyroBias.array() + errorBudget.SampleFrequency * GyroARWRadDraws.array()
211 + errorBudget.SampleFrequency * (Mg * InputDeltaThetaBodyWRTInertialInBody).array();
212
213 // Add errors to gyro data
214 Eigen::Vector3d OutputNoisyDeltaThetaBodyWRTInertialInBody =
215 InputDeltaThetaBodyWRTInertialInBody.array()
216 + NominalDeltaT * OutputBodyFrameRateGyroErrors.array();
217
218 // Take the noisy delta-theta's calculated above in double precision
219 // and quantize them based on the gyro spec.
220 quantizedGyroOutput =
221 errorBudget.GyroDeltaThetaQuantizationRadians
222 * ((OutputNoisyDeltaThetaBodyWRTInertialInBody + GyroQuantizationResiduals).array()
223 / errorBudget.GyroDeltaThetaQuantizationRadians)
224 .round();
225 // Calculate residuals from quantization process and later add them to
226 // the next gyro measurements to be quantized
227 GyroQuantizationResiduals = OutputNoisyDeltaThetaBodyWRTInertialInBody
228 + GyroQuantizationResiduals - quantizedGyroOutput;
229
230 // Calculate totality of errors added to gyro data:
231 OutputBodyFrameRateGyroErrors =
232 errorBudget.SampleFrequency
233 * (quantizedGyroOutput - InputDeltaThetaBodyWRTInertialInBody).array();
234
235 returnValue.noisyDeltaV = quantizedAccelOutput;
236 returnValue.noisyDeltaTheta = quantizedGyroOutput;
237 returnValue.deltaVErrors = OutputBodyFrameAccelErrors;
238 returnValue.deltaThetaErrors = OutputBodyFrameRateGyroErrors;
239
240 return returnValue;
241}
