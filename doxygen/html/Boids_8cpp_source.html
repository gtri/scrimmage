101bool Boids::step_autonomy(double t, double dt) {
102 // Find neighbors that are within field-of-view and within comms range
103 std::vector<ID> rtree_neighbors;
104 rtree_->neighbors_in_range(state_->pos(), rtree_neighbors, comms_range_);
105
106 // Remove neighbors that are not within field of view
107 for (auto it = rtree_neighbors.begin(); it != rtree_neighbors.end();
108 /* no inc */) {
109
110 // Ignore own position / id
111 if (it->id() == parent_->id().id()) {
112 it = rtree_neighbors.erase(it);
113 } else if (state_->InFieldOfView(*(*contacts_)[it->id()].state(), fov_az_, fov_el_)) {
114 // The neighbor is "in front"
115 ++it;
116 } else {
117 // The neighbor is "behind." Remove it.
118 it = rtree_neighbors.erase(it);
119 }
120 }
121
122 // move-to-goal behavior
123 Eigen::Vector3d v_goal = (goal_ - state_->pos()).normalized();
124
125 // Steer to avoid local neighbors
126 // Align with neighbors
127 // Cohesion: move towards average position of neighbors
128 // (i.e., Find centroid of neighbors)
129 double heading = 0;
130 Eigen::Vector3d align(0, 0, 0);
131 Eigen::Vector3d centroid(0, 0, 0);
132
133 // Compute repulsion vector from each robot (team/nonteam)
134 std::vector<Eigen::Vector3d> O_team_vecs;
135 std::vector<Eigen::Vector3d> O_nonteam_vecs;
136
137 for (ID id : rtree_neighbors) {
138 bool is_team = (id.team_id() == parent_->id().team_id());
139
140 StatePtr other_state = (*contacts_)[id.id()].state();
141
142 // Calculate vector pointing from own position to other
143 Eigen::Vector3d diff = other_state->pos() - state_->pos();
144 double dist = diff.norm();
145
146 // Calculate magnitude of repulsion vector
147 double min_range = is_team ? minimum_team_range_ : minimum_nonteam_range_;
148 double O_mag = 0;
149 if (dist > sphere_of_influence_) {
150 O_mag = 0;
151 } else if (min_range < dist && dist <= sphere_of_influence_) {
152 O_mag = (sphere_of_influence_ - dist) / (sphere_of_influence_ - min_range);
153 } else if (dist <= min_range) {
154 O_mag = 1e10;
155 }
156
157 // Calculate repulsion vector
158 Eigen::Vector3d O_dir = -O_mag * diff.normalized();
159 if (is_team) {
160 O_team_vecs.push_back(O_dir);
161 } else {
162 O_nonteam_vecs.push_back(O_dir);
163 }
164
165 // Calculate centroid of team members and heading alignment
166 if (is_team) {
167 centroid = centroid + other_state->pos();
168 align += other_state->vel().normalized();
169 heading += other_state->quat().yaw();
170 }
171 }
172
173 Eigen::Vector3d align_vec(0, 0, 0);
174 if (rtree_neighbors.size() > 0) {
175 centroid = centroid / static_cast<double>(rtree_neighbors.size());
176 align = align / static_cast<double>(rtree_neighbors.size());
177 heading /= static_cast<double>(rtree_neighbors.size());
178 align_vec << cos(heading), sin(heading), 0;
179 }
180
181 // Make sure alignment vector is well-behaved
182 align = align_vec; // TODO
183 Eigen::Vector3d v_align_normed = align.normalized();
184 double v_align_norm = align.norm();
185 if (v_align_normed.hasNaN()) {
186 v_align_normed = Eigen::Vector3d::Zero();
187 v_align_norm = 0;
188 }
189
190 // Normalize each team repulsion vector and sum
191 Eigen::Vector3d O_team_vec(0, 0, 0);
192 for (Eigen::Vector3d v : O_team_vecs) {
193 if (v.hasNaN()) {
194 continue; // ignore misbehaved vectors
195 }
196 O_team_vec += v;
197 }
198
199 // Normalize each nonteam repulsion vector and sum
200 Eigen::Vector3d O_nonteam_vec(0, 0, 0);
201 for (Eigen::Vector3d v : O_nonteam_vecs) {
202 if (v.hasNaN()) {
203 continue; // ignore misbehaved vectors
204 }
205 O_nonteam_vec += v;
206 }
207
208 // Apply gains to independent behaviors
209 Eigen::Vector3d v_goal_w_gain = v_goal * w_goal_;
210 Eigen::Vector3d O_team_vec_w_gain = O_team_vec * w_avoid_team_;
211 Eigen::Vector3d O_nonteam_vec_w_gain = O_nonteam_vec * w_avoid_nonteam_;
212 Eigen::Vector3d v_centroid_w_gain = (centroid - state_->pos()).normalized() * w_centroid_;
213 Eigen::Vector3d v_align_w_gain = v_align_normed * w_align_;
214
215 double sum_norms = v_goal_w_gain.norm() + O_team_vec_w_gain.norm() + O_nonteam_vec_w_gain.norm()
216 + v_centroid_w_gain.norm() + v_align_norm;
217
218 Eigen::Vector3d v_sum = (v_goal_w_gain + O_team_vec_w_gain + O_nonteam_vec_w_gain
219 + v_centroid_w_gain + v_align_w_gain)
220 / sum_norms;
221
222 // Scale velocity to max speed:
223 Eigen::Vector3d vel_result = v_sum * max_speed_;
224
225 if (rtree_neighbors.size() > 0) {
226
227 velocity_controller(vel_result);
228
229 if (show_shapes_) {
230 ShapePtr sphere(new scrimmage_proto::Shape);
231 sphere->set_opacity(0.1);
232 sc::set(sphere->mutable_color(), 0, 255, 0);
233 sphere->mutable_sphere()->set_radius(sphere_of_influence_);
234 sc::set(sphere->mutable_sphere()->mutable_center(), state_->pos());
235 draw_shape(sphere);
236
237 // Draw resultant vector:
238 ShapePtr line(new scrimmage_proto::Shape);
239 line->set_opacity(0.75);
240 sc::set(line->mutable_color(), 255, 255, 0);
241 sc::set(line->mutable_line()->mutable_start(), state_->pos());
242 sc::set(line->mutable_line()->mutable_end(), vel_result + state_->pos());
243 draw_shape(line);
244 }
245 } else {
246 velocity_controller(v_goal);
247 }
248 return true;
249}

251void Boids::velocity_controller(Eigen::Vector3d& v) {
252 // Convert to spherical coordinates:
253 double desired_heading = atan2(v(1), v(0));
254 double desired_pitch = atan2(v(2), v.head<2>().norm());
255
256 vars_.output(io_vel_idx_, max_speed_);
257 vars_.output(io_turn_rate_idx_, Angles::angle_pi(desired_heading - state_->quat().yaw()));
258 vars_.output(io_pitch_rate_idx_, Angles::angle_pi(desired_pitch + state_->quat().pitch()));
259
260 vars_.output(io_heading_idx_, desired_heading);
261 vars_.output(io_altitude_idx_, v(2));
262 vars_.output(io_desired_speed_idx_, max_speed_);
263
264 double norm = v.norm();
265 double ratio = (max_speed_ / 2) / std::max(norm, 1.0);
266 if (norm != 0 && ratio < 1) {
267 v *= ratio;
268 }
269
270 vars_.output(io_vel_x_idx_, v(0));
271 vars_.output(io_vel_y_idx_, v(1));
272 vars_.output(io_vel_z_idx_, v(2));
273}
