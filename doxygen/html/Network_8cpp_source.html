107bool Network::step(
108 std::map<std::string, std::list<NetworkDevicePtr>>& pubs,
109 std::map<std::string, std::list<NetworkDevicePtr>>& subs) {
110 reachable_map_.clear();
111
112 // Reset msg pub / sub counts
113 for (auto& kv : pub_counts_) {
114 kv.second = 0;
115 }
116 for (auto& kv : sub_counts_) {
117 kv.second = 0;
118 }
119
120 // Get references to the "all topics" string for faster access later
121 auto it_all_pub = pub_counts_.find("*");
122 auto it_all_sub = sub_counts_.find("*");
123
124 // number of messages delivered (may be >1 if delivered with delay)
125 int n_delivered = 0;
126
127 // For all publisher topic names
128 for (auto& pub_kv : pubs) {
129 std::string topic = pub_kv.first;
130
131 // Get reference to this specific topic in the pub_counts_ map for
132 // faster access later
133 auto it_pub_topic = pub_counts_.find(topic);
134 bool valid_pub_topic = (it_pub_topic != pub_counts_.end());
135
136 // Get reference to this specific topic in the sub_counts_ map for
137 // faster access later
138 auto it_sub_topic = sub_counts_.find(topic);
139 bool valid_sub_topic = (it_sub_topic != sub_counts_.end());
140
141 // For all publisher devices with topic name
142 for (NetworkDevicePtr& pub : pub_kv.second) {
143 pub->enforce_queue_size();
144
145 auto msgs = pub->pop_msgs<MessageBase>();
146 if (msgs.empty()) {
147 continue;
148 }
149
150 if (monitor_all_pubs_) {
151 // Accumulate published message counts on all topics
152 it_all_pub->second += msgs.size();
153 }
154
155 if (valid_pub_topic) {
156 // Accumulate published message counts on specific topic
157 it_pub_topic->second += msgs.size();
158 }
159
160 // For all subscribers on this topic
161 for (NetworkDevicePtr& sub : subs[topic]) {
162
163 if (sub->undelivered_msg_list_size() > 0) {
164 // deliver undelivered messages if delay time has passed
165 n_delivered = sub->deliver_undelivered_msg(time_->t(), is_stochastic_delay_);
166 if (monitor_all_subs_) {
167 // Accumulate received msg counts on all topics
168 it_all_sub->second += n_delivered;
169 }
170 if (valid_sub_topic) {
171 // Accumulate received msg counts on specific topic
172 it_sub_topic->second += n_delivered;
173 }
174 }
175
176 if (is_reachable(pub->plugin(), sub->plugin())) {
177 for (auto& msg : msgs) {
178 if (is_successful_transmission(pub->plugin(), sub->plugin())) {
179
180 double msg_delay = get_transmission_delay();
181
182 if (msg_delay < 0) {
183
184 msg->time = time_->t();
185 sub->add_msg(msg);
186
187 if (monitor_all_subs_) {
188 // Accumulate received message counts on all topics
189 it_all_sub->second += 1;
190 }
191
192 if (valid_sub_topic) {
193 // Accumulate received message counts on specific topic
194 it_sub_topic->second += 1;
195 }
196 } else {
197 // put msg in subs undelivered msg queue
198 msg->time = time_->t() + msg_delay;
199 sub->add_undelivered_msg(msg, is_stochastic_delay_);
200 }
201 }
202 }
203 }
204 }
205 }
206 }
207
208 // Enforce queue sizes, if necessary
209 for (auto& sub_kv : subs) {
210 // For all publisher devices with topic name
211 for (NetworkDevicePtr& sub : sub_kv.second) {
212 sub->enforce_queue_size();
213 }
214 }
215
216 if (write_csv_) {
217 CSV::Pairs pairs;
218 pairs.push_back(std::make_pair("t", time_->t()));
219
220 for (auto& kv : pub_counts_) {
221 std::string header = kv.first + "_Pub_Count";
222 pairs.push_back(std::make_pair(header, kv.second));
223 }
224
225 for (auto& kv : sub_counts_) {
226 std::string header = kv.first + "_Sub_Count";
227 pairs.push_back(std::make_pair(header, kv.second));
228 }
229 csv_.append(pairs);
230 }
231
232 return true;
233}
