96bool DubinsAirplane::step(double t, double dt) {
97 // Get inputs and saturate
98 speed_ = boost::algorithm::clamp(vars_.input(desired_speed_idx_), speed_min_, speed_max_);
99 pitch_ = boost::algorithm::clamp(vars_.input(desired_pitch_idx_), pitch_min_, pitch_max_);
100 roll_ = boost::algorithm::clamp(vars_.input(desired_roll_idx_), roll_min_, roll_max_);
101
102 x_[X] = state_->pos()(0);
103 x_[Y] = state_->pos()(1);
104 x_[Z] = state_->pos()(2);
105
106 // Directly set the vehicle's roll and pitch
107 x_[ROLL] = roll_;
108 x_[PITCH] = pitch_;
109 x_[YAW] = state_->quat().yaw();
110
111 double prev_x = x_[X];
112 double prev_y = x_[Y];
113 double prev_z = x_[Z];
114
115 ode_step(dt);
116
117 // Compute velocity (change in position)
118 state_->vel()(0) = (x_[X] - prev_x) / dt;
119 state_->vel()(1) = (x_[Y] - prev_y) / dt;
120 state_->vel()(2) = (x_[Z] - prev_z) / dt;
121
122 // Update state position
123 state_->pos()(0) = x_[X];
124 state_->pos()(1) = x_[Y];
125 state_->pos()(2) = x_[Z];
126
127 // Update state orientation
128 state_->quat().set(x_[ROLL], x_[PITCH], x_[YAW]);
129
130 if (write_csv_) {
131 // Log state to CSV
132 csv_.append(
133 sc::CSV::Pairs{
134 {"t", t},
135 {"x", state_->pos()(0)},
136 {"y", state_->pos()(1)},
137 {"z", state_->pos()(2)},
138 {"roll", state_->quat().roll()},
139 {"pitch", state_->quat().pitch()},
140 {"yaw", state_->quat().yaw()},
141 {"speed", speed_}});
142 }
143 return true;
144}

146void DubinsAirplane::model(const vector_t& x, vector_t& dxdt, double t) {
147 // The DubinsAirplane equations of motion are from the following paper:
148 // Owen, Mark, Randal W. Beard, and Timothy W. McLain. "Implementing dubins
149 // airplane paths on fixed-wing uavs." Handbook of unmanned aerial
150 // vehicles. Springer, Dordrecht, 2015. 1677-1701.
151
152 double xy_speed = speed_ * cos(x[PITCH]);
153 dxdt[X] = xy_speed * cos(x[YAW]);
154 dxdt[Y] = xy_speed * sin(x[YAW]);
155 dxdt[Z] = -speed_ * sin(x[PITCH]);
156
157 dxdt[ROLL] = 0;
158 dxdt[PITCH] = 0;
159
160 if (std::abs(speed_) < std::numeric_limits<double>::epsilon()) {
161 dxdt[YAW] = 0;
162 } else {
163 dxdt[YAW] = -g_ / speed_ * tan(roll_);
164 }
165}
