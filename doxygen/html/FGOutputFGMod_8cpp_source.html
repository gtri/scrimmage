61static void htond(double& x) {
62 if (isLittleEndian) {
63 int* Double_Overlay;
64 int Holding_Buffer;
65
66 // Double_Overlay = (int *) &x;
67 // cppcheck-suppress invalidPointerCast
68 Double_Overlay = reinterpret_cast<int*>(&x);
69 Holding_Buffer = Double_Overlay[0];
70
71 Double_Overlay[0] = htonl(Double_Overlay[1]);
72 Double_Overlay[1] = htonl(Holding_Buffer);
73 } else {
74 return;
75 }
76}

126void FGOutputFGMod::PacketInit(FGPacketVersion version) {
127 switch (version) {
128 case FGPacketVersion::v24: {
129 dataLength = sizeof(FGNetFDM1)
130 + sizeof(FGNetFDM3); // FS 2020.2 or earlier does not use FGNetFDM2
131 // Ptr Arithmatic to set net3 to the next byte after net1 ends
132 net3 = reinterpret_cast<FGNetFDM3*>(net1 + 1);
133 break;
134 }
135 case FGPacketVersion::v25: {
136 dataLength = s;
137 net2 = reinterpret_cast<FGNetFDM2*>(net1 + 1);
138 net3 = reinterpret_cast<FGNetFDM3*>(net2 + 1);
139 break;
140 }
141 }

165 net1->longitude = Propagate->GetLocation().GetLongitude(); //
166 net1->latitude = Propagate->GetLocation().GetLatitude(); // geocentric (radians)
167
168 net1->altitude = Propagate->GetAltitudeASL() * 0.3048; // altitude, above sea level (meters)
169 net1->agl = static_cast<float>(
170 Propagate->GetDistanceAGL() * 0.3048); // altitude, above ground level (meters)
171
172 net1->phi = static_cast<float>(Propagate->GetEuler(ePhi)); // roll (radians)
173 net1->theta = static_cast<float>(Propagate->GetEuler(eTht)); // pitch (radians)
174 net1->psi = static_cast<float>(Propagate->GetEuler(ePsi)); // yaw or true heading (radians)
175
176 net1->alpha = static_cast<float>(Auxiliary->Getalpha()); // angle of attack (radians)
177 net1->beta = static_cast<float>(Auxiliary->Getbeta()); // side slip angle (radians)
178
179 // Velocities
180 net1->phidot = static_cast<float>(Auxiliary->GetEulerRates(ePhi)); // roll rate (radians/sec)
181 net1->thetadot =
182 static_cast<float>(Auxiliary->GetEulerRates(eTht)); // pitch rate (radians/sec)
183 net1->psidot = static_cast<float>(Auxiliary->GetEulerRates(ePsi)); // yaw rate (radians/sec)
184 net1->vcas = static_cast<float>(Auxiliary->GetVcalibratedKTS()); // VCAS, knots
185 net1->climb_rate = static_cast<float>(Propagate->Gethdot()); // altitude rate, ft/sec
186 net1->v_north = static_cast<float>(Propagate->GetVel(eNorth)); // north vel in NED frame, fps
187 net1->v_east = static_cast<float>(Propagate->GetVel(eEast)); // east vel in NED frame, fps
188 net1->v_down = static_cast<float>(Propagate->GetVel(eDown)); // down vel in NED frame, fps
189 // ---ADD METHOD TO CALCULATE THESE TERMS---
190 net1->v_body_u = static_cast<float>(Propagate->GetUVW(1)); // ECEF speed in body axis
191 net1->v_body_v = static_cast<float>(Propagate->GetUVW(2)); // ECEF speed in body axis

199 // Stall
200 net1->stall_warning = 0.0; // 0.0 - 1.0 indicating the amount of stall
201 net1->slip_deg =
202 static_cast<float>(Auxiliary->Getbeta(inDegrees)); // slip ball deflection, deg
203
204 net1->num_engines = std::min(
205 static_cast<size_t>(FG_MAX_ENGINES),
206 Propulsion->GetNumEngines()); // Number of valid engines
207
208 for (i = 0; i < net1->num_engines; i++) {
209 auto engine = Propulsion->GetEngine(i);
210 if (engine->GetRunning())
211 net1->eng_state[i] = 2; // Engine state running
212 else if (engine->GetCranking())
213 net1->eng_state[i] = 1; // Engine state cranking
214 else
215 net1->eng_state[i] = 0; // Engine state off
216
217 switch (engine->GetType()) {
218 case (FGEngine::etRocket):
219 break;
220 case (FGEngine::etPiston): {
221 auto piston_engine = static_pointer_cast<FGPiston>(engine);
222 net1->rpm[i] = static_cast<float>(piston_engine->getRPM());
223 net1->fuel_flow[i] = static_cast<float>(piston_engine->getFuelFlow_gph());
224 net1->fuel_px[i] = 0; // Fuel pressure, psi (N/A in current model)
225 net1->egt[i] = static_cast<float>(piston_engine->GetEGT());
226 net1->cht[i] = static_cast<float>(piston_engine->getCylinderHeadTemp_degF());
227 net1->mp_osi[i] = static_cast<float>(piston_engine->getManifoldPressure_inHg());
228 net1->oil_temp[i] = static_cast<float>(piston_engine->getOilTemp_degF());
229 net1->oil_px[i] = static_cast<float>(piston_engine->getOilPressure_psi());
230 net1->tit[i] = 0; // Turbine Inlet Temperature (N/A for piston)
231 } break;
232 case (FGEngine::etTurbine):
233 break;
234 case (FGEngine::etTurboprop):
235 break;
236 case (FGEngine::etElectric):
237 net1->rpm[i] =
238 static_cast<float>(static_pointer_cast<FGElectric>(engine)->getRPM());
239 break;
240 case (FGEngine::etUnknown):
241 break;
242 }
243 }
244
245 net1->num_tanks = std::min(
