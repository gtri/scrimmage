61bool SimpleQuadrotorControllerLQR::step(double t, double dt) {
62 Eigen::Vector3d des_pos = desired_state_->pos();
63 double des_yaw = desired_state_->quat().yaw();
64
65 Eigen::Vector3d pos = parent()->state_belief()->pos();
66 Eigen::Vector3d vel = parent()->state_belief()->vel();
67 double yaw = parent()->state_belief()->quat().yaw();
68 double xy_speed = vel.head<2>().norm();
69
70 double yaw_dot = std::isnan(prev_yaw_) ? 0 : sc::Angles::angle_diff_rad(yaw, prev_yaw_) / dt;
71 prev_yaw_ = yaw;
72
73 // LQR Altitude Controller:
74 double q1 = 1;
75 double z_thrust = -1.0 / q1 * (pos(2) - des_pos(2)) - sqrt(2.0 / q1) * vel(2);
76
77 // LQR Heading Controller:
78 double q2 = 1;
79 double turn_force = -1.0 / q2 * sc::Angles::angle_pi(yaw - des_yaw) - sqrt(2.0 / q2) * yaw_dot;
80
81 // If not close to x/y position, use forward velocity:
82 double dist = (des_pos - pos).head<2>().norm();
83 vel_pid_.set_setpoint((dist < 10) ? 0 : dist / 10);
84
85 double ctrl_vel = vel_pid_.step(dt, xy_speed);
86
87 u_(0) = boost::algorithm::clamp(ctrl_vel, -max_vel_, max_vel_);
88 u_(1) = 0;
89 u_(2) = z_thrust;
90 u_(3) = turn_force;
91
92 return true;
93}
