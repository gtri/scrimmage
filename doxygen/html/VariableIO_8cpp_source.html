93int VariableIO::add_input_variable(const std::string& var) {
94 // If the variable already exists, return its existing index
95 declared_input_variables_.insert(var);
96 auto it = input_variable_index_.find(var);
97 if (it != input_variable_index_.end()) {
98 return it->second;
99 }
100
101 // If it doesn't exist, add it to the map and resize input vector
102 int idx = next_input_variable_index_;
103 input_variable_index_[var] = idx;
104 ++next_input_variable_index_;
105 *input_ = Eigen::VectorXd::Zero(input_variable_index_.size());
106 return idx;
107}

109int VariableIO::add_output_variable(const std::string& var) {
110 // If the variable already exists, return its existing index
111 declared_output_variables_.insert(var);
112 int idx;
113 auto it = output_variable_index_.find(var);
114 if (it != output_variable_index_.end()) {
115 idx = it->second;
116 } else {
117 // If it doesn't exist, it will be ignored by the next vector
118 idx = output_variable_index_.size();
119 output_variable_index_[var] = idx;
120 }
121 return idx;
122}

147void VariableIO::output(int i, double x) {
148 // The plugin writing to the output doesn't know if the index it is writing
149 // to is within the bounds of the input of the next plugin. Connect is
150 // called before output to assign the shared ptr output to point to the
151 // shared pointer input of the next plugin.
152 if (i < output_->size())
153 (*output_)(i) = x;
154}
