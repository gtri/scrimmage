252std::list<scrimmage_proto::Line> points_to_lines(const std::list<Eigen::Vector3d>& points) {
253
254 // Stop one item before the end of the points list
255 auto stop = std::prev(points.end());
256
257 std::list<scrimmage_proto::Line> lines;
258 for (auto it = points.begin(); it != stop; ++it) {
259 scrimmage_proto::Line line;
260 set(line.mutable_start(), *it);
261 set(line.mutable_end(), *std::next(it));
262 lines.push_back(line);
263 }
264 return lines;
265}

280std::shared_ptr<scrimmage_proto::Frame> create_frame(
281 double time,
282 std::shared_ptr<ContactMap>& contacts) {
283 std::shared_ptr<scrimmage_proto::Frame> frame(new scrimmage_proto::Frame());
284 frame->set_time(time);
285
286 for (auto& kv : *contacts) {
287 StatePtr& state = kv.second.state();
288 Contact::Type type = kv.second.type();
289 const ID& id = kv.second.id();
290
291 scrimmage_proto::Contact* contact = frame->add_contact();
292 scrimmage_proto::ID* sp_id = contact->mutable_id();
293
294 set(contact->mutable_state(), state);
295
296 switch (type) {
297 case Contact::Type::AIRCRAFT:
298 contact->set_type(scrimmage_proto::AIRCRAFT);
299 break;
300 case Contact::Type::QUADROTOR:
301 contact->set_type(scrimmage_proto::QUADROTOR);
302 break;
303 case Contact::Type::SPHERE:
304 contact->set_type(scrimmage_proto::SPHERE);
305 break;
306 case Contact::Type::MESH:
307 contact->set_type(scrimmage_proto::MESH);
308 break;
309 default:
310 contact->set_type(scrimmage_proto::UNKNOWN);
311 break;
312 }
313
314 contact->set_active(kv.second.active());
315
316 sp_id->set_id(id.id());
317 sp_id->set_sub_swarm_id(id.sub_swarm_id());
318 sp_id->set_team_id(id.team_id());
319 }
320 return frame;
321}

323Contact proto_2_contact(const scrimmage_proto::Contact& proto_contact) {
324 Contact contact;

327 if (proto_contact.type() == scrimmage_proto::AIRCRAFT) {

329 } else if (proto_contact.type() == scrimmage_proto::QUADROTOR) {
330 contact.set_type(Contact::Type::QUADROTOR);
331 } else if (proto_contact.type() == scrimmage_proto::SPHERE) {
332 contact.set_type(Contact::Type::SPHERE);
333 } else if (proto_contact.type() == scrimmage_proto::MESH) {
334 contact.set_type(Contact::Type::MESH);
335 } else {
336 contact.set_type(Contact::Type::UNKNOWN);
337 }
338
339 StatePtr state = proto_2_state(proto_contact.state());
340
341 contact.set_state(state);
342
343 return contact;
