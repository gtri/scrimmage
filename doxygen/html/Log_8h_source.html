189 template <class T, class T2>
190 bool parse_proto(T factory, T2& container, const std::string& path, const std::string& prefix) {
191 int ct = 0;
192 while (true) {
193 auto temp_path = path + "/" + prefix + "_" + std::to_string(ct) + ".bin";
194 ct++;
195 std::fstream input(temp_path, std::ios::in | std::ios::binary);
196
197 if (!input.good()) {
198 break;
199 }
200
201 auto proto_obj = factory();
202 if (!proto_obj->ParseFromIstream(&input)) {
203 std::cerr << "Failed to parse " + prefix + " in file " << temp_path << std::endl;
204 }
205 container.push_back(proto_obj);
206 }
207 return true;
208 }
