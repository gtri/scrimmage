202std::shared_ptr<BoundaryBase> Boundary::make_boundary(const scrimmage_proto::Shape& shape) {
203 std::shared_ptr<BoundaryBase> boundary = nullptr;
204 switch (shape.oneof_type_case()) {
205 case sp::Shape::kTriangle:
206 break;
207 case sp::Shape::kPlane:
208 boundary = std::make_shared<Plane>(shape);
209 break;
210 case sp::Shape::kArrow:
211 break;
212 case sp::Shape::kCone:
213 break;
214 case sp::Shape::kLine:
215 break;
216 case sp::Shape::kPolygon:
217 break;
218 case sp::Shape::kPolydata:
219 break;
220 case sp::Shape::kCuboid:
221 boundary = std::make_shared<Cuboid>(shape);
222 break;
223 case sp::Shape::kPointcloud:
224 break;
225 case sp::Shape::kCircle:
226 break;
227 case sp::Shape::kSphere:
228 boundary = std::make_shared<Sphere>(shape);
229 break;
230 case sp::Shape::kText:
231 break;
232 default:
233 break;
234 }
235
236 if (boundary != nullptr) {
237 sc::ShapePtr shape_ptr = std::make_shared<scrimmage_proto::Shape>(shape);
238 boundary->set_shape(shape_ptr);
239 }
240
241 return boundary;
242}
