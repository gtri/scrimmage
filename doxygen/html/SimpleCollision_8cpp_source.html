81bool SimpleCollision::step_entity_interaction(std::list<EntityPtr>& ents, double t, double dt) {
82 if (startup_collisions_only_) {
83 return true;
84 }
85
86 // Account for entities "colliding"
87 for (EntityPtr ent1 : ents) {
88 Eigen::Vector3d p1 = ent1->state_truth()->pos();
89 for (EntityPtr ent2 : ents) {
90 // ignore distance between itself
91 if (ent1->id().id() == ent2->id().id())
92 continue;
93
94 // ignore collisions that have already occurred this time-step
95 if (!ent1->is_alive() || !ent2->is_alive())
96 continue;
97
98 Eigen::Vector3d p2 = ent2->state_truth()->pos();
99
100 double dist = (p1 - p2).norm();
101 if (dist < collision_range_) {
102 if (enable_team_collisions_ && ent1->id().team_id() == ent2->id().team_id()) {
103
104 ent1->collision();
105 ent2->collision();
106
107 auto msg = std::make_shared<Message<sm::TeamCollision>>();
108 msg->data.set_entity_id_1(ent1->id().id());
109 msg->data.set_entity_id_2(ent2->id().id());
110 team_collision_pub_->publish(msg);
111
112 } else if (
113 enable_non_team_collisions_ && ent1->id().team_id() != ent2->id().team_id()) {
114 ent1->collision();
115 ent2->collision();
116
117 auto msg = std::make_shared<Message<sm::NonTeamCollision>>();
118 msg->data.set_entity_id_1(ent1->id().id());
119 msg->data.set_entity_id_2(ent2->id().id());
120 non_team_collision_pub_->publish(msg);
121 }
122 }
123 }
124 }
125 return true;
126}

128bool SimpleCollision::collision_exists(std::list<EntityPtr>& ents, Eigen::Vector3d& p) {
129 if (ents.empty()) {
130 return false;
131 } else {
132 if (init_alt_deconflict_) {
133 for (EntityPtr ent : ents) {
134 if (std::abs(p(2) - ent->state_truth()->pos()(2)) <= startup_collision_range_) {
135 return true;
136 }
137 }
138 } else {
139 if (ents.front()->autonomies().empty()) {
140 return false;
141 } else {
142 std::vector<ID> neighbors;
143 RTreePtr rtree = ents.front()->autonomies().front()->rtree();
144 rtree->neighbors_in_range(p, neighbors, startup_collision_range_);
145 return !neighbors.empty();
146 }
147 }
148 }
149 return false;
150}
