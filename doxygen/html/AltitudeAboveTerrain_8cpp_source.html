67void AltitudeAboveTerrain::init(std::map<std::string, std::string>& params) {
68 // Use the same generator as the parent so that the simulation is
69 // completely deterministic with respect to the simulation seed.
70 gener_ = parent_->random()->gener();
71
72 std::vector<double> vec;
73 if (str2container(sc::get<std::string>("altitude_noise", params, "0.0, 1.0"), ", ", vec, 2)) {
74 noise_ = parent_->random()->make_rng_normal(vec[0], vec[1]);
75 } else {
76 noise_ = parent_->random()->make_rng_normal(0.0, 1.0);
77 }
78
79 pub_true_ = advertise(
80 "GlobalNetwork",
81 "/" + std::to_string(parent_->id().id()) + "/TrueAltitudeAboveTerrain");
82 pub_noise_ = advertise("LocalNetwork", "AltitudeAboveTerrain");
83
84 auto terrain_map_cb = [&](auto& msg) {
85 map_ = std::make_shared<scrimmage::interaction::TerrainMap>(msg->data);
86 };
87 subscribe<scrimmage_msgs::Terrain>("GlobalNetwork", "Terrain", terrain_map_cb);
88}

90bool AltitudeAboveTerrain::step() {
91 // Wait until we have received a valid terrain map
92 if (map_ == nullptr)
93 return true;
94
95 boost::optional<double> height =
96 map_->height_at(parent_->state_truth()->pos()(0), parent_->state_truth()->pos()(1));
97 if (height) {
98 double alt = parent_->state_truth()->pos()(2) - *height;
99
100 // Create and send the true altitude
101 auto msg_true = std::make_shared<sc::Message<double>>();
102 msg_true->data = alt;
103 pub_true_->publish(msg_true);
104
105 // Generate a noisy version of the altitude
106 auto msg_noise = std::make_shared<sc::Message<double>>();
107 msg_noise->data = alt + (*noise_)(*gener_);
108 pub_noise_->publish(msg_noise);
109 } else {
110 std::cout << "AltitudeAboveTerrain: vehicle out of terrain map bounds" << endl;
111 }
112 return true;
113}
