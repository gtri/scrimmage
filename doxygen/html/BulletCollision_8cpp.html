400 if (it_object != objects_.end()) {
401 sc::EntityPtr& ent = kv.second;
402 it_object->second.object->getWorldTransform().setOrigin(btVector3(
403 (btScalar)ent->state_truth()->pos()(0),

407 if (show_collision_shapes_) {
408 sc::set(
409 it_object->second.shape->mutable_sphere()->mutable_center(),
410 ent->state_truth()->pos());

412 }
413 }
414 }
415
416 // If an entity no longer exists in the ent map, remove it's shape from the
417 // bullet environment
418 for (auto& kv : objects_) {
419 auto it_ent = id_to_ent_map_->find(kv.first);
420 if (it_ent == id_to_ent_map_->end()) {
421 remove_object(kv.first);
422 }
423 }

425 // For each entity's ray-based sensors, compute point clouds
426 for (auto& kv : pc_descs_) {
427 sc::EntityPtr& own_ent = (*id_to_ent_map_)[kv.first];
428 Eigen::Vector3d own_pos = own_ent->state_truth()->pos();
429
430 // For each ray sensor on a single entity
431 for (auto& kv2 : kv.second) {
432 sensor::RayTrace::PointCloud& pc = kv2.second->point_cloud;
433
434 if (t > (pc.max_sample_rate + kv2.second->last_update_time)) {
435 kv2.second->last_update_time = time_->t();

452 // Transform sensor's origin to world coordinates
453 Eigen::Vector4d sensor_pos = tf_m * Eigen::Vector4d(0, 0, 0, 1);
454 Eigen::Vector3d sensor_pos_w = sensor_pos.head<3>() + own_pos;
455
456 // Transform ray's end point to world coordinates
457 Eigen::Vector4d ray =
458 tf_m
459 * Eigen::Vector4d(original_ray(0), original_ray(1), original_ray(2), 1);

462 // Create bullet vectors
463 btVector3 btFrom(sensor_pos_w(0), sensor_pos_w(1), sensor_pos_w(2));

466 // Perform ray casting

468 bt_collision_world->rayTest(btFrom, btTo, res);

470 // Points in the RayTrace message are defined with respect to
471 // the LIDAR sensor's coordinate frame. Use original ray's
472 // direction, shorten to length of detection ray, if a
473 // collision occurred.
474 Eigen::Vector3d ray_end;
475 if (res.hasHit()) {
476 Eigen::Vector3d hit_point(
477 res.m_hitPointWorld.x(),
478 res.m_hitPointWorld.y(),
479 res.m_hitPointWorld.z());
480 Eigen::Vector3d return_vec =
481 original_ray.normalized() * (hit_point - sensor_pos_w).norm();

483 RayTrace::PCPoint(
484 return_vec,
485 255,
486 ((return_vec.norm() > msg->data.max_range)
487 || (return_vec.norm() < msg->data.min_range))));
488 ray_end = hit_point;
489 } else {
490 msg->data.points.push_back(RayTrace::PCPoint(original_ray, 255, true));

492 }

612bool BulletCollision::get_ray_tracing(
613 scrimmage::MessageBasePtr request,
614 scrimmage::MessageBasePtr& response) {
615 auto request_cast = std::dynamic_pointer_cast<sc::Message<RayTrace::PointCloudWithId>>(request);
616
617 if (request_cast == nullptr) {
618 LOG_ERROR("Could not cast to sc::Message<RayTrace::PointCloudId> request");
619 return false;
620 }
621
622 int entity_id = request_cast->data.entity_id;
623 std::string sensor_name = request_cast->data.sensor_name;
624 sc::EntityPtr& own_ent = (*id_to_ent_map_)[entity_id];
625 Eigen::Vector3d own_pos = own_ent->state_truth()->pos();
626 // Compute transformation matrix from entity's frame to sensor's
627 // frame.
628 sc::SensorPtr& sensor = own_ent->sensors()[sensor_name];
629 Eigen::Matrix4d tf_m =
630 own_ent->state_truth()->tf_matrix(false) * sensor->transform()->tf_matrix();
631 auto response_cast = std::make_shared<sc::Message<RayTrace::PointCloud>>();
632 response_cast->data.max_range = request_cast->data.max_range;
633 response_cast->data.min_range = request_cast->data.min_range;
634 response_cast->data.rays = request_cast->data.get_rays();
635
636 // Create the points from the rays
637 for (auto& ray : response_cast->data.rays) {
638 Eigen::Vector3d r(response_cast->data.max_range, 0, 0);
639 sc::Quaternion rot_vert(Eigen::Vector3d(0, 1, 0), ray.elevation_rad);
640 r = rot_vert.rotate(r);
641
642 sc::Quaternion rot_horiz(Eigen::Vector3d(0, 0, 1), ray.azimuth_rad);
643 r = rot_horiz.rotate(r);
644
645 request_cast->data.points.push_back(RayTrace::PCPoint(r));
646 }
647
648 // For each ray in the sensor
649 for (RayTrace::PCPoint& pcpoint : request_cast->data.points) {
650 Eigen::Vector3d original_ray = pcpoint.point;
651 // Transform sensor's origin to world coordinates
652 Eigen::Vector4d sensor_pos = tf_m * Eigen::Vector4d(0, 0, 0, 1);
653 Eigen::Vector3d sensor_pos_w = sensor_pos.head<3>() + own_pos;
654
655 Eigen::Vector3d ray_w;
656 if (request_cast->data.world_frame == false) {
657 // Transform ray's end point to world coordinates
658 Eigen::Vector4d ray =
659 tf_m * Eigen::Vector4d(original_ray(0), original_ray(1), original_ray(2), 1);
660 ray_w = ray.head<3>() + own_pos;
661 } else {
662 // Already in world frame
663 ray_w = original_ray;
664 }
665
666 // Create bullet vectors
667 btVector3 btFrom(sensor_pos_w(0), sensor_pos_w(1), sensor_pos_w(2));
668 btVector3 btTo(ray_w(0), ray_w(1), ray_w(2));
669
670 // Perform ray casting
671 btCollisionWorld::ClosestRayResultCallback res(btFrom, btTo);
672 bt_collision_world->rayTest(btFrom, btTo, res);
673
674 // Points in the RayTrace message are defined with respect to
675 // the LIDAR sensor's coordinate frame. Use original ray's
676 // direction, shorten to length of detection ray, if a
677 // collision occurred.
678 Eigen::Vector3d ray_end;
679 if (res.hasHit()) {
680 Eigen::Vector3d hit_point(
681 res.m_hitPointWorld.x(),
682 res.m_hitPointWorld.y(),
683 res.m_hitPointWorld.z());
684 Eigen::Vector3d return_vec =
685 original_ray.normalized() * (hit_point - sensor_pos_w).norm();
686 response_cast->data.points.push_back(
687 RayTrace::PCPoint(
688 return_vec,
689 255,
690 ((return_vec.norm() > response_cast->data.max_range)
691 || (return_vec.norm() < response_cast->data.min_range))));
692 ray_end = hit_point;
693 } else {
694 response_cast->data.points.push_back(RayTrace::PCPoint(original_ray, 255, true));
695 ray_end = ray_w;
696 }
697 }
698 // Save the response
699 response = response_cast;
700
701 // Success
702 return true;
703}
