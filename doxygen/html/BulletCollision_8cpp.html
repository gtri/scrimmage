398 auto it_object = objects_.find(kv.first);
399 if (it_object != objects_.end()) {
400 sc::EntityPtr& ent = kv.second;
401 it_object->second.object->getWorldTransform().setOrigin(btVector3(
402 (btScalar)ent->state_truth()->pos()(0),
403 (btScalar)ent->state_truth()->pos()(1),

406 if (show_collision_shapes_) {
407 sc::set(
408 it_object->second.shape->mutable_sphere()->mutable_center(),
409 ent->state_truth()->pos());

411 }
412 }
413 }
414
415 // If an entity no longer exists in the ent map, remove it's shape from the
416 // bullet environment
417 for (auto& kv : objects_) {
418 auto it_ent = id_to_ent_map_->find(kv.first);
419 if (it_ent == id_to_ent_map_->end()) {
420 remove_object(kv.first);
421 }
422 }
423
424 // For each entity's ray-based sensors, compute point clouds
425 for (auto& kv : pc_descs_) {
426 sc::EntityPtr& own_ent = (*id_to_ent_map_)[kv.first];
427 Eigen::Vector3d own_pos = own_ent->state_truth()->pos();
428
429 // For each ray sensor on a single entity
430 for (auto& kv2 : kv.second) {
431 sensor::RayTrace::PointCloud& pc = kv2.second->point_cloud;
432
433 if (t > (pc.max_sample_rate + kv2.second->last_update_time)) {
434 kv2.second->last_update_time = time_->t();

436 auto msg = std::make_shared<sc::Message<RayTrace::PointCloud>>();

442 // frame.

448 unsigned int i = 0;

450 Eigen::Vector3d original_ray = pcpoint.point;
451 // Transform sensor's origin to world coordinates
452 Eigen::Vector4d sensor_pos = tf_m * Eigen::Vector4d(0, 0, 0, 1);
453 Eigen::Vector3d sensor_pos_w = sensor_pos.head<3>() + own_pos;
454
455 // Transform ray's end point to world coordinates
456 Eigen::Vector4d ray =
457 tf_m
458 * Eigen::Vector4d(original_ray(0), original_ray(1), original_ray(2), 1);
459 Eigen::Vector3d ray_w = ray.head<3>() + own_pos;

462 btVector3 btFrom(sensor_pos_w(0), sensor_pos_w(1), sensor_pos_w(2));

465 // Perform ray casting

467 bt_collision_world->rayTest(btFrom, btTo, res);
468
469 // Points in the RayTrace message are defined with respect to
470 // the LIDAR sensor's coordinate frame. Use original ray's
471 // direction, shorten to length of detection ray, if a
472 // collision occurred.
473 Eigen::Vector3d ray_end;
474 if (res.hasHit()) {
475 Eigen::Vector3d hit_point(
476 res.m_hitPointWorld.x(),
477 res.m_hitPointWorld.y(),
478 res.m_hitPointWorld.z());
479 Eigen::Vector3d return_vec =
480 original_ray.normalized() * (hit_point - sensor_pos_w).norm();
481 msg->data.points.push_back(
482 RayTrace::PCPoint(
483 return_vec,
484 255,
485 ((return_vec.norm() > msg->data.max_range)
486 || (return_vec.norm() < msg->data.min_range))));
487 ray_end = hit_point;
488 } else {
489 msg->data.points.push_back(RayTrace::PCPoint(original_ray, 255, true));

491 }

498 } else {

611bool BulletCollision::get_ray_tracing(
612 scrimmage::MessageBasePtr request,
613 scrimmage::MessageBasePtr& response) {
614 auto request_cast = std::dynamic_pointer_cast<sc::Message<RayTrace::PointCloudWithId>>(request);
615
616 if (request_cast == nullptr) {
617 std::cout << "Could not cast to sc::Message<RayTrace::PointCloudId> request" << std::endl;
618 return false;
619 }
620
621 int entity_id = request_cast->data.entity_id;
622 std::string sensor_name = request_cast->data.sensor_name;
623 sc::EntityPtr& own_ent = (*id_to_ent_map_)[entity_id];
624 Eigen::Vector3d own_pos = own_ent->state_truth()->pos();
625 // Compute transformation matrix from entity's frame to sensor's
626 // frame.
627 sc::SensorPtr& sensor = own_ent->sensors()[sensor_name];
628 Eigen::Matrix4d tf_m =
629 own_ent->state_truth()->tf_matrix(false) * sensor->transform()->tf_matrix();
630 auto response_cast = std::make_shared<sc::Message<RayTrace::PointCloud>>();
631 response_cast->data.max_range = request_cast->data.max_range;
632 response_cast->data.min_range = request_cast->data.min_range;
633 response_cast->data.rays = request_cast->data.get_rays();
634
635 // Create the points from the rays
636 for (auto& ray : response_cast->data.rays) {
637 Eigen::Vector3d r(response_cast->data.max_range, 0, 0);
638 sc::Quaternion rot_vert(Eigen::Vector3d(0, 1, 0), ray.elevation_rad);
639 r = rot_vert.rotate(r);
640
641 sc::Quaternion rot_horiz(Eigen::Vector3d(0, 0, 1), ray.azimuth_rad);
642 r = rot_horiz.rotate(r);
643
644 request_cast->data.points.push_back(RayTrace::PCPoint(r));
645 }
646
647 // For each ray in the sensor
648 for (RayTrace::PCPoint& pcpoint : request_cast->data.points) {
649 Eigen::Vector3d original_ray = pcpoint.point;
650 // Transform sensor's origin to world coordinates
651 Eigen::Vector4d sensor_pos = tf_m * Eigen::Vector4d(0, 0, 0, 1);
652 Eigen::Vector3d sensor_pos_w = sensor_pos.head<3>() + own_pos;
653
654 Eigen::Vector3d ray_w;
655 if (request_cast->data.world_frame == false) {
656 // Transform ray's end point to world coordinates
657 Eigen::Vector4d ray =
658 tf_m * Eigen::Vector4d(original_ray(0), original_ray(1), original_ray(2), 1);
659 ray_w = ray.head<3>() + own_pos;
660 } else {
661 // Already in world frame
662 ray_w = original_ray;
663 }
664
665 // Create bullet vectors
666 btVector3 btFrom(sensor_pos_w(0), sensor_pos_w(1), sensor_pos_w(2));
667 btVector3 btTo(ray_w(0), ray_w(1), ray_w(2));
668
669 // Perform ray casting
670 btCollisionWorld::ClosestRayResultCallback res(btFrom, btTo);
671 bt_collision_world->rayTest(btFrom, btTo, res);
672
673 // Points in the RayTrace message are defined with respect to
674 // the LIDAR sensor's coordinate frame. Use original ray's
675 // direction, shorten to length of detection ray, if a
676 // collision occurred.
677 Eigen::Vector3d ray_end;
678 if (res.hasHit()) {
679 Eigen::Vector3d hit_point(
680 res.m_hitPointWorld.x(),
681 res.m_hitPointWorld.y(),
682 res.m_hitPointWorld.z());
683 Eigen::Vector3d return_vec =
684 original_ray.normalized() * (hit_point - sensor_pos_w).norm();
685 response_cast->data.points.push_back(
686 RayTrace::PCPoint(
687 return_vec,
688 255,
689 ((return_vec.norm() > response_cast->data.max_range)
690 || (return_vec.norm() < response_cast->data.min_range))));
691 ray_end = hit_point;
692 } else {
693 response_cast->data.points.push_back(RayTrace::PCPoint(original_ray, 255, true));
694 ray_end = ray_w;
695 }
696 }
697 // Save the response
698 response = response_cast;
699
700 // Success
701 return true;
702}
