81bool GraphInteraction::init(

89 // If the data tag has been set for this plugin in the mission file, get

91 std::map<std::string, std::string> data_params;
92 if (sc::parse_autonomy_data(plugin_params, data_params)) {
93 FileSearch file_search;
94 graph_file_name = sc::get<std::string>("graph_file", data_params, default_file_name);
95 if (graph_file_name != default_file_name) {
96 std::string graph_ext = fs::path(graph_file_name).extension().string();
97 file_search
98 .find_file(graph_file_name, graph_ext, "SCRIMMAGE_DATA_PATH", graph_file_name);
99 }
100 }
101
102 vis_graph_ = sc::get<bool>("visualize_graph", plugin_params, true);
103 id_ = sc::get<int>("id", plugin_params, 1);
104 if (graph_file_name == default_file_name)
105 return true;
106
107 std::ifstream graph_file(graph_file_name);
108
109 boost::dynamic_properties dp(boost::ignore_other_properties);
110 dp.property("osmid", boost::get(&VertexProperties::osmid, g_));
111 dp.property("x", boost::get(&VertexProperties::x, g_));
112 dp.property("y", boost::get(&VertexProperties::y, g_));
113
114 dp.property("geometry", boost::get(&EdgeProperties::geometry, g_));
115 dp.property("length", boost::get(&EdgeProperties::length, g_));
116 dp.property("osmid", boost::get(&EdgeProperties::osmid, g_));
117 dp.property("name", boost::get(&EdgeProperties::name, g_));
118
119 if (graph_file.is_open()) {
120 boost::read_graphml(graph_file, g_, dp);
121 LOG_INFO("read graph file");
122 graph_file.close();
123 }
124
125 std::map<int64_t, Eigen::Vector3d> nodes;
126 std::map<int64_t, int64_t> boost_vert_to_osmid;
127
128 auto graph_msg = std::make_shared<sc::Message<sm::Graph>>();
129 graph_msg->data.set_id(id_);
130 for (auto vs = boost::vertices(g_); vs.first != vs.second; ++vs.first) {
131 // double longitude = boost::get(&VertexProperties::x, g_, *vs.first);
132 // double latitude = boost::get(&VertexProperties::y, g_, *vs.first);
133 // int64_t this_osmid =
134 // boost::get(&VertexProperties::osmid, g_, *vs.first);
135 // NOTE: the above three lines are equivalent to the below three lines
136 double longitude = g_[*vs.first].x;
137 double latitude = g_[*vs.first].y;
138 double this_osmid = g_[*vs.first].osmid;
139 boost_vert_to_osmid[*vs.first] = this_osmid;
140
141 double x, y, z;
142 parent_->projection()->Forward(latitude, longitude, 0.0, x, y, z);
143 nodes[this_osmid] = Eigen::Vector3d(x, y, z);
144 auto node_ptr = graph_msg->data.add_nodes();
145 node_ptr->set_id(this_osmid);
146 sc::set(node_ptr->mutable_point(), nodes[this_osmid]);
147 }
148
149 for (auto es = boost::edges(g_); es.first != es.second; ++es.first) {
150 int64_t id_start = boost_vert_to_osmid[boost::source(*es.first, g_)];
151 int64_t id_end = boost_vert_to_osmid[boost::target(*es.first, g_)];
152
153 auto edge_ptr = graph_msg->data.add_edges();
154 edge_ptr->set_start_node_id(id_start);
155 edge_ptr->set_end_node_id(id_end);
156 edge_ptr->set_weight(g_[*es.first].length);
157 edge_ptr->set_label(g_[*es.first].name);
158 }
159 pub_graph_->publish(graph_msg);
160
161 if (vis_graph_) {
162 auto node_idx_to_pos = nodes_idxs_to_pos_map(graph_msg->data);
163 auto ptr = std::static_pointer_cast<EntityPlugin>(shared_from_this());
164 bool draw_node_labels = sc::get<bool>("draw_node_labels", plugin_params, true);
165 auto draw = draw_node_labels ? DrawNodeLabels::YES : DrawNodeLabels::NO;
166 draw_graph(graph_msg->data, node_idx_to_pos, draw, ptr);
167 }
168
169 return true;
170}
