71template <class T>
72T scale(const T& input, const T& in_min, const T& in_max, const T& out_min, const T& out_max) {
73 T result = input;
74 // Check for input/output min/max bounds
75 if (in_min > in_max || out_min > out_max) {
76 // cout << "Error: in_min > in_max || out_min > out_max" << endl;
77 return 0;
78 }
79
80 // Saturate input
81 if (result < in_min) {
82 result = in_min;
83 } else if (result > in_max) {
84 result = in_max;
85 }
86
87 double scale_factor = (out_max - out_min) / (in_max - in_min);
88 return (result - in_min) * scale_factor + out_min;
89}
