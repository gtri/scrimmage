39class AuctionAssign(sc.Autonomy):

41 def init(self, params):

55 def step_autonomy(self, t, dt):
56 self.action = sc.Action()
57 self.action.action_type = sc.Action.ActionType.None
58 self.action.target_id = 0
59
60 # if someone started an auction then send in a bid
61 for msg_in in self.subs["StartAuction"].msg_list:
62 print("{} received StartAuction from {}" \
63 .format(self.id.id, msg_in.sender))
64 data = BidAuction(originator_id=self.id.id, bid=np.random.rand())
65 msg_out = sc.Message(self.id.id, data.SerializeToString(), data)
66 self.pubs["BidAuction"].publish(msg_out)
67
68 if self.auction_started:
69 for msg_in in self.subs["BidAuction"].msg_list:
70
71 data = msg_in.data
72
73 print("{} received BidAuction from {}: bid = {:.2f}" \
74 .format(self.id.id, msg_in.sender, data.bid))
75 if data.bid > self.max_bid:
76 self.max_bid = data.bid
77 self.max_bid_champ = msg_in.sender
78
79 if not self.auction_started and self.id.id == 1:
80 print("{} is sending StartAuction msg".format(self.id.id))
81 msg_out = sc.Message(self.id.id)
82 self.pubs["StartAuction"].publish(msg_out)
83 self.auction_started = True
84 self.auction_in_prog = True
85 self.auction_start_time = t
86
87 if (self.auction_in_prog and
88 t > self.auction_start_time + self.auction_max_time and
89 self.id.id == 1):
90
91 self.auction_in_prog = False
92 print("="*60)
93 print("Auction Complete")
94 print("Max Bidder: {} - Bid: {:.2f}" \
95 .format(self.max_bid_champ, self.max_bid))
96
97 return True
