113void SimpleCollisionMetrics::calc_team_scores() {
114 double end_time = -std::numeric_limits<double>::infinity();
115 double beg_time = std::numeric_limits<double>::infinity();
116 for (auto& kv : scores_) {
117 double temp_beg_time = kv.second.flight_time_start();
118 double temp_end_time = kv.second.flight_time_end();
119 if (temp_beg_time < beg_time) {
120 beg_time = temp_beg_time;
121 }
122 if (temp_end_time > end_time) {
123 end_time = temp_end_time;
124 }
125 }
126
127 // If the end_time is <= beginning time, all entities survived, need
128 // to use the ending simulation time.
129 if (end_time <= beg_time) {
130 end_time = time_->t();
131 }
132
133 double max_flight_time = end_time - beg_time;
134
135 // Create the score, if necessary
136 for (auto& team_id : teams_) {
137 if (team_coll_scores_.count(team_id) == 0) {
138 SimpleCollisionScore score;
139 score.set_weights(params_);
140 // Set the max flight time for each score:
141 score.set_max_flight_time(max_flight_time);
142 team_coll_scores_[team_id] = score;
143 }
144 }
145
146 for (auto& kv : scores_) {
147 SimpleCollisionScore& score = kv.second;
148
149 // If the end time is less than start time, the flight time was never
150 // set. Use the ending simulation time.
151 if (score.flight_time_end() <= score.flight_time_start()) {
152 score.set_flight_time_end(end_time);
153 }
154
155 int team_id = (*id_to_team_map_)[kv.first];
156
157 team_coll_scores_[team_id].add_non_team_collisions(score.non_team_collisions());
158 team_coll_scores_[team_id].add_team_collisions(score.team_collisions());
159 team_coll_scores_[team_id].add_ground_collisions(score.ground_collisions());
160 team_coll_scores_[team_id].add_flight_time(score.flight_time());
161 team_coll_scores_[team_id].increment_entity_count();
162 }
163
164 for (auto& kv : team_coll_scores_) {
165 int team_id = kv.first;
166 SimpleCollisionScore& score = kv.second;
167 team_metrics_[team_id]["entity_count"] = score.entity_count();
168 team_metrics_[team_id]["flight_time"] = score.flight_time();
169 team_metrics_[team_id]["flight_time_norm"] = score.flight_time_norm();
170 team_metrics_[team_id]["non_team_coll"] = score.non_team_collisions();
171 team_metrics_[team_id]["team_coll"] = score.team_collisions();
172 team_metrics_[team_id]["ground_coll"] = score.ground_collisions();
173
174 team_scores_[team_id] = score.score();
175 }
