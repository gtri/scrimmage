114void SimpleCollisionMetrics::calc_team_scores() {
115 double end_time = -std::numeric_limits<double>::infinity();
116 double beg_time = std::numeric_limits<double>::infinity();
117 for (auto& kv : scores_) {
118 double temp_beg_time = kv.second.flight_time_start();
119 double temp_end_time = kv.second.flight_time_end();
120 if (temp_beg_time < beg_time) {
121 beg_time = temp_beg_time;
122 }
123 if (temp_end_time > end_time) {
124 end_time = temp_end_time;
125 }
126 }
127
128 // If the end_time is <= beginning time, all entities survived, need
129 // to use the ending simulation time.
130 if (end_time <= beg_time) {
131 end_time = time_->t();
132 }
133
134 double max_flight_time = end_time - beg_time;
135
136 // Create the score, if necessary
137 for (auto& team_id : teams_) {
138 if (team_coll_scores_.count(team_id) == 0) {
139 SimpleCollisionScore score;
140 score.set_weights(params_);
141 // Set the max flight time for each score:
142 score.set_max_flight_time(max_flight_time);
143 team_coll_scores_[team_id] = score;
144 }
145 }
146
147 for (auto& kv : scores_) {
148 SimpleCollisionScore& score = kv.second;
149
150 // If the end time is less than start time, the flight time was never
151 // set. Use the ending simulation time.
152 if (score.flight_time_end() <= score.flight_time_start()) {
153 score.set_flight_time_end(end_time);
154 }
155
156 int team_id = (*id_to_team_map_)[kv.first];
157
158 team_coll_scores_[team_id].add_non_team_collisions(score.non_team_collisions());
159 team_coll_scores_[team_id].add_team_collisions(score.team_collisions());
160 team_coll_scores_[team_id].add_ground_collisions(score.ground_collisions());
161 team_coll_scores_[team_id].add_flight_time(score.flight_time());
162 team_coll_scores_[team_id].increment_entity_count();
163 }
164
165 for (auto& kv : team_coll_scores_) {
166 int team_id = kv.first;
167 SimpleCollisionScore& score = kv.second;
168 team_metrics_[team_id]["entity_count"] = score.entity_count();
169 team_metrics_[team_id]["flight_time"] = score.flight_time();
170 team_metrics_[team_id]["flight_time_norm"] = score.flight_time_norm();
171 team_metrics_[team_id]["non_team_coll"] = score.non_team_collisions();
172 team_metrics_[team_id]["team_coll"] = score.team_collisions();
173 team_metrics_[team_id]["ground_coll"] = score.ground_collisions();
174
175 team_scores_[team_id] = score.score();
176 }
177
178 // list the headers we want put in the csv file
179 headers_.push_back("entity_count");
180 headers_.push_back("flight_time");
181 headers_.push_back("flight_time_norm");
182 headers_.push_back("non_team_coll");
183 headers_.push_back("team_coll");
184 headers_.push_back("ground_coll");
