112void SimpleCollisionMetrics::calc_team_scores() {
113 double end_time = -std::numeric_limits<double>::infinity();
114 double beg_time = std::numeric_limits<double>::infinity();
115 for (auto& kv : scores_) {
116 double temp_beg_time = kv.second.flight_time_start();
117 double temp_end_time = kv.second.flight_time_end();
118 if (temp_beg_time < beg_time) {
119 beg_time = temp_beg_time;
120 }
121 if (temp_end_time > end_time) {
122 end_time = temp_end_time;
123 }
124 }
125
126 // If the end_time is <= beginning time, all entities survived, need
127 // to use the ending simulation time.
128 if (end_time <= beg_time) {
129 end_time = time_->t();
130 }
131
132 double max_flight_time = end_time - beg_time;
133
134 // Create the score, if necessary
135 for (auto& team_id : teams_) {
136 if (team_coll_scores_.count(team_id) == 0) {
137 SimpleCollisionScore score;
138 score.set_weights(params_);
139 // Set the max flight time for each score:
140 score.set_max_flight_time(max_flight_time);
141 team_coll_scores_[team_id] = score;
142 }
143 }
144
145 for (auto& kv : scores_) {
146 SimpleCollisionScore& score = kv.second;
147
148 // If the end time is less than start time, the flight time was never
149 // set. Use the ending simulation time.
150 if (score.flight_time_end() <= score.flight_time_start()) {
151 score.set_flight_time_end(end_time);
152 }
153
154 int team_id = (*id_to_team_map_)[kv.first];
155
156 team_coll_scores_[team_id].add_non_team_collisions(score.non_team_collisions());
157 team_coll_scores_[team_id].add_team_collisions(score.team_collisions());
158 team_coll_scores_[team_id].add_ground_collisions(score.ground_collisions());
159 team_coll_scores_[team_id].add_flight_time(score.flight_time());
160 team_coll_scores_[team_id].increment_entity_count();
161 }
162
163 for (auto& kv : team_coll_scores_) {
164 int team_id = kv.first;
165 SimpleCollisionScore& score = kv.second;
166 team_metrics_[team_id]["entity_count"] = score.entity_count();
167 team_metrics_[team_id]["flight_time"] = score.flight_time();
168 team_metrics_[team_id]["flight_time_norm"] = score.flight_time_norm();

170 team_metrics_[team_id]["team_coll"] = score.team_collisions();
171 team_metrics_[team_id]["ground_coll"] = score.ground_collisions();
