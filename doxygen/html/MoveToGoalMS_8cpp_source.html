65void MoveToGoalMS::init(std::map<std::string, std::string>& params) {
66 if (sc::get("use_initial_heading", params, false)) {
67 Eigen::Vector3d rel_pos = Eigen::Vector3d::UnitX() * 1e6;
68 Eigen::Vector3d unit_vector = rel_pos.normalized();
69 unit_vector = state_->quat().rotate(unit_vector);
70 wp_local_ = state_->pos() + unit_vector * rel_pos.norm();
71 } else {
72 std::vector<double> goal_vec;
73 if (sc::get_vec<double>("goal", params, ", ", goal_vec, 3)) {
74 wp_local_ = sc::vec2eigen(goal_vec);
75 } else {
76 cout << "Failed to parse MoveToGoalMS' initial goal" << endl;
77 }
78 }
79
80 // Initialize PID controller class
81 speed_pid_.set_parameters(
82 sc::get("p_gain", params, 0.5),
83 sc::get("i_gain", params, 0.0),
84 sc::get("d_gain", params, 0.0));
85 speed_pid_.set_integral_band(sc::get("integral_band", params, 0.0));
86 speed_pid_.set_setpoint(0.0);
87
88 // Convert XYZ goal to lat/lon/alt
89 double lat, lon, alt;
90 parent_->projection()->Reverse(wp_local_(0), wp_local_(1), wp_local_(2), lat, lon, alt);
91
92 wp_ = Waypoint(lat, lon, alt);
93 wp_.set_time(0);
94 wp_.set_quat(scrimmage::Quaternion(0, 0, 0));
95 wp_.set_position_tolerance(1);
96 wp_.set_quat_tolerance(1);

99 wp_ = msg->data;
100 parent_->projection()->Forward(
101 wp_.latitude(),
102 wp_.longitude(),
103 wp_.altitude(),
104 wp_local_(0),
105 wp_local_(1),
106 wp_local_(2));
107 };
