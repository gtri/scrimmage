170std::list<cv::Rect> MapGen2D::find_rectangles(cv::Mat& img, int threshold) {
171 // Make sure image is gray and apply threshold
172 cv::Mat gray;
173 cv::cvtColor(img, gray, cv::COLOR_BGRA2GRAY);
174
175 cv::Mat thresh;
176 cv::threshold(gray, thresh, std::floor(threshold * 255), 255, cv::THRESH_BINARY_INV);
177
178 cv::Mat img_rects = img.clone();
179
180 std::list<cv::Rect> rects;
181
182 cv::Mat I = thresh.clone();
183
184 // Scan image looking for non-zero pixel
185 for (int i = 0; i < I.rows; i++) {
186 for (int j = 0; j < I.cols; j++) {
187 if (I.at<uchar>(i, j) > 0) {
188 // Non-zero pixel found, search for largest rectangle of
189 // non-zero pixels
190
191 bool end_c_found = false;
192 int end_c = I.cols;
193 int end_r = I.rows;
194
195 // Scan rows and columns looking for a zero-valued pixel
196 for (int r = i; r < end_r; r++) {
197 for (int c = j; c < end_c; c++) {
198 if (I.at<uchar>(r, c) == 0) {
199 // Zero pixel found
200 if (!end_c_found) {
201 // Use this column number as end column of
202 // rectangle
203 end_c_found = true;
204 end_c = c;
205 } else {
206 // Use this row number as end row of rectangle
207 end_r = r;
208 }
209 }
210 }
211
212 if (end_c == I.cols) {
213 // If end_c is still the same as I.cols, then the
214 // rectangle extended to the end of the image.
215 end_c_found = true;
216 }
217 }
218
219 cv::Rect rect(j, i, end_c - j, end_r - i);
220 cv::rectangle(img_rects, rect, cv::Scalar(0, 0, 255), 1, 8, 0);
221 cv::Mat roi = I(rect);
222 roi.setTo(0);
223
224 rects.push_back(rect);
225 }
226 }
227 }
228
229 if (enable_map_boundary_) {
230 rects.push_back(cv::Rect(0, 0, img.cols, 1)); // top rect
231 rects.push_back(cv::Rect(0, 0, 1, img.rows)); // left rect
232 rects.push_back(cv::Rect(img.cols, 0, 1, img.rows)); // right rect
233 rects.push_back(cv::Rect(0, img.rows, img.cols, 1)); // bottom rect
234 }
235
236 if (show_map_debug_) {
237 cout << "Number of rectangles: " << rects.size() << endl;
238 cv::imshow("Original", img);
239 cv::imshow("Gray", gray);
240 cv::imshow("Thresh", thresh);
241 cv::imshow("Rects", img_rects);
242 cv::waitKey(0);
243 }
244 return rects;
245}
