66Waypoint::Waypoint(const scrimmage_msgs::Waypoint& wp)
67 : id_(wp.id()),
68 time_(wp.time()),
69 position_tolerance_(wp.position_tolerance()),
70 quat_tolerance_(wp.quat_tolerance()),
71 tolerance_in_2d_(wp.tolerance_in_2d()) {
72 sc::set(quat_, wp.quat());
73
74 auto get_alt = [](const sc_msgs::PositionLLA& lla) {
75 double alt = 0;
76 if (lla.alt_oneof_case() == sc_msgs::PositionLLA::kAltMsl) {
77 alt = lla.alt_msl();
78 } else {
79 std::cout << "Waypoints only support alt_msl()!" << std::endl;
80 }
81 return alt;
82 };
83
84 if (wp.pos_oneof_case() == sc_msgs::Waypoint::kLla) {
85 latitude_ = wp.lla().latitude();
86 longitude_ = wp.lla().longitude();
87 altitude_ = get_alt(wp.lla());
88 } else if (wp.pos_oneof_case() == sc_msgs::Waypoint::kXyz) {
89 // Convert to lat/lon/alt given origin
90 GeographicLib::LocalCartesian proj(
91 wp.xyz().origin().latitude(),
92 wp.xyz().origin().longitude(),
93 get_alt(wp.xyz().origin()));
94 proj.Reverse(
95 wp.xyz().xyz().x(),
96 wp.xyz().xyz().y(),
97 wp.xyz().xyz().z(),
98 latitude_,
99 longitude_,
100 altitude_);
101 }
102}

118bool Waypoint::is_within_tolerance(
119 const scrimmage::StatePtr& state,
120 const std::shared_ptr<GeographicLib::LocalCartesian>& proj) {
121
122 Eigen::Vector3d pos;
123 proj->Forward(latitude_, longitude_, altitude_, pos(0), pos(1), pos(2));
124 double dist = tolerance_in_2d_ ? (state->pos().head<2>() - pos.head<2>()).norm()
125 : (state->pos() - pos).norm();
126 return dist <= position_tolerance_;
127}
