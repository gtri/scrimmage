56void GraphvizFSM::init(std::map<std::string, std::string>& params) {
57 print_current_state_ = sc::get<bool>("print_current_state", params, print_current_state_);
58 std::string graph_str = sc::get<std::string>("graphviz_fsm", params, "");
59
60 // Replace any ' with " characters. This allows the graphviz_fsm string to
61 // include single quotes instead of having to use &quot;
62 boost::replace_all(graph_str, "'", "\"");
63
64 boost::dynamic_properties dp(boost::ignore_other_properties);
65
66 dp.property("node_id", boost::get(&DotVertex::name, fsm_graph_));
67
68 label_ = get(label_t(), fsm_graph_);
69 dp.property("label", label_);

71 if (!boost::read_graphviz(graph_str, fsm_graph_, dp)) {
72 cout << "Failed to parse graphviz fms" << endl;
73 return;
74 }
75
76 bool create_fsm_pdf = sc::get<bool>("generate_fsm_pdf", params, false);
77 if (create_fsm_pdf) {
78 std::ofstream gv_file("./fsm.gv");
79 gv_file << graph_str;
80 gv_file.close();
81
82 int result = system("dot -Tpdf fsm.gv -o fsm.pdf");
83 if (result != 0) {
84 cout << "Error during FSM pdf creation. Make sure the dot program "
85 << "is installed." << endl;
86 }
87 }

89 std::string qi_name = sc::get<std::string>("initial_node_name", params, "qi");
90
91 // Find the initial state
92 typedef boost::graph_traits<graph_t>::vertex_iterator vertex_iter;
93 std::pair<vertex_iter, vertex_iter> vp;
94 for (vp = boost::vertices(fsm_graph_); vp.first != vp.second; ++vp.first) {
95 boost::graph_traits<graph_t>::vertex_descriptor v = *vp.first;
96 if (qi_name == fsm_graph_[v].name) {
97 current_state_ = v;
98 }
99 }
100
101 std::string state_topic = sc::get<std::string>("state_topic_name", params, "State");
102 std::string event_topic = sc::get<std::string>("event_topic_name", params, "Event");
103 std::string network_name = sc::get<std::string>("network_name", params, "LocalNetwork");
104
105 // Setup publisher on the state topic
106 state_pub_ = advertise(network_name, state_topic);
107
108 // Subscribe to event topic
109 auto event_callback = [&](scrimmage::MessagePtr<std::string> msg) {
110 // If the event received is a trigger for the current state, transition
111 // to the next state.
112 typename boost::graph_traits<graph_t>::out_edge_iterator ei, ei_end;
113 for (boost::tie(ei, ei_end) = boost::out_edges(current_state_, fsm_graph_); ei != ei_end;
114 ++ei) {
115 if (label_(*ei) == msg->data) {
116 this->update_state_info(current_state_, boost::target(*ei, fsm_graph_));
117 }
118 }
119 };
120 subscribe<std::string>(network_name, event_topic, event_callback);
121
122 return;

125bool GraphvizFSM::step_autonomy(double t, double dt) {
126 // If the current state only has one out edge (degree == 1) and doesn't
127 // have a trigger label to its next state, immediately move to that next
128 // state since this is just a designator for the "initial" state.
129 if (boost::out_degree(current_state_, fsm_graph_) == 1) {
130 typename boost::graph_traits<graph_t>::out_edge_iterator ei, ei_end;
131 for (boost::tie(ei, ei_end) = boost::out_edges(current_state_, fsm_graph_); ei != ei_end;
132 ++ei) {
133 if (label_(*ei) == std::string("")) {
134 this->update_state_info(current_state_, boost::target(*ei, fsm_graph_));
135 }
136 }
137 }
138 return true;
139}
