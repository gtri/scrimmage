100 void compute_dots() {
101 u = normZ;
102 v = -1 * normX;
103 w = normX;
104 s = normY;
105 r = -1 * normY;
106
107 // up or down
108 u_dot_P0 = u.dot(center_ - points_[0]);
109
110 // inside bounds of plane
111 w_dot_P0 = w.dot(center_ - points_[0]);
112 s_dot_P1 = s.dot(center_ - points_[1]);
113 v_dot_P2 = v.dot(center_ - points_[2]);
114 r_dot_P3 = r.dot(center_ - points_[3]);
115 }

117 bool contains(Eigen::Vector3d p) override {
118 double u_dot_p = u.dot(p - center_);
119 double w_dot_p = w.dot(p - center_);
120 double v_dot_p = v.dot(p - center_);
121 double r_dot_p = r.dot(p - center_);
122 double s_dot_p = s.dot(p - center_);
123
124 if ((u_dot_p > u_dot_P0) && (w_dot_p > w_dot_P0) && (v_dot_p > v_dot_P2)
125 && (s_dot_p > s_dot_P1) && (r_dot_p > r_dot_P3)) {
126 return true;
127 }
128 return false;
129 }

141 // Compute min / max values for x, y, z
142 Eigen::Vector3d mins(

144 std::numeric_limits<double>::infinity(),

151 for (int i = 0; i < 3; i++) {
