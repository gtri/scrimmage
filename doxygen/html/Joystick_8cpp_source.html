124bool Joystick::step(double t, double dt, VariableIO& vars) {
125 int bytes = read(joy_fd_, &js_, sizeof(struct js_event));
126 if (bytes == -1) {
127 // nop, avoid unused variable warning
128 }
129
130 // see what to do with the event
131 bool button_changed = false;
132 switch (js_.type & ~JS_EVENT_INIT) {
133 case JS_EVENT_AXIS:
134 axis_[js_.number] = js_.value;
135 break;
136 case JS_EVENT_BUTTON:
137 button_[js_.number] = js_.value;
138 button_changed = true;
139 break;
140 }
141
142 if (print_js_values_) {
143 for (int x = 0; x < num_of_axis_; x++) {
144 printf("%d: %6d ", x, axis_[x]);
145 }
146
147 for (int x = 0; x < num_of_buttons_; x++) {
148 printf("B%d: %d ", x, button_[x]);
149 }
150 printf(" \r");
151 fflush(stdout);
152 }
153
154 for (AxisScale axis_tf : axis_tfs_) {
155 vars.output(axis_tf.vector_index(), axis_tf.scale(axis_[axis_tf.axis_index()]));
156 }
157
158 if (publish_button_state_) {
159 // If the button state has changed, publish a new button message
160 if (button_changed) {
161 auto msg = std::make_shared<sc::Message<std::vector<bool>>>();
162 msg->data.resize(num_of_buttons_);
163
164 for (int i = 0; i < num_of_buttons_; i++) {
165 msg->data[i] = button_[i];
166 }
167
168 auto it = std::mismatch(msg->data.begin(), msg->data.end(), prev_button_state_.begin());
169 if (it.first != msg->data.end()) {
170 pub_buttons_->publish(msg);
171 prev_button_state_ = msg->data;
172 }
173 }
174 }
175
176 return true;
177}
