96bool RigidBody6DOF::step(double time, double dt) {
97 if (ctrl_u_ == nullptr) {
98 std::shared_ptr<Controller> ctrl =
99 std::dynamic_pointer_cast<Controller>(parent_->controllers().back());
100 if (ctrl) {
101 ctrl_u_ = ctrl->u();
102 }
103 }
104
105 if (ctrl_u_ == nullptr) {
106 return false;
107 }
108
109 // Need to saturate state variables before model runs
110 x_[U] = clamp(x_[U], min_velocity_, max_velocity_);
111
112 x_[P] = clamp(x_[U], -0.001, 0.001);
113 x_[Q] = clamp(x_[U], -0.001, 0.001);
114 x_[R] = clamp(x_[U], -0.001, 0.001);
115
116 ode_step(dt);
117
118 // Normalize quaternion
119 quat_local_.w() = x_[q0];
120 quat_local_.x() = x_[q1];
121 quat_local_.y() = x_[q2];
122 quat_local_.z() = x_[q3];
123 quat_local_.normalize();
124
125 x_[q0] = quat_local_.w();
126 x_[q1] = quat_local_.x();
127 x_[q2] = quat_local_.y();
128 x_[q3] = quat_local_.z();
129
130 // Convert local coordinates to world coordinates
131 state_->quat() = quat_world_ * quat_local_;
132 state_->pos() << x_[Xw], x_[Yw], x_[Zw];
133
134 return true;
135}

137void RigidBody6DOF::model(const vector_t& x, vector_t& dxdt, double t) {
138 double thrust = (*ctrl_u_)(THRUST);
139 double elevator = (*ctrl_u_)(ELEVATOR);
140 double aileron = (*ctrl_u_)(AILERON);
141 double rudder = (*ctrl_u_)(RUDDER);
142
143 double F_thrust = thrust;
144
145 double F_x = F_thrust;
146
147 double m = 1.0;
148 double g = 0; // -9.8; no gravity
149 double theta = 0;
150
151 double Ixx = 1;
152 double Iyy = 1;
153 double Izz = 1;
154 double Ixz = 0;
155
156 // TODO: Should these be cached from previous run or should the current dxdt
157 // versions be used?
158 double U_dot = dxdt[U];
159 double V_dot = dxdt[V];
160 double W_dot = dxdt[W];
161 double P_dot = dxdt[P];
162 double Q_dot = dxdt[Q];
163 double R_dot = dxdt[R];
164
165 dxdt[U] = x[V] * x[R] - x[W] * x[Q] - g * sin(theta) + F_x / m;
166 dxdt[V] = 0; // x[W]*x[P] - x[U]*x[R] + g*sin(phi)*cos(theta) + F_y / m;
167 dxdt[W] = 0; // x[U]*x[Q] - x[V]*x[P] + g*cos(phi)*cos(theta) + F_z / m;
168
169 double L = aileron + Ixx * P_dot - Ixz * R_dot - Ixz * x[P] * x[Q] + (Izz - Iyy) * x[R] * x[Q];
170 double M =
171 elevator + Iyy * Q_dot + (Ixx - Izz) * x[P] * x[R] + (Ixz * (pow(x[P], 2) - pow(x[R], 2)));
172 double N = rudder + Izz * R_dot - Ixz * P_dot + (Iyy - Ixx) * x[P] * x[Q] + Ixz * x[Q] * x[R];
173
174 double L_tic = L + Ixz * x[P] * x[Q] - (Izz - Iyy) * x[R] * x[Q];
175 double N_tic = N - (Iyy - Ixx) * x[P] * x[Q] - Ixz * x[R] * x[Q];
176
177 dxdt[P] = (L_tic * Izz - N_tic * Ixz) / (Ixx * Izz - pow(Ixz, 2));
178 dxdt[Q] = (M - (Ixx - Izz) * x[P] * x[R]) - Ixz * (pow(x[P], 2) - pow(x[R], 2)) / Iyy;
179 dxdt[R] = (N_tic * Ixx + L_tic * Ixz) / (Ixx * Izz - pow(Ixz, 2));
180
181 double lambda = 1 - (pow(x[q0], 2) + pow(x[q1], 2) + pow(x[q2], 2) + pow(x[q3], 2));
182 dxdt[q0] = -0.5 * (x[q1] * x[P] + x[q2] * x[Q] + x[q3] * x[R]) + lambda * x[q0];
183 dxdt[q1] = +0.5 * (x[q0] * x[P] + x[q2] * x[R] - x[q3] * x[Q]) + lambda * x[q1];
184 dxdt[q2] = +0.5 * (x[q0] * x[Q] + x[q3] * x[P] - x[q1] * x[R]) + lambda * x[q2];
185 dxdt[q3] = +0.5 * (x[q0] * x[R] + x[q1] * x[Q] - x[q2] * x[P]) + lambda * x[q3];
186
187 // Local position / velocity to global
188 // Normalize quaternion
189 sc::Quaternion quat(x[q0], x[q1], x[q2], x[q3]);
190 quat.w() = x[q0];
191 quat.x() = x[q1];
192 quat.y() = x[q2];
193 quat.z() = x[q3];
194
195 quat = quat_world_ * quat;
196 quat.normalize();
197
198 // Convert local positions and velocities into global coordinates
199 Eigen::Matrix3d rot = quat.toRotationMatrix();
200
201 Eigen::Vector3d vel_local(x[U], x[V], x[W]);
202 Eigen::Vector3d vel_world = rot * vel_local;
203 dxdt[Xw] = vel_world(0);
204 dxdt[Yw] = vel_world(1);
205 dxdt[Zw] = vel_world(2);
206
207 Eigen::Vector3d acc_local(U_dot, V_dot, W_dot);
208 Eigen::Vector3d acc_world = rot * acc_local;
209 dxdt[Uw] = acc_world(0);
210 dxdt[Vw] = acc_world(1);
211 dxdt[Ww] = acc_world(2);
212}
