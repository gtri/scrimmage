495
496 ConfigParse cv_parse;
497 bool mesh_found, texture_found;
498 auto it = info.find("visual_model");

501 }
502
503 find_model_properties(
504 it->second,
505 cv_parse,
506 *file_search_,
507 overrides,
508 visual_,
509 mesh_found,
510 texture_found);
511
512 // Set the entity color. Use the team color by default
513 std::vector<int> color;
514 auto it_color = info.find("color");

516 } else {

518 }
519 set(visual_->mutable_color(), color[0], color[1], color[2]);
520
521 std::string visual_model = boost::to_upper_copy(info["visual_model"]);
522 if (mesh_found) {
523 type_ = Contact::Type::MESH;
524 visual_->set_visual_mode(
525 texture_found ? scrimmage_proto::ContactVisual::TEXTURE

527 } else if (visual_model == std::string("QUADROTOR")) {
528 type_ = Contact::Type::QUADROTOR;
529 visual_->set_visual_mode(scrimmage_proto::ContactVisual::COLOR);
530 } else if (visual_model == std::string("AIRCRAFT")) {
531 type_ = Contact::Type::AIRCRAFT;
532 visual_->set_visual_mode(scrimmage_proto::ContactVisual::COLOR);
533 } else if (visual_model == std::string("SPHERE")) {
534 type_ = Contact::Type::SPHERE;
535 visual_->set_visual_mode(scrimmage_proto::ContactVisual::COLOR);
536 } else {
537 type_ = Contact::Type::SPHERE;
538 visual_->set_visual_mode(scrimmage_proto::ContactVisual::COLOR);
539 }

541 return true;

544bool Entity::ready() {
545 auto all_ready = [&](auto& rng, auto& func) {
546 return std::all_of(rng.begin(), rng.end(), func);
547 };
548
549 auto single_ready = [&](auto& plugin) { return plugin->ready(); };
550 auto values_single_ready = [&](auto& kv) { return kv.second->ready(); };
551
552 bool autonomies_ready = all_ready(autonomies_, single_ready);
553 bool controllers_ready = all_ready(controllers_, single_ready);
554 bool sensors_ready = all_ready(sensors_, values_single_ready);
555 bool motion_model_ready = motion_model_ == nullptr || motion_model_->ready();
556
557 return autonomies_ready && controllers_ready && sensors_ready && motion_model_ready;
558}

728bool Entity::call_service(
729 scrimmage::MessageBasePtr req,
730 scrimmage::MessageBasePtr& res,
731 const std::string& service_name) {
732 auto it = services_.find(service_name);
733 if (it == services_.end()) {
734 // First check for a global service of this name
735 bool found = global_services_->call_service(req, res, service_name);
736 if (!found) {
737 std::cout << "request for service (" << service_name << ") that does not exist"
738 << std::endl;
739 std::cout << "services are: ";
740 for (auto& kv : services_) {
741 std::cout << kv.first << ", ";
742 }
743 std::cout << std::endl;
744 return false;
745 } else {
746 return true;
747 }
748 }
749
750 Service& service = it->second;
751 bool success = service(req, res);
752
753 if (!success) {
754 std::cout << "call to " << service_name << " failed" << std::endl;
755 return false;
756 } else {
757 return true;
758 }
