475bool Entity::parse_visual(
476 std::map<std::string, std::string>& info,
477 MissionParsePtr mp,
478 std::map<std::string, std::string>& overrides) {
479 visual_->set_id(id_.id());
480 visual_->set_opacity(1.0);
481
482 ConfigParse cv_parse;
483 bool mesh_found, texture_found;
484 auto it = info.find("visual_model");
485 if (it == info.end()) {
486 return true;
487 }
488
489 find_model_properties(
490 it->second,
491 cv_parse,
492 *file_search_,
493 overrides,
494 visual_,
495 mesh_found,
496 texture_found);
497
498 // Set the entity color. Use the team color by default
499 std::vector<int> color;
500 auto it_color = info.find("color");
501 if (it_color != info.end() and str2container(it_color->second, ", ", color, 3)) {
502 } else {
503 set(color, mp->team_info()[id_.team_id()].color);
504 }
505 set(visual_->mutable_color(), color[0], color[1], color[2]);
506
507 std::string visual_model = boost::to_upper_copy(info["visual_model"]);
508 if (mesh_found) {
509 type_ = Contact::Type::MESH;
510 visual_->set_visual_mode(
511 texture_found ? scrimmage_proto::ContactVisual::TEXTURE
512 : scrimmage_proto::ContactVisual::COLOR);
513 } else if (visual_model == std::string("QUADROTOR")) {
514 type_ = Contact::Type::QUADROTOR;
515 visual_->set_visual_mode(scrimmage_proto::ContactVisual::COLOR);
516 } else if (visual_model == std::string("AIRCRAFT")) {
517 type_ = Contact::Type::AIRCRAFT;
518 visual_->set_visual_mode(scrimmage_proto::ContactVisual::COLOR);
519 } else if (visual_model == std::string("SPHERE")) {
520 type_ = Contact::Type::SPHERE;
521 visual_->set_visual_mode(scrimmage_proto::ContactVisual::COLOR);
522 } else {
523 type_ = Contact::Type::SPHERE;
524 visual_->set_visual_mode(scrimmage_proto::ContactVisual::COLOR);
525 }
526
527 return true;
528}

530bool Entity::ready() {
531 auto all_ready = [&](auto& rng, auto& func) {
532 return std::all_of(rng.begin(), rng.end(), func);
533 };
534

705 scrimmage::MessageBasePtr req,
706 scrimmage::MessageBasePtr& res,
707 const std::string& service_name) {
708 auto it = services_.find(service_name);
709 if (it == services_.end()) {
710 // First check for a global service of this name
711 bool found = global_services_->call_service(req, res, service_name);
712 if (!found) {
713 std::cout << "request for service (" << service_name << ") that does not exist"
714 << std::endl;
715 std::cout << "services are: ";
716 for (auto& kv : services_) {
717 std::cout << kv.first << ", ";
718 }
719 std::cout << std::endl;
720 return false;
721 } else {
722 return true;
723 }
724 }
725
726 Service& service = it->second;
727 bool success = service(req, res);
728
729 if (!success) {
730 std::cout << "call to " << service_name << " failed" << std::endl;
731 return false;
732 } else {
733 return true;
