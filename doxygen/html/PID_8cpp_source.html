143double PID::step(const double& dt, const double& measurement) {
144 error_ = setpoint_ - measurement;
145 if (is_angle_) {
146 error_ = Angles::angle_pi(error_);
147 }
148
149 if (std::abs(error_) > integral_band_) {
150 integral_ = 0;
151 } else {
152 integral_ += error_ * dt;
153 }
154
155 derivative_ = (error_ - prev_error_) / std::max(1.0e-9, dt);
156 double u = kp_ * error_ + ki_ * integral_ + kd_ * derivative_;
157 prev_error_ = error_;
158
159 return enable_output_limits_ ? clamp(u, output_min_, output_max_) : u;
160}
