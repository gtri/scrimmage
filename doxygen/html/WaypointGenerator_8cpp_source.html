67void WaypointGenerator::init(std::map<std::string, std::string>& params) {
68 waypoint_color_ = sc::str2container<std::vector<int>>(
69 sc::get<std::string>("waypoint_color", params, "255,0,0"),
70 ",");
71
72 show_waypoints_ = sc::get<bool>("show_waypoints", params, false);
73
74 std::string waypoints = sc::get<std::string>("waypoints", params, "");
75
76 std::vector<std::vector<std::string>> vecs;
77 if (!sc::get_vec_of_vecs(waypoints, vecs)) {
78 LOG_ERROR("Failed to parse waypoints:" << waypoints);
79 } else {
80 for (std::vector<std::string> vec : vecs) {
81 if (vec.size() != 10) {
82 LOG_ERROR("Invalid waypoint: " << waypoints);
83 continue;
84 }
85
86 double lat = std::stod(vec[2]);
87 double lon = std::stod(vec[3]);
88 double alt = std::stod(vec[4]);
89
90 std::string type = vec[0];
91 std::transform(type.begin(), type.end(), type.begin(), ::toupper);
92 if (type != "XYZ" && type != "GPS") {
93 LOG_ERROR("Invalid waypoint type: " << type);
94 } else if (type == "XYZ") {
95 // Convert to XYZ point to GPS
96 parent_->projection()->Reverse(lat, lon, alt, lat, lon, alt);
97 }
98
99 Waypoint wp(lat, lon, alt);
100 wp.set_time(std::stod(vec[1]));
101 scrimmage::Quaternion quat(
102 sc::Angles::deg2rad(std::stod(vec[5])),
103 sc::Angles::deg2rad(std::stod(vec[6])),
104 sc::Angles::deg2rad(std::stod(vec[7])));
105 wp.set_quat(quat);
106 wp.set_position_tolerance(std::stod(vec[8]));
107 wp.set_quat_tolerance(std::stod(vec[9]));
108
109 wp_list_.waypoints().push_back(wp);
110 }
111 }
112
113 std::string mode = sc::get<std::string>("mode", params, "follow_once");
114 if (mode == "follow_once") {
115 wp_list_.set_mode(WaypointList::WaypointMode::follow_once);
116 } else if (mode == "back_and_forth") {
117 wp_list_.set_mode(WaypointList::WaypointMode::back_and_forth);
118 } else if (mode == "loiter") {
119 wp_list_.set_mode(WaypointList::WaypointMode::loiter);
120 } else if (mode == "racetrack") {
121 wp_list_.set_mode(WaypointList::WaypointMode::racetrack);
122 } else {
123 LOG_WARN("WaypointGenerator: Invalid mode. Defaulting to follow_once");
124 wp_list_.set_mode(WaypointList::WaypointMode::follow_once);
125 }
126
127 std::string network_name = sc::get<std::string>("network_name", params, "GlobalNetwork");
128 std::string topic_name = sc::get<std::string>("topic_name", params, "WaypointList");
129 waypoint_list_pub_ = advertise(network_name, topic_name);
130
131 position_x_idx_ = vars_.declare("position_x", VariableIO::Direction::Out);
132 position_y_idx_ = vars_.declare("position_y", VariableIO::Direction::Out);
133 position_z_idx_ = vars_.declare("position_z", VariableIO::Direction::Out);
134}

136bool WaypointGenerator::step_autonomy(double t, double dt) {
137 auto msg = std::make_shared<sc::Message<WaypointList>>();
138
139 std::list<Waypoint>::iterator it = wp_list_.waypoints().begin();
140 while (it != wp_list_.waypoints().end()) {
141 if (t >= it->time()) {
142 msg->data.set_mode(wp_list_.mode());
143 msg->data.waypoints().push_back(*it);
144
145 // Erase this waypoint from the main list
146 std::list<Waypoint>::iterator it_del = it;
147 it = wp_list_.waypoints().erase(it_del);
148 } else {
149 ++it;
150 }
151 }
152
153 // Publish the waypoint list message if it has waypoints
154 if (msg->data.waypoints().size() > 0) {
155 if (show_waypoints_) {
156 draw_waypoints(msg->data);
157 }
158 waypoint_list_pub_->publish(msg);
159 }
160
161 return true;
162}

164void WaypointGenerator::draw_waypoints(WaypointList& wp_list) {
165 for (Waypoint wp : wp_list.waypoints()) {
166 double x, y, z;
167 parent_->projection()->Forward(wp.latitude(), wp.longitude(), wp.altitude(), x, y, z);
168 vars_.output(position_x_idx_, x);
169 vars_.output(position_y_idx_, y);
170 vars_.output(position_z_idx_, z);
171
172 auto sphere = std::make_shared<scrimmage_proto::Shape>();
173 sphere->set_opacity(0.25);
174 sphere->set_persistent(true);
175 sc::set(
176 sphere->mutable_color(),
177 waypoint_color_[0],
178 waypoint_color_[1],
179 waypoint_color_[2]);
180
181 sphere->mutable_sphere()->set_radius(wp.position_tolerance());
182 sc::set(sphere->mutable_sphere()->mutable_center(), x, y, z);
183 draw_shape(sphere);
184 }
