306 Eigen::Vector3d gravity_vector(0, 0, -mass_ * g_);
307 Eigen::Vector3d F_weight = state_->quat().rotate_reverse(gravity_vector);
308
309 // Calculate force from drag (TODO: Check source)
310 Eigen::Vector3d vel_body(x_[U], x_[V], x_[W]);
311 double vel_mag = vel_body.norm();
312 Eigen::Vector3d F_drag = vel_body * (-0.5 * c_D_ * vel_mag);
313
314 // Calculate normal force
315 force_ext_body_[2] = force_ext_body_[2] - F_thrust[2];
316 if (force_ext_body_[2] > 0) { // only true if contacting ground
317 force_ext_body_[2] =
318 -1.0 * (F_weight[2]) - F_thrust[2]; // correctly calculate normal force
319 }
320
321 if (force_ext_body_[2] < 0) { // normal force only pushes up from the ground
322 force_ext_body_[2] = 0.0;
323 }
324
325 // Calculate total force
326 Eigen::Vector3d F_total = F_thrust + F_weight + F_drag + force_ext_body_;
327
328 dxdt[U] = x[V] * x[R] - x[W] * x[Q] + F_total(0) / mass_;
329 dxdt[V] = x[W] * x[P] - x[U] * x[R] + F_total(1) / mass_;
330 dxdt[W] = x[U] * x[Q] - x[V] * x[P] + F_total(2) / mass_;
331
332 // Calculate moments from thrust
333 Eigen::Vector3d Moments_thrust(0, 0, 0); // L, M, N
334
335 for (unsigned int i = 0; i < rotors_.size(); i++) {
336 double length = rotors_[i].offset_length();
337 double xy_angle = rotors_[i].xy_angle();
338
339 Moments_thrust(0) += rotor_thrust(i) * length * sin(xy_angle);
340 Moments_thrust(1) -= rotor_thrust(i) * length * cos(xy_angle);
341 Moments_thrust(2) += rotor_torque(i);
342 }
343
344 Eigen::Vector3d Moments_total = Moments_thrust;
345
346 Eigen::Vector3d pqr(x_[P], x_[Q], x_[R]);
347 Eigen::Vector3d pqr_dot = I_inv_ * (Moments_total - pqr.cross(I_ * pqr));
348 dxdt[P] = pqr_dot(0);
349 dxdt[Q] = pqr_dot(1);
350 dxdt[R] = pqr_dot(2);
351
352 // Update quaternion from angular accelerations
353 double lambda = 1 - (pow(x[q0], 2) + pow(x[q1], 2) + pow(x[q2], 2) + pow(x[q3], 2));
354 dxdt[q0] = -0.5 * (x[q1] * x[P] + x[q2] * x[Q] + x[q3] * x[R]) + lambda * x[q0];
355 dxdt[q1] = +0.5 * (x[q0] * x[P] + x[q2] * x[R] - x[q3] * x[Q]) + lambda * x[q1];
356 dxdt[q2] = +0.5 * (x[q0] * x[Q] + x[q3] * x[P] - x[q1] * x[R]) + lambda * x[q2];
357 dxdt[q3] = +0.5 * (x[q0] * x[R] + x[q1] * x[Q] - x[q2] * x[P]) + lambda * x[q3];
358
359 // Local position / velocity to global
360 // Normalize quaternion
361 sc::Quaternion quat(x[q0], x[q1], x[q2], x[q3]);
362 quat.w() = x[q0];
363 quat.x() = x[q1];
364 quat.y() = x[q2];
365 quat.z() = x[q3];
366 quat.normalize();
367
368 Eigen::Vector3d vel_local(x[U], x[V], x[W]);
369 Eigen::Vector3d vel_world = quat.rotate(vel_local);
370 dxdt[Xw] = vel_world(0);
371 dxdt[Yw] = vel_world(1);
372 dxdt[Zw] = vel_world(2);
373
375
376 Eigen::Vector3d acc_local = F_total / mass_; // this is in body frame
377 Eigen::Vector3d acc_world = quat.rotate(acc_local);
378 dxdt[Uw] = acc_world(0);
379 dxdt[Vw] = acc_world(1);
380 dxdt[Ww] = acc_world(2);
381}
