61bool SimpleCapture::init(

65
66 enable_team_captures_ = sc::get<bool>("enable_team_captures", plugin_params, true);
67 enable_non_team_captures_ = sc::get<bool>("enable_non_team_captures", plugin_params, true);
68
69 // Setup publishers
70 team_capture_pub_ = advertise("GlobalNetwork", "TeamCapture");
71 non_team_capture_pub_ = advertise("GlobalNetwork", "NonTeamCapture");
72
73 // Setup capture callback
74 auto capture_cb = [&](scrimmage::MessagePtr<sm::CaptureEntity> msg) {
75 int source_id = msg->data.source_id();
76 int target_id = msg->data.target_id();
77
78 // if target_id is already present in already_captured_,
79 // capture_result.second will be false
80 auto capture_result = already_captured_.emplace(target_id);
81
82 if (capture_result.second) {
83 sc::EntityPtr& src = (*id_to_ent_map_)[source_id];
84 sc::EntityPtr& dst = (*id_to_ent_map_)[target_id];
85 if ((src->state_truth()->pos() - dst->state_truth()->pos()).norm() <= capture_range_) {
86 if (enable_team_captures_ && src->id().team_id() == dst->id().team_id()) {
87 dst->collision();
88
89 auto msg = std::make_shared<sc::Message<sm::TeamCapture>>();
90 msg->data.set_source_id(src->id().id());
91 msg->data.set_target_id(dst->id().id());
92 team_capture_pub_->publish(msg);
93
94 } else if (
95 enable_non_team_captures_ && src->id().team_id() != dst->id().team_id()) {
96 dst->collision();
97
98 auto msg = std::make_shared<sc::Message<sm::NonTeamCapture>>();
99 msg->data.set_source_id(src->id().id());
100 msg->data.set_target_id(dst->id().id());
101 non_team_capture_pub_->publish(msg);
102 }
103 }
104 }
105 };
106 subscribe<sm::CaptureEntity>("GlobalNetwork", "CaptureEntity", capture_cb);
107 return true;
108}
