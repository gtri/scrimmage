62bool load_plugins(
63 const std::list<PluginOverrides>& plugin_names_overrides,
64 const std::string& parent_class_name,

66 const std::set<std::string>& plugin_tags = {},
67 PluginManagerPtr mgr = std::make_shared<PluginManager>(),
68 FileSearchPtr file_search = std::make_shared<FileSearch>()) {
69 bool all_loaded = true;
70 for (const auto& plugin : plugin_names_overrides) {
71 ConfigParse config_parse;
72
73 PluginStatus<T> status = mgr->make_plugin<T>(
74 parent_class_name,
75 plugin.name,
76 *file_search,
77 config_parse,
78 plugin.overrides,
79 plugin_tags);
80
81 if (status.status == PluginStatus<T>::cast_failed) {
82 cout << "Failed to open: " << parent_class_name << ": " << plugin.name << endl;
83 all_loaded = false;
84 } else if (status.status == PluginStatus<T>::parse_failed) {
85 cout << "Failed to parse: " << parent_class_name << ": " << plugin.name << endl;
86 all_loaded = false;
87 } else if (status.status == PluginStatus<T>::loaded) {
88 auto it_name = config_parse.params().find("name");
89 std::string plugin_name =
90 it_name == config_parse.params().end() ? plugin.name : it_name->second;
91
92 plugins_and_configs[plugin_name] = PluginAndConfig<T>(status.plugin, config_parse);
93 }
94 all_loaded &= (status.status == PluginStatus<T>::loaded);
95 }
96 return all_loaded;
97}
