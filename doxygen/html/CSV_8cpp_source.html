45// Function to escape special characters in a CSV field
46std::string escape_csv_field(std::string_view field) {
47 // If the field contains a comma, a quote, or a newline, wrap it in quotes
48 bool need_to_escape = field.find_first_of(",\"\n") != std::string::npos;
49 if (!need_to_escape) {
50 return std::string(field);
51 }
52
53 std::ostringstream escaped;
54 escaped << '"';
55 for (char c : field) {
56 if (c == '"') {
57 escaped << "\"\""; // Escape double quotes as ""
58 } else {
59 escaped << c;
60 }
61 }
62 escaped << '"';
63 return escaped.str();
64}

66// Function to unescape a single CSV field
67std::string unescape_csv_field(std::string_view field) {
68 // If the field starts and ends with a quote, it is escaped
69 bool is_quoted_string = field.size() >= 2 && field.front() == '"' && field.back() == '"';
70 if (!is_quoted_string) {
71 return std::string(field);
72 }
73
74 std::ostringstream unescaped;
75 bool isQuotePair = false;
76
77 // Process the part within the quotes
78 for (size_t i = 1; i < field.size() - 1; ++i) {
79 char c = field[i];
80 if (isQuotePair) {
81 if (c == '"') { // This is the double quote -> unescape it
82 unescaped << '"';
83 isQuotePair = false;
84 } else {
85 // CSV is invalid if we find a singe quote within a single quoted field
86 throw std::runtime_error(
87 "Invalid CSV: field wrapped by quotes but quote inside is not escaped properly "
88 "(ex. \"field \"\"with\"\" quote\"): "
89 + std::string(field));
90 }
91 } else {
92 if (c == '"') {
93 // First quote of a pair
94 isQuotePair = true;
95 } else {
96 unescaped << c; // Regular character
97 }
98 }
99 }
100
101 // If `isQuotePair` is still true here, it means the quotes were improperly escaped
102 if (isQuotePair) {
103 throw std::runtime_error("Invalid CSV: unbalanced quotes in field");
104 }
105
106 return unescaped.str();
107}

109// Function to split a CSV string by some token while respecting escape sequences of
110// 2 double quotes """"
111std::vector<std::string> split_csv_string(std::string_view row, const char token) {
112 std::vector<std::string> fields; // Stores the extracted fields
113 std::ostringstream currentField; // Accumulates characters for the current field
114 bool insideQuotes = false; // Tracks if we're inside a quoted field
115
116 for (size_t i = 0; i < row.size(); ++i) {
117 char c = row[i];
118
119 if (insideQuotes) {
120 if (c == '"') {
121 // If we see a double-quote while inside quotes, check for escape (i.e., `""`)
122 if (i + 1 < row.size() && row[i + 1] == '"') {
123 currentField << "\"\"";
124 ++i; // Skip the next quote (because this is an escape sequence)
125 } else {
126 // Exiting the quoted field
127 insideQuotes = false;
128 currentField << c;
129 }
130 } else {
131 // Append any other character inside quotes
132 // This is where newlines and commas can appear
133 // as valid text inside a cell
134 currentField << c;
135 }
136 } else {
137 if (c == token) {
138 // token outside quotes: field delimiter
139 fields.push_back(currentField.str()); // Add the current field to the list
140 currentField.str(""); // Reset for the next field
141 currentField.clear(); // Clear any error flags on the object
142 } else if (c == '"') {
143 // Start of a quoted field
144 currentField << c;
145 insideQuotes = true;
146 } else {
147 // Regular character outside quotes
148 currentField << c;
149 }
150 }
151 }
152
153 // Add the last field (if any)
154 fields.push_back(currentField.str());
155
156 return fields;
157}

196
197 for (const auto& pair : pairs) {
198 auto it = column_headers_.find(pair.first);
199 if (it == column_headers_.end()) {
200 LOG_WARN("Warning: column header doesn't exist: " << pair.first);

204

207 LOG_WARN("File isn't open. Can't write CSV");

211 }
212

317bool CSV::read_csv_from_string(const std::string& csv_str, const bool& contains_header) {
318 table_.clear();
319 column_headers_.clear();
320
321 std::vector<std::string> line_tokens = split_csv_string(csv_str, '\n');
322 int row_num = 0;
323 for (unsigned int line_num = 0; line_num < line_tokens.size(); line_num++) {
324 std::string line = line_tokens[line_num];
325
326 if (line == "")
327 continue; // Ignore lines that are empty
328
329 if (line_num == 0 && contains_header) {
330 std::vector<std::string> headers = split_csv_string(line, ',');
331 int i = 0;
332 for (const auto& str : headers) {
333 column_headers_[unescape_csv_field(str)] = i;
334 i++;
335 }
336 } else {
337 std::vector<std::string> tokens = split_csv_string(line, ',');
338 for (unsigned int i = 0; i < tokens.size(); i++) {
339 table_[row_num][i] = unescape_csv_field(tokens[i]);
340 }
341
342 // If this is the first line and the file doesn't contain a header,
343 // populate the column headers with indices based on the number of
344 // comma separated values
345 if (row_num == 0 && !contains_header) {
346 for (unsigned int i = 0; i < tokens.size(); i++) {
347 column_headers_[std::to_string(i)] = i;
348 }
349 }
350
351 // Print a warning if the number of comma separated values doesn't
352 // match the number of columns
353 if (column_headers_.size() != tokens.size()) {
354 LOG_WARN(
355 "Warning the number of values ("
356 << tokens.size() << ") on line number " << row_num
357 << " doesn't match the number of column headers: " << column_headers_.size());
358 }
359 row_num++;
360 }
361 }
362 return true;
363}
