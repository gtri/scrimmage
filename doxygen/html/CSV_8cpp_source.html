159std::string CSV::row_to_string(const int& row) const {
160 std::string result = "";
161
162 // Initialize a vector with no value string. Iterate over
163 // column_headers, use column index to fill in column for values.
164 std::vector<std::string> values(column_headers_.size(), no_value_str_);
165 auto it_row = table_.find(row);
166 for (auto& kv : it_row->second) {
167 if (static_cast<int64_t>(kv.second) == kv.second) {
168 values[kv.first] = std::to_string(static_cast<int64_t>(kv.second));
169 } else if (static_cast<double>(kv.second) == kv.second) {
170 // default precision values for double are not enough in many cases
171 std::ostringstream conv;
172 if (double_is_fixed_) {
173 conv << std::fixed;
174 }
175 if (double_is_scientific_) {
176 conv << std::scientific;
177 }
178 conv << std::setprecision(double_precision_) << kv.second;
179 values[kv.first] = conv.str();
180 } else {
181 values[kv.first] = std::to_string(kv.second);
182 }
183 }
184
185 // Append the rows to the resultant string
186 unsigned int i = 0;
187 for (std::string str : values) {
188 result += str;
189
190 if (i + 1 < values.size()) {
191 result += ",";
192 }
193 i++;
194 }
195 return result;
196}

217bool CSV::read_csv_from_string(const std::string& csv_str, const bool& contains_header) {
218 table_.clear();
219 column_headers_.clear();
220
221 std::vector<std::string> line_tokens;
222 boost::split(line_tokens, csv_str, boost::is_any_of("\n"));
223 int row_num = 0;
224 for (unsigned int line_num = 0; line_num < line_tokens.size(); line_num++) {
225 // Strip "whitespace" characters.
226 std::string line = remove_whitespace(line_tokens[line_num]);
227
228 if (line == "")
229 continue; // Ignore lines that are empty
230
231 if (line_num == 0 && contains_header) {
232 std::list<std::string> headers = get_csv_line_elements(line);
233 int i = 0;
234 for (auto& str : headers) {
235 column_headers_[str] = i;
236 i++;
237 }
238 } else {
239 std::vector<std::string> tokens;
240 boost::split(tokens, line, boost::is_any_of(","));
241 for (unsigned int i = 0; i < tokens.size(); i++) {
242 table_[row_num][i] = std::stod(tokens[i]);
243 }
244
245 // If this is the first line and the file doesn't contain a header,
246 // populate the column headers with indices based on the number of
247 // comma separated values
248 if (row_num == 0 && not contains_header) {
249 for (unsigned int i = 0; i < tokens.size(); i++) {
250 column_headers_[std::to_string(i)] = i;
251 }
252 }
253
254 // Print a warning if the number of comma separated values doesn't
255 // match the number of columns
256 if (column_headers_.size() != tokens.size()) {
257 cout << "Warning the number of values (" << tokens.size() << ") on line number "
258 << row_num
259 << " doesn't match the number of column headers: " << column_headers_.size()
260 << endl;
261 }
262 row_num++;
263 }
264 }
265 return true;
266}

293std::list<std::string> CSV::get_csv_line_elements(const std::string& str) {
294 std::list<std::string> elems;
295 std::vector<std::string> tokens;
296 boost::split(tokens, str, boost::is_any_of(","));
297 for (unsigned int i = 0; i < tokens.size(); i++) {
298 // Remove whitespace
299 tokens[i].erase(
300 std::remove_if(
301 tokens[i].begin(),
302 tokens[i].end(),
303 [](unsigned char x) { return std::isspace(x); }),
304 tokens[i].end());
305
306 if (tokens[i] != "") {
307 elems.push_back(tokens[i]);
308 }
309 }
310 return elems;
311}
