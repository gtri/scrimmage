43void WaypointListProcessor::set_waypoint_list(const scrimmage_msgs::WaypointList& wp_list) {
44 returning_stage_ = false;
45
46 // Construct a list of Waypoints from protobuf waypoint list
47 wp_list_.clear();
48 for (int i = 0; i < wp_list.waypoints().size(); i++) {
49 wp_list_.push_back(Waypoint(wp_list.waypoints(i)));
50 }
51
52 // Filter the waypoint list to ensure that two waypoints in succession
53 // are not equal. Some planners generate duplicate waypoints.
54 auto wps_are_close = [](auto& wp0, auto& wp1) {
55 auto numbers_are_close = [](const double& a, const double& b) {
56 return std::abs(a - b) < std::numeric_limits<double>::epsilon();
57 };
58 return numbers_are_close(wp0.latitude(), wp1.latitude())
59 && numbers_are_close(wp0.longitude(), wp1.longitude())
60 && numbers_are_close(wp0.altitude(), wp1.altitude());
61 };
62 wp_list_.unique(wps_are_close);
63
64 // Set the wp iterator to point to the beginning of the list
65 // prev_wp_it_ = wp_list_.begin();
66 curr_wp_it_ = wp_list_.begin();
67 next_wp_it_ = next_waypoint(curr_wp_it_);
68}

70WaypointListProcessor::Status WaypointListProcessor::process(
71 const scrimmage::StatePtr& state,
72 const std::shared_ptr<GeographicLib::LocalCartesian>& proj) {
73 if (wp_list_.size() == 0) {
74 return Status::Empty;
75 }
76
77 Waypoint& curr_wp_lla = *curr_wp_it_;
78
79 // If the current state isn't within tolerance, the waypoint is unchanged
80 if (not curr_wp_lla.is_within_tolerance(state, proj)) {
81 return Status::Unchanged;
82 }
83
84 prev_wp_it_ = curr_wp_it_;
85 curr_wp_it_ = next_waypoint(curr_wp_it_);
86 next_wp_it_ = next_waypoint(curr_wp_it_);
87
88 return Status::Changed;
89}

91std::list<Waypoint>::iterator WaypointListProcessor::next_waypoint(
92 const std::list<Waypoint>::iterator& it_wp) {
93 std::list<Waypoint>::iterator it_next;
94 switch (mode_) {
95 case scrimmage_msgs::WaypointList::FOLLOW_ONCE:
96 it_next = std::next(it_wp);
97 break;
98
99 case scrimmage_msgs::WaypointList::BACK_AND_FORTH:
100 it_next = returning_stage_ ? std::prev(it_wp) : std::next(it_wp);
101
102 if (it_next == wp_list_.begin()) {
103 returning_stage_ = false;
104 } else if (it_next == wp_list_.end()) {
105 it_next = it_wp;
106 returning_stage_ = true;
107 }
108 break;
109
110 case scrimmage_msgs::WaypointList::LOITER:
111 // TODO : Not implemented yet
112 break;
113
114 case scrimmage_msgs::WaypointList::LOOP:
115 // Continuous repeat of waypoints
116 it_next = std::next(curr_wp_it_);
117
118 if (it_next == wp_list_.end()) {
119 it_next = wp_list_.begin();
120 }
121 break;
122
123 default:
124 std::string msg = "Waypoint mode not defined yet.";
125 throw std::runtime_error(msg);
126 }
127 return it_next;
128}
