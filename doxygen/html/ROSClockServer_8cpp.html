115void ROSClockServer::publish_clock_msg(const double& t) {
116 // Add the current simulation time to the system time at the start of the
117 // simulation.
118 std::chrono::time_point<std::chrono::system_clock, std::chrono::duration<double>>
119 sim_time_point = sim_start_time_ + std::chrono::duration<double>(t);
120
121 // Convert the time_point into a duration that represents double time since
122 // the epoch.
123 std::chrono::duration<double> time_since_epoch = sim_time_point.time_since_epoch();
124 double sim_time = time_since_epoch.count();
125
126 // Extract the seconds and nanoseconds to construct the ros::Time message
127 uint32_t sec = std::floor(sim_time);
128 uint32_t nsec = (sim_time - static_cast<double>(sec)) * 1e9;
129
130 // Create the ros Clock message and publish it
131 rosgraph_msgs::Clock clock_msg;
132 clock_msg.clock = ros::Time(sec, nsec);
133
134 pub_mutex_.lock();
135 // Check the publisher init since rosmaster might not have started yet (see
136 // check_rosmaster()).
137 if (clock_pub_) {
138 clock_pub_.publish(clock_msg);
139 }
140 pub_mutex_.unlock();
141}
