50int PluginManager::check_library(std::string lib_path) {
51 void* lib_handle;
52#ifdef __APPLE__
53 lib_handle = dlopen(lib_path.c_str(), RTLD_LAZY | RTLD_GLOBAL);
54#else
55 lib_handle = dlopen(lib_path.c_str(), RTLD_LAZY | RTLD_LOCAL);
56#endif
57 if (!lib_handle) {
58 LOG_ERROR("Failed to open " << lib_path << ", dlopen failed: " << dlerror());
59 return -1;
60 }
61
62 char* error;
63
64 // Extract the plugin type
65 plugin_type_t type_func;
66 type_func = (plugin_type_t)dlsym(lib_handle, "plugin_type");
67 if ((error = dlerror()) != NULL) {
68 // fputs(error, stderr);
69 LOG_WARN(lib_path << " doesn't contain 'plugin_type'");
70 dlclose(lib_handle);
71 return 1;
72 }
73 std::string plugin_type((*type_func)());
74
75 // Extract the plugin name
76 plugin_name_t name_func;
77 name_func = (plugin_name_t)dlsym(lib_handle, "plugin_name");
78 if ((error = dlerror()) != NULL) {
79 // fputs(error, stderr);
80 LOG_WARN(lib_path << " doesn't contain 'plugin_name'");
81 dlclose(lib_handle);
82 return 3;
83 }
84 std::string plugin_name((*name_func)());
85
86 // Ensure the maker function exists, but we don't need to call it
87 dlsym(lib_handle, "maker");
88 if ((error = dlerror()) != NULL) {
89 fputs(error, stderr);
90 LOG_WARN(lib_path << " doesn't contain 'maker'");
91 dlclose(lib_handle);
92 return 4;
93 }
94
95 PluginInfo info;
96 info.name = plugin_name;
97 info.type = plugin_type;
98 info.path = lib_path;
99 info.handle = lib_handle;
100
101 plugins_info_[plugin_type][plugin_name] = info;
102
103 // dlclose(lib_handle);
104
105 return 0;
106}

151PluginPtr PluginManager::make_plugin_helper(std::string& plugin_type, std::string& plugin_name) {
152 auto it = plugins_info_.find(plugin_type);
153 if (it != plugins_info_.end()) {
154 auto it2 = it->second.find(plugin_name);
155 if (it2 != it->second.end()) {
156 it2->second.returned = true;
157
158 if (reload_ && it2->second.handle) {
159 dlclose(it2->second.handle);
160 }
161
162 PluginPtr (*maker_func)(void);
163 // cppcheck-suppress cstyleCast
164 maker_func = (PluginPtr (*)(void))dlsym(it2->second.handle, "maker");
165 char* error;
166 if ((error = dlerror()) != NULL) {
167 LOG_ERROR("dlsym failed for plugin '" << plugin_name << "': " << error);
168 return nullptr;
169 } else {
170 return (*maker_func)();
171 }
172 }
173 }
174 return nullptr;
175}

177std::map<std::string, std::unordered_set<std::string>> PluginManager::get_commits() {
178 std::map<std::string, std::unordered_set<std::string>> commits;
179 std::string sha;
180 for (auto& kv : plugins_info_) {
181 for (auto& kv2 : kv.second) {
182 scrimmage::PluginInfo& plugin_info = kv2.second;
183 if (!plugin_info.returned)
184 continue;
185
186 std::string path = boost::filesystem::path(plugin_info.path).parent_path().string();
187
188 if ((sha = scrimmage::get_sha(path)) != "") {
189 commits[sha].insert(plugin_info.name);
190 }
191 }
192 }
193 return commits;
194}

196void PluginManager::find_matching_plugins(
197 const std::string& plugin_name_so,
198 std::unordered_map<std::string, std::list<std::string>>& so_files,
199 std::list<std::string>& plugins) {
200
201 auto it = so_files.find(std::string("lib") + plugin_name_so + LIB_EXT);
202 if (it != so_files.end()) {
203 for (std::string& full_fname : it->second) {
204 if (check_library(full_fname) == 0) {
205 plugins.push_back(full_fname);
206 }
207 }
208 if (plugins.size() > 0) {
209 so_files.erase(it);
210 }
211 }
212}
