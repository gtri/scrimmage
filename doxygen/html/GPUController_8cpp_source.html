69KernelBuildOpts GPUController::get_kernel_build_opts(
70 const std::map<std::string, std::string>& attributes) const {
71 KernelBuildOpts opts;
72 opts.single_precision = scrimmage::get<bool>("single_precision", attributes, false);
73
74 std::string include_dirs_str{
75 scrimmage::get<std::string>("include_dirs", attributes, "kernels")};
76 std::string src_dirs_str{scrimmage::get<std::string>("src_dirs", attributes, "kernels")};
77 std::string preferred_platforms_str{scrimmage::get<std::string>("platforms", attributes, "")};
78
79 std::vector<std::string> include_dirs =
80 scrimmage::str2container<std::vector<std::string>>(include_dirs_str, ",");
81 std::vector<std::string> src_dirs =
82 scrimmage::str2container<std::vector<std::string>>(src_dirs_str, ",");
83
84 opts.preferred_platforms =
85 scrimmage::str2container<std::vector<std::string>>(preferred_platforms_str, ",");
86
87 auto to_path = [&](std::string filename,
88 const fs::path& kernel_dir) -> std::optional<fs::path> {
89 filename = scrimmage::expand_user(filename);
90 if (!fs::exists(filename) && !kernel_dir.empty()) {
91 // The kernel_dir is not part of the recursive iteration below. Explicitly
92 // check if it matches
93 if (kernel_dir.stem() == filename) {
94 return std::optional<fs::path>{kernel_dir};
95 }
96 fs::recursive_directory_iterator kernel_dir_it{kernel_dir};
97 for (auto dir_entry : kernel_dir_it) {
98 if (dir_entry.is_directory() && dir_entry.path().stem() == filename) {
99 return std::optional<fs::path>{dir_entry.path()};
100 }
101 }
102 }
103 return std::nullopt;
104 };
105
106 std::vector<std::optional<fs::path>> include_dir_opts;
107 std::vector<std::optional<fs::path>> src_dir_opts;
108
109 for (const fs::path& kernel_dir : kernel_dirs_) {
110 std::transform(
111 include_dirs.cbegin(),
112 include_dirs.cend(),
113 std::back_inserter(include_dir_opts),
114 [&](const std::string& filename) { return to_path(filename, kernel_dir); });
115 std::transform(
116 src_dirs.cbegin(),
117 src_dirs.cend(),
118 std::back_inserter(src_dir_opts),
119 [&](const std::string& filename) { return to_path(filename, kernel_dir); });
120 }
121
122 for (auto include_dir_opt : include_dir_opts) {
123 if (include_dir_opt.has_value()) {
124 opts.include_dirs.push_back(include_dir_opt.value());
125 }
126 }
127
128 for (auto src_dir_opt : src_dir_opts) {
129 if (src_dir_opt.has_value()) {
130 opts.src_dirs.push_back(src_dir_opt.value());
131 }
132 }
133
134 return opts;
135}

137cl::Program::Sources GPUController::read_kernels(const std::vector<fs::path>& kernel_src_dirs) {
138 auto kernel_src_to_cl_src = [&](KernelSource kernel_src) {
139#ifndef CL_HPP_ENABLE_PROGRAM_CONSTRUCTION_FROM_ARRAY_COMPATIBILITY
140 return kernel_src.src_;
141#else
142 return std::pair<const char*, std::size_t>{kernel_src.src_.c_str(), kernel_src.size()};
143#endif
144 };
145
146 cl::Program::Sources ret_srcs;
147 std::unordered_set<KernelSource, KernelSource::Hash, KernelSource::EqualTo> kernel_srcs;
148 for (auto kernel_dir : kernel_src_dirs) {
149 fs::directory_iterator kernel_dir_it{kernel_dir};
150 for (auto dir_entry : kernel_dir_it) {
151 std::ifstream ifstream;
152 std::stringstream buffer;
153 fs::path path = dir_entry.path();
154 if (path.extension() != ".cl") {
155 continue;
156 }
157 ifstream.open(path);
158 if (ifstream.fail()) {
159 std::stringstream ss;
160 ss << "Error reading kernel file '" << path << "'";
161 LOG_ERROR(ss.str());
162 } else {
163 buffer << ifstream.rdbuf();
164 std::string src = buffer.str();
165 kernel_srcs.emplace(path.stem(), src);
166 }
167 }
168 }
169
170 std::transform(
171 kernel_srcs.cbegin(),
172 kernel_srcs.cend(),
173 std::back_inserter(ret_srcs),
174 kernel_src_to_cl_src);
175 return ret_srcs;
176}

217std::optional<std::pair<cl::Kernel, cl::CommandQueue>> GPUController::build_kernel(
218 const std::string& kernel_name,
219 const KernelBuildOpts& opts) {
220 cl_int err;
221 std::optional<cl::Device> device_opt = pick_device(opts);
222 if (!device_opt.has_value()) {
223 return std::nullopt;
224 }
225 cl::Device& device = device_opt.value();
226 cl::Program::Sources cl_srcs;
227 std::string compiler_opts;
228
229 cl_srcs = read_kernels(opts.src_dirs);
230
231 cl::Context context{device, nullptr, nullptr, nullptr, &err};
232 if (OpenCLUtils::check_error(err, __FILE__, __LINE__, "Unable to Initalize OpenCL Context")) {
233 return std::nullopt;
234 }
235
236 cl::Program program = cl::Program{context, cl_srcs, &err};
237 if (OpenCLUtils::check_error(err, __FILE__, __LINE__, "Unable to initalize OpenCL Program")) {
238 return std::nullopt;
239 }
240
241 if (opts.single_precision) {
242 compiler_opts += " -D SINGLE_PRECISION";
243 }
244 for (const std::filesystem::path& include_dir : opts.include_dirs) {
245 compiler_opts += " -I " + include_dir.string();
246 }
247
248 err = program.build(device, compiler_opts.c_str(), nullptr, nullptr);
249
250 // Display Compiler Error messages, if any.
251 if (err != 0) {
252 std::vector<std::pair<cl::Device, std::string>> build_logs =
253 program.getBuildInfo<CL_PROGRAM_BUILD_LOG>();
254 std::string build_log = "";
255 for (auto device_build_log : build_logs) {
256 build_log += "Build Log for " + get_device_name(device_build_log.first) + "\n\n"
257 + device_build_log.second + "\n";
258 }
259 OpenCLUtils::check_error(
260 err,
261 __FILE__,
262 __LINE__,
263 "Error Building Kernels: Output Build Log:\n" + build_log);
264 return std::nullopt;
265 }
266 cl::Kernel kernel{program, kernel_name.c_str(), &err};
267 if (OpenCLUtils::check_error(
268 err,
269 __FILE__,
270 __LINE__,
271 "Error creating Kernel \'" + kernel_name + "\'")) {
272 return std::nullopt;
273 }
274
275 cl::CommandQueue queue{context, device, 0, &err};
276 if (OpenCLUtils::check_error(err, __FILE__, __LINE__, "Error creating command Queue")) {
277 return std::nullopt;
278 }
279
280 return std::pair<cl::Kernel, cl::CommandQueue>{kernel, queue};

283std::optional<cl::Device> GPUController::pick_device(const KernelBuildOpts& opts) {
284 std::vector<cl::Platform> platforms;
285 cl::Platform::get(&platforms);
286 std::map<std::string, std::vector<cl::Device>> device_map;
287
288 // Choose devices that can support the precision we want
289 for (cl::Platform& platform : platforms) {
290 std::vector<cl::Device> devices;
291 std::string platform_name = platform.getInfo<CL_PLATFORM_NAME>();
292 platform.getDevices(CL_DEVICE_TYPE_GPU, &devices);
293 for (auto it = devices.begin(); it != devices.end(); ++it) {
294 if (opts.single_precision || OpenCLUtils::supports_fp64(*it)) {
295 device_map[platform_name].push_back(*it);
296 }
297 }
298 }
299
300 if (device_map.size() == 0 && !opts.single_precision) {
301 LOG_WARN("Warning: No Platform found that supports double precision. Try enabling single precision.");
302 return std::nullopt;
303 }
304
305 std::vector<cl::Device> usable_devices;
306 if (opts.preferred_platforms.empty()) {
307 for (auto platform_name_devices : device_map) {
308 const std::vector<cl::Device>& platform_devices = platform_name_devices.second;
309 if (!platform_devices.empty()) {
310 usable_devices.insert(
311 usable_devices.end(),
312 platform_devices.begin(),
313 platform_devices.end());
314 }
315 }
316 } else {
317 for (std::string preferred_platform : opts.preferred_platforms) {
318 if (device_map.count(preferred_platform) == 0) {
319 std::stringstream ss;
320 ss << "Warning: No Platform named " << preferred_platform << " found.";
321 LOG_WARN(ss.str());
322 } else {
323 std::vector<cl::Device>& platform_devices = device_map[preferred_platform];
324 usable_devices.insert(
325 usable_devices.end(),
326 platform_devices.begin(),
327 platform_devices.end());
328 }
329 }
330 }
331
332 if (usable_devices.empty()) {
333 return std::nullopt;
334 }
335
336 using ClockFreqDevice = std::pair<cl_uint, cl::Device>;
337 std::vector<std::pair<cl_uint, cl::Device>> clock_freq_devices;
338
339 std::transform(
340 usable_devices.begin(),
341 usable_devices.end(),
342 std::back_inserter(clock_freq_devices),
343 [](cl::Device& device) {
344 cl_int err;
345 cl_uint max_clock_freq = device.getInfo<CL_DEVICE_MAX_CLOCK_FREQUENCY>(&err);
346 if (OpenCLUtils::check_error(
347 err,
348 __FILE__,
349 __LINE__,
350 "Error querying device MAX_CLOCK_FREQUENCY")) {
351 max_clock_freq =
352 std::numeric_limits<cl_uint>::min(); // Still need to return something,
353 // but will not picked as the device
354 // with the highest clock frequency
355 }
356 return std::pair{max_clock_freq, device};
357 });
358
359 auto max_it = std::max_element(
360 clock_freq_devices.cbegin(),
361 clock_freq_devices.cend(),
362 [](const ClockFreqDevice& rhs, const ClockFreqDevice& lhs) {
363 return rhs.first < lhs.first;
364 });
365 if (max_it == clock_freq_devices.end()) {
366 return std::nullopt;
367 }
368 return std::make_optional(max_it->second);
369}
