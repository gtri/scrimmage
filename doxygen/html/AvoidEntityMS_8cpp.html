72void AvoidEntityMS::avoidance_vectors(ContactMap& contacts, std::vector<Eigen::Vector3d>& O_vecs) {
73 for (auto it = contacts.begin(); it != contacts.end(); it++) {
74 // Ignore own position / id
75 if (it->second.id().id() == parent_->id().id()) {
76 continue;
77 }
78
79 if (!avoid_non_team_ && it->second.id().team_id() != parent_->id().team_id()) {
80 continue;
81 }
82
83 Eigen::Vector3d diff = it->second.state()->pos() - state_->pos();
84
85 double O_mag = 0;
86 double dist = diff.norm();
87
88 if (dist > sphere_of_influence_) {
89 O_mag = 0;
90 } else if (minimum_range_ < dist && dist <= sphere_of_influence_) {
91 O_mag = (sphere_of_influence_ - dist) / (sphere_of_influence_ - minimum_range_);
92 } else if (dist <= minimum_range_) {
93 O_mag = 1e10;
94 }
95
96 Eigen::Vector3d O_dir = -O_mag * diff.normalized();
97 O_vecs.push_back(O_dir);
98 }
