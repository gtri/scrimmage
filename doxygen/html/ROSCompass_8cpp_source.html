59void ROSCompass::init(std::map<std::string, std::string>& params) {
60 // Setup robot namespace
61 vehicle_name_ = sc::get<std::string>("vehicle_name", params, "none");

63 ros_namespace_ = sc::get<std::string>("ros_namespace_prefix", params, "robot");
64 ros_namespace_ += std::to_string(parent_->id().id());
65 } else {
66 ros_namespace_ = vehicle_name_;
67 }

69 if (!ros::isInitialized()) {

76 // Create Publisher

80bool ROSCompass::step() {
81 // Obtain current state information
82 sc::StatePtr& state = parent_->state_truth();
83
84 // Scrimmage is in ENU so all outputs are in ENU (East North Up).
85
86 // Get rotation vector
87 // Rotate Reverse: Get rotation between orientation of the body and ENU orientation of the north
88 // pole (0, 1, 0)
89 Eigen::Vector3d rotation = state->quat().rotate_reverse(Eigen::Vector3d(0.0, 1.0, 0.0));
90
91 // Fill Compass Message. We're using the magnetic field message, but the info is the rotation
92 // vector.
93 sensor_msgs::MagneticField compass_msg;
94 compass_msg.magnetic_field.x = rotation.x();
95 compass_msg.magnetic_field.y = rotation.y();
96 compass_msg.magnetic_field.z = rotation.z();
97
98 // Header
99 std_msgs::Header header; // empty header
100 // TODO: header.frame = ? // No system for ROS Frame ID's yet
101 header.stamp = ros::Time::now(); // time
102 compass_msg.header = header;
103
104 // Publish Compass information
105 compass_pub_.publish(compass_msg);
106
107 return true;
108}
