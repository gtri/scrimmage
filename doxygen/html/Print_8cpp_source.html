134const char* Print::getWarningLevel(PrintEnums::WARN_LEVEL e) {
135 const std::map<PrintEnums::WARN_LEVEL, const char*> GetAsStrings{
136 {PrintEnums::WARN_LEVEL::DEV, "DEV"},
137 {PrintEnums::WARN_LEVEL::INFO, "INFO"},
138 {PrintEnums::WARN_LEVEL::WARNING, "WARN"},
139 {PrintEnums::WARN_LEVEL::ERROR, "ERROR"}};

154std::string Print::formatTime(double time) {
155 if (std::isnan(time)) {
156 return "-----.---";
157 } else {
158 std::ostringstream o;
159 o << std::internal << std::fixed << std::setprecision(3) << std::setw(9)
160 << std::setfill('0') << time;
161 return o.str();
162 }
163}

165std::string Print::formatMsg(PrintEnums::WARN_LEVEL level, PrintData& data, std::string msg) {
166 std::ostringstream pre;
167 std::ostringstream formattedmsg;
168 std::string entitySection = "";
169
170 // add entity_id if available
171 if (data.entity_id_ > 0) {
172 entitySection = "[Entity: " + std::to_string(data.entity_id_) + "]";
173 }
174
175 // create our prefix
176 pre << formatTime(data.time_) << "s [" << data.name_ << "]" << entitySection << "["
177 << Print::getWarningLevel(level) << "]: ";
178
179 // remove last character if it's a newline
180 if (msg.back() == '\n') {
181 msg.pop_back();
182 }
183
184 // iterate and place the prefix at the beginning of each newline
185 formattedmsg << pre.str();
186 for (auto i = msg.cbegin(); i != msg.cend(); ++i) {
187 formattedmsg << *i;
188 // add prefix after each newline
189 if (*i == '\n') {
190 formattedmsg << pre.str();
191 }
192 }
193
194 return formattedmsg.str();
195}

211 PrintEnums::WARN_LEVEL level,
212 PrintData& data,
213 std::string msg) {

215
216 o << formatMsg(level, data, msg) << std::endl;
217
218 if (write_file()) {

221 if (write_console()) {

223 } // write to console
