63bool GoToWaypoint::step_autonomy(double t, double dt) {
64 if (waypoint_status_) {
65
66 sc::State ns = *(parent_->state_belief());
67
68 double lat = (waypoint_[2] == "X") ? ns.pos()(0) : stod(waypoint_[2]);
69 double lon = (waypoint_[3] == "Y") ? ns.pos()(1) : stod(waypoint_[3]);
70 double alt = (waypoint_[4] == "Z") ? ns.pos()(2) : stod(waypoint_[4]);
71
72 std::string type = waypoint_[0];
73 if (type != "XYZ" && type != "GPS") {
74 std::cout << "Invalid waypoint type: " << type << std::endl;
75 } else if (type == "XYZ") {
76 parent_->projection()->Reverse(lat, lon, alt, lat, lon, alt);
77 }
78
79 Waypoint wp(lat, lon, alt);
80 wp.set_time(std::stod(waypoint_[1]));
81 scrimmage::Quaternion quat(
82 sc::Angles::deg2rad(std::stod(waypoint_[5])),
83 sc::Angles::deg2rad(std::stod(waypoint_[6])),
84 sc::Angles::deg2rad(std::stod(waypoint_[7])));
85 wp.set_quat(quat);
86 wp.set_position_tolerance(std::stod(waypoint_[8]));
87 wp.set_quat_tolerance(std::stod(waypoint_[9]));
88
89 wp_list_.waypoints().push_back(wp);
90
91 publish_waypoint(t);
92 }
93
94 return true;
95}

97void GoToWaypoint::publish_waypoint(double t) {
98 auto msg = std::make_shared<sc::Message<WaypointList>>();
99
100 std::list<Waypoint>::iterator it = wp_list_.waypoints().begin();
101 while (it != wp_list_.waypoints().end()) {
102 if (t >= it->time()) {
103 msg->data.set_mode(wp_list_.mode());
104 msg->data.waypoints().push_back(*it);
105
106 // Erase this waypoint from the main list
107 std::list<Waypoint>::iterator it_del = it;
108 it = wp_list_.waypoints().erase(it_del);
109 } else {
110 ++it;
111 }
112 }
113
114 // Publish the waypoint list message if it has waypoints
115 if (msg->data.waypoints().size() > 0) {
116 waypoint_list_pub_->publish(msg);
117 }
118}
