60bool find_terrain_files(
61 std::string terrain_name,
62 ConfigParse& terrain_parse,
63 std::shared_ptr<scrimmage_proto::UTMTerrain>& utm_terrain) {
64 if (utm_terrain == nullptr) {
65 LOG_ERROR("utm_terrain is null.");
66 return false;
67 }
68
69 utm_terrain->set_terrain_name(terrain_name);
70
71 // Parse the motion model xml file
72 terrain_parse.set_required("polydata");
73 terrain_parse.set_required("texture");
74 terrain_parse.set_required("system");
75 terrain_parse.set_required("hemisphere");
76 terrain_parse.set_required("zone");
77
78 FileSearch file_search;
79 std::map<std::string, std::string> overrides; // empty, no overrides
80 if (terrain_parse.parse(overrides, terrain_name, "SCRIMMAGE_DATA_PATH", file_search)) {
81 std::string polydata_file =
82 terrain_parse.directory() + "/" + terrain_parse.params()["polydata"];
83 std::string texture_file =
84 terrain_parse.directory() + "/" + terrain_parse.params()["texture"];
85 std::string extrusion_file =
86 terrain_parse.directory() + "/" + terrain_parse.params()["extrusion"];
87 std::string extrusion_property = terrain_parse.params()["extrusion_property"];
88 if (fs::exists(extrusion_file) && fs::is_regular_file(extrusion_file)
89 && !extrusion_property.empty()) {
90 utm_terrain->set_extrusion_file(extrusion_file);
91 utm_terrain->set_extrusion_property(extrusion_property);
92 utm_terrain->set_enable_extrusion(true);
93 }
94 if (fs::exists(polydata_file) && fs::exists(texture_file)
95 && fs::is_regular_file(polydata_file) && fs::is_regular_file(texture_file)) {
96 utm_terrain->set_texture_file(texture_file);
97 utm_terrain->set_poly_data_file(polydata_file);
98 return true;
99 }
100 LOG_WARN("Terrain '" << terrain_name << "' files missing - polydata: "
101 << polydata_file << ", texture: " << texture_file);
102 } else {
103 const char* data_path = std::getenv("SCRIMMAGE_DATA_PATH");
104 LOG_WARN("Terrain '" << terrain_name << "' config not found in SCRIMMAGE_DATA_PATH="
105 << (data_path ? data_path : "(not set)"));
106 }
107 return false;
108}

110bool find_model_properties(
111 std::string model_name,
112 ConfigParse& cv_parse,
113 FileSearch& file_search,
114 std::map<std::string, std::string>& overrides,
115 std::shared_ptr<scrimmage_proto::ContactVisual>& cv,
116 bool& mesh_found,
117 bool& texture_found) {
118 if (cv == nullptr) {
119 LOG_ERROR("Contact Visual is null.");
120 return false;
121 }
122
123 mesh_found = texture_found = false;
124
125 cv->set_name(model_name);
126
127 cv_parse.set_required("model");
128
129 if (cv_parse.parse(overrides, model_name, "SCRIMMAGE_DATA_PATH", file_search)) {
130 std::string model_file = cv_parse.directory() + "/" + cv_parse.params()["model"];
131 std::string texture_file = cv_parse.directory() + "/" + cv_parse.params()["texture"];
132
133 if (fs::exists(model_file) && fs::is_regular_file(model_file)) {
134 cv->set_model_file(model_file);
135 mesh_found = true;
136 }
137 if (fs::exists(texture_file) && fs::is_regular_file(texture_file)) {
138 cv->set_texture_file(texture_file);
139 texture_found = true;
140 }
141 }
142
143 auto it_info = cv_parse.params().find("visual_scale");
144 if (it_info != cv_parse.params().end()) {
145 cv->set_scale(std::stod(it_info->second));
146 } else {
147 cv->set_scale(1.0);
148 }
149
150 bool valid_rotate = false;
151 it_info = cv_parse.params().find("visual_rpy");
152 if (it_info != cv_parse.params().end()) {
153 std::vector<double> temp_rotate;
154 if (str2container(it_info->second, " ", temp_rotate, 3)) {
155 valid_rotate = true;
156 for (auto it = temp_rotate.begin(); it != temp_rotate.end(); it++) {
157 cv->add_rotate(*it);
158 }
159 } else {
160 LOG_WARN("Error parsing: visual_rpy");
161 }
162 }
163 if (!valid_rotate) {
164 for (int i = 0; i < 3; i++) {
165 cv->add_rotate(0);
166 }
167 }
168 return (mesh_found && texture_found);
169}

191std::string expand_user(std::string path) {
192 if (not path.empty() and path[0] == '~') {
193 assert(path.size() == 1 or path[1] == '/'); // or other error handling
194 char const* home = std::getenv("HOME");
195 if (home or (home = std::getenv("USERPROFILE"))) {
196 path.replace(0, 1, home);
197 } else {
198 char const *hdrive = std::getenv("HOMEDRIVE"), *hpath = std::getenv("HOMEPATH");
199 assert(hdrive); // or other error handling
200 assert(hpath);
201 path.replace(0, 1, std::string(hdrive) + hpath);
202 }
203 }
204 return path;

278bool get_vec_of_vecs(
279 const std::string& str,
280 std::vector<std::vector<std::string>>& out,
281 const std::string& delims) {
282
283 // Of the form...
284 // [a b c d]
285 // [e f g h]
286 // [h i j k]
287 out.clear();
288
289 std::string str_copy(str);
290 remove_leading_spaces(str_copy);
291 remove_trailing_spaces(str_copy);
292
293 // First split based on user defined delimiters (typically ", ")
294 std::vector<std::string> tokens;
295 boost::algorithm::split(tokens, str_copy, boost::is_any_of(delims), boost::token_compress_on);
296
297 // The tokens vector consists of the delimited items. Break up each vector
298 // based on the start "[" and stop "]" delimiters. However, we will allow
299 // one-level of nested start and stop delimiters.
300 std::vector<std::string> items;
301 int open_bracket_count = 0;
302 std::string item = "";
303 for (std::string token : tokens) {
304
305 // Remove whitespace
306 token.erase(
307 std::remove_if(
308 token.begin(),
309 token.end(),
310 [](unsigned char x) { return std::isspace(x); }),
311 token.end());
312
313 // cout << token << endl;
314
315 for (unsigned int i = 0; i < token.size(); i++) {
316 if (token[i] == '[') {
317 if (open_bracket_count > 0) {
318 item += "[";
319 }
320 ++open_bracket_count;
321 } else if (token[i] == ']') {
322 if (open_bracket_count > 1) {
323 item += "]";
324 }
325 --open_bracket_count;
326
327 if (open_bracket_count == 0) {
328 items.push_back(item);
329 out.push_back(items);
330 items.clear();
331 item = "";
332 }
333 } else if (open_bracket_count == 2) {
334 item += std::string(1, token[i]);
335 } else if (open_bracket_count > 0) {
336 item += std::string(1, token[i]);
337 }
338 }
339
340 if (open_bracket_count == 1) {
341 items.push_back(item);
342 item.clear();
343 } else if (open_bracket_count == 0) {
344 out.push_back(items);
345 item.clear();
346 items.clear();
347 }
348 }
349
350 // Remove any empty strings from the individual vectors
351 for (auto& vec : out) {
352 vec.erase(
353 std::remove_if(vec.begin(), vec.end(), [](auto& str) { return str.empty(); }),
354 vec.end());
355 }
356
357 // Remove any empty vectors from the out vector of vectors
358 out.erase(
359 std::remove_if(out.begin(), out.end(), [](auto& vec) { return vec.empty(); }),
360 out.end());
361
362 return true;
363}

369bool set_pid_gains(sc::PID& pid, std::string str, bool is_angle) {
370 std::vector<std::string> str_vals;
371 boost::split(str_vals, str, boost::is_any_of(","));
372
373 if (str_vals.size() != 4) {
374 LOG_ERROR("PID gains string '" << str << "' invalid format (expected: p,i,d,i_lim)");
375 return false;
376 } else {
377 try {
378 double p = std::stod(str_vals[0]);
379 double i = std::stod(str_vals[1]);
380 double d = std::stod(str_vals[2]);
381 pid.set_parameters(p, i, d);
382
383 if (is_angle) {
384 double i_lim = sc::Angles::deg2rad(std::stod(str_vals[3]));
385 pid.set_integral_band(i_lim);
386 pid.set_is_angle(true);
387 } else {
388 double i_lim = std::stod(str_vals[3]);
389 pid.set_integral_band(i_lim);
390 pid.set_is_angle(false);
391 }
392 } catch (const std::exception& e) {
393 LOG_ERROR("PID gains: failed to parse values from '" << str << "': " << e.what());
394 return false;
395 }
396 }
397 return true;
398}
