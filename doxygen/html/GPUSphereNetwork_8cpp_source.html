90bool GPUSphereNetwork::step(
91 std::map<std::string, std::list<NetworkDevicePtr>>& pubs,
92 std::map<std::string, std::list<NetworkDevicePtr>>& subs) {
93 using EntityIdPair = std::pair<int, int>;
94
95 std::map<int, StatePtr> states;
96 std::map<int, EntityPtr> ents;
97
98 std::map<int, std::map<std::string, std::list<NetworkDevicePtr>>> ent_pubs;
99 std::map<int, std::map<std::string, std::list<NetworkDevicePtr>>> ent_subs;
100
101 for (const auto& kv : pubs) {
102 const std::string& topic = kv.first;
103 for (NetworkDevicePtr device : kv.second) {
104
105 /*
106 * TODO: Network devices are not removed from the pubusbs when
107 * an entity is destroyed. This should happen in the sim control,
108 * unless there is a good reason why this should not happen
109 */
110 EntityPtr entity = device->plugin()->parent();
111 if (entity == nullptr) {
112 continue;
113 }
114 StatePtr state = entity->state_truth();
115 int id = entity->id().id();
116 states[id] = state;
117 ents[id] = entity;
118 ent_pubs[id][topic].push_back(device);
119 }
120 }
121
122 for (const auto& kv : subs) {
123 const std::string& topic = kv.first;
124 for (NetworkDevicePtr device : kv.second) {
125 EntityPtr entity = device->plugin()->parent();
126 if (entity == nullptr) {
127 continue;
128 }
129 StatePtr state = entity->state_truth();
130 int id = entity->id().id();
131 states[id] = state;
132 ents[id] = entity;
133 ent_subs[id][topic].push_back(device);
134 }
135 }
136
137 std::size_t num_entities = states.size();
138 if (num_entities <= 1) {
139 return true;
140 }
141
142 std::set<EntityIdPair> proximity_pairs = utils_->proximity_pairs(states);
143
144 // Mapping of an individual pub to all the subs it needs to send messages to
145 std::map<NetworkDevicePtr, std::vector<NetworkDevicePtr>> pub_to_subs;
146 using TopicNetworkDevices = std::map<std::string, std::list<NetworkDevicePtr>>;
147
148 const auto add_pub_subs =
149 [&pub_to_subs](const TopicNetworkDevices& pubs, const TopicNetworkDevices& subs) -> void {
150 for (const auto& kv : pubs) {
151 const std::string& topic_name = kv.first;
152 const std::list<NetworkDevicePtr>& topic_pubs = kv.second;
153 const std::list<NetworkDevicePtr>& topic_subs = subs.at(topic_name);
154
155 for (NetworkDevicePtr topic_pub : topic_pubs) {
156 std::vector<NetworkDevicePtr>& subs = pub_to_subs[topic_pub];
157 subs.insert(subs.end(), topic_subs.cbegin(), topic_subs.cend());
158 }
159 }
160 };
161
162 for (const EntityIdPair& prox_pair : proximity_pairs) {
163 EntityPtr ent1 = ents.at(prox_pair.first);
164 EntityPtr ent2 = ents.at(prox_pair.second);
165
166 const std::map<std::string, std::list<NetworkDevicePtr>>& pubs1 = ent_pubs[prox_pair.first];
167 const std::map<std::string, std::list<NetworkDevicePtr>>& subs1 = ent_subs[prox_pair.first];
168
169 const std::map<std::string, std::list<NetworkDevicePtr>>& pubs2 =
170 ent_pubs[prox_pair.second];
171 const std::map<std::string, std::list<NetworkDevicePtr>>& subs2 =
172 ent_subs[prox_pair.second];
173
174 add_pub_subs(pubs1, subs2);
175 add_pub_subs(pubs2, subs1);
176 }
177
178 for (auto& kv : pub_to_subs) {
179 NetworkDevicePtr pub = kv.first;
180 std::vector<NetworkDevicePtr>& subs = kv.second;
181 pub->enforce_queue_size();
182 const std::list<MessageBasePtr> msgs = pub->pop_msgs<MessageBase>();
183 if (msgs.empty()) {
184 continue;
185 }
186
187 for (NetworkDevicePtr sub : subs) {
188 if (sub->undelivered_msg_list_size() > 0) {
189 // deliver undelivered messages if delay time has passed
190 sub->deliver_undelivered_msg(time_->t(), is_stochastic_delay_);
191 }
192 for (auto& msg : msgs) {
193 if (is_successful_transmission(pub->plugin(), sub->plugin())) {
194 double msg_delay = get_transmission_delay();
195
196 if (msg_delay < 0) {
197 msg->time = time_->t();
198 sub->add_msg(msg);
199 } else {
200 // put msg in subs undelivered msg queue
201 msg->time = time_->t() + msg_delay;
202 sub->add_undelivered_msg(msg, is_stochastic_delay_);
203 }
204 }
205 }
206 sub->enforce_queue_size();
207 }
208 }
209
210 // I kind of hate this. Clear all messages that were not processed.
211 for (auto& kv : pubs) {
212 for (auto pub : kv.second) {
213 pub->clear_msg_list();
214 }
215 }
216
217 return true;
218}

276bool GPUSphereNetwork::is_reachable(
277 const scrimmage::EntityPluginPtr& pub_plugin,
278 const scrimmage::EntityPluginPtr& sub_plugin) {
279 // Never reachable if plugin's entity was destroyed
280 if (pub_plugin->parent() == nullptr || sub_plugin->parent() == nullptr)
281 return false;
282 // If the publisher and subscriber have the same parent, it is reachable
283 if (pub_plugin->parent() == sub_plugin->parent())
284 return true;
285
286 // Is the publisher to subscriber already in the reachable map?
287 int pub_id = pub_plugin->parent()->id().id();
288 int sub_id = sub_plugin->parent()->id().id();
289
290 // Find the publisher in the mapping
291 auto it_pub = reachable_map_.find(pub_id);
292 if (it_pub != reachable_map_.end()) {
293 // See if the publisher has a reachability mapping for the subscriber
294 auto it_sub = it_pub->second.find(sub_id);
295 if (it_sub != it_pub->second.end()) {
296 return it_sub->second;
297 }
298 }
299
300 // No reachability mapping exists. Find all entity IDs that are reachable.
301 std::vector<sc::ID> neigh;
302 rtree_->neighbors_in_range(pub_plugin->parent()->state_truth()->pos(), neigh, range_);
303
304 // Add the entity IDs to the publisher's reachability set
305 // Look for the subscriber ID
306 reachable_map_[pub_id][sub_id] = false; // Default to not reachable
307 reachable_map_[sub_id][pub_id] = false; // Default to not reachable
308 for (sc::ID id : neigh) {
309 auto ent_neighbor = id_to_ent_map_->find(id.id());
310 if (ent_neighbor == id_to_ent_map_->end()) {
311 std::stringstream ss;
312 ss << "Warning: Sphere Network entity id doesn't exists: " << id.id();
313 LOG_WARN(ss.str());
314 continue;
315 }
316 if (not filter_comms_plane_) {
317 reachable_map_[pub_id][id.id()] = true;
318 reachable_map_[id.id()][pub_id] = true;
319 } else if (within_planar_boundary(
320 pub_plugin->parent()->state_truth()->pos()[2],
321 ent_neighbor->second->state_truth()->pos()[2])) {
322 reachable_map_[pub_id][id.id()] = true;
323 reachable_map_[id.id()][pub_id] = true;
324 }
325 }
326 return reachable_map_[pub_id][sub_id];
327}

335// Return true if the plane does not block communications
336bool GPUSphereNetwork::within_planar_boundary(double z1, double z2) {
337 bool both_above = (z1 >= (comms_boundary_altitude_ - comms_boundary_epsilon_))
338 && (z2 >= (comms_boundary_altitude_ - comms_boundary_epsilon_));
339 bool both_below = (z1 <= (comms_boundary_altitude_ + comms_boundary_epsilon_))
340 && (z2 <= (comms_boundary_altitude_ + comms_boundary_epsilon_));
341 return (both_above || both_below);
342}
