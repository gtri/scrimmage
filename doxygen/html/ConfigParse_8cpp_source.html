63void ConfigParse::recursive_params(
64 rx::xml_node<char>* root,
65 const std::map<std::string, std::string>& overrides,
66 std::map<std::string, std::string>& params,
67 const std::string& prev) {
68 // End condition
69 if (root == 0 || root->name() == std::string("") || root->name() == std::string(" ")) {
70 return;
71 }
72
73 // Process this node
74 std::string name = root->name();
75 if (prev != "") {
76 name = prev + "/" + name;
77 }
78
79 if (params.count(name) > 0) {
80 // Name already exists, consider this a list. Does the size
81 // parameter exist yet?
82 std::string name_size = name + ":size";
83 int size = 2;
84 if (params.count(name_size) > 0) {
85 // It does exist, grab the current size and increment.
86 size = std::stoi(params[name_size]) + 1;
87 }
88 name = name + "_" + std::to_string(size - 1);
89 params[name_size] = std::to_string(size);
90 }
91
92 if (overrides.count(name)) {
93 params[name] = overrides.at(name);
94 } else {
95 params[name] = root->value();
96 }
97
98 // Recurse (depth-first)
99 recursive_params(root->first_node(), overrides, params, name);
100
101 // Recurse (sibling node)
102 recursive_params(root->next_sibling(), overrides, params, prev);
103}

105bool ConfigParse::parse(
106 const std::map<std::string, std::string>& overrides,
107 std::string filename,
108 std::string env_var,
109 FileSearch& file_search,
110 bool verbose) {
111
112 std::string result = "";
113 bool status = file_search.find_file(filename, "xml", env_var, result, verbose);
114 if (!status) {
115 if (boost::algorithm::to_lower_copy(filename) != "sphere") {
116 LOG_WARN("Failed to find configuration: " << filename);
117 }
118 return false;
119 } else if (verbose) {
120 LOG_INFO("ConfigParse: found " << result);
121 }
122 filename_ = result;
123
124 rx::xml_document<> doc;
125 std::ifstream file(filename_.c_str());
126 std::stringstream buffer;
127 buffer << file.rdbuf();
128 file.close();
129 std::string content(buffer.str());
130 doc.parse<0>(&content[0]);
131
132 rx::xml_node<>* config_node = doc.first_node("params");
133 if (config_node == 0) {
134 LOG_ERROR("Missing tag: params");
135 return false;
136 }
137
138 params_.clear();
139 params_["XML_DIR"] = this->directory() + "/";
140 params_["XML_FILENAME"] = filename_;
141
142 recursive_params(config_node->first_node(), overrides, params_, "");
143
144 // Determine if there were any overrides (XML attributes) specified in the
145 // mission file that weren't declared in the Plugin's XML
146 // file. Automatically add these overrides to the params block.
147 for (auto& kv : overrides) {
148 if (params_.count(kv.first) == 0) {
149 params_[kv.first] = kv.second;
150 }
151 }
152
153 for (std::string& node_name : required_) {
154 if (params_.count(node_name) == 0) {
155 LOG_ERROR("Config file is missing XML tag: " << node_name);
156 return false;
157 }
158 }
159 return true;
160}
