60bool GroundCollision::init(

67 // Determine ground collision z-value. If ground_collision_altitude is

78 y,
79 z);
80 ground_collision_z_ = z;

89bool GroundCollision::step_entity_interaction(std::list<sc::EntityPtr>& ents, double t, double dt) {
90 // Account for entities colliding with
91 for (sc::EntityPtr ent : ents) {
92 if (team_ != "all" && std::stoi(team_) != ent->id().team_id()) {
93 continue;
94 }
95
96 if (ent->is_alive() && ent->state_truth()->pos()(2) <= ground_collision_z_) {
97 if (remove_on_collision_) {
98 ent->collision();
99 } else {
100 // Apply a normal force to motion model in opposite direction
101 // of gravity.
102 double force = ent->motion()->mass() * ent->motion()->gravity_magnitude();
103 ent->motion()->set_external_force(Eigen::Vector3d(0, 0, force));
104 // StatePtr &s = ent->state_truth();

111 msg->data.set_entity_id(ent->id().id());
112 collision_pub_->publish(msg);
