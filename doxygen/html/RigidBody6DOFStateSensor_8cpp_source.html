61void RigidBody6DOFStateSensor::init(std::map<std::string, std::string>& params) {
62 // Use the same generator as the parent so that the simulation is
63 // completely deterministic with respect to the simulation seed.
64 gener_ = parent_->random()->gener();
65
66 // Create three independent gaussian noise generators. They will use the
67 // same generator seed.
68 for (int i = 0; i < 3; i++) {
69 std::string tag_name = "pos_noise_" + std::to_string(i);
70 std::vector<double> vec;
71 bool status = get_vec(tag_name, params, " ", vec, 2);
72 if (status) {
73 pos_noise_.push_back(parent_->random()->make_rng_normal(vec[0], vec[1]));
74 } else {
75 pos_noise_.push_back(parent_->random()->make_rng_normal(0, 1));
76 }
77 }
